<script type="text/x-server-plugin">
/* ── BGN Go Fish server ─────────────────────────────────────────
   Rooms: 5-letter codes. Host = Player 0, guest = Player 1.
   The server holds the full deck and both hands; the only
   authority on the game. On every change it sends each player a
   personalised snapshot (their own hand face-up, the opponent as
   a card count) so hands stay hidden. Hand size comes from the
   creating client (rulesConfig.handSize, default 7). Durable room
   state lives in the plugin `state` bytes (versioned layout).   */
const SLOT=1024;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const SUITS=["spades","hearts","clubs","diamonds"];
const RANKCH="A23456789TJQK";   // 13 ranks; index 9 = Ten

let rooms=new Map();        // code -> room (live cache)
const conns=new Map();      // conn.id -> {conn, code, role}

function rankName(i){ return i===9?"10":RANKCH[i]; }
function cardFromByte(b){ return {id:b, suit:SUITS[Math.floor(b/13)], rank:rankName(b%13)}; }
function nowSec(){ return Math.floor(Date.now()/1000); }

/* ── durable state layout (one SLOT per room) ──────────────────
   +0     status  0=free 1=waiting 2=playing 3=finished
   +1..+8 room code (8 ascii, zero padded)
   +9     turn (0/1)
   +10    winner (0 none, 1 p0, 2 p1, 3 draw)
   +11    first (0/1 — who asks first next deal)
   +12..+13 uint16 deck length
   +14..+15 uint16 hand0 length
   +16..+17 uint16 hand1 length
   +18..+21 uint32 lastActive (unix secs)
   +22..+23 uint16 seq (action sequence, for client logs)
   +24..+25 uint16 books0
   +26..+27 uint16 books1
   +28    handSize (cards dealt to each player, 3-13)
   +29    layout version (=1; mismatch slots are reclaimed)
   +30..  deck bytes, then hand0 bytes, then hand1 bytes
   (card bytes 0..51, encoded rankIndex + suitIndex*13)          */
function off(i){ return 1+i*SLOT; }
function w16(o,p,v){ state[o+p]=(v>>8)&255; state[o+p+1]=v&255; }
function r16(o,p){ return (state[o+p]<<8)|state[o+p+1]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function readBytes(o,p,n){ const a=[]; for(let k=0;k<n;k++)a.push(state[o+p+k]); return a; }
function readLastAct(i){ const o=off(i); return (state[o+18]<<24)|(state[o+19]<<16)|(state[o+20]<<8)|state[o+21]; }
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+9]=room.turn;
  state[o+10]=room.winner;
  state[o+11]=room.first;
  w16(o,12,room.deck.length);
  w16(o,14,room.h0.length);
  w16(o,16,room.h1.length);
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+18]=(la>>>24)&255; state[o+19]=(la>>>16)&255; state[o+20]=(la>>>8)&255; state[o+21]=la&255;
  w16(o,22,room.seq);
  w16(o,24,room.books0);
  w16(o,26,room.books1);
  state[o+28]=room.handSize;
  state[o+29]=1;
  let p=o+30;
  for(const c of room.deck)state[p++]=c;
  for(const c of room.h0)state[p++]=c;
  for(const c of room.h1)state[p++]=c;
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const o=off(i);
    const st=state[o];
    if(st===0)continue;
    if(st===1||state[o+29]!==1){ state.fill(0,o,o+SLOT); continue; }   // abandoned/new-layout slots reclaimed
    const deckLen=r16(o,12), h0=r16(o,14), h1=r16(o,16);
    rooms.set(codeAt(i),{
      slot:i,code:codeAt(i),status:st,turn:state[o+9],winner:state[o+10],first:state[o+11],
      handSize:Math.max(3,Math.min(13,state[o+28]||7)),
      deck:readBytes(o,30,deckLen), h0:readBytes(o,30+deckLen,h0), h1:readBytes(o,30+deckLen+h0,h1),
      books0:r16(o,24),books1:r16(o,26),seq:r16(o,22),lastAct:readLastAct(i),
      hostId:-1,guestId:-1,lastAction:null
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const o=off(i);
    const st=state[o];
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }

/* ── game logic ── */
function deal(room){
  const ids=[];
  for(let i=0;i<52;i++)ids.push(i);
  for(let i=51;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); const t=ids[i]; ids[i]=ids[j]; ids[j]=t; }
  const hs=room.handSize;
  room.deck=ids.slice(hs*2);
  room.h0=ids.slice(0,hs);
  room.h1=ids.slice(hs,hs*2);
  room.turn=room.first;
  room.winner=0;
  room.books0=0; room.books1=0;
  room.status=2;
  room.lastAct=nowSec();
  room.lastAction={seq:++room.seq,type:"deal",from:room.first};
}
function layBooksServer(hand,room,p){
  const counts={};
  for(const id of hand)counts[id%13]=(counts[id%13]||0)+1;
  const laid=[];
  for(const r in counts)if(counts[r]>=4)laid.push(+r);
  let n=0;
  for(const r of laid){
    n++;
    const keep=[];
    for(const id of hand)if(id%13!==r)keep.push(id);
    hand.length=0; for(const x of keep)hand.push(x);
  }
  if(p===0)room.books0+=n; else room.books1+=n;
  return laid;
}
function refill(room){
  const hand=room.turn===0?room.h0:room.h1;
  let drew=0;
  while(hand.length===0&&room.deck.length){
    hand.push(room.deck.pop());
    drew++;
    layBooksServer(hand,room,room.turn);
  }
  if(drew){
    room.lastAction={seq:++room.seq,type:"refill",from:room.turn,drew};
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
  }
}
function checkEnd(room){
  if(room.status!==2)return false;
  if(room.books0+room.books1>=13){
    room.status=3;
    room.winner=room.books0>room.books1?1:room.books1>room.books0?2:3;
    return true;
  }
  if(room.deck.length===0&&noTradesLeft(room)){
    room.status=3;
    room.winner=room.books0>room.books1?1:room.books1>room.books0?2:3;
    return true;
  }
  return false;
}
function noTradesLeft(room){
  if(room.deck.length>0)return false;
  const p0={},p1={};
  for(const id of room.h0)p0[id%13]=1;
  for(const id of room.h1)p1[id%13]=1;
  for(const r in p0)if(p1[r])return false;
  return true;
}
function advanceUntilReady(room){
  let guard=0;
  while(guard++<12){
    refill(room);
    if(room.status!==2)return;
    const hand=room.turn===0?room.h0:room.h1;
    if(hand.length)return;
    room.turn=1-room.turn;
    if(checkEnd(room))return;
  }
}
function doAsk(room,fromIdx,rank,toIdx){
  const giveHand=fromIdx===0?room.h0:room.h1;
  const toHand=toIdx===0?room.h0:room.h1;
  const given=toHand.filter(id=>id%13===rank);
  if(given.length){
    for(const id of given)toHand.splice(toHand.indexOf(id),1);
    for(const id of given)giveHand.push(id);
    const laid=layBooksServer(giveHand,room,fromIdx);
    room.turn=fromIdx;                       // asker keeps their turn
    room.lastAction={seq:++room.seq,type:"give",from:fromIdx,to:toIdx,rank,gave:given.length,laid:laid.length};
  }else{
    let drawn=-1,lucky=false;
    if(room.deck.length){
      drawn=room.deck.pop();
      giveHand.push(drawn);
      lucky=(drawn%13)===rank;
    }
    const laid=lucky?layBooksServer(giveHand,room,fromIdx):[];
    room.lastAction={seq:++room.seq,type:"fish",from:fromIdx,to:toIdx,rank,drawn,lucky,laid:laid.length};
    room.turn=lucky?fromIdx:toIdx;
  }
  room.lastAct=nowSec();
  if(!checkEnd(room))advanceUntilReady(room);
  saveRoom(room);
  broadcast(room);
}

/* ── snapshots (personalised — hands stay hidden) ── */
function snapFor(room,i){
  const hand=i===0?room.h0:room.h1;
  return {
    code:room.code,status:room.status,you:i,turn:room.turn,winner:room.winner,
    books:[room.books0,room.books1],
    myHand:hand.map(cardFromByte),
    oppCount:(i===0?room.h1:room.h0).length,
    deckCount:room.deck.length,
    lastAction:room.lastAction,
    oppIn:room.hostId!==-1&&room.guestId!==-1
  };
}
function broadcast(room){
  for(const info of conns.values()){
    if(info.code!==room.code)continue;
    const i=info.role==="host"?0:1;
    try{ info.conn.send(JSON.stringify({t:"snap",snap:snapFor(room,i)})); }catch(e){}
  }
}

/* ── RPCs ── */
self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    let hs=7;
    try{ const d=JSON.parse(data||"{}"); const n=Number(d.handSize); if(Number.isFinite(n))hs=Math.floor(n); }catch(e){}
    hs=Math.max(3,Math.min(13,hs));
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room={slot,code,status:1,turn:0,winner:0,first:Math.floor(Math.random()*2),
      deck:[],h0:[],h1:[],books0:0,books1:0,seq:0,lastAct:nowSec(),handSize:hs,
      hostId:conn.id,guestId:-1,lastAction:{seq:0,type:"wait"}};
    rooms.set(code,room);
    conns.set(conn.id,{conn,code,role:"host",lastAsk:0});
    saveRoom(room);
    return JSON.stringify({ok:true,code,role:"host",snap:snapFor(room,0)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)deal(room);
    conns.set(conn.id,{conn,code,role,lastAsk:0});
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true,role,snap:snapFor(room,role==="host"?0:1)});
  },
  getRoom({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    return JSON.stringify({ok:true,snap:snapFor(room,roleOf(conn)==="host"?0:1)});
  },
  ask({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const info=conns.get(conn.id);
    if(!info||info.code!==room.code)return JSON.stringify({ok:false,err:"not_in_room"});
    const myIdx=info.role==="host"?0:1;
    if(room.turn!==myIdx)return JSON.stringify({ok:false,err:"not_your_turn"});
    const rank=obj.rank;
    if(typeof rank!=="number"||rank<0||rank>12||Math.floor(rank)!==rank)return JSON.stringify({ok:false,err:"bad_rank"});
    const hand=myIdx===0?room.h0:room.h1;
    if(!hand.some(id=>id%13===rank))return JSON.stringify({ok:false,err:"dont_have_rank"});
    const t=nowSec();
    if(t-info.lastAsk<1)return JSON.stringify({ok:false,err:"slow_down"});
    info.lastAsk=t;
    doAsk(room,myIdx,rank,1-myIdx);
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const role=roleOf(conn);
    const oppIn=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(!oppIn)return JSON.stringify({ok:false,err:"opp_gone"});
    room.first=1-room.first;
    room.status=1;
    room.h0=[]; room.h1=[]; room.deck=[];
    room.books0=0; room.books1=0; room.winner=0;
    deal(room);
    room.lastAct=nowSec();
    saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      const role=roleOf(conn);
      room.status=3;
      room.winner=role==="host"?2:1;
      room.lastAction={seq:++room.seq,type:"resign",from:role==="host"?0:1};
      room.lastAct=nowSec();
      saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    const oppIn=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(oppIn)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3;
    room.winner=role==="host"?1:2;
    room.lastAction={seq:++room.seq,type:"resign",from:role==="host"?1:0};
    room.lastAct=nowSec();
    saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      const role=roleOf(conn);
      room.status=3;
      room.winner=role==="host"?2:1;
      room.lastAction={seq:++room.seq,type:"resign",from:role==="host"?0:1};
      room.lastAct=nowSec();
      saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  }
};
self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);
};

loadRooms();
</script>

function startGame(){
  const n=game.mode==="ai"?1+game.numAI:2;
  const cards=shuffle(deck52);
  game.deck=cards.slice(HAND_SIZE*n);
  game.hands=[]; game.books=[];
  for(let p=0;p<n;p++){ game.hands.push(cards.slice(p*HAND_SIZE,p*HAND_SIZE+HAND_SIZE)); game.books.push([]); }
  game.turn=0; game.over=false; game.winner=null;
  game.pendingRank=null; game.aiBusy=false; game.askedFor=[];
  game.myIndex=0; game.deckCount=game.deck.length; game.oppIn=true;
  clearLog();
  logMsg("<b>Deal!</b> "+n+" players, "+HAND_SIZE+" cards each. "+turnStarterText(0));
  sfx("deal");
  renderAll();
}
function layBooks(p){
  const hand=game.hands[p];
  const laid=[];
  for(const r of distinctRanks(hand)){
    if(countRank(hand,r)>=4){
      for(let i=hand.length-1;i>=0;i--)if(hand[i].rank===r)hand.splice(i,1);
      game.books[p].push(r);
      laid.push(r);
    }
  }
  return laid;
}
function refillHand(){
  let drew=0;
  while(game.hands[game.turn].length===0&&game.deck.length){
    game.hands[game.turn].push(game.deck.pop());
    drew++;
    layBooks(game.turn);
  }
  if(drew)logMsg(playerName(game.turn)+" drew "+(drew>1?drew+" cards":"a card")+" to refill an empty hand.");
  game.deckCount=game.deck.length;
}
function ensureTurnHand(){
  if(game.hands[game.turn].length>0)return;
  refillHand();
  if(game.hands[game.turn].length===0)logMsg(playerName(game.turn)+" has no cards and the pond is empty.");
}
function doAsk(fromIdx,toIdx,rank){
  const giver=game.hands[toIdx];
  const given=giver.filter(c=>c.rank===rank);
  for(const c of given)giver.splice(giver.indexOf(c),1);
  game.askedFor.push({from:fromIdx,to:toIdx,rank});
  if(given.length){
    const taker=game.hands[fromIdx];
    for(const c of given)taker.push(c);
    const laid=layBooks(fromIdx);
    return {gave:given,fish:false,drawn:null,laid};
  }
  let drawn=null,lucky=false;
  if(game.deck.length){
    drawn=game.deck.pop();
    game.hands[fromIdx].push(drawn);
    lucky=drawn.rank===rank;
    game.deckCount=game.deck.length;
  }
  const laid=lucky?layBooks(fromIdx):[];
  return {gave:[],fish:true,drawn,laid,lucky};
}
function checkEnd(){
  const tot=game.books.reduce((s,b)=>s+b.length,0);
  if(tot>=13)return true;
  if(noTradesLeft())return true;
  return false;
}
function noTradesLeft(){
  if(game.deck.length>0)return false;
  const owners=new Map();
  for(let p=0;p<game.hands.length;p++){
    for(const c of game.hands[p]){
      if(!owners.has(c.rank))owners.set(c.rank,new Set());
      owners.get(c.rank).add(p);
    }
  }
  for(const s of owners.values())if(s.size>1)return false;
  return true;
}
function finishGame(){
  game.over=true;
  const counts=game.books.map(b=>b.length);
  const max=Math.max(...counts);
  game.winner=counts.map((c,i)=>c===max?i:null).filter(x=>x!==null);
  sfx(game.winner.indexOf(meIndex())!==-1 ? "win" : "lose");
}
function applyAsk(fromIdx,toIdx,rank){
  const res=doAsk(fromIdx,toIdx,rank);
  const pn=playerName(fromIdx), tn=playerName(toIdx);
  const rl=rankLabel(rank);
  game.pendingRank=null;
  if(res.gave.length){
    logMsg(pn+" asked "+tn+" for "+rl+"s — got "+res.gave.length+"!");
    if(res.laid.length){ logMsg(pn+" laid down a book of "+res.laid.join("s, ")+"s!"); sfx("tada"); }
    else sfx("coin");
    if(checkEnd()){ finishGame(); renderAll(); return; }
    ensureTurnHand();
    if(checkEnd()){ finishGame(); renderAll(); return; }
    if(game.hands[fromIdx].length===0){ nextTurn(); return; }
    flashBanner("Got "+res.gave.length+" "+rl+(res.gave.length>1?"s":"")+(res.laid.length?" — book laid!":"")+" Ask again.");
    renderAll();
    if(game.mode==="ai"&&game.turn!==0)scheduleAI();
  }else if(res.drawn){
    if(res.lucky){
      logMsg(pn+" asked "+tn+" for "+rl+"s — go fish, but drew a "+rl+"! Lucky!");
      if(res.laid.length){ logMsg(pn+" laid down a book of "+rl+"s!"); sfx("tada"); }
      else sfx("tada");
      if(checkEnd()){ finishGame(); renderAll(); return; }
      ensureTurnHand();
      if(checkEnd()){ finishGame(); renderAll(); return; }
      if(game.hands[fromIdx].length===0){ nextTurn(); return; }
      fishFlash(res.drawn);
      flashBanner("Lucky — you fished a "+rl+". Ask again!");
      renderAll();
      if(game.mode==="ai"&&game.turn!==0)scheduleAI();
    }else{
      logMsg(pn+" asked "+tn+" for "+rl+"s — go fish! Drew "+cardName(res.drawn)+".");
      if(checkEnd()){ finishGame(); renderAll(); return; }
      sfx("draw");
      fishFlash(res.drawn);
      nextTurn();
    }
  }else{
    logMsg(pn+" asked "+tn+" for "+rl+"s — go fish! (pond empty)");
    if(checkEnd()){ finishGame(); renderAll(); return; }
    sfx("warn");
    nextTurn();
  }
}
function nextTurn(){
  game.pendingRank=null;
  game.turn=(game.turn+1)%numPlayers();
  startTurn();
}
function startTurn(){
  game.pendingRank=null;
  for(let guard=0;guard<24;guard++){
    refillHand();
    if(checkEnd()){ finishGame(); renderAll(); return; }
    if(game.hands[game.turn].length>0)break;
    if(game.deck.length>0)break;
    game.turn=(game.turn+1)%numPlayers();
  }
  renderAll();
  if(game.mode==="ai"&&game.turn!==0&&!game.over)scheduleAI();
}

/* ══════════ AI ══════════ */
function aiChooseRank(p){
  const groups={};
  for(const c of game.hands[p])groups[c.rank]=(groups[c.rank]||0)+1;
  const entries=Object.entries(groups);
  const max=Math.max(...entries.map(e=>e[1]));
  const top=entries.filter(e=>e[1]===max).map(e=>e[0]);
  return top[Math.floor(Math.random()*top.length)];
}
function aiChooseTarget(p,rank){
  const cands=[];
  for(let q=0;q<numPlayers();q++)if(q!==p&&game.hands[q].length>0)cands.push(q);
  if(!cands.length)return -1;
  const remembered=[];
  for(let k=game.askedFor.length-1;k>=0;k--){
    const a=game.askedFor[k];
    if(a.rank===rank&&cands.indexOf(a.from)!==-1&&remembered.indexOf(a.from)===-1)remembered.push(a.from);
  }
  const holders=cands.filter(q=>countRank(game.hands[q],rank)>0);
  const pool=remembered.length?remembered:(holders.length?holders:cands);
  return pool[Math.floor(Math.random()*pool.length)];
}
function scheduleAI(){
  if(game.aiBusy)return;
  game.aiBusy=true;
  renderAll();
  setTimeout(function(){
    game.aiBusy=false;
    if(game.mode!=="ai"||game.over)return;
    if(game.turn===0)return;
    const p=game.turn;
    refillHand();
    if(checkEnd()){ finishGame(); renderAll(); return; }
    if(game.hands[p].length===0){ nextTurn(); return; }
    const rank=aiChooseRank(p);
    const to=aiChooseTarget(p,rank);
    if(to===-1){ nextTurn(); return; }
    applyAsk(p,to,rank);
  }, 850+Math.random()*500);
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — sharing the code lets a friend sit opposite you.":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }
  catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
}
function cardFromByteServer(id){ return {id:id, suit:SUITS[Math.floor(id/13)], rank:RANKS[id%13]}; }
function pushServerLog(a){
  if(a.type==="wait")return;
  if(a.type==="deal"){ clearLog(); logMsg("<b>Deal!</b> "+HAND_SIZE+" cards each. "+turnStarterText(a.from)); sfx("deal"); return; }
  if(a.type==="refill"){ logMsg(playerName(a.from)+" drew "+(a.drew>1?a.drew+" cards":"a card")+" to refill an empty hand."); sfx("draw"); return; }
  if(a.type==="give"){
    const r=rankLabel(a.rank);
    logMsg(playerName(a.from)+" asked "+playerName(a.to)+" for "+r+"s — got "+a.gave+"!"+(a.laid?" Book laid!":""));
    sfx(a.laid?"tada":"coin");
    return;
  }
  if(a.type==="fish"){
    const r=rankLabel(a.rank);
    if(a.drawn>=0&&a.lucky)logMsg(playerName(a.from)+" asked "+playerName(a.to)+" for "+r+"s — go fish, but drew a "+r+"! Lucky!");
    else if(a.drawn>=0)logMsg(playerName(a.from)+" asked "+playerName(a.to)+" for "+r+"s — go fish! Drew "+cardName(cardFromByteServer(a.drawn))+".");
    else logMsg(playerName(a.from)+" asked "+playerName(a.to)+" for "+r+"s — go fish! (pond empty)");
    sfx(a.lucky?"tada":"draw");
    return;
  }
  if(a.type==="resign"){ logMsg(playerName(a.from)+" resigned."); return; }
}
function applyServerSnap(s){
  game.mode="online";
  game.myIndex=s.you;
  game.status=s.status;
  game.turn=s.turn;
  game.hands=[[],[]];
  game.hands[s.you]=s.myHand;
  game.handLens=[0,0];
  game.handLens[1-s.you]=s.oppCount;
  game.booksN=s.books;
  game.deckCount=s.deckCount;
  game.oppIn=s.oppIn;
  game.winner=s.winner;
  const wasOver=game.over;
  game.over=s.status===3;
  game.lockAsk=false;
  game.pendingRank=null;
  if(game.over&&!wasOver)sfx(s.winner===s.you?"win":"lose");
  if(s.lastAction&&s.lastAction.seq>game.lastSeq){
    game.lastSeq=s.lastAction.seq;
    pushServerLog(s.lastAction);
    if(s.lastAction.type==="fish"&&s.lastAction.drawn>=0)fishFlash(cardFromByteServer(s.lastAction.drawn));
  }
  renderAll();
}

  lobbyCtn.hidden=true;
  gameCtn.hidden=false;

async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom",JSON.stringify({handSize:HAND_SIZE}));
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=r.code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
async function sendAsk(rank,toIdx){
  const rankIdx=RANKS.indexOf(rank);
  if(rankIdx<0)return;
  game.lockAsk=true;
  renderAll();
  try{
    const r=await rpc("ask",JSON.stringify({code:game.roomCode,rank:rankIdx,to:toIdx}));
    if(!r.ok){
      game.lockAsk=false;
      if(r.err==="not_in_room")rejoinRoom();
      flashBanner(r.err==="not_your_turn"?"Not your turn yet.":(r.err==="dont_have_rank"?"You don't hold that rank.":"Ask rejected ("+r.err+")."));
    }
  }catch(e){
    game.lockAsk=false;
    flashBanner("Couldn't reach the server — reconnecting…");
  }
}
