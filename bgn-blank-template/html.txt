<script type="text/x-server-plugin">
/* ── BGN table server ─────────────────────────────────────────────
   Generic two-player rooms: host + guest, 5-letter codes. The
   server sequences turns (host first unless a rematch flips it),
   stores ONE full-game snapshot per room (opaque string submitted
   by whichever player is on turn), and tracks winner/reason for
   resign, forfeit and end-of-game. No game logic here — the
   snapshot is whatever the game decides it should be.
   Durable state in the plugin `state` bytes (versioned layout).  */
const SLOT=1024;
const STATE_VERSION=4;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const MAX_PAYLOAD=2000;

let rooms=new Map();           // code -> room object (live cache)
const conns=new Map();         // conn.id -> {code, role}
const moveTimes=new Map();     // conn.id -> last move ms (rate limit)
const chatTimes=new Map();     // conn.id -> last chat ms (rate limit)
const CHAT_BANNED=["nigger","nigga","faggot","kys","kill yourself","retard","heil hitler","tranny"];
function cleanText(s,max){ return String(s||"").trim().replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").slice(0,max); }
function isChatBanned(t){ const s=t.toLowerCase(); for(let i=0;i<CHAT_BANNED.length;i++){ if(s.indexOf(CHAT_BANNED[i])!==-1) return true; } return false; }

/* ── durable state (slot layout) ──
   slot byte 0: status 0=free 1=waiting 2=playing 3=finished
   slot +1..+8: room code (8 ASCII, zero padded)
   slot +9: first (0=host moves first, 1=guest; rematch flips)
   slot +10: state version
   slot +11: reserved
   slot +12..+15: uint32 lastActive (unix secs)
   slot +16: winner 0=none 1=host 2=guest 3=draw
   slot +17: reason 0=none 1=resign 2=forfeit 3=end
   slot +18..+19: uint16 turn counter (completed turns)
   slot +20..+21: uint16 snapshot bytes length
   slot +22..+23: reserved
   slot +24..: snapshot bytes (utf8, latest only)                */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+12]<<24)|(state[o+13]<<16)|(state[o+14]<<8)|state[o+15]; }
function slotStatus(i){ return state[off(i)]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function utf8Encode(s){ const out=[]; for(let i=0;i<s.length;i++){ const c=s.charCodeAt(i); if(c<128)out.push(c); else if(c<2048)out.push(192|(c>>6),128|(c&63)); else out.push(224|(c>>12),128|((c>>6)&63),128|(c&63)); } return out; }
function utf8Decode(bytes){ let s="",i=0; while(i<bytes.length){ const b=bytes[i]; if(b<128){ s+=String.fromCharCode(b); i++; } else if(b<224){ s+=String.fromCharCode(((b&31)<<6)|(bytes[i+1]&63)); i+=2; } else { s+=String.fromCharCode(((b&15)<<12)|((bytes[i+1]&63)<<6)|(bytes[i+2]&63)); i+=3; } } return s; }
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+9]=room.first;
  state[o+10]=STATE_VERSION;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+12]=(la>>>24)&255; state[o+13]=(la>>>16)&255; state[o+14]=(la>>>8)&255; state[o+15]=la&255;
  state[o+16]=room.winner==="host"?1:room.winner==="guest"?2:room.winner==="draw"?3:0;
  state[o+17]=room.reason==="resign"?1:room.reason==="forfeit"?2:room.reason==="end"?3:0;
  const tn=Math.max(0,Math.min(0xFFFF,room.turn));
  state[o+18]=(tn>>8)&255; state[o+19]=tn&255;
  const mb=room.snapshot?utf8Encode(room.snapshot):[];
  const len=Math.min(SLOT-24,mb.length);
  state[o+20]=(len>>8)&255; state[o+21]=len&255;
  if(len>0)state.set(mb.slice(0,len),o+24);
}
const REASON=["","resign","forfeit","end"];
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(state[o+10]!==STATE_VERSION){ state.fill(0,o,o+SLOT); continue; }  // old/foreign schema → reclaim
    if(st===1){ state.fill(0,o,o+SLOT); continue; }                        // abandoned waiting rooms are reclaimed
    const snapLen=(state[o+20]<<8)|state[o+21];
    rooms.set(codeAt(i),{
      slot:i, code:codeAt(i), status:st,
      first:state[o+9], hostId:-1, guestId:-1,
      turn:(state[o+18]<<8)|state[o+19],
      snapshot:snapLen?utf8Decode(state.slice(o+24,o+24+snapLen)):null,
      winner:state[o+16]===1?"host":state[o+16]===2?"guest":state[o+16]===3?"draw":null,
      reason:REASON[state[o+17]]||null,
      lastAct:readLastAct(i)
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── helpers ── */
function topic(code){ return "tbl:"+code; }
function snap(room){ return {code:room.code,status:room.status,snapshot:room.snapshot,first:room.first,turn:room.turn,hostIn:room.hostId!==-1,guestIn:room.guestId!==-1,winner:room.winner,reason:room.reason}; }
function broadcast(room){ try{ pubsub.publish(topic(room.code), JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }
function turnRoleFor(n,first){ return ((n+(first?1:0))%2)===0?"host":"guest"; }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room={slot,code,status:1,first:0,hostId:conn.id,guestId:-1,turn:0,snapshot:null,winner:null,reason:null,lastAct:nowSec()};
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host"});
    return JSON.stringify({ok:true,code,role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)room.status=2;
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role});
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true,role,snap:snap(room)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    return JSON.stringify({ok:true,snap:snap(room)});
  },
  move({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(turnRoleFor(room.turn,room.first)!==role)return JSON.stringify({ok:false,err:"not_your_turn"});
    const now=Date.now();
    if(now-(moveTimes.get(conn.id)||0)<300)return JSON.stringify({ok:false,err:"rate_limited"});
    moveTimes.set(conn.id,now);
    if(typeof obj.evt!=="number"||obj.evt!==room.turn)return JSON.stringify({ok:false,err:"stale"});
    if(typeof obj.data!=="string"||obj.data.length<1||obj.data.length>MAX_PAYLOAD)return JSON.stringify({ok:false,err:"bad_move"});
    room.snapshot=obj.data;
    room.turn++;
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true});
  },
  endGame({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(["host","guest","draw"].indexOf(obj.winner)===-1)return JSON.stringify({ok:false,err:"bad_request"});
    room.status=3; room.winner=obj.winner; room.reason="end";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"guest":"host"; room.reason="resign";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    room.status=2; room.turn=0; room.snapshot=null; room.winner=null; room.reason=null;
    room.first=1-room.first;
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3; room.winner=role; room.reason="forfeit";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"guest":"host"; room.reason="forfeit";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  chat({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const now=Date.now();
    if(now-(chatTimes.get(conn.id)||0)<700)return JSON.stringify({ok:false,err:"rate_limited"});
    chatTimes.set(conn.id,now);
    const text=cleanText(obj.text,160);
    if(!text)return JSON.stringify({ok:false,err:"empty"});
    if(isChatBanned(text))return JSON.stringify({ok:false,err:"blocked"});
    pubsub.publish(topic(room.code),JSON.stringify({t:"chat",role,text,ts:nowSec()}));
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  conns.delete(conn.id);
  moveTimes.delete(conn.id);
  chatTimes.delete(conn.id);
  if(!info)return;
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

function resultText(){
  if(game.winner==="draw")return "Draw — the match is over.";
  const w=roleName(game.winner);
  const youWon=game.mode==="online"&&game.winner===game.myRole;
  if(youWon){
    if(game.reason==="resign")return "You win — your opponent resigned!";
    if(game.reason==="forfeit")return "You win — your opponent left!";
    return "You win!";
  }
  if(game.mode==="online"&&game.winner!==game.myRole){
    if(game.reason==="resign")return "You lose — you resigned.";
    if(game.reason==="forfeit")return "You lose — you left the match.";
    return "You lose.";
  }
  if(game.reason==="resign")return w+" wins by resignation.";
  if(game.reason==="forfeit")return w+" wins — the other side left.";
  return w+" wins!";
}

/* ══════════ turn timer (local & online; 0 to disable) ══════════ */
const TIMER_SECS=60;
const turnTimerEl=document.getElementById("turnTimerEl");
let timerRemain=0, timerInt=null, timerKey="";
function timerFmt(s){ const m=Math.floor(s/60), r=s%60; return m+":"+String(r).padStart(2,"0"); }
function timerStop(){ if(timerInt){ clearInterval(timerInt); timerInt=null; } }
function timerStart(){ if(TIMER_SECS<=0)return; timerStop(); timerRemain=TIMER_SECS; timerInt=setInterval(timerTick,1000); timerTick(); }
function timerTick(){
  timerRemain--;
  if(turnTimerEl){ turnTimerEl.textContent="⏱ "+timerFmt(Math.max(0,timerRemain)); turnTimerEl.classList.toggle("low",timerRemain<=10); }
  if(timerRemain<=0){
    timerStop();
    flashBanner("Time's up — pass auto-sent.");
    if(!game.over&&!game.locked)passTurn();
  }
}
function syncTimer(){
  const running = game.mode!=="idle" && game.mode!==null && !game.over && !game.locked;
  if(!running){ timerStop(); if(turnTimerEl)turnTimerEl.hidden=true; return; }
  if(turnTimerEl)turnTimerEl.hidden=false;
  const key = game.mode==="online" ? "O:"+game.serverTurn : "L:"+game.log.length;
  if(game.mode==="online" && turnRoleFor(game.serverTurn)!==game.myRole){ timerStop(); if(turnTimerEl)turnTimerEl.textContent="⏱ —"; return; }
  if(game.mode==="ai" && aiThinking){ timerStop(); if(turnTimerEl)turnTimerEl.textContent="⏱ —"; return; }
  if(key!==timerKey){ timerKey=key; timerStart(); }
}
function refreshTurn(){
  const r=game.mode==="online"?turnRoleFor(game.serverTurn):turnRoleFor(game.log.length);
  turnEl.textContent=game.over?"Match over":(game.mode==="ai"?(r==="host"?"Your move — pass when ready.":"Bot thinking…"):(game.mode==="online"?(r===game.myRole?"Your move — "+roleName(r)+" to move":roleName(r)+" to move"):roleName(r)+" to move"));

/* ══════════ play vs AI (mode "ai") ══════════
   The human always plays Host; the AI is Guest. This template ships
   with a passive AI that simply passes the turn — build your own
   computer player by overriding AI_DO_TURN() (return a string
   describing what the AI did, or null to log nothing). The turn
   flow, banner, turn timer, and saving are all wired up for you. */
let aiThinking=false, aiTimer=null;
function AI_DO_TURN(){
  return "passed the turn";
}
function aiPassTurn(){
  if(game.mode!=="ai"||game.over)return;
  if(turnRoleFor(game.log.length)!=="guest")return;
  aiThinking=true;
  refreshTurn(); updateBanner(); updateButtons(); syncTimer();
  aiTimer=setTimeout(function(){
    aiTimer=null;
    aiThinking=false;
    if(game.mode!=="ai"){ renderAll(); return; }
    let msg="passed the turn";
    try{ const m=AI_DO_TURN(); if(m)msg=m; }catch(e){ console.error("AI error:",e); }
    addLog("guest",msg);
    sfx("turn");
    renderAll();
    saveGame();
  },900);
}

/* ══════════ save / continue (local table only) ══════════
   This template ships without game logic, so the local save captures
   the table itself (turn log + match state). Builders with a real game

const SAVE_KEY="bgn_save_blank";
function saveGame(){
  if(game.mode!=="local"&&game.mode!=="ai")return;
  if(game.over){ deleteSave(); return; }
  try{ localStorage.setItem(SAVE_KEY, JSON.stringify({v:1,mode:game.mode,log:game.log,winner:game.winner,reason:game.reason,lastLoggedTurn:game.lastLoggedTurn,savedAt:Date.now()})); }catch(e){}
}
function hasSave(){ try{ return !!localStorage.getItem(SAVE_KEY); }catch(e){ return false; } }
function deleteSave(){ try{ localStorage.removeItem(SAVE_KEY); }catch(e){} }
function refreshContinuePanel(){
  if(!continuePanel)return;
  const s=hasSave();
  continuePanel.hidden=!s;
  if(!s)return;
  let info="A saved table was found.";
  try{ const o=JSON.parse(localStorage.getItem(SAVE_KEY)); if(o&&o.savedAt)info="Saved "+new Date(o.savedAt).toLocaleTimeString()+" — turn "+((o.log||[]).length+1)+"."; }catch(e){}
  continueInfoEl.textContent=info;
}
function continueGame(){
  let o; try{ o=JSON.parse(localStorage.getItem(SAVE_KEY)); }catch(e){ o=null; }
  if(!o||!Array.isArray(o.log)){ flashBanner("No saved table found."); return; }
  resetTable();
  game.mode=o.mode==="ai"?"ai":"local";
  game.log=o.log.slice();
  game.lastLoggedTurn=o.lastLoggedTurn||0;
  game.winner=o.winner||null; game.reason=o.reason||"";
  updateNames();
  roomBox.hidden=true; chatBox.hidden=true; claimBtn.hidden=true;
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  sfx("deal");
  renderAll();
  if(game.mode==="ai"&&turnRoleFor(game.log.length)==="guest")aiPassTurn();
}

/* ══════════ turn action ══════════ */
function passTurn(){
  if(game.over||game.locked)return;
  if(game.mode==="ai"){
    if(turnRoleFor(game.log.length)!=="host")return;
    addLog("host","passed the turn");
    sfx("turn");
    saveGame();
    aiPassTurn();
    return;
  }
  if(game.mode==="local"){
    addLog(turnRoleFor(game.log.length),"passed the turn");
    sfx("turn");
    renderAll();
    saveGame();
    return;
  }
  if(turnRoleFor(game.serverTurn)!==game.myRole)return;
  game.locked=true;
  updateButtons();
  const evt=game.serverTurn;
  rpc("move",JSON.stringify({code:game.roomCode,evt:evt,data:"T"+(evt+1)})).then(function(r){
    game.locked=false;
    if(!r||!r.ok){ flashBanner("Couldn't pass the turn ("+(r&&r.err?r.err:"error")+")."); }
    sfx("turn");
    return refreshFromServer();
  }).catch(function(){
    game.locked=false;
    flashBanner("Couldn't reach the server — reconnecting…");
  });
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }
  catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true; game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){ if(game.mode==="online")onServerMsg(ev.data); });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null; connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — sharing code with a friend starts the match":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="chat")addChat(msg.role,msg.text,msg.ts);
}
function refreshFromServer(){
  return rpc("getRoom",game.roomCode).then(function(r){
    if(r&&r.snap)applyServerSnap(r.snap);
  }).catch(function(){});
}
function applyServerSnap(s){
  game.serverTurn=s.turn;
  game.first=s.first;
  game.status=s.status;
  game.oppPresent=game.myRole==="host"?s.guestIn:s.hostIn;
  while(game.lastLoggedTurn<s.turn){
    const r=turnRoleFor(game.lastLoggedTurn);
    addLog(r,"passed the turn");
    game.lastLoggedTurn++;
  }
  if(s.status===3){
    game.over=true;
    game.winner=s.winner;
    game.reason=s.reason||"";
    if(game.winner==="draw")sfx("draw2");
    else if(game.winner===game.myRole)sfx("win");
    else sfx("lose");
  }else{
    game.over=false; game.winner=null; game.reason="";
  }
  renderAll();
}
function enterOnlineGame(){
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  roomBox.hidden=false;
  chatBox.hidden=false;
  roomCodeEl.textContent=game.roomCode;
  updateNames(); renderAll();
}
async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    resetTable();
    game.mode="online"; game.roomCode=r.code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
    sfx("deal");
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    resetTable();
    game.mode="online"; game.roomCode=code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
    sfx("deal");
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
