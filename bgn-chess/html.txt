/* ── BGN Chess server ────────────────────────────────────────────
   Rooms: 5-letter codes. Host = WHITE and moves first; guest = BLACK.
   The server re-runs the full chess rules engine to validate every
   move (castling, en passant, promotion, check/checkmate included),
   enforces turn order, detects draws, and broadcasts a snapshot to
   both players on every change. Durable room state is kept in the
   plugin `state` bytes (versioned binary layout).                */
const WHITE=1, BLACK=2;
const EMPTY=0, W_P=1,W_N=2,W_B=3,W_R=4,W_Q=5,W_K=6, B_P=7,B_N=8,B_B=9,B_R=10,B_Q=11,B_K=12;
const FILES="abcdefgh";
const KNIGHT_D=[17,15,10,6,-6,-10,-15,-17];
const KING_D=[9,8,7,1,-1,-7,-8,-9];
const DIAG=[[1,1],[1,-1],[-1,1],[-1,-1]];
const LINE=[[1,0],[-1,0],[0,1],[0,-1]];
const MATE=100000, FIFTY=100;
const STATE_VERSION=2;
const SLOT=1024;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";

const CHAT_BANNED=["nigger","nigga","faggot","kys","kill yourself","retard","heil hitler","tranny"];
function cleanText(s,max){ return String(s||"").trim().replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").slice(0,max); }
function isChatBanned(t){ const s=t.toLowerCase(); for(let i=0;i<CHAT_BANNED.length;i++){ if(s.indexOf(CHAT_BANNED[i])!==-1) return true; } return false; }

/* ── rules engine (mirrors client — keep in sync) ── */
function colorOf(p){ return p>=1&&p<=6?WHITE:(p>=7?BLACK:EMPTY); }
function typeOf(p){ return p>=1&&p<=6?p:p-6; }
function opp(c){ return c===WHITE?BLACK:WHITE; }
function inB(s){ return s>=0&&s<64; }
function sqName(s){ return FILES[s&7]+(8-(s>>3)); }
function initialBoard(){
  const b=new Array(64).fill(EMPTY);
  const back=[W_R,W_N,W_B,W_Q,W_K,W_B,W_N,W_R];
  for(let c=0;c<8;c++){ b[56+c]=back[c]; b[48+c]=W_P; b[8+c]=B_P; b[c]=back[c]+6; }
  return b;
}
function makeState(){ return {board:initialBoard(), turn:WHITE, castle:15, ep:-1, half:0, full:1}; }
function isAttacked(state,sq,by){
  const b=state.board, r=sq>>3, c=sq&7;
  if(by===WHITE){
    if(inB(sq+7)&&c>0&&b[sq+7]===W_P)return true;
    if(inB(sq+9)&&c<7&&b[sq+9]===W_P)return true;
  }else{
    if(inB(sq-9)&&c>0&&b[sq-9]===B_P)return true;
    if(inB(sq-7)&&c<7&&b[sq-7]===B_P)return true;
  }
  const kn=by===WHITE?W_N:B_N, kg=by===WHITE?W_K:B_K;
  for(const d of KNIGHT_D){ const t=sq+d; if(!inB(t))continue; const tr=t>>3,tc=t&7;
    if((Math.abs(tr-r)===2&&Math.abs(tc-c)===1)||(Math.abs(tr-r)===1&&Math.abs(tc-c)===2)){ if(b[t]===kn)return true; } }
  for(const d of KING_D){ const t=sq+d; if(!inB(t))continue; const tr=t>>3,tc=t&7; if(Math.max(Math.abs(tr-r),Math.abs(tc-c))===1&&b[t]===kg)return true; }
  const bq=by===WHITE?W_B:B_B, rq=by===WHITE?W_R:B_R, qq=by===WHITE?W_Q:B_Q;
  for(const pair of DIAG){ let rr=r+pair[0], cc=c+pair[1];
    while(rr>=0&&rr<8&&cc>=0&&cc<8){ const p=b[rr*8+cc]; if(p!==EMPTY){ if(p===bq||p===qq)return true; break; } rr+=pair[0]; cc+=pair[1]; } }
  for(const pair of LINE){ let rr=r+pair[0], cc=c+pair[1];
    while(rr>=0&&rr<8&&cc>=0&&cc<8){ const p=b[rr*8+cc]; if(p!==EMPTY){ if(p===rq||p===qq)return true; break; } rr+=pair[0]; cc+=pair[1]; } }
  return false;
}
function pseudoMoves(state,from){
  const b=state.board, p=b[from], c=colorOf(p), t=typeOf(p);
  const res=[], r=from>>3, col=from&7;
  function add(to){ res.push({from:from,to:to}); }
  function addPawn(to,promo){ if(promo){ res.push({from:from,to:to,promo:5},{from:from,to:to,promo:4},{from:from,to:to,promo:3},{from:from,to:to,promo:2}); } else res.push({from:from,to:to}); }
  if(t===1){
    const dir=c===WHITE?-8:8, promoRow=c===WHITE?0:7;
    const one=from+dir;
    if(inB(one)&&b[one]===EMPTY){
      addPawn(one,(one>>3)===promoRow);
      if((from>>3)===(c===WHITE?6:1)){ const two=from+2*dir; if(b[two]===EMPTY)addPawn(two,false); }
    }
    for(const dc of [col-1,col+1]){
      if(dc<0||dc>7)continue;
      const tgt=from+dir+(dc-col);
      if(inB(tgt)){
        if(colorOf(b[tgt])===opp(c))addPawn(tgt,(tgt>>3)===promoRow);
        else if(state.ep===tgt)res.push({from:from,to:tgt});
      }
    }
  }else if(t===2){
    for(const d of KNIGHT_D){ const t=from+d; if(!inB(t))continue; const tr=t>>3,tc=t&7;
      if(!((Math.abs(tr-r)===2&&Math.abs(tc-col)===1)||(Math.abs(tr-r)===1&&Math.abs(tc-col)===2)))continue;
      const q=b[t]; if(!q||colorOf(q)!==c)add(t); }
  }else if(t===3||t===5){
    for(const pair of DIAG){ let rr=r+pair[0], cc=col+pair[1];
      while(rr>=0&&rr<8&&cc>=0&&cc<8){ const s=rr*8+cc, q=b[s]; if(q===EMPTY)add(s); else { if(colorOf(q)!==c)add(s); break; } rr+=pair[0]; cc+=pair[1]; } }
  }
  if(t===4||t===5){
    for(const pair of LINE){ let rr=r+pair[0], cc=col+pair[1];
      while(rr>=0&&rr<8&&cc>=0&&cc<8){ const s=rr*8+cc, q=b[s]; if(q===EMPTY)add(s); else { if(colorOf(q)!==c)add(s); break; } rr+=pair[0]; cc+=pair[1]; } }
  }else if(t===6){
    for(const d of KING_D){ const t=from+d; if(!inB(t))continue; const tr=t>>3,tc=t&7; if(Math.max(Math.abs(tr-r),Math.abs(tc-col))!==1)continue; const q=b[t]; if(!q||colorOf(q)!==c)add(t); }
    const ec=opp(c);
    if(!isAttacked(state,from,ec)){
      if(c===WHITE){
        if((state.castle&1)&&b[61]===EMPTY&&b[62]===EMPTY&&!isAttacked(state,61,ec)&&!isAttacked(state,62,ec))add(62);
        if((state.castle&2)&&b[57]===EMPTY&&b[58]===EMPTY&&b[59]===EMPTY&&!isAttacked(state,58,ec)&&!isAttacked(state,59,ec))add(58);
      }else{
        if((state.castle&4)&&b[5]===EMPTY&&b[6]===EMPTY&&!isAttacked(state,5,ec)&&!isAttacked(state,6,ec))add(6);
        if((state.castle&8)&&b[1]===EMPTY&&b[2]===EMPTY&&b[3]===EMPTY&&!isAttacked(state,2,ec)&&!isAttacked(state,3,ec))add(2);
      }
    }
  }
  return res;
}
function makeMove(state,m){
  const b=state.board.slice();
  const p=state.board[m.from], c=colorOf(p), t=typeOf(p);
  const captured=b[m.to]!==EMPTY;
  const ns={board:b, turn:state.turn, castle:state.castle, ep:-1, half:state.half+1, full:state.full};
  const HK={56:2,63:1,60:3,0:8,7:4,4:12};
  if(HK[m.from]!==undefined)ns.castle&=~(HK[m.from]);
  if(HK[m.to]!==undefined)ns.castle&=~(HK[m.to]);
  if(t===1||captured||m.to===state.ep)ns.half=0;
  b[m.from]=EMPTY;
  if(m.to===state.ep&&t===1)b[m.to+(c===WHITE?8:-8)]=EMPTY;
  if(m.promo)b[m.to]=c===WHITE?m.promo:m.promo+6;
  else b[m.to]=p;
  if(t===6&&Math.abs(m.to-m.from)===2){
    if(m.to===62){ b[61]=b[63]; b[63]=EMPTY; }
    else if(m.to===58){ b[59]=b[56]; b[56]=EMPTY; }
    else if(m.to===6){ b[5]=b[7]; b[7]=EMPTY; }
    else if(m.to===2){ b[3]=b[0]; b[0]=EMPTY; }
  }
  if(t===1&&Math.abs(m.to-m.from)===16)ns.ep=(m.from+m.to)>>1;
  ns.turn=opp(c);
  if(c===BLACK)ns.full++;
  return ns;
}
function kingOf(state,c){ const k=c===WHITE?W_K:B_K; for(let i=0;i<64;i++)if(state.board[i]===k)return i; return -1; }
function legalMoves(state){
  const b=state.board, c=state.turn, res=[], ec=opp(c);
  for(let i=0;i<64;i++){
    if(colorOf(b[i])!==c)continue;
    for(const m of pseudoMoves(state,i)){
      const ns=makeMove(state,m);
      if(!isAttacked(ns,kingOf(ns,c),ec))res.push(m);
    }
  }
  return res;
}
function inCheck(state){ const k=kingOf(state,state.turn); return k>=0&&isAttacked(state,k,opp(state.turn)); }
function posKey(st){ return st.board.join(",")+"|"+st.turn+"|"+st.castle+"|"+st.ep; }
function parseMove(s){
  if(typeof s!=="string")return null;
  const parts=s.split(",");
  if(parts.length<2||parts.length>3)return null;
  const out=[];
  for(const x of parts){ if(!/^\d{1,2}$/.test(x))return null; out.push(+x); }
  if(out[0]<0||out[0]>63||out[1]<0||out[1]>63||out[0]===out[1])return null;
  if(out.length===3&&(out[2]<2||out[2]>5))return null;
  return {from:out[0], to:out[1], promo:out.length===3?out[2]:null};
}
function encodeMove(m){ return m.from+","+m.to+(m.promo?","+m.promo:""); }
function stateFromMoves(moves){ let st=makeState(); for(const enc of moves){ const m=parseMove(enc); if(m)st=makeMove(st,{from:m.from,to:m.to,promo:m.promo||undefined}); } return st; }
function repetitionCount(moves){
  const keys=[posKey(makeState())];
  let st=makeState();
  for(const enc of moves){ const m=parseMove(enc); if(m)st=makeMove(st,{from:m.from,to:m.to,promo:m.promo||undefined}); keys.push(posKey(st)); }
  const cur=keys[keys.length-1]; let n=0;
  for(const k of keys)if(k===cur)n++;
  return n;
}
function insufficientMaterial(state){
  let minors=0, bishops=0, bi=-1;
  for(let i=0;i<64;i++){ const p=state.board[i]; if(!p)continue; const t=typeOf(p); if(t===6)continue;
    if(t===5||t===4)return false;
    if(t===2)minors++;
    else if(t===3){ minors++; bishops++; if(bi===-1)bi=i; }
    else return false;
  }
  if(minors<=1)return true;
  if(minors===2&&bishops===2){
    let b2=-1; for(let i=0;i<64;i++){ const p=state.board[i]; if(typeOf(p)===3&&i!==bi){ b2=i; break; } }
    return (((bi>>3)+(bi&7))&1)===(((b2>>3)+(b2&7))&1);
  }
  return false;
}
function afterMoveResult(state,moves){
  const mv=legalMoves(state);
  if(!mv.length){
    if(inCheck(state))return {winner: state.turn===WHITE?"black":"white", reason:"checkmate"};
    return {winner:"draw", reason:"stalemate"};
  }
  if(state.half>=FIFTY)return {winner:"draw", reason:"fifty"};
  if(insufficientMaterial(state))return {winner:"draw", reason:"insufficient"};
  if(repetitionCount(moves)>=3)return {winner:"draw", reason:"threefold"};
  return null;
}
function colorForHost(h){ return h===0?WHITE:BLACK; }

/* ── durable state (slot layout) ──
   slot byte 0: status 0=free 1=waiting 2=playing 3=finished
   slot +1..+8: room code (8 ASCII, zero padded)
   slot +9: hostColor (0=white)
   slot +10: state version (2 = chess)
   slot +11: reserved
   slot +12..+15: uint32 lastActive (unix secs)
   slot +16..+17: uint16 halfmove clock
   slot +18..+19: uint16 moves bytes length
   slot +20: winner 0=none 1=white 2=black 3=draw
   slot +21..+23: reserved
   slot +24..: move bytes [len][from][to][promo?] per move          */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+12]<<24)|(state[o+13]<<16)|(state[o+14]<<8)|state[o+15]; }
function slotStatus(i){ return state[off(i)]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function encodeMoves(list){
  const out=[];
  for(const s of list){ const m=parseMove(s); if(!m)continue; out.push(m.promo?3:2, m.from, m.to); if(m.promo)out.push(m.promo); }
  return out;
}
function decodeMoves(bytes){
  const out=[]; let i=0;
  while(i<bytes.length){
    const len=bytes[i++];
    if(!len||i+len>bytes.length)break;
    const f=bytes[i], t=bytes[i+1];
    if(len===3)out.push(f+","+t+","+bytes[i+2]);
    else out.push(f+","+t);
    i+=len;
  }
  return out;
}
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+9]=room.hostColor;
  state[o+10]=STATE_VERSION;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+12]=(la>>>24)&255; state[o+13]=(la>>>16)&255; state[o+14]=(la>>>8)&255; state[o+15]=la&255;
  state[o+16]=(room.half>>8)&255; state[o+17]=room.half&255;
  const mb=encodeMoves(room.moves);
  const len=Math.min(SLOT-24,mb.length);
  state[o+18]=(len>>8)&255; state[o+19]=len&255;
  state[o+20]=room.winner==="white"?1:room.winner==="black"?2:room.winner==="draw"?3:0;
  if(len>0)state.set(mb.slice(0,len),o+24);
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(state[o+10]!==STATE_VERSION){ state.fill(0,o,o+SLOT); continue; }  // old/foreign schema → reclaim
    if(st===1){ state.fill(0,o,o+SLOT); continue; }                        // abandoned waiting rooms are reclaimed
    const movesLen=(state[o+18]<<8)|state[o+19];
    const moves=decodeMoves(state.slice(o+24,o+24+movesLen));
    const win=state[o+20];
    rooms.set(codeAt(i),{
      slot:i, code:codeAt(i), status:st,
      hostColor:state[o+9], hostId:-1, guestId:-1,
      moves, winner:win===1?"white":win===2?"black":win===3?"draw":null, reason:null,
      half:(state[o+16]<<8)|state[o+17],
      lastAct:readLastAct(i)
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── helpers ── */
function topic(code){ return "ch:"+code; }
function snap(room){ return {code:room.code,status:room.status,moves:room.moves,hostColor:room.hostColor,hostIn:room.hostId!==-1,guestIn:room.guestId!==-1,winner:room.winner,reason:room.reason}; }
function broadcast(room){ try{ pubsub.publish(topic(room.code), JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room={slot,code,status:1,hostColor:0,hostId:conn.id,guestId:-1,moves:[],winner:null,reason:null,half:0,lastAct:nowSec()};
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host"});
    return JSON.stringify({ok:true,code,color:colorForHost(room.hostColor),role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)room.status=2;
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role});
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true,color:colorForHost(role==="host"?room.hostColor:1-room.hostColor),role,snap:snap(room)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    return JSON.stringify({ok:true,snap:snap(room)});
  },
  move({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const expectedTurn=room.moves.length%2===0?"host":"guest";
    if(role!==expectedTurn)return JSON.stringify({ok:false,err:"not_your_turn"});
    if(typeof obj.evt!=="number"||obj.evt!==room.moves.length)return JSON.stringify({ok:false,err:"stale"});
    const mv=parseMove(obj.move);
    if(!mv)return JSON.stringify({ok:false,err:"bad_move"});
    const st=stateFromMoves(room.moves);
    const mover=colorOf(st.board[mv.from]);
    if(mover===EMPTY)return JSON.stringify({ok:false,err:"no_piece"});
    if(mover!==colorForHost(role==="host"?room.hostColor:1-room.hostColor))return JSON.stringify({ok:false,err:"not_your_piece"});
    let found=false;
    for(const lm of legalMoves(st)){ if(lm.from===mv.from&&lm.to===mv.to&&(lm.promo||null)===(mv.promo||null)){ found=true; break; } }
    if(!found)return JSON.stringify({ok:false,err:"illegal"});
    room.moves.push(obj.move);
    room.lastAct=nowSec();
    const ns=makeMove(st,{from:mv.from,to:mv.to,promo:mv.promo||undefined});
    room.half=ns.half;
    const res=afterMoveResult(ns,room.moves);
    if(res){ room.status=3; room.winner=res.winner; room.reason=res.reason; }
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"black":"white"; room.reason="resign";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    room.status=2; room.moves=[]; room.half=0; room.winner=null; room.reason=null;
    room.hostColor=1-room.hostColor;
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3; room.winner=role==="host"?"white":"black"; room.reason="forfeit";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"black":"white"; room.reason="forfeit";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  chat({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const now=Date.now();
    if(now-(chatTimes.get(conn.id)||0)<700)return JSON.stringify({ok:false,err:"rate_limited"});
    chatTimes.set(conn.id,now);
    const text=cleanText(obj.text,160);
    if(!text)return JSON.stringify({ok:false,err:"empty"});
    if(isChatBanned(text))return JSON.stringify({ok:false,err:"blocked"});
    pubsub.publish(topic(room.code),JSON.stringify({t:"chat",role,text,ts:nowSec()}));
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  conns.delete(conn.id);
  moveTimes.delete(conn.id);
  chatTimes.delete(conn.id);
  if(!info)return;
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

function aiSearch(state,depth,alpha,beta){
  if((++aiNodes&255)===0){ if(Date.now()>aiDeadline||aiNodes>220000)aiAbort=true; }

  const moves=legalMoves(state);
  if(!moves.length)return inCheck(state)?-MATE+depth:0;
  if(depth===0)return qsearch(state,alpha,beta);
  orderMoves(state,moves);
  let best=-Infinity;
  for(const m of moves){
    const s=-aiSearch(makeMove(state,m),depth-1,-beta,-alpha);
    if(s>best)best=s;
    if(s>alpha)alpha=s;
    if(alpha>=beta)break;
    if(aiAbort)break;
  }
  return best;
}
function aiChoose(state,thinkMs){
  aiDeadline=Date.now()+thinkMs; aiAbort=false; aiNodes=0;
  let moves=legalMoves(state);
  if(!moves.length)return null;
  orderMoves(state,moves);
  let completed=null;
  const maxDepth=4;
  for(let d=1;d<=maxDepth;d++){
    aiAbort=false;
    let iterBest=null, iterScore=-Infinity;
    for(const m of moves){
      const s=-aiSearch(makeMove(state,m),d-1,-Infinity,Infinity);
      if(aiAbort)break;
      if(iterBest===null||s>iterScore){ iterScore=s; iterBest=m; }
    }
    if(aiAbort)break;
    completed=iterBest;
    if(d<maxDepth&&completed)moves=[completed].concat(moves.filter(function(m){ return m!==completed; }));
    if(Date.now()>aiDeadline)break;
  }
  return completed||moves[0];
}

/* ══════════ board rendering ══════════ */
function buildBoard(){
  boardEl.innerHTML="";
  cellEls=[];
  for(let i=0;i<64;i++){
    const r=i>>3,c=i&7;
    const el=document.createElement("div");
    el.className="cell "+(((r+c)&1)?"dark":"light");
    el.dataset.i=i;
    el.addEventListener("click",function(){ onCellClick(+this.dataset.i); });
    if(c===0){ const s=document.createElement("span"); s.className="coord rk "+(((r+c)&1)?"on-dark":"on-light"); s.textContent=8-r; el.appendChild(s); }
    if(r===7){ const s=document.createElement("span"); s.className="coord fl "+(((r+c)&1)?"on-dark":"on-light"); s.textContent=FILES[c]; el.appendChild(s); }
    boardEl.appendChild(el);
    cellEls.push(el);
  }
}
function renderBoard(){
  const st=game.state, b=st.board;
  const targets={}, captures={};
  if(!game.over&&game.selected!==-1){
    for(const m of game.legal){
      if(b[m.to]!==EMPTY||m.to===st.ep)captures[m.to]=true;
      else targets[m.to]=true;
    }
  }
  let checkSq=-1;
  if(inCheck(st))checkSq=kingOf(st,st.turn);
  for(let i=0;i<64;i++){
    const el=cellEls[i];
    el.classList.toggle("target",!!targets[i]);
    el.classList.toggle("capture",!!captures[i]);
    el.classList.toggle("selected",i===game.selected);
    el.classList.toggle("lastmove",!!game.lastMove&&(i===game.lastMove.from||i===game.lastMove.to));
    el.classList.toggle("check",i===checkSq);
    const p=b[i];
    if(p)el.innerHTML='<div class="piece '+(colorOf(p)===WHITE?"w":"b")+'">'+GLYPHS[p]+'</div>';
    else el.innerHTML="";
  }
}

function turnName(col){ return col===WHITE?"White":"Black"; }
function refreshTurn(){
  const tn=turnName(game.state.turn);
  turnEl.textContent = game.over ? "Game over"
    : (game.mode==="online" ? (game.state.turn===game.myColor?"Your move — "+tn+" to move":tn+" to move")
      : tn+" to move");
  pWhite.classList.toggle("on",!game.over&&game.state.turn===WHITE);
  pBlack.classList.toggle("on",!game.over&&game.state.turn===BLACK);
  if(game.over){ pWhite.classList.remove("on"); pBlack.classList.remove("on"); }
}
function renderChips(el,types,col){
  el.innerHTML="";
  for(const t of types){
    const s=document.createElement("span");
    s.className="mini-piece "+col;
    s.textContent=GLYPHS[col==="w"?t:t+6];
    el.appendChild(s);
  }
}
function refreshCaptured(){
  const cs=capturedStats(game.state);
  renderChips(whiteChipsEl,cs.byW,"w");
  renderChips(blackChipsEl,cs.byB,"b");
  const diff=cs.matW-cs.matB;
  whiteCapEl.textContent="captured "+cs.byW.length+(diff>0?"  +"+diff:"");
  blackCapEl.textContent="captured "+cs.byB.length+(diff<0?"  +"+(-diff):"");
}
function updateLog(){
  logEl.innerHTML="";
  if(!game.moves.length){ logEl.innerHTML='<div class="muted small">No moves yet.</div>'; return; }
  const sans=[]; let st=makeState();
  for(const enc of game.moves){
    const m=parseMove(enc);
    sans.push(m?sanMove(st,m):"…");
    if(m)st=makeMove(st,{from:m.from,to:m.to,promo:m.promo||undefined});
  }
  const start=Math.max(0,Math.ceil(sans.length/2)-22);
  const frag=document.createDocumentFragment();
  for(let p=start;p<Math.ceil(sans.length/2);p++){
    const div=document.createElement("div");
    div.className="log-item";
    const w=sans[p*2], b=sans[p*2+1];
    div.innerHTML="<b>"+(p+1)+".</b> "+w+(b?"  <span class='resp'>"+b+"</span>":"");
    frag.appendChild(div);
  }
  logEl.appendChild(frag);
}
function reasonText(){
  if(game.winner==="draw"){
    if(game.reason==="stalemate")return "Draw — stalemate. No legal moves.";
    if(game.reason==="fifty")return "Draw — fifty-move rule.";
    if(game.reason==="insufficient")return "Draw — insufficient material.";
    if(game.reason==="threefold")return "Draw — threefold repetition.";
    return "Draw.";
  }
  const w=game.winner===WHITE?"White":"Black";
  if(game.mode==="online"){
    const you=game.winner===game.myColor;
    if(game.reason==="forfeit")return you?"You win — your opponent left!":"You lose — your opponent resigned.";
    if(game.reason==="resign")return you?"You win — your opponent resigned!":"You lose — you resigned.";
    if(game.reason==="checkmate")return you?"Checkmate — you win!":"Checkmate — you lose.";
    return you?"You win!":"You lose.";
  }
  if(game.reason==="checkmate")return "Checkmate — "+w+" wins!";
  return w+" wins.";
}
function updateBanner(){
  let msg="";
  if(game.over){
    msg=reasonText();
  }else if(game.aiThinking){
    msg="The machine is thinking…";
  }else if(game.mode==="online"){
    if(!game.sockOpen)msg="Connecting…";
    else if(game.status===1)msg="Waiting for an opponent to join…";
    else if(!game.oppPresent)msg="Your opponent disconnected — they can rejoin with the code. Claim the win anytime.";
    else if(game.state.turn===game.myColor)msg="Your move.";
    else msg="Waiting for your opponent…";
  }else{
    if(inCheck(game.state))msg=turnName(game.state.turn)+" is in check!";
  }
  bannerEl.textContent=msg;
  if(!fsOverlay.hidden)fsStatusEl.textContent=msg;
}

    newGameBtn.style.display="";
    menuBtn.textContent="Main menu";

  }
}

/* ══════════ flow / turn logic ══════════ */
function isMyTurn(){
  if(game.over||game.locked)return false;
  if(game.mode==="local")return true;
  if(game.mode==="ai")return game.state.turn===WHITE&&!game.aiThinking;
  return game.sockOpen&&game.status===2&&game.state.turn===game.myColor;
}
function onCellClick(i){
  if(game.over||game.locked||game.pendingPromo)return;
  if(!isMyTurn())return;
  const st=game.state;
  if(colorOf(st.board[i])===st.turn){
    const mv=legalMoves(st).filter(function(m){ return m.from===i; });
    game.selected=-1; game.legal=[];
    if(mv.length){ game.selected=i; game.legal=mv; }
    renderBoard();
    return;
  }
  if(game.selected!==-1){
    for(const m of game.legal){
      if(m.to===i){
        if(m.promo){ game.pendingPromo=m; promoCtn.hidden=false; return; }
        executeMove(m);
        return;
      }
    }
  }
  game.selected=-1; game.legal=[];
  renderBoard();
}
function executeMove(m){
  game.selected=-1; game.legal=[]; game.pendingPromo=null;
  promoCtn.hidden=true;
  const encoded=encodeMove(m);
  sfx("turn");
  if(game.mode==="online"){ sendMove(encoded,m); return; }
  game.state=makeMove(game.state,m);
  game.moves.push(encoded);
  game.lastMove=m;
  finishTurn();
}
function checkEnd(){
  const st=game.state;
  if(!legalMoves(st).length){
    if(inCheck(st))return {over:true, winner:opp(st.turn), reason:"checkmate"};
    return {over:true, winner:"draw", reason:"stalemate"};
  }
  if(st.half>=FIFTY)return {over:true, winner:"draw", reason:"fifty"};
  if(insufficientMaterial(st))return {over:true, winner:"draw", reason:"insufficient"};
  if(repetitionCount(game.moves)>=3)return {over:true, winner:"draw", reason:"threefold"};
  return {over:false};
}
function finishTurn(){
  game.selected=-1; game.legal=[];
  const res=checkEnd();
  if(res.over){ endGame(res.winner,res.reason); return; }
  renderAll();
  if(game.mode==="ai"&&game.state.turn===BLACK&&!game.over)scheduleAI();
  if(game.mode!=="online")saveGame();
}
function scheduleAI(){
  game.aiThinking=true;
  updateBanner();
  setTimeout(function(){
    if(game.mode!=="ai"||game.over)return;
    const m=aiChoose(game.state,600);
    game.aiThinking=false;
    if(game.over)return;
    if(!m)return;
    executeMove(m);
  },50);
}
function endGame(winner,reason){
  game.over=true; game.winner=winner; game.reason=reason;
  game.selected=-1; game.legal=[]; game.pendingPromo=null;
  game.aiThinking=false;
  promoCtn.hidden=true;
  if(winner==="draw")sfx("draw2");
  else if(game.mode==="online")sfx(winner===game.myColor?"win":"lose");
  else if(game.mode==="ai")sfx(winner===WHITE?"win":"lose");
  else sfx("win");
  if(game.mode==="local"||game.mode==="ai")deleteSave();
  renderAll();
}

function resetMatch(){
  game.state=makeState();
  game.moves=[];
  game.over=false; game.winner=null; game.reason="";
  game.selected=-1; game.legal=[]; game.lastMove=null;
  game.locked=false; game.pendingPromo=null; game.aiThinking=false;
}

  whiteNameEl.textContent=mode==="ai"?"You":"White";

function exitOnline(showLobby){

  game.sockOpen=false;
  game.roomCode=null; game.myColor=null; game.myRole=null; game.mode=null;

  if(showLobby){ gameCtn.hidden=true; lobbyCtn.hidden=false; }
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{
    game.socket=rootRef.createServerSocket();
  }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — sharing code with a friend starts the match":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="chat")addChat(msg.role,msg.text,msg.ts);
}
function applyServerSnap(s){
  game.hostColor=s.hostColor;
  game.myColor=colorForHost(game.myRole==="host"?s.hostColor:(1-s.hostColor));
  game.status=s.status;
  game.moves=s.moves.slice();
  game.state=stateFromMoves(game.moves);
  game.turn=s.moves.length%2===0?WHITE:BLACK;
  game.locked=false;
  game.selected=-1; game.legal=[]; game.pendingPromo=null;
  promoCtn.hidden=true;
  game.lastMove=s.moves.length?parseMove(s.moves[s.moves.length-1]):null;
  game.oppPresent=game.myColor===WHITE?s.guestIn:s.hostIn;
  updateNames();
  if(s.status===3){
    game.over=true;
    game.winner=s.winner==="draw"?"draw":(s.winner==="white"?WHITE:BLACK);
    game.reason=s.reason||"";
  }else{
    game.over=false; game.winner=null; game.reason="";
  }
  renderAll();
}
function enterOnlineGame(){
  lobbyCtn.hidden=true;
  gameCtn.hidden=false;
  roomBox.hidden=false;
  chatBox.hidden=false;
  roomCodeEl.textContent=game.roomCode;
  game.selected=-1; game.legal=[]; game.locked=false;
  updateNames();
  renderAll();
  updateButtons();
  sfx("deal");
}
async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=r.code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
function sendMove(encoded,m){
  game.locked=true;
  game.state=makeMove(game.state,m);
  game.moves.push(encoded);
  game.lastMove=m;
  game.selected=-1; game.legal=[]; game.pendingPromo=null;
  promoCtn.hidden=true;
  renderAll();
  const evt=game.moves.length-1;
  rpc("move",JSON.stringify({code:game.roomCode,evt:evt,move:encoded})).then(function(r){
    if(!r||!r.ok){
      game.locked=false;
      flashBanner("Move rejected ("+(r&&r.err?r.err:"error")+")");
      return rpc("getRoom",game.roomCode).then(function(s){
        if(s&&s.snap)applyServerSnap(s.snap);
      });
    }
  }).catch(function(){
    game.locked=false;
    flashBanner("Couldn't send your move — reconnecting…");
  });
}
function exitToLobby(){
  if(game.mode==="online"){
    if(game.socket&&game.sockOpen){
      rpc("leave",game.roomCode).catch(function(){});
    }
    exitOnline(true);
    refreshContinuePanel();
  }else{
    game.mode=null;
    gameCtn.hidden=true;
    lobbyCtn.hidden=false;
    refreshContinuePanel();
  }
}

/* ══════════ table chat (online) ══════════ */
function esc(s){ return String(s).replace(/[&<>"']/g,function(c){ return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]; }); }
function fmtTime(ts){ const d=new Date((ts||0)*1000); return String(d.getHours()).padStart(2,"0")+":"+String(d.getMinutes()).padStart(2,"0"); }
function chatName(role){
  if(role===game.myRole)return "You";
  return game.myColor===WHITE?"Black":"White";
}
function addChat(role,text,ts){
  const row=document.createElement("div");
  row.className="chat-item"+(role===game.myRole?" mine":"");
  row.innerHTML="<b>"+esc(chatName(role))+"</b>"+esc(text)+'<span class="chat-time">'+fmtTime(ts)+'</span>';
  chatMsgsEl.appendChild(row);
  chatMsgsEl.scrollTop=chatMsgsEl.scrollHeight;
  if(role!==game.myRole)sfx("draw");
}
function sendChat(){
  const t=chatInput.value.trim().slice(0,160);
  if(!t||game.mode!=="online"||!game.sockOpen)return;
  chatInput.value="";
  rpc("chat",JSON.stringify({code:game.roomCode,text:t})).then(function(r){
    if(r&&!r.ok)flashBanner("Chat blocked ("+r.err+").");
  }).catch(function(){});
}

/* ══════════ save / continue (hotseat & AI) ══════════
   The board is rebuilt from the move list (stateFromMoves), so the
   save just stores the moves plus how the match ended. */
const SAVE_KEY="bgn_save_chess";
function saveGame(){
  if(game.mode!=="local"&&game.mode!=="ai")return;
  if(game.over){ deleteSave(); return; }
  try{ localStorage.setItem(SAVE_KEY, JSON.stringify({v:1,mode:game.mode,moves:game.moves,over:false,winner:null,reason:null,savedAt:Date.now()})); }catch(e){}
}
function hasSave(){ try{ return !!localStorage.getItem(SAVE_KEY); }catch(e){ return false; } }
function deleteSave(){ try{ localStorage.removeItem(SAVE_KEY); }catch(e){} }
function refreshContinuePanel(){
  if(!continuePanel)return;
  const s=hasSave();
  continuePanel.hidden=!s;
  if(!s)return;
  let info="A saved match was found.";
  try{ const o=JSON.parse(localStorage.getItem(SAVE_KEY)); if(o&&o.savedAt)info="Saved "+new Date(o.savedAt).toLocaleTimeString()+" — move "+((o.moves||[]).length+1)+"."; }catch(e){}
  continueInfoEl.textContent=info;
}
function restoreMatch(o){
  resetMatch();
  game.mode=o.mode==="ai"?"ai":"local";
  game.moves=o.moves.slice();
  game.state=stateFromMoves(game.moves);
  game.over=!!o.over; game.winner=o.winner||null; game.reason=o.reason||null;
  whiteNameEl.textContent=game.mode==="ai"?"You":"White";
  blackNameEl.textContent=game.mode==="ai"?"The Machine":"Black";
  roomBox.hidden=true; chatBox.hidden=true; claimBtn.hidden=true;
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  sfx("deal");
  renderAll();
  if(game.over)endGame(game.winner,game.reason);
  else if(game.mode==="ai"&&game.state.turn===BLACK)scheduleAI();
}
function continueGame(){
  let o; try{ o=JSON.parse(localStorage.getItem(SAVE_KEY)); }catch(e){ o=null; }
  if(!o||!Array.isArray(o.moves)){ flashBanner("No saved match found."); return; }
  restoreMatch(o);
}

const fsCfg={canvas:null,resize:null};
function fsRefit(){ if(fsCfg.resize)try{ fsCfg.resize(); }catch(e){} renderAll(); }
function openFullscreen(){
  if(!fsCfg.canvas||!fsOverlay.hidden)return;
  const el=fsCfg.canvas;
  if(!el._fsOrig)el._fsOrig={parent:el.parentNode,next:el.nextSibling};
  fsBoardWrap.appendChild(el);
  fsOverlay.hidden=false;
  fsStatusEl.textContent=bannerEl.textContent||"Full screen map";
  document.body.style.overflow="hidden";
  fsRefit();
  try{ if(fsOverlay.requestFullscreen)fsOverlay.requestFullscreen().catch(function(){}); }catch(e){}
}
function closeFullscreen(){
  if(fsOverlay.hidden)return;
  fsOverlay.hidden=true;
  document.body.style.overflow="";
  if(document.fullscreenElement)try{ document.exitFullscreen().catch(function(){}); }catch(e){}
  const el=fsCfg.canvas;
  if(el&&el._fsOrig){
    try{ el._fsOrig.parent.insertBefore(el,el._fsOrig.next); }catch(e){ el._fsOrig.parent.appendChild(el); }
  }
  fsRefit();
  renderAll();
}
window.bgnFullscreen={
  register:function(o){
    if(!o||!o.canvas)return;
    fsCfg.canvas=o.canvas;
    if(typeof o.resize==="function")fsCfg.resize=o.resize;
    if(o.tile)fsTileCtn.appendChild(o.tile);
    if(o.meeple)fsMeepleCtn.appendChild(o.meeple);
    fsOpenBtn.hidden=false;
  },
  open:openFullscreen,
  close:closeFullscreen,
  isOpen:function(){ return !fsOverlay.hidden; }
};
fsOpenBtn.addEventListener("click",openFullscreen);
fsExitBtn.addEventListener("click",closeFullscreen);
window.addEventListener("resize",function(){ if(!fsOverlay.hidden)fsRefit(); });
