app.applySave=function(d){
  if(!d||!d.game||typeof game.deserialize!=="function")return false;
  if(!game.deserialize(d.game))return false;

  app.elapsed=d.e||0;
  app.status="playing";
  app.startTs=Date.now()-app.elapsed*1000;
  app.renderControls();
  return true;
};

  var d;
  try{ d=JSON.parse(str); }catch(e){ return "That doesn't look like valid JSON."; }
  if(!d||typeof d!=="object"||Array.isArray(d))return "That doesn't look like a game export.";

function cloneWithStyles(src){
  var clone=src.cloneNode(true);
  clone.setAttribute("xmlns","http://www.w3.org/1999/xhtml");
  var PROPS=["display","position","top","left","width","height","min-width","min-height","aspect-ratio","box-sizing",
    "padding-top","padding-right","padding-bottom","padding-left","margin-top","margin-right","margin-bottom","margin-left",
    "grid-template-columns","grid-template-rows","gap","grid-gap","align-items","justify-content","justify-items","align-content","flex-direction","flex-wrap",
    "background-color","background-image","background-size","background-position","background-repeat",
    "border-radius",
    "border-top-width","border-top-style","border-top-color","border-right-width","border-right-style","border-right-color",
    "border-bottom-width","border-bottom-style","border-bottom-color","border-left-width","border-left-style","border-left-color",
    "color","font-family","font-size","font-weight","font-style","line-height","letter-spacing","text-transform","text-align",
    "box-shadow","opacity","overflow","cursor","user-select","text-shadow","outline"];
  function walk(el,c){
    if(!el||!c)return;
    var cs=window.getComputedStyle(el),style="";
    for(var i=0;i<PROPS.length;i++){
      var v=cs.getPropertyValue(PROPS[i]);
      if(v)style+=PROPS[i]+":"+v+";";
    }
    c.setAttribute("style",style);
    for(var k=0;k<el.children.length;k++)walk(el.children[k],c.children[k]);
  }
  walk(src,clone);
  return clone;
}
app.renderBoardImage=function(){
  var el=(typeof game.captureTarget==="function")?game.captureTarget():boardEl;
  var w=el.offsetWidth,h=el.offsetHeight;
  if(!w||!h)return Promise.resolve(null);
  var scale=3;
  var svg='<svg xmlns="http://www.w3.org/2000/svg" width="'+w+'" height="'+h+'">'+
    '<foreignObject width="100%" height="100%">'+new XMLSerializer().serializeToString(cloneWithStyles(el))+'</foreignObject></svg>';
  return new Promise(function(resolve){
    var img=new Image();
    img.onload=function(){
      var c=document.createElement("canvas");
      c.width=Math.round(w*scale); c.height=Math.round(h*scale);
      var ctx=c.getContext("2d");
      ctx.scale(scale,scale);
      ctx.drawImage(img,0,0);
      resolve(c);
    };
    img.onerror=function(){ resolve(null); };
    img.src="data:image/svg+xml;charset=utf-8,"+encodeURIComponent(svg);
  });
};
app.renderFramedImage=function(){
  return app.renderBoardImage().then(function(boardCanvas){
    if(!boardCanvas)return null;
    var s=3; /* supersample factor — matches the board canvas */
    var bw=boardCanvas.width/s, bh=boardCanvas.height/s;
    var side=48, top=84, bottom=64;
    var W=bw+side*2, H=bh+top+bottom;
    var c=document.createElement("canvas");
    c.width=Math.round(W*s); c.height=Math.round(H*s);
    var ctx=c.getContext("2d");
    function rr(x,y,w,h,r){
      ctx.beginPath();
      ctx.moveTo(x+r,y); ctx.arcTo(x+w,y,x+w,y+h,r);
      ctx.arcTo(x+w,y+h,x,y+h,r); ctx.arcTo(x,y+h,x,y,r);
      ctx.arcTo(x,y,x+w,y,r); ctx.closePath();
    }
    /* background — matches the page's dark radial theme */
    var g=ctx.createRadialGradient(W/2*s,0,0,W/2*s,0,Math.max(W,H)*s);
    g.addColorStop(0,"#1b1730"); g.addColorStop(1,"#0a0912");
    ctx.fillStyle=g; ctx.fillRect(0,0,W*s,H*s);
    /* the board itself (blitted at native resolution) */
    ctx.drawImage(boardCanvas,side*s,top*s);
    /* gold frame */
    ctx.strokeStyle="#d4af37"; ctx.lineWidth=3*s;
    rr((side-14)*s,(top-14)*s,(bw+28)*s,(bh+28)*s,10*s); ctx.stroke();
    ctx.strokeStyle="rgba(212,175,55,.3)"; ctx.lineWidth=1*s;
    rr((side-22)*s,(top-22)*s,(bw+44)*s,(bh+44)*s,14*s); ctx.stroke();
    /* title + subtitle */
    ctx.fillStyle="#f6e6a4";
    ctx.font="600 "+Math.round(30*s)+"px Georgia, serif";
    ctx.textAlign="center"; ctx.textBaseline="middle";
    ctx.fillText(app.gameName().toUpperCase(),W/2*s,(top/2-10)*s);
    ctx.fillStyle="#8f721d";
    ctx.font="700 "+Math.round(12*s)+"px ui-sans-serif, system-ui, sans-serif";
    ctx.fillText(((DIFFS[app.diff]?DIFFS[app.diff].label:app.diff)+" · "+new Date().toISOString().slice(0,10)).toUpperCase(),W/2*s,(top/2+11)*s);
    /* footer strip */
    ctx.strokeStyle="rgba(212,175,55,.35)"; ctx.lineWidth=1*s;
    ctx.beginPath(); ctx.moveTo(side*s,(H-bottom)*s); ctx.lineTo((W-side)*s,(H-bottom)*s); ctx.stroke();
    /* URL (bottom-left) */
    ctx.fillStyle="#8b87a0";
    ctx.font="600 "+Math.round(14*s)+"px ui-sans-serif, system-ui, sans-serif";
    ctx.textAlign="left"; ctx.textBaseline="middle";
    var url="https://perchance.org/"+(window.generatorName||"");
    var fy=((H-bottom)+(bottom/2))*s;
    var badgeW=96,badgeGap=24;
    var avail=(W-side-8)-(side+8)-badgeW-badgeGap;
    var urlW=ctx.measureText(url).width;
    if(urlW>avail){
      var f=Math.max(9,Math.floor(14*avail/urlW));
      ctx.font="600 "+Math.round(f*s)+"px ui-sans-serif, system-ui, sans-serif";
    }
    ctx.fillText(url,(side+8)*s,fy);
    /* BGN badge (bottom-right) */
    var bh2=36,bx=W-side-badgeW-10,by=H-bottom+((bottom-bh2)/2);
    ctx.fillStyle="rgba(212,175,55,.12)";
    rr(bx*s,by*s,badgeW*s,bh2*s,9*s); ctx.fill();
    ctx.strokeStyle="#d4af37"; ctx.lineWidth=1.5*s;
    rr(bx*s,by*s,badgeW*s,bh2*s,9*s); ctx.stroke();
    ctx.fillStyle="#f6e6a4";
    ctx.font="800 "+Math.round(19*s)+"px Georgia, serif";
    ctx.textAlign="center"; ctx.textBaseline="middle";
    ctx.fillText("BGN",(bx+badgeW/2)*s,(by+bh2/2+1)*s);
    return c;
  });
};

function downloadBlob(name,mime,blob){
  var url=URL.createObjectURL(blob);
  var a=document.createElement("a");
  a.href=url; a.download=name;

app.submitScore=function(){
  if(app.winTimeMs<3000){ winMsgEl.textContent="That was too fast to count — solve it for real!"; return; }
  var name=(winNameInput.value||"").trim().slice(0,20)||"Anonymous";
  localStorage.setItem(app.meta.saveKey+"-name",name);
  var entry={n:name,t:app.winTimeMs,d:new Date().toISOString().slice(0,10),h:app.hints};
  winMsgEl.textContent="Publishing…";
  Promise.resolve().then(function(){
    if(rootRef.uploadPlugin&&app.meta.leaderboardFile)return rootRef.uploadPlugin.editable.get(app.meta.leaderboardFile);
    return null;
  }).then(function(fresh){
    if(fresh){
      try{
        var b=JSON.parse(fresh);
        for(var k in boards)if(b[k]&&Array.isArray(b[k]))boards[k]=b[k].slice(0,10);
      }catch(e){}
    }
    var list=(boards[app.diff]||[]).concat(entry).sort(function(a,b){return a.t-b.t;}).slice(0,10);
    boards[app.diff]=list;
    app.renderLeaderboard();
    if(!rootRef.uploadPlugin||!app.meta.leaderboardFile){ winMsgEl.textContent="Score saved locally."; return; }
    var knownKey=localStorage.getItem(app.meta.saveKey+"-lbKey")||app.meta.leaderboardEditKey;
    var data=JSON.stringify(boards);
    function createFresh(){
      return rootRef.uploadPlugin.editable.set(app.meta.leaderboardFile,data).then(function(created){
        if(created&&created.editKey){
          localStorage.setItem(app.meta.saveKey+"-lbKey",created.editKey);
          winMsgEl.textContent="Published! For a board every player can update, paste this editKey into main.pjs: "+created.editKey;
        }else if(created&&created.error){
          winMsgEl.textContent="Couldn't publish to the shared board ("+created.error+") — score kept locally.";
        }else{
          winMsgEl.textContent="Couldn't publish to the shared board — score kept locally.";
        }
      });
    }
    var p=knownKey?rootRef.uploadPlugin.editable.set(app.meta.leaderboardFile,data,{editKey:knownKey}):createFresh();
    p.then(function(res){
      if(!res)return;
      if(res&&res.editKey){
        localStorage.setItem(app.meta.saveKey+"-lbKey",res.editKey);
        winMsgEl.textContent="Published to the leaderboard!";
        return;
      }
      if(res&&res.error==="invalid_edit_key")return createFresh();
      if(res&&res.error)winMsgEl.textContent="Couldn't publish to the shared board ("+res.error+") — score kept locally.";
      else winMsgEl.textContent="Published to the leaderboard!";
    }).catch(function(){
      winMsgEl.textContent="Couldn't reach the leaderboard — score kept locally.";
    });

/* ── board + rules ── */
var LINES=[[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];
function winnerOf(cells){
  for(var i=0;i<LINES.length;i++){
    var a=LINES[i][0],b=LINES[i][1],c=LINES[i][2];
    if(cells[a]&&cells[a]===cells[b]&&cells[a]===cells[c])return cells[a];
  }
  return null;
}
function lineOf(cells){
  for(var i=0;i<LINES.length;i++){
    var a=LINES[i][0],b=LINES[i][1],c=LINES[i][2];
    if(cells[a]&&cells[a]===cells[b]&&cells[a]===cells[c])return LINES[i];
  }
  return null;
}
function emptyCells(cells){
  var e=[];
  for(var i=0;i<cells.length;i++)if(!cells[i])e.push(i);
  return e;
}

/* ── minimax: O (computer) maximizes, X minimizes ── */
function minimax(cells,turn){
  var w=winnerOf(cells);
  if(w==="O")return 1;
  if(w==="X")return -1;
  var e=emptyCells(cells);
  if(!e.length)return 0;
  var scores=[];
  for(var i=0;i<e.length;i++){
    var c=cells.slice(); c[e[i]]=turn;
    scores.push(minimax(c,turn==="X"?"O":"X"));
  }
  return turn==="O"?Math.max.apply(null,scores):Math.min.apply(null,scores);
}

/* ── computer's move ── */
function chooseMove(){
  var level=DIFFS[app.diff]?DIFFS[app.diff].ai:"block";
  var e=emptyCells(S.cells);
  if(!e.length)return -1;
  if(level==="random")return e[Math.floor(Math.random()*e.length)];
  if(level==="block"){
    var i;
    for(i=0;i<e.length;i++){ var t=S.cells.slice(); t[e[i]]="O"; if(winnerOf(t)==="O")return e[i]; }
    for(i=0;i<e.length;i++){ var t2=S.cells.slice(); t2[e[i]]="X"; if(winnerOf(t2)==="X")return e[i]; }
    if(!S.cells[4])return 4;
    var corners=[0,2,6,8].filter(function(n){return !S.cells[n];});
    if(corners.length)return corners[Math.floor(Math.random()*corners.length)];
    return e[Math.floor(Math.random()*e.length)];
  }
  var best=-1,bestScore=-Infinity;
  for(var j=0;j<e.length;j++){
    var t3=S.cells.slice(); t3[e[j]]="O";
    var sc=minimax(t3,"X");
    if(sc>bestScore){ bestScore=sc; best=e[j]; }
  }
  return best;
}

/* ── state ── */
var S={cells:null,turn:"X",mode:"ai",over:false,result:null,line:null,moves:0,history:[],suggestion:null,aiTimer:null};
function resetState(){
  S.cells=["","","","","","","","",""];
  S.turn="X"; S.over=false; S.result=null; S.line=null; S.moves=0;
  S.history=[]; S.suggestion=null;
  clearTimeout(S.aiTimer); S.aiTimer=null;
}
function scheduleAI(){
  if(S.mode!=="ai"||S.over||S.turn!=="O")return;
  clearTimeout(S.aiTimer);
  app.setStatus("Computer is thinking…");
  var d=DIFFS[app.diff]?DIFFS[app.diff].delay:cfg("aiDelayMs",420);
  S.aiTimer=setTimeout(function(){
    if(app.status!=="playing"){ scheduleAI(); return; }
    if(S.over||S.mode!=="ai"||S.turn!=="O"){ if(!S.over)app.setStatus("Your turn."); return; }
    var m=chooseMove();
    if(m<0){ app.setStatus("Computer is stuck — your turn."); return; }
    doMove(m);
    app.save();
    app.completeIfWon();
    if(!S.over)app.setStatus("Your turn.");
  },d);
}
function doMove(i){
  if(!S.cells||S.over||S.cells[i])return false;
  S.cells[i]=S.turn;
  S.history.push(i);
  S.moves++;
  S.suggestion=null;
  var w=winnerOf(S.cells);
  if(w){ S.over=true; S.result=w; S.line=lineOf(S.cells); app.setStatus(w+" wins!"); }
  else if(emptyCells(S.cells).length===0){ S.over=true; S.result="draw"; S.line=null; app.setStatus("It's a draw!"); }
  else{
    S.turn=S.turn==="X"?"O":"X";
    if(S.mode==="2p")app.setStatus(S.turn+" to move.");
  }
  renderBoard(); renderHud();
  return true;
}

/* ── hint: mark the player's best square ── */
function suggestMove(){
  var e=emptyCells(S.cells);
  var best=-1,bestScore=Infinity;
  for(var i=0;i<e.length;i++){
    var c=S.cells.slice(); c[e[i]]="X";
    var sc=minimax(c,"O");
    if(sc<bestScore){ bestScore=sc; best=e[i]; }
  }
  return best;
}
function revealHintFn(){
  if(app.status!=="playing")return;
  if(S.mode==="2p"){ app.setStatus("Hints are for computer games only."); return; }
  if(S.over){ app.setStatus("This game is already over."); return; }
  if(S.turn!=="X"){ app.setStatus("It's not your turn."); return; }
  var i=suggestMove();
  if(i<0)return;
  S.suggestion=i;
  app.addHint(app.meta.hint1Cost);
  renderBoard();
  app.setStatus("Hint: play the ✦ square (+"+Math.round(app.meta.hint1Cost/1000)+"s).");
  var t=setTimeout(function(){ if(S.suggestion===i){ S.suggestion=null; renderBoard(); } },3000);
}

/* ── undo last move (⌫ / erase) ── */
function undoFn(){
  if(app.status!=="playing")return;
  if(!S.history.length){ app.setStatus("Nothing to undo."); return; }
  clearTimeout(S.aiTimer); S.aiTimer=null;
  S.history.pop();
  if(S.mode==="ai"&&S.history.length&&S.cells[S.history[S.history.length-1]]==="O")S.history.pop();
  S.cells=["","","","","","","","",""];
  for(var i=0;i<S.history.length;i++)S.cells[S.history[i]]=i%2===0?"X":"O";
  S.moves=S.history.length;
  S.turn=S.history.length%2===0?"X":"O";
  S.over=false; S.result=null; S.line=null; S.suggestion=null;
  renderBoard(); renderHud();
  app.setStatus("Move undone.");
  if(S.mode==="ai"&&S.turn==="O")scheduleAI();
}

    else{ S.over=false; S.result=null; S.line=null; }
    renderBoard(); renderHud();
    setTimeout(function(){
      if(S.over&&app.status==="playing")app.completeIfWon();
      else if(!S.over&&S.mode==="ai"&&S.turn==="O")scheduleAI();
    },0);
    return true;
  }
};
