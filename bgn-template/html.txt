<script>
(function(){
  "use strict";

  /* ── read config from main.pjs (with safe defaults) ── */
  function val(n, d){ 
    if(n==null) return d;
    var v = (n && typeof n.evaluateItem==="function") ? n.evaluateItem() : n.evaluateItem;
    return v==null ? d : String(v);
  }
  var rootRef = window.root || {};
  var accent = val(rootRef.accentColor, "") || "#d4af37";
  var heroImg = val(rootRef.heroImage, "").trim();
  document.body.style.setProperty("--bgn-accent", accent);
  var hero = document.getElementById("gameHero");
  if(heroImg && hero){ hero.classList.add("has-img"); hero.style.setProperty("--bgn-hero-img", "url('"+heroImg+"')"); }

  /* ── demo dice roller (delete me) ── */
  var PIPS = {1:[5],2:[3,7],3:[1,5,9],4:[1,3,7,9],5:[1,3,5,7,9],6:[1,3,4,6,7,9]};
  var dieEl = document.getElementById("demoDie");
  var logEl = document.getElementById("rollLog");
  function showDie(n){
    dieEl.classList.remove("num");
    dieEl.innerHTML = "";
    if(n<=6){ PIPS[n].forEach(function(p){ var s=document.createElement("span"); s.className="d c"+p; dieEl.appendChild(s); }); }
    else { dieEl.classList.add("num"); dieEl.textContent = n; }
  }
  function roll(sides){
    var n = 1 + Math.floor(Math.random()*sides);
    dieEl.classList.remove("shake"); void dieEl.offsetWidth; dieEl.classList.add("shake");
    setTimeout(function(){ showDie(n); }, 200);
    return n;
  }
  function log(msg){
    logEl.textContent = msg + (logEl.textContent.indexOf("Your rolls")===0 ? "" : " · "+logEl.textContent);
  }
  var d6Btn = document.getElementById("d6Btn");
  var d20Btn = document.getElementById("d20Btn");
  if(d6Btn) d6Btn.onclick = function(){ log("d6 → "+roll(6)); };
  if(d20Btn) d20Btn.onclick = function(){ log("d20 → "+roll(20)); };
})();
</script>
