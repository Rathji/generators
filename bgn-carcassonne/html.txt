<script type="text/x-server-plugin">
/* ── BGN table server ─────────────────────────────────────────────
   Generic two-player rooms: host + guest, 5-letter codes. The
   server sequences turns (host first unless a rematch flips it),
   stores ONE full-game snapshot per room (opaque string submitted
   by whichever player is on turn), and tracks winner/reason for
   resign, forfeit and end-of-game. No game logic here — the
   snapshot is whatever the game decides it should be.
   Durable state in the plugin `state` bytes (versioned layout).  */
const SLOT=1024;
const STATE_VERSION=4;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const MAX_PAYLOAD=2000;

let rooms=new Map();           // code -> room object (live cache)
const conns=new Map();         // conn.id -> {code, role}
const moveTimes=new Map();     // conn.id -> last move ms (rate limit)
const chatTimes=new Map();     // conn.id -> last chat ms (rate limit)
const CHAT_BANNED=["nigger","nigga","faggot","kys","kill yourself","retard","heil hitler","tranny"];
function cleanText(s,max){ return String(s||"").trim().replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").slice(0,max); }
function isChatBanned(t){ const s=t.toLowerCase(); for(let i=0;i<CHAT_BANNED.length;i++){ if(s.indexOf(CHAT_BANNED[i])!==-1) return true; } return false; }

/* ── durable state (slot layout) ──
   slot byte 0: status 0=free 1=waiting 2=playing 3=finished
   slot +1..+8: room code (8 ASCII, zero padded)
   slot +9: first (0=host moves first, 1=guest; rematch flips)
   slot +10: state version
   slot +11: reserved
   slot +12..+15: uint32 lastActive (unix secs)
   slot +16: winner 0=none 1=host 2=guest 3=draw
   slot +17: reason 0=none 1=resign 2=forfeit 3=end
   slot +18..+19: uint16 turn counter (completed turns)
   slot +20..+21: uint16 snapshot bytes length
   slot +22..+23: reserved
   slot +24..: snapshot bytes (utf8, latest only)                */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+12]<<24)|(state[o+13]<<16)|(state[o+14]<<8)|state[o+15]; }
function slotStatus(i){ return state[off(i)]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function utf8Encode(s){ const out=[]; for(let i=0;i<s.length;i++){ const c=s.charCodeAt(i); if(c<128)out.push(c); else if(c<2048)out.push(192|(c>>6),128|(c&63)); else out.push(224|(c>>12),128|((c>>6)&63),128|(c&63)); } return out; }
function utf8Decode(bytes){ let s="",i=0; while(i<bytes.length){ const b=bytes[i]; if(b<128){ s+=String.fromCharCode(b); i++; } else if(b<224){ s+=String.fromCharCode(((b&31)<<6)|(bytes[i+1]&63)); i+=2; } else { s+=String.fromCharCode(((b&15)<<12)|((bytes[i+1]&63)<<6)|(bytes[i+2]&63)); i+=3; } } return s; }
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+9]=room.first;
  state[o+10]=STATE_VERSION;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+12]=(la>>>24)&255; state[o+13]=(la>>>16)&255; state[o+14]=(la>>>8)&255; state[o+15]=la&255;
  state[o+16]=room.winner==="host"?1:room.winner==="guest"?2:room.winner==="draw"?3:0;
  state[o+17]=room.reason==="resign"?1:room.reason==="forfeit"?2:room.reason==="end"?3:0;
  const tn=Math.max(0,Math.min(0xFFFF,room.turn));
  state[o+18]=(tn>>8)&255; state[o+19]=tn&255;
  const mb=room.snapshot?utf8Encode(room.snapshot):[];
  const len=Math.min(SLOT-24,mb.length);
  state[o+20]=(len>>8)&255; state[o+21]=len&255;
  if(len>0)state.set(mb.slice(0,len),o+24);
}
const REASON=["","resign","forfeit","end"];
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(state[o+10]!==STATE_VERSION){ state.fill(0,o,o+SLOT); continue; }  // old/foreign schema → reclaim
    if(st===1){ state.fill(0,o,o+SLOT); continue; }                        // abandoned waiting rooms are reclaimed
    const snapLen=(state[o+20]<<8)|state[o+21];
    rooms.set(codeAt(i),{
      slot:i, code:codeAt(i), status:st,
      first:state[o+9], hostId:-1, guestId:-1,
      turn:(state[o+18]<<8)|state[o+19],
      snapshot:snapLen?utf8Decode(state.slice(o+24,o+24+snapLen)):null,
      winner:state[o+16]===1?"host":state[o+16]===2?"guest":state[o+16]===3?"draw":null,
      reason:REASON[state[o+17]]||null,
      lastAct:readLastAct(i)
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── helpers ── */
function topic(code){ return "tbl:"+code; }
function snap(room){ return {code:room.code,status:room.status,snapshot:room.snapshot,first:room.first,turn:room.turn,hostIn:room.hostId!==-1,guestIn:room.guestId!==-1,winner:room.winner,reason:room.reason}; }
function broadcast(room){ try{ pubsub.publish(topic(room.code), JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }
function turnRoleFor(n,first){ return ((n+(first?1:0))%2)===0?"host":"guest"; }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room={slot,code,status:1,first:0,hostId:conn.id,guestId:-1,turn:0,snapshot:null,winner:null,reason:null,lastAct:nowSec()};
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host"});
    return JSON.stringify({ok:true,code,role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)room.status=2;
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role});
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true,role,snap:snap(room)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    return JSON.stringify({ok:true,snap:snap(room)});
  },
  move({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(turnRoleFor(room.turn,room.first)!==role)return JSON.stringify({ok:false,err:"not_your_turn"});
    const now=Date.now();
    if(now-(moveTimes.get(conn.id)||0)<300)return JSON.stringify({ok:false,err:"rate_limited"});
    moveTimes.set(conn.id,now);
    if(typeof obj.evt!=="number"||obj.evt!==room.turn)return JSON.stringify({ok:false,err:"stale"});
    if(typeof obj.data!=="string"||obj.data.length<1||obj.data.length>MAX_PAYLOAD)return JSON.stringify({ok:false,err:"bad_move"});
    room.snapshot=obj.data;
    room.turn++;
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true});
  },
  endGame({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(["host","guest","draw"].indexOf(obj.winner)===-1)return JSON.stringify({ok:false,err:"bad_request"});
    room.status=3; room.winner=obj.winner; room.reason="end";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"guest":"host"; room.reason="resign";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    room.status=2; room.turn=0; room.snapshot=null; room.winner=null; room.reason=null;
    room.first=1-room.first;
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3; room.winner=role; room.reason="forfeit";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"guest":"host"; room.reason="forfeit";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  chat({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const now=Date.now();
    if(now-(chatTimes.get(conn.id)||0)<700)return JSON.stringify({ok:false,err:"rate_limited"});
    chatTimes.set(conn.id,now);
    const text=cleanText(obj.text,160);
    if(!text)return JSON.stringify({ok:false,err:"empty"});
    if(isChatBanned(text))return JSON.stringify({ok:false,err:"blocked"});
    pubsub.publish(topic(room.code),JSON.stringify({t:"chat",role,text,ts:nowSec()}));
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  conns.delete(conn.id);
  moveTimes.delete(conn.id);
  chatTimes.delete(conn.id);
  if(!info)return;
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

function buildDeck(seed){

}

function resultText(){
  if(game.winner==="draw")return "Draw — the match is over.";
  if(game.mode!=="online"){
    if(game.winner==="draw")return "Draw — final scores are level!";
    return roleName(game.winner)+" wins!";
  }
  const w=roleName(game.winner);
  const youWon=game.winner===game.myRole;
  if(youWon){
    if(game.reason==="resign")return "You win — your opponent resigned!";
    if(game.reason==="forfeit")return "You win — your opponent left!";
    return "You win!";
  }
  if(game.winner!==game.myRole){
    if(game.reason==="resign")return "You lose — you resigned.";
    if(game.reason==="forfeit")return "You lose — you left the match.";
    return "You lose.";
  }
  if(game.reason==="resign")return w+" wins by resignation.";
  if(game.reason==="forfeit")return w+" wins — the other side left.";
  return w+" wins!";
}

function noLegalAnyRotation(){
  if(!game.st||!game.board||game.currentTile==null)return false;

    for(let r=0;r<4;r++){ if(isLegal(game.currentTile,r,xy[0],xy[1],game.board)){ ok=true; break; } }
    if(ok)return false;

}
function adjacentCells(){

  }
  return set;
}

      const feat=td.f[fi];
      const node={idx:nodes.length,ti:ti,ft:feat.t,fi:fi,parent:nodes.length,x:t.x,y:t.y,rot:t.rot,lp:feat.p,g:feat.g||0,m:null};
      if(t.m&&t.m.t===feat.t&&t.m.f===fi)node.m=t.m;
      nodes.push(node);
      if(feat.t===0)by.c.push(node.idx);
      else if(feat.t===1)by.r.push(node.idx);
      else if(feat.t===2)by.m=node.idx;
      else by.f.push(node.idx);
      if(feat.p)for(const p of feat.p)pm[p]=node.idx;
    }
    byTile.push(by);
    posMap.push(pm);
    cellMap.set(key(t.x,t.y),ti);
  }
  const find=function(a){ let r=a; while(nodes[r].parent!==r)r=nodes[r].parent; let c=a; while(c!==r){const n=nodes[c].parent; nodes[c].parent=r; c=n;} return r; };
  const union=function(a,b){ const ra=find(a), rb=find(b); if(ra!==rb)nodes[rb].parent=ra; };
  for(let ti=0;ti<st.placed.length;ti++){
    const t=st.placed[ti];
    for(let d=0;d<4;d++){
      const nti=cellMap.get(key(t.x+DIRS[d][0],t.y+DIRS[d][1]));
      if(nti===undefined)continue;
      const nb=st.placed[nti];
      for(let k=0;k<3;k++){
        const wa=EDGE_POS[SIDES[d]][k];
        const wb=OPP_POS[wa];
        const la=localP(wa,t.rot), lb=localP(wb,nb.rot);
        const ai=posMap[ti][la], bi=posMap[nti][lb];
        if(ai===undefined||bi===undefined)continue;
        const na=nodes[ai], nbg=nodes[bi];
        if(na.ft!==nbg.ft)continue;
        if(na.ft===0||na.ft===1||na.ft===3)union(ai,bi);
      }
    }
  }
  return {nodes:nodes,byTile:byTile,posMap:posMap,cellMap:cellMap,find:find,union:union};
}
function gather(graph,root,st){
  const g=graph;
  const comp={root:root,ft:-1,nodes:[],tiles:new Set(),coats:0,open:0,meeps:[],neighbors:0};
  for(const n of g.nodes){
    if(g.find(n.idx)!==root)continue;
    comp.nodes.push(n);
    comp.ft=n.ft;
    comp.tiles.add(n.ti);
    if(n.ft===0)comp.coats+=n.g;
    if(n.m)comp.meeps.push(n.m);
  }
  if(comp.ft===0){
    for(const n of comp.nodes){
      for(const lp of n.lp){
        const wc=worldP(lp,n.rot);
        if(CROSS_IDX[wc]===undefined)continue;
        const d=CROSS_IDX[wc];
        const nti=g.cellMap.get(key(n.x+DIRS[d][0],n.y+DIRS[d][1]));
        if(nti===undefined){ comp.open++; continue; }
        const nb=st.placed[nti];
        const nbl=localP(OPP_CROSS[wc],nb.rot);
        const other=g.posMap[nti][nbl];
        if(other===undefined||g.nodes[other].ft!==0)comp.open++;
      }
    }
  }else if(comp.ft===1){
    for(const n of comp.nodes){
      for(const ep of n.lp){
        if(ep===0)continue;
        const wc=worldP(ep,n.rot);
        const d=CROSS_IDX[wc];
        const nti=g.cellMap.get(key(n.x+DIRS[d][0],n.y+DIRS[d][1]));
        if(nti===undefined)comp.open++;
      }
    }
  }else if(comp.ft===2){
    const n=comp.nodes[0];
    for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
      if(dx===0&&dy===0)continue;
      if(g.cellMap.has(key(n.x+dx,n.y+dy)))comp.neighbors++;
    }
  }
  return comp;
}

/* ══════════ RULES ══════════ */
function edgeCharFor(id,rot,side){ return TILES[id].e[((SIDES.indexOf(side)-rot)%4+4)%4]; }
function edgeChar(t,side){ return TILES[t.id].e[((SIDES.indexOf(side)-t.rot)%4+4)%4]; }
function isLegal(id,rot,x,y,bd){
  if(bd.has(key(x,y)))return false;
  let has=false;
  for(let d=0;d<4;d++){
    const nb=bd.get(key(x+DIRS[d][0],y+DIRS[d][1]));
    if(!nb)continue;
    has=true;
    const e1=TILES[id].e[((d-rot)%4+4)%4];
    const e2=TILES[nb.id].e[(((d+2)%4-nb.rot)%4+4)%4];
    if(e1!==e2)return false;
  }
  return has;
}
function legalCells(id,rot,bd){
  const cells=new Set();
  for(const ckey of bd.keys()){
    const xy=ckey.split(",").map(Number);
    for(let d=0;d<4;d++)cells.add(key(xy[0]+DIRS[d][0],xy[1]+DIRS[d][1]));
  }
  const out=[];
  for(const ckey of cells){
    if(bd.has(ckey))continue;
    const xy=ckey.split(",").map(Number);
    if(isLegal(id,rot,xy[0],xy[1],bd))out.push(xy);
  }
  return out;
}
function legalRotations(id,x,y,bd){
  const ok=[];
  for(let r=0;r<4;r++)if(isLegal(id,r,x,y,bd))ok.push(r);
  return ok;
}
function scoreTile(ti,st){
  const bd=buildBoard(st.placed);
  const graph=buildGraph(st,bd);
  const t=st.placed[ti];
  const handled=new Set();
  const events=[];
  const finish=function(root){
    if(handled.has(root))return;
    const comp=gather(graph,root,st);
    if(comp.ft===0&&comp.open===0){ handled.add(root); events.push(award(comp,st)); }
    else if(comp.ft===1&&comp.open===0){ handled.add(root); events.push(award(comp,st)); }
    else if(comp.ft===2&&comp.neighbors===8){ handled.add(root); events.push(award(comp,st)); }
  };
  for(const n of graph.nodes){
    if(n.ti!==ti)continue;
    finish(graph.find(n.idx));
  }
  for(let dy=-1;dy<=1;dy++)for(let dx=-1;dx<=1;dx++){
    const nti=graph.cellMap.get(key(t.x+dx,t.y+dy));
    if(nti===undefined)continue;
    const mn=graph.byTile[nti].m;
    if(mn<0)continue;
    finish(graph.find(mn));
  }
  return events;
}
function award(comp,st){
  const ft=comp.ft;
  let points=0;
  if(ft===0)points=2*comp.tiles.size+2*comp.coats;
  else if(ft===1)points=comp.tiles.size;
  else points=9;
  const counts=[0,0];
  for(const mp of comp.meeps)counts[mp.p]++;
  const max=Math.max(counts[0],counts[1]);
  const winners=max>0?[0,1].filter(function(p){return counts[p]===max;}):[];
  for(const n of comp.nodes){ if(n.m)st.placed[n.ti].m=null; }
  return {type:ft,points:points,winners:winners,tiles:comp.tiles.size,coats:comp.coats,meeples:comp.meeps.length};
}

/* ══════════ REPLAY — deterministic re-derivation of scores & events ══════════
   Both the mover and every receiver run this on the same snapshot, so scores,
   returned meeples and logs always agree. */
function replay(st){
  const placed=st.placed.map(function(t){ return {id:t.id,rot:t.rot,x:t.x,y:t.y,dp:t.dp,m:t.m?{p:t.m.p,t:t.m.t,f:t.m.f}:null}; });
  const prefix={placed:placed};
  const scores=[0,0];
  const byMove={};
  for(let i=1;i<placed.length;i++){
    const events=scoreTile(i,prefix);
    byMove[placed[i].dp]=events;
    for(const e of events){ for(const w of e.winners)scores[w]+=e.points; }
  }
  return {scores:scores,byMove:byMove,placed:placed};
}
function countMeeples(placed){
  const c=[0,0];
  for(const t of placed){ if(t.m)c[t.m.p]++; }
  return c;
}
function meeplesLeft(p){ return MEEPLE_LIMIT-game.meeples[p]; }

/* ══════════ ENDGAME / FIELD SCORING ══════════ */
function finalScoring(st){
  const placed=st.placed.map(function(t){ return {id:t.id,rot:t.rot,x:t.x,y:t.y,dp:t.dp,m:t.m?{p:t.m.p,t:t.m.t,f:t.m.f}:null}; });
  st={seed:st.seed,deckPos:st.deckPos,first:st.first,placed:placed};
  const bd=buildBoard(st.placed);
  const graph=buildGraph(st,bd);
  const scores=[0,0];
  const events=[];
  const roots={};
  for(const n of graph.nodes)roots[graph.find(n.idx)]=true;
  const comps={};
  for(const r of Object.keys(roots)){
    const comp=gather(graph,parseInt(r,10),st);
    const k=parseInt(r,10);
    comps[k]=comp;
  }
  const byMove2={};
  for(const k of Object.keys(comps)){
    const comp=comps[k];
    if(comp.ft===0&&comp.open>0){
      const ev=award(comp,st);
      for(const w of ev.winners)scores[w]+=ev.points;
      events.push({type:comp.ft,points:ev.points,winners:ev.winners,tiles:comp.tiles.size,coats:comp.coats,field:false});
    }else if(comp.ft===1&&comp.open>0){
      const ev=award(comp,st);
      for(const w of ev.winners)scores[w]+=ev.points;
      events.push({type:comp.ft,points:ev.points,winners:ev.winners,tiles:comp.tiles.size,field:false});
    }else if(comp.ft===2&&comp.neighbors<8){
      const ev=award(comp,st);
      for(const w of ev.winners)scores[w]+=ev.points;
      events.push({type:comp.ft,points:ev.points,winners:ev.winners,tiles:1,field:false});
    }else if(comp.ft===3){
      // fields: 3 pts per adjacent COMPLETED city, majority
      const adjCities=new Set();
      for(const n of comp.nodes){
        for(const pF of n.lp){
          const p1=pF===12?1:pF+1, p2=pF===1?12:pF-1;
          for(const pC of [p1,p2]){
            const other=graph.posMap[n.ti][pC];
            if(other===undefined)continue;
            const on=graph.nodes[other];
            if(on.ft!==0)continue;
            const cityComp=gather(graph,graph.find(other),st);
            if(cityComp.open===0)adjCities.add(graph.find(other));
          }
        }
      }
      const n=adjCities.size;
      if(n>0){
        const ev=award(comp,st);
        const pts=3*n;
        for(const w of ev.winners)scores[w]+=pts;
        events.push({type:3,points:pts,winners:ev.winners,fields:n,field:true});
      }
    }
  }
  return {scores:scores,events:events,comps:comps};
}
/* ══════════ STATE ENCODING (online snapshot, ≤2000 chars) ══════════ */
function encodeState(st){
  return "S"+st.seed+";"+st.deckPos+";"+st.first+";"
    +st.placed.map(function(t){
      return t.id+":"+t.rot+":"+(t.x+32)+":"+(t.y+32)+":"+t.dp+":"+(t.m?t.m.p+""+t.m.t+""+t.m.f:"0");
    }).join(";");
}
function decodeState(s){
  try{
    const parts=s.split(";");
    const h=parts[0];
    const seed=parseInt(h.slice(1),10);
    const deckPos=parseInt(parts[1],10);
    const first=parseInt(parts[2],10);
    const placed=parts.slice(3).map(function(seg){
      const a=seg.split(":");
      let m=null;
      if(a[5]!=="0")m={p:+a[5].charAt(0),t:+a[5].charAt(1),f:+a[5].charAt(2)};
      return {id:+a[0],rot:+a[1],x:+a[2]-32,y:+a[3]-32,dp:+a[4],m:m};
    });
    return {seed:seed,deckPos:deckPos,first:first,placed:placed};
  }catch(e){ return null; }
}
function validateState(st,deck){
  if(!st||!Array.isArray(st.placed)||st.placed.length<1)return "bad";
  const p0=st.placed[0];
  if(p0.id!==START_TILE||p0.dp!==-1||p0.x!==0||p0.y!==0||p0.rot!==0)return "bad_start";
  if(st.seed!==game.seed)return "seed_mismatch";
  const bd=new Map();
  let maxDp=-1;
  for(let i=0;i<st.placed.length;i++){
    const t=st.placed[i];
    const ck=key(t.x,t.y);
    if(bd.has(ck))return "dup_cell";
    if(t.id<0||t.id>=TILES.length||t.rot<0||t.rot>3)return "bad_tile";
    if(t.dp>=0){
      if(t.dp>=deck.length||deck[t.dp]!==t.id)return "deck_mismatch";
      maxDp=Math.max(maxDp,t.dp);
    }
    if(t.m){
      if(t.m.p<0||t.m.p>1)return "bad_meeple";
      const td=TILES[t.id];
      let cnt=0;
      for(const f of td.f)if(f.t===t.m.t)cnt++;
      if(t.m.f<0||t.m.f>=cnt)return "bad_meeple";
    }
    if(i>0&&!isLegal(t.id,t.rot,t.x,t.y,bd))return "illegal:"+i;
    bd.set(ck,t);
  }
  if(st.deckPos!==maxDp+1||st.deckPos>deck.length)return "bad_deckpos";
  return "ok";
}

/* ══════════ TURN ENGINE ══════════ */
function startTileEntry(){ return {id:START_TILE,rot:0,x:0,y:0,dp:-1,m:null}; }
function isOver(){ return game.st.deckPos>=game.deck.length; }
function initFreshGame(){
  const seed=game.mode==="online"
    ? hashStr(game.roomCode+":"+game.first+":carc")
    : ((Math.random()*0x7fffffff)>>>0);
  game.seed=seed;
  game.deck=buildDeck(seed);
  game.st={seed:seed,deckPos:0,first:game.first,placed:[startTileEntry()]};
  game.board=buildBoard(game.st.placed);
  game.appliedDeckPos=0;
  game.currentTile=null;
  game.rotation=0;
  game.phase="idle";
  game.pending=null;
  game.hoverCell=null;
  game.lastPlaced=null;
  game.scores=[0,0];
  game.meeples=[MEEPLE_LIMIT,MEEPLE_LIMIT];
  game.over=false;
  game.winner=null;
  game.reason="";
  applyState(game.st,{silent:true});
  onTurnStart();
}
function canAct(){
  if(!game.st||game.over)return false;
  const t=turnRoleFor(game.st.deckPos);
  if(game.mode==="online")return t===game.myRole&&game.status===2&&game.sockOpen;
  if(game.mode==="ai")return t==="host";
  return true;
}
function onTurnStart(){
  if(!game.st||game.over)return;
  const t=turnRoleFor(game.st.deckPos);
  if(game.mode==="ai"&&t==="guest"){ game.currentTile=null; game.phase="idle"; aiTurnSoon(); renderAll(); return; }
  const mine=canAct();
  if(mine){
    game.currentTile=game.deck[game.st.deckPos];
    game.rotation=0;
    game.phase="place";
    game.pending=null;
    sfx("deal");
  }else{
    game.currentTile=null;
    game.phase="idle";
    game.pending=null;
  }
  renderAll();
}
function applyState(st,opts){
  if(!game.deck||game.seed!==st.seed){
    game.seed=st.seed;
    game.deck=buildDeck(st.seed);
    game.appliedDeckPos=0;
  }
  game.st=st;
  game.board=buildBoard(st.placed);
  game.first=st.first;
  const rp=replay(st);
  game.scores=rp.scores;
  game.meeples=countMeeples(st.placed);
  const v=st.deckPos;
  const silent=opts&&opts.silent;
  if(!silent&&v>game.appliedDeckPos){
    for(let k=game.appliedDeckPos;k<v;k++)logMove(k,st,rp);
    game.appliedDeckPos=v;
  }else if(v<game.appliedDeckPos){
    game.appliedDeckPos=v;
  }
  const finished=isOver();
  if(finished&&!game.over){
    game.over=true;
    const fs=finalScoring(st);
    game.scores=[game.scores[0]+fs.scores[0], game.scores[1]+fs.scores[1]];
    logFinalScoring(fs);
    game.winner=game.scores[0]>game.scores[1]?"host":game.scores[1]>game.scores[0]?"guest":"draw";
    game.reason="end";
    sfx(game.winner==="draw"?"draw2":(game.mode==="online"&&game.winner===game.myRole)?"win":"lose");
  }
  renderAll();
}
function logMove(k,st,rp){
  const role=turnRoleFor(k);
  const drawn=game.deck[k];
  const placedEntry=null;
  for(const t of st.placed){ if(t.dp===k){ /* found */ } }
  const entry=st.placed.filter(function(t){return t.dp===k;})[0]||null;
  const tileName=tileTypeName(drawn);
  if(!entry){
    addLog(role,"drew a "+tileName+" and discarded it — no legal placement.");
    return;
  }
  addLog(role,"placed a "+tileName+" at ("+entry.x+", "+entry.y+")");
  const evs=rp.byMove[k]||[];
  for(const e of evs)addLog(role,eventText(e));
  if(entry.m)addLog(role,"put a follower on a "+(entry.m.t===3?"field":FEAT_NAME[entry.m.t]));
}
function eventText(e){
  if(e.field)return "🌾 Field scored — "+formatWinners(e.winners)+" take "+e.points+" pts ("+e.fields+" city"+(e.fields>1?"s":"")+")";
  const nm=FEAT_NAME[e.type];
  return FEAT_ICON[e.type]+" "+nm.charAt(0).toUpperCase()+nm.slice(1)+" completed — "+formatWinners(e.winners)+" score "+e.points+" pts";
}
function formatWinners(ws){
  if(!ws.length)return "nobody";
  return ws.map(function(p){return roleName(p==="host"?"host":p===0?"host":"guest");}).join(" & ");
}
function tileTypeName(id){
  const e=TILES[id].e;
  if(e.indexOf("C")>=0){
    const c=(e.match(/C/g)||[]).length;
    if(c===4)return "all-city tile";
    if(c===3)return "three-sided city tile";
    if(c===2)return e[0]===e[2]?"two-city tile":"corner city tile";
    const hasRoad=e.indexOf("R")>=0;
    return hasRoad?"city and road tile":"city tile";
  }
  if(TILES[id].f.some(function(f){return f.t===2;}))return "monastery tile";
  const r=(e.match(/R/g)||[]).length;
  if(r===1)return "road tile";
  if(r===2)return e[0]===e[2]?"straight road tile":"curved road tile";
  if(r===3)return "road junction tile";
  return "crossroads tile";
}
function logFinalScoring(fs){
  addLog(game.winner==="draw"?"host":"guest","— End of game — final scoring:");
  for(const e of fs.events)addLog(game.winner==="draw"?"guest":"host",eventText(e));
  addLog("guest","Final: "+game.scores[0]+" – "+game.scores[1]+"  ("+roleName(game.winner)+" wins)");
}

/* ══════════ MEepLE CANDIDATES ══════════ */
function candidateFeatures(ts){
  const ti=ts.placed.length-1;
  const bd=buildBoard(ts.placed);
  const graph=buildGraph(ts,bd);
  const out=[];
  const t=ts.placed[ti];
  const td=TILES[t.id];
  for(let fi=0;fi<td.f.length;fi++){
    const feat=td.f[fi];
    const node=graph.nodes.filter(function(n){return n.ti===ti&&n.ft===feat.t&&n.fi===fi;})[0];
    if(!node)continue;
    const root=graph.find(node.idx);
    const comp=gather(graph,root,ts);
    if(comp.meeps.length>0)continue;
    if(comp.ft===0&&comp.open===0)continue;
    if(comp.ft===1&&comp.open===0)continue;
    if(comp.ft===2&&comp.neighbors===8)continue;
    out.push({ft:feat.t,fi:fi,comp:comp});
  }
  return out;
}
function cloneState(st){
  return {seed:st.seed,deckPos:st.deckPos,first:st.first,
    placed:st.placed.map(function(t){return {id:t.id,rot:t.rot,x:t.x,y:t.y,dp:t.dp,m:t.m?{p:t.m.p,t:t.m.t,f:t.m.f}:null};})};
}
function finalizeState(ts){
  const ti=ts.placed.length-1;
  scoreTile(ti,ts);
  ts.deckPos++;
  return ts;
}

/* ══════════ MOVE FLOW ══════════ */
function tryPlace(x,y){
  if(!canAct()||game.phase!=="place"||game.currentTile==null)return;
  const id=game.currentTile, rot=game.rotation;
  if(!isLegal(id,rot,x,y,game.board)){ flashBanner("That tile doesn't fit there — edges must match."); sfx("draw"); return; }
  const entry={id:id,rot:rot,x:x,y:y,dp:game.st.deckPos,m:null};
  const ts=cloneState(game.st);
  ts.placed.push(entry);
  if(game.mode==="online"){
    game.pending={entry:entry,ts:ts};
    game.phase="meeple";
    game.currentTile=null;
    game.board=buildBoard(ts.placed);
    sfx("place");
    renderBoard();
    renderMeepleCtn();
    return;
  }
  game.phase="meeple";
  game.currentTile=null;
  game.st=ts;
  game.board=buildBoard(ts.placed);
  game.lastPlaced={x:x,y:y};
  sfx("place");
  renderBoard();
  renderMeepleCtn();
}
function chooseMeeple(ft,fi){
  if(game.phase!=="meeple")return;
  let ts,entry;
  if(game.mode==="online"&&game.pending){
    ts=game.pending.ts;
    entry=game.pending.entry;
  }else{
    ts=game.st;
    entry=ts.placed[ts.placed.length-1];
  }
  if(meeplesLeft(playerIdx())>0){
    entry.m={p:playerIdx(),t:ft,f:fi};
  }
  finishTurn(ts);
}
function skipMeeple(){
  if(game.phase!=="meeple")return;
  const ts=game.mode==="online"&&game.pending?game.pending.ts:game.st;
  finishTurn(ts);
}
function playerIdx(){ return game.mode==="ai"?"host"===turnRoleFor(game.st.deckPos)?0:1:(turnRoleFor(game.st.deckPos)==="host"?0:1); }
function finishTurn(ts){
  finalizeState(ts);
  if(game.mode==="online"){
    game.locked=true;
    const evt=game.serverTurn;
    rpc("move",JSON.stringify({code:game.roomCode,evt:evt,data:encodeState(ts)})).then(function(r){
      game.locked=false;
      if(!r||!r.ok){ flashBanner("Couldn't send your move ("+(r&&r.err?r.err:"error")+")."); game.phase="place"; game.currentTile=game.deck[game.st.deckPos]; renderAll(); return; }
      game.phase="idle";
      game.pending=null;
      if(isOver()){ const w=ts.scores?winnerFrom(ts):null; rpc("endGame",JSON.stringify({code:game.roomCode,winner:winnerFrom(ts)})).catch(function(){}); }
      refreshFromServer();
    }).catch(function(){
      game.locked=false;
      flashBanner("Couldn't reach the server — reconnecting…");
      game.phase="place"; game.currentTile=game.deck[game.st.deckPos]; renderAll();
    });
    return;
  }
  game.phase="idle";
  game.pending=null;
  applyState(ts,{});
  saveGame();
  onTurnStart();
}
function winnerFrom(ts){
  const rp=replay(ts);
  return rp.scores[0]>rp.scores[1]?"host":rp.scores[1]>rp.scores[0]?"guest":"draw";
}

function adjCitiesCount(graph,node,st){
  const set=new Set();
  for(const pF of node.lp){
    const p1=pF===12?1:pF+1, p2=pF===1?12:pF-1;
    for(const pC of [p1,p2]){
      const other=graph.posMap[node.ti][pC];
      if(other===undefined)continue;
      if(graph.nodes[other].ft!==0)continue;
      set.add(graph.find(other));
    }
  }
  return set.size;
}
function evaluatePlacement(ts,entry,botP){
  const opp=1-botP;
  const hs=cloneState(ts);
  hs.placed.push(entry);
  const bd=buildBoard(hs.placed);
  const ti=hs.placed.length-1;
  const graph=buildGraph(hs,bd);
  const td=TILES[entry.id];
  let val=0;
  const events=scoreTile(ti,hs);
  for(const e of events){
    const pts=e.points;
    if(e.winners.indexOf(botP)>=0)val+=pts*1.7;
    else if(e.winners.indexOf(opp)>=0)val-=pts*1.7;
    else val+=pts*0.5;
  }
  const nodeFor=function(fi,ft){
    for(const n of graph.nodes){ if(n.ti===ti&&n.ft===ft&&n.fi===fi)return n; }
    return null;
  };
  for(let fi=0;fi<td.f.length;fi++){
    const feat=td.f[fi];
    const node=nodeFor(fi,feat.t);
    if(!node)continue;
    const comp=gather(graph,graph.find(node.idx),hs);
    const complete=comp.ft===0?comp.open===0:comp.ft===1?comp.open===0:(comp.ft===2&&comp.neighbors===8);
    if(complete)continue;
    let myC=0,opC=0;
    for(const mm of comp.meeps){ if(mm.p===botP)myC++; else if(mm.p===opp)opC++; }
    const size=comp.tiles.size;
    if(comp.ft===0){
      if(myC>0)val+=1.4*size+2;
      else if(opC>0)val-=0.7*size;
      else val+=0.5*size+1;
    }else if(comp.ft===1){
      if(myC>0)val+=0.7*size;
      else if(opC>0)val-=0.25*size;
      else val+=0.2*size;
    }else if(comp.ft===2){
      val+=2.6;
    }else{
      val+=0.15*adjCitiesCount(graph,node,hs);
    }
  }
  let mval=0;
  for(let fi=0;fi<td.f.length;fi++){
    const feat=td.f[fi];
    const node=nodeFor(fi,feat.t);
    if(!node)continue;
    const comp=gather(graph,graph.find(node.idx),hs);
    if(comp.meeps.length>0)continue;
    const complete=comp.ft===0?comp.open===0:comp.ft===1?comp.open===0:(comp.ft===2&&comp.neighbors===8);
    if(complete)continue;
    let v=0;
    if(comp.ft===0){
      const size=comp.tiles.size;
      let opC=0; for(const mm of comp.meeps){ if(mm.p===opp)opC++; }
      if(opC>0)v=0.9*size+1.5;
      else v=Math.min(4,0.55*size+2);
    }else if(comp.ft===1)v=1.1;
    else if(comp.ft===2)v=2.4;
    else v=0.45*adjCitiesCount(graph,node,hs);
    if(v>mval)mval=v;
  }
  if(mval>0)val+=Math.min(mval,4)*0.9;
  return val;
}
function aiTurnSoon(){
  if(game.aiTimer)clearTimeout(game.aiTimer);
  game.aiTimer=setTimeout(function(){
    game.aiTimer=setTimeout(function(){ aiDoTurn(); },game.fastPlay?80+Math.random()*120:480+Math.random()*650);
  },game.fastPlay?100:520);
}
function aiDoTurn(){
  game.aiTimer=null;
  if(game.mode!=="ai"||game.over||!game.st)return;
  const ts=cloneState(game.st);
  const id=game.deck[ts.deckPos];
  const bd0=buildBoard(ts.placed);
  const placements=[];
  for(const ckey of adjacentCells()){
    const xy=ckey.split(",").map(Number);
    for(let r=0;r<4;r++){
      if(isLegal(id,r,xy[0],xy[1],bd0))placements.push({x:xy[0],y:xy[1],rot:r});
    }
  }
  if(!placements.length){
    ts.deckPos++;
    applyState(ts,{});
    saveGame();
    onTurnStart();
    return;
  }
  const botP=1;
  let best=null,bestV=-1e9;
  for(const p of placements){
    const v=evaluatePlacement(ts,{id:id,rot:p.rot,x:p.x,y:p.y,dp:ts.deckPos,m:null},botP)+Math.random()*0.4;
    if(best===null||v>bestV){ bestV=v; best=p; }
  }
  const entry={id:id,rot:best.rot,x:best.x,y:best.y,dp:ts.deckPos,m:null};
  ts.placed.push(entry);
  const cands=candidateFeatures(ts);
  const free=meeplesLeft(botP);
  const deckLeft=game.deck.length-ts.deckPos;
  const endIsNear=deckLeft/2<=free;
  let bestM=null,bestMv=0;
  for(const c of cands){
    const ft=c.ft;
    if(ft===3){
      if(free<=1&&!endIsNear)continue;
      if(free<=3&&!endIsNear&&Math.random()<0.7)continue;
      if(Math.random()<0.55)continue;
    }
    let v=ft===0?2.2+c.comp.tiles.size:ft===1?1.3:ft===2?2.6:0.4;
    if(ft===0)v-=0.08*c.comp.open;
    else if(ft===1)v-=0.25*c.comp.open;
    if(ft!==2&&free<=2&&!endIsNear)v-=1.2;
    if(v>bestMv){ bestMv=v; bestM=c; }
  }
  if(free>0&&bestM&&bestMv>0)entry.m={p:botP,t:bestM.ft,f:bestM.fi};
  aiPlayAnim(ts,entry);
}
function aiPlayAnim(ts,entry){
  if(game.over||game.mode!=="ai"||!game.st){
    finalizeState(ts);
    applyState(ts,{}); saveGame(); onTurnStart(); return;
  }
  if(game.fastPlay){
    game.phase="idle";
    sfx("turn");
    finalizeState(ts);
    applyState(ts,{});
    saveGame();
    onTurnStart();
    return;
  }
  const meeple=entry.m?{p:entry.m.p,t:entry.m.t,f:entry.m.f}:null;
  game.aiAnim={ts:ts,entry:entry,meeple:meeple,tileT:0,meepleT:null,start:performance.now()};
  game.phase="aianim";
  sfx("turn");
  renderAll();
  const TILE_MS=430, MEEPLE_MS=340;
  function commit(){
    game.aiAnim=null;
    game.phase="idle";
    finalizeState(ts);
    applyState(ts,{});
    saveGame();
    onTurnStart();
  }
  function step(now){
    const a=game.aiAnim;
    if(!a){ return; }
    if(game.over||!game.st){ commit(); return; }
    const et=now-a.start;
    if(et<TILE_MS){
      a.tileT=easeOutCubic(Math.max(0,Math.min(1,et/TILE_MS)));
      renderBoard();
      requestAnimationFrame(step);
      return;
    }
    if(a.meeple&&et<TILE_MS+MEEPLE_MS){
      if(a.meepleT==null)updateBanner();
      a.tileT=1;
      a.meepleT=easeOutBack(Math.max(0,Math.min(1,(et-TILE_MS)/MEEPLE_MS)));
      renderBoard();
      requestAnimationFrame(step);
      return;
    }
    commit();
  }
  requestAnimationFrame(step);
}
function easeOutCubic(t){ return 1-Math.pow(1-t,3); }
function easeOutBack(t){ const c1=1.70158,c3=c1+1; return 1+c3*Math.pow(t-1,3)+c1*Math.pow(t-1,2); }

function rotp(u,v,k){ for(let i=0;i<k;i++){ const nu=1-v, nv=u; u=nu; v=nv; } return [u,v]; }
function drawUnitPath(ctx,S,cmds,close){
  ctx.beginPath();
  for(const c of cmds){
    if(c[0]==="M")ctx.moveTo(c[1]*S,c[2]*S);
    else if(c[0]==="L")ctx.lineTo(c[1]*S,c[2]*S);
    else if(c[0]==="Q")ctx.quadraticCurveTo(c[1]*S,c[2]*S,c[3]*S,c[4]*S);
    else if(c[0]==="A")ctx.arc(c[1]*S,c[2]*S,c[3]*S,c[4],c[5]);
  }
  if(close)ctx.closePath();
}

function rotCmds(cmds,k){
  const out=[];
  for(const c of cmds){
    const o=[c[0]];
    for(let i=1;i<c.length;i+=2){ const r=rotp(c[i],c[i+1],k); o.push(r[0],r[1]); }
    out.push(o);
  }
  return out;
}

function cityGeom(crossings){
  const ns=crossings.slice().sort(function(a,b){return a-b;});
  if(ns.length===1){
    const k=(4-ns[0])%4;
    return {fills:[rotCmds(bandFill(0),k)],walls:[rotCmds(wallCmds(0),k)],meeple:rotPts([0.5,0.16],k),pennant:rotPts([0.5,0.24],k)};
  }
  if(ns.length===2){
    if(ns[1]-ns[0]===1||(ns[0]===0&&ns[1]===3)){
      const k=(ns[0]===0&&ns[1]===3)?3:ns[0];
      const fill=[["M",0,GD],["L",0.58,GD],["Q",0.64,GD,0.64,GD+0.06],["L",0.64,1],["L",1,1],["L",1,0],["L",0,0]];
      const wall=[["M",0,GD],["L",0.58,GD],["Q",0.64,GD,0.64,GD+0.06],["L",0.64,1]];
      return {fills:[rotCmds(fill,k)],walls:[rotCmds(wall,k)],meeple:rotPts([0.72,0.26],k),pennant:rotPts([0.8,0.18],k)};
    }
    const k=ns[0];
    return {fills:[rotCmds(bandFill(0),k),rotCmds(bandFill(2),k)],walls:[rotCmds(wallCmds(0),k),rotCmds(wallCmds(2),k)],meeple:rotPts([0.5,0.16],k),pennant:rotPts([0.5,0.24],k)};
  }
  if(ns.length===3){
    const missing=[0,1,2,3].find(function(s){return ns.indexOf(s)<0;});
    const k=(4-missing)%4;
    return {fills:[rotCmds(bandFill(0),k),rotCmds(bandFill(1),k),rotCmds(bandFill(3),k)],walls:[rotCmds(wallCmds(0),k),rotCmds(wallCmds(1),k),rotCmds(wallCmds(3),k)],meeple:rotPts([0.5,0.52],k),pennant:rotPts([0.34,0.5],k)};

}
function edgeMid(e){ if(e===2)return [0.5,0]; if(e===5)return [1,0.5]; if(e===8)return [0.5,1]; if(e===11)return [0,0.5]; return [0.5,0.5]; }
function roadPath(ep){
  if(ep.indexOf(0)>=0){
    const arms=[];
    for(const e of ep){ if(e===0)continue; arms.push(edgeMid(e)); }
    return {arms:arms};
  }
  const a=ep[0],b=ep[1];
  const ea=edgeMid(a),eb=edgeMid(b);
  if((a===2&&b===8)||(a===5&&b===11))return {seg:[ea,eb]};
  const pull=0.55;
  const inw=function(p){ const dx=0.5-p[0], dy=0.5-p[1]; const L=Math.hypot(dx,dy)||1e-6; return [dx/L*pull,dy/L*pull]; };
  const va=inw(ea), vb=inw(eb);
  return {curve:[ea,[ea[0]+va[0],ea[1]+va[1]],[eb[0]+vb[0],eb[1]+vb[1]],eb]};
}
function roadMid(ep){
  const p=roadPath(ep);
  if(p.curve){ const c=p.curve, t=0.5, u=1-t;
    return [u*u*u*c[0][0]+3*u*u*t*c[1][0]+3*u*t*t*c[2][0]+t*t*t*c[3][0],
            u*u*u*c[0][1]+3*u*u*t*c[1][1]+3*u*t*t*c[2][1]+t*t*t*c[3][1]]; }
  if(p.arc){ const a=(p.arc.a0+p.arc.a1)/2; return [0.5+0.5*Math.cos(a),0.5+0.5*Math.sin(a)]; }
  if(p.seg){ return [(p.seg[0][0]+p.seg[1][0])/2,(p.seg[0][1]+p.seg[1][1])/2]; }
  const e=p.arms[0];
  return [(0.5+e[0])/2,(0.5+e[1])/2];
}

  else {
    for(const arm of path.arms){
      ctx.moveTo(0.5*S,0.5*S); ctx.lineTo(arm[0]*S,arm[1]*S);
    }
  }
  ctx.stroke();
}

  }
  const pIdx=playerIdx();

  if(ghost){

    if(ft===0){
      const cr=f.p.filter(function(p){return CROSS_IDX[p]!==undefined;}).map(function(p){return CROSS_IDX[p];});
      pt=cityGeom(cr).meeple; gs=Math.max(8,S*0.42);

        const arm=path.arms[0];
        pt=[0.5+(arm[0]-0.5)*0.62,0.5+(arm[1]-0.5)*0.62]; gs=Math.max(7,S*0.36);

    ctx.globalAlpha=0.7;
    ctx.drawImage(ghost,pt[0]*S-gs/2,pt[1]*S-gs/2,gs,gs);
  }
  ctx.restore();
}
function drawCityHouses(ctx,td,fi,S,g){
  let s=(Math.imul(td._v^fi+7919,73856093)^Math.imul(fi+1,19349663))>>>0;
  const rnd=function(){ s=(Math.imul(s,1664525)+1013904223)>>>0; return s/4294967296; };
  const n=g.courtyard?12:7;
  const roofCols=["#b85c3a","#a04a2e","#c26b46","#8f5f3c","#9a5a44"];
  const wallCols=["#d9c08d","#cdae75","#e0c99a","#c4a56c"];
  for(let i=0;i<n;i++){
    const x=0.16+rnd()*0.68, y=0.16+rnd()*0.68;
    if(g.courtyard){

    ctx.strokeRect(-W/2,-H/2,W,H);
    ctx.fillStyle=roofCols[i%roofCols.length];

    ctx.lineTo(W/2+1,-H/2);

  }
}
function drawCren(ctx,S,side){
  ctx.fillStyle=COL.cityWall;
  const w=0.085,gap=0.05,h=0.05;
  let x=0.015;
  while(x+w<=1.0){
    let px,py;
    if(side===0){ px=x; py=0; }
    else if(side===1){ px=1-w; py=x; }
    else if(side===2){ px=x; py=1-h; }
    else { px=0; py=x; }
    ctx.fillRect(px*S,py*S,w*S,h*S);
    x+=w+gap;
  }
}

function boardTransform(cssW,cssH){
  let minX=0,maxX=0,minY=0,maxY=0;
  if(game.st&&game.st.placed.length){
    for(const t of game.st.placed){ if(t.x<minX)minX=t.x; if(t.x>maxX)maxX=t.x; if(t.y<minY)minY=t.y; if(t.y>maxY)maxY=t.y; }
  }
  const pad=2;
  const cols=(maxX-minX)+2*pad, rows=(maxY-minY)+2*pad;
  const S=Math.max(24,Math.floor(Math.min(cssW/cols,cssH/rows)));
  const ox=(cssW-cols*S)/2+(pad-minX)*S;
  const oy=(cssH-rows*S)/2+(pad-minY)*S;
  return {S:S,ox:ox,oy:oy};
}
function cellRect(tr,x,y){ return {x:tr.ox+x*tr.S,y:tr.oy+y*tr.S,w:tr.S,h:tr.S}; }
function drawCellOverlay(ctx,tr,x,y,fill,border){
  const r=cellRect(tr,x,y);
  ctx.fillStyle=fill;
  ctx.fillRect(r.x+1,r.y+1,r.w-2,r.h-2);
  if(border!=="transparent"){
    ctx.strokeStyle=border; ctx.lineWidth=2;
    ctx.strokeRect(r.x+1.5,r.y+1.5,r.w-3,r.h-3);
  }
}
function renderBoard(){
  const ctx=boardCanvas.getContext("2d");
  const dpr=boardCanvas._dpr||1;
  const cssW=boardCanvas.width/dpr, cssH=boardCanvas.height/dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.clearRect(0,0,cssW,cssH);
  ctx.fillStyle=COL.boardBg;
  ctx.fillRect(0,0,cssW,cssH);
  if(!game.st){ renderBoardWaiting(ctx,cssW,cssH); return; }
  const tr=boardTransform(cssW,cssH);
  game._tr=tr;
  for(const t of game.st.placed)drawTile(ctx,t,tr.S,tr.ox,tr.oy);
  if(game.pending)drawTile(ctx,game.pending.entry,tr.S,tr.ox,tr.oy,0.8);
  if(game.phase==="meeple"&&game.hoverMeeple){
    const mpT=(game.mode==="online"&&game.pending)?game.pending.entry:(game.st.placed[game.st.placed.length-1]);
    if(mpT&&mpT.id!=null){
      const td=TILES[mpT.id];
      const f=td.f[game.hoverMeeple.fi];
      if(f&&f.t===game.hoverMeeple.ft){
        ctx.save();
        ctx.translate(tr.ox+(mpT.x+0.5)*tr.S,tr.oy+(mpT.y+0.5)*tr.S);
        ctx.rotate(mpT.rot*Math.PI/2);
        ctx.translate(-tr.S/2,-tr.S/2);
        drawFeatureHighlight(ctx,td,game.hoverMeeple.ft,game.hoverMeeple.fi,tr.S);
        ctx.restore();
      }
    }
  }
  if(game.aiAnim){
    const a=game.aiAnim;
    const e=a.entry;
    const cx=tr.ox+(e.x+0.5)*tr.S, cy=tr.oy+(e.y+0.5)*tr.S;
    if(a.meepleT==null){
      const sc=0.35+0.65*a.tileT;
      ctx.save();
      ctx.translate(cx,cy); ctx.scale(sc,sc); ctx.translate(-cx,-cy);
      drawTile(ctx,e,tr.S,tr.ox,tr.oy,0.45+0.55*a.tileT);
      ctx.restore();
    }else if(a.meeple){
      const sc2=easeOutBack(a.meepleT);
      ctx.save();
      ctx.translate(cx,cy); ctx.rotate(e.rot*Math.PI/2);
      ctx.scale(sc2,sc2);
      ctx.translate(-tr.S/2,-tr.S/2);
      drawTileBase(ctx,TILES[e.id],tr.S);
      drawMeeple(ctx,TILES[e.id],a.meeple,tr.S);
      ctx.restore();
    }else{
      drawTile(ctx,e,tr.S,tr.ox,tr.oy,1);
    }
  }
  if(game.phase==="place"&&canAct()&&game.currentTile!=null){
    const legalSet=new Set();
    for(const p of legalCells(game.currentTile,game.rotation,game.board))legalSet.add(key(p[0],p[1]));
    for(const ckey of adjacentCells()){
      const xy=ckey.split(",").map(Number);
      if(legalSet.has(ckey))drawCellOverlay(ctx,tr,xy[0],xy[1],COL.legal,COL.legalBorder);
      else if(legalRotations(game.currentTile,xy[0],xy[1],game.board).length===0)drawCellOverlay(ctx,tr,xy[0],xy[1],COL.illegal,"transparent");
    }
  }
  if(game.hoverCell&&game.phase==="place"&&canAct()){
    const hx=game.hoverCell[0],hy=game.hoverCell[1];
    const legal=game.currentTile!=null&&isLegal(game.currentTile,game.rotation,hx,hy,game.board);
    drawCellOverlay(ctx,tr,hx,hy,"transparent",legal?COL.hover:"rgba(255,90,90,.8)");
  }
  if(game.lastPlaced&&!game.over){
    const r=cellRect(tr,game.lastPlaced.x,game.lastPlaced.y);
    ctx.strokeStyle=COL.last; ctx.lineWidth=2.5; ctx.setLineDash([5,4]);
    ctx.strokeRect(r.x+1,r.y+1,r.w-2,r.h-2);
    ctx.setLineDash([]);
  }
}

/* ══════════ ONLINE CLIENT (template plumbing) ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }
  catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true; game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){ if(game.mode==="online")onServerMsg(ev.data); });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null; connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — share the code with a friend to start the match":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="chat")addChat(msg.role,msg.text,msg.ts);
}
function refreshFromServer(){
  return rpc("getRoom",game.roomCode).then(function(r){
    if(r&&r.snap)applyServerSnap(r.snap);
  }).catch(function(){});
}
function applyServerSnap(s){
  game.serverTurn=s.turn;
  game.first=s.first;
  game.status=s.status;
  game.oppPresent=game.myRole==="host"?s.guestIn:s.hostIn;
  if(s.status===3){
    game.over=true;
    game.winner=s.winner;
    game.reason=s.reason||"";
    if(game.winner==="draw")sfx("draw2");
    else if(game.winner===game.myRole)sfx("win");
    else sfx("lose");
  }else{
    game.over=false; game.winner=null; game.reason="";
  }
  if(s.snapshot){
    const st=decodeState(s.snapshot);
    if(st){
      const verr=validateState(st,game.deck);
      if(verr!=="ok"){ flashBanner("⚠ Unexpected state from server ("+verr+")."); }
      else applyState(st,{});
    }else{
      flashBanner("⚠ Couldn't parse the game state.");
    }
  }else if(s.status===2){
    initFreshGame();
  }
  renderAll();
  if(!game.over&&game.mode==="online")onTurnStart();
}
function enterOnlineGame(){
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  gameHero.hidden=true;
  roomBox.hidden=false;
  chatBox.hidden=false;
  roomCodeEl.textContent=game.roomCode;
  updateNames(); renderAll();
}
async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    resetTable();
    game.mode="online"; game.roomCode=r.code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
    sfx("deal");
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    resetTable();
    game.mode="online"; game.roomCode=code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
    sfx("deal");
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}

/* ══════════ SAVE / CONTINUE (local & AI) ══════════ */
const SAVE_KEY="bgn_save_carcassonne";
function saveGame(){
  if(game.mode!=="local"&&game.mode!=="ai")return;
  if(game.over){ deleteSave(); return; }
  try{
    localStorage.setItem(SAVE_KEY, JSON.stringify({
      v:2, mode:game.mode, st:game.st, seed:game.seed, appliedDeckPos:game.appliedDeckPos,
      first:game.first, tileInHand:(game.phase==="place"&&game.currentTile!=null), rotation:game.rotation,
      savedAt:Date.now()
    }));
  }catch(e){}
}
function hasSave(){ try{ return !!localStorage.getItem(SAVE_KEY); }catch(e){ return false; } }
function deleteSave(){ try{ localStorage.removeItem(SAVE_KEY); }catch(e){} }
function refreshContinuePanel(){
  if(!continuePanel)return;
  const s=hasSave();
  continuePanel.hidden=!s;
  if(!s)return;
  let info="A saved table was found.";
  try{ const o=JSON.parse(localStorage.getItem(SAVE_KEY)); if(o&&o.savedAt)info="Saved "+new Date(o.savedAt).toLocaleTimeString()+" — turn "+(o.st?(o.st.deckPos+1):1)+"."; }catch(e){}
  continueInfoEl.textContent=info;
}
function continueGame(){
  let o; try{ o=JSON.parse(localStorage.getItem(SAVE_KEY)); }catch(e){ o=null; }
  if(!o||!o.st||!Array.isArray(o.st.placed)){ flashBanner("No saved table found."); return; }
  if(game.mode==="online")exitOnline(false);
  resetTable();
  game.mode=o.mode==="ai"?"ai":"local";
  game.seed=o.seed!=null?o.seed:o.st.seed;
  game.deck=buildDeck(game.seed);
  game.appliedDeckPos=o.appliedDeckPos||0;
  game.st=o.st;
  game.first=o.st.first||0;
  updateNames();
  roomBox.hidden=true; chatBox.hidden=true; claimBtn.hidden=true;
  fastPlayToggle.hidden=game.mode!=="ai";
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  gameHero.hidden=true;
  sfx("deal");
  applyState(game.st,{silent:true});
  if(o.tileInHand&&!game.over){
    game.currentTile=game.deck[game.st.deckPos];
    game.rotation=o.rotation||0;
    game.phase="place";
  }else{
    onTurnStart();
  }
  renderAll();
}
function restoreFromObject(o){
  resetTable();
  game.mode=o.mode==="ai"?"ai":"local";
  game.seed=o.seed!=null?o.seed:o.st.seed;
  game.deck=buildDeck(game.seed);
  game.appliedDeckPos=o.appliedDeckPos||0;
  game.st=o.st;
  game.first=o.st.first||0;
  updateNames();
  roomBox.hidden=true; chatBox.hidden=true; claimBtn.hidden=true;
  fastPlayToggle.hidden=game.mode!=="ai";
  lobbyCtn.hidden=true; gameCtn.hidden=false;
  gameHero.hidden=true;
  sfx("deal");
  applyState(game.st,{silent:true});
  if(o.tileInHand&&!game.over){
    game.currentTile=game.deck[game.st.deckPos];
    game.rotation=o.rotation||0;
    game.phase="place";
  }else{
    onTurnStart();
  }
  renderAll();
}

/* ══════════ BOARD INPUT ══════════ */
function dprOf(){ return boardCanvas._dpr||1; }
function cssPoint(e){
  const r=boardCanvas.getBoundingClientRect();
  const sx=boardCanvas.width/(Math.max(1,r.width)*dprOf());
  const sy=boardCanvas.height/(Math.max(1,r.height)*dprOf());
  return { x:(e.clientX-r.left)*sx, y:(e.clientY-r.top)*sy };
}
function onBoardClick(e){
  if(!game._tr||!game.st)return;
  if(!canAct()||game.phase!=="place")return;
  const p=cssPoint(e);
  const x=Math.floor((p.x-game._tr.ox)/game._tr.S);
  const y=Math.floor((p.y-game._tr.oy)/game._tr.S);
  tryPlace(x,y);
}
function onBoardMove(e){
  if(!game._tr||!game.st)return;
  const p=cssPoint(e);
  const x=Math.floor((p.x-game._tr.ox)/game._tr.S);
  const y=Math.floor((p.y-game._tr.oy)/game._tr.S);
  game.hoverCell=[x,y];
  renderBoard();
}
function resizeBoard(){
  const wrap=boardCanvas.parentElement;
  const w=wrap.clientWidth||640;
  const h=wrap.clientHeight||Math.max(320,Math.min(560,Math.round(w*0.8)));
  const dpr=window.devicePixelRatio||1;
  const nw=Math.round(w*dpr), nh=Math.round(h*dpr);
  if(nw===boardCanvas.width&&nh===boardCanvas.height)return;
  boardCanvas.width=nw;
  boardCanvas.height=nh;
  boardCanvas._dpr=dpr;
  renderBoard();
}
function openFullscreen(){
  if(!fsOverlay.hidden)return;
  fsOverlay.hidden=false;
  fsStatusEl.textContent=bannerEl.textContent||"Full screen map";
  fsActionsCtn.appendChild(rotateBtn);

  document.body.style.overflow="hidden";
  resizeBoard();
  renderAll();
  try{ if(fsOverlay.requestFullscreen)fsOverlay.requestFullscreen().catch(function(){}); }catch(e){}
}
function closeFullscreen(){

  document.querySelector(".carc-tilebox").appendChild(rotateBtn);

  resizeBoard();
  renderAll();
}
