// ============================================================================
//  github-data-plugin  —  read from & push to public GitHub repositories
//  by Rathji
//
//  READ:
//    gh = \{import:github-data-plugin\}
//    let text = await root.gh("owner/repo", "path/file.txt");       // raw file text
//    let info = await root.gh("octocat/Hello-World");               // repo info
//    let json = await root.gh("owner/repo", "data.json", {format:"json"});
//    let list = await root.gh("owner/repo", "src", {format:"list"});
//    let tree = await root.gh("owner/repo", null, {format:"tree"});
//
//  PUSH (write access + token required):
//    root.gh.setToken("github_pat_...");                            // once per browser
//    let r = await root.gh.push("owner/repo", {
//      "main.pjs": "...", "index.html": "..."                       // object: path -> content
//    });
//    let r2 = await root.gh.push("owner/repo", [{path, content}, ...], {message, branch});
//    await root.gh.pushFile("owner/repo", "path.txt", "contents");
//
//  BACKUP THIS GENERATOR:
//    let r = await root.gh.backupGenerator("owner/repo");           // pushes main.pjs + index.html
//    // (optional) {generatorName: "my-gen"} to back up another generator
//
//  The push result: { ok:true, branch, commitSha, files:[...], message, url }
//  (push/pushFile/backupGenerator THROW on failure — wrap in try/catch.)
//
//  RESULT SHAPE (reads) — the awaited result is a String object:
//    String(result)  — file text or pretty-printed JSON
//    result.text / .data / .json / .error / .ok / .repo / .path / .branch
//
//  RENDER INTO AN ELEMENT:
//    root.gh.into("elId", "owner/repo", "path/file.txt");
//
//  OPTIONS:  { path, branch, format }   format: "text" | "json" | "info" | "list" | "tree"
//
//  TOKENS & SECURITY (IMPORTANT)
//    * Pushing requires a GitHub Personal Access Token. Public repos still need
//      write access. Create a fine-grained token scoped to ONLY the target repo
//      with "Contents: Read and write", and set an expiry.
//    * Your token is stored ONLY in your own browser's localStorage (this origin),
//      and sent ONLY to api.github.com. It is never hardcoded, logged, or sent
//      anywhere else. Never put a token in code, prompts, or public files.
//    * Anyone who obtains your token can write to your repos — treat it like a password.
// ============================================================================

  if(window.__ghDataPlugin7483621) return window.__ghDataPlugin7483621;

  const RAW_BASE = "https://raw.githubusercontent.com";
  const API_BASE = "https://api.github.com/repos";

  let _token = null;
  try { _token = localStorage.getItem("github-data-plugin-token") || null; } catch(e) {}

  function authHeaders() {
    let h = { Accept: "application/vnd.github+json", "X-GitHub-Api-Version": "2022-11-28" };
    if(_token) h.Authorization = "Bearer " + _token;
    return h;
  }

  async function ghFetch(url, init) {
    let res = await fetch(url, Object.assign({ headers: authHeaders() }, init));
    if(!res.ok) {
      let detail = "";
      try {
        let body = await res.clone().json();
        if(body && body.message) detail = " — " + body.message;
      } catch(e) {}
      throw new Error("HTTP " + res.status + " " + res.statusText + detail);
    }
    return res;
  }

  async function apiJson(url, init) {
    let res = await ghFetch(url, init);
    return await res.json();
  }

  function parseRepo(repo) {
    repo = String(repo || "").trim()
      .replace(/^https?:\/\/github\.com\//, "")
      .replace(/\.git$/, "")
      .replace(/\/+$/, "");
    let parts = repo.split("/").filter(Boolean);
    if(parts.length !== 2 || !/^[a-zA-Z0-9_.-]+$/.test(parts[0]) || !/^[a-zA-Z0-9_.-]+$/.test(parts[1])) {
      throw new Error("Invalid repo '" + repo + "'. Expected format: owner/repo");
    }
    return parts[0] + "/" + parts[1];
  }

  function normalizePath(path) {
    return String(path || "").replace(/^\/+|\/+$/g, "");
  }

  function normBranch(branch) {
    branch = String(branch || "").trim();
    return branch ? branch : null;
  }

  async function rawFetch(repo, path, branch) {
    path = encodeURI(path);
    let candidates = [];
    let b0 = normBranch(branch);
    if(b0) candidates.push(b0);
    candidates.push("main", "master", "HEAD");
    let tried = [];
    let lastErr = null;
    for(let b of [...new Set(candidates)]) {
      let res = null;
      try { res = await fetch(RAW_BASE + "/" + repo + "/" + b + "/" + path); }
      catch(e) { lastErr = e; }
      if(res && res.ok) return { text: await res.text(), branch: b };
      tried.push(b);
    }
    if(lastErr) throw lastErr;
    throw new Error("404 — " + repo + "/" + path + " not found (tried branches: " + tried.join(", ") + ")");
  }

  async function apiGet(url) {
    return await apiJson(url);
  }

  async function apiWithBranch(urlBuilder, branch) {
    let candidates = [];
    let b0 = normBranch(branch);
    if(b0) candidates.push(b0);
    candidates.push("main", "master");
    let tried = [];
    for(let b of [...new Set(candidates)]) {
      try {
        let data = await apiGet(urlBuilder(b));
        return { data, branch: b };
      } catch(e) {
        if(!String(e.message).startsWith("HTTP 404")) throw e;
        tried.push(b);
      }
    }
    throw new Error("404 — branch not found (tried: " + tried.join(", ") + ")");
  }

  async function fetchData(opts) {
    let repo = parseRepo(opts.repo);
    let path = normalizePath(opts.path);
    let format = String(opts.format || "").toLowerCase();
    if(format === "raw" || format === "file" || format === "text") format = "text";
    if(format !== "json" && format !== "info" && format !== "list" && format !== "tree") format = "text";
    if(format === "text" && !path) format = "info";

    if(format === "info") {
      let data = await apiGet(API_BASE + "/" + repo);
      return { data, json: data, format, repo, path, branch: data.default_branch || null, text: JSON.stringify(data, null, 2) };
    }

    if(format === "list") {
      let encPath = encodeURI(path);
      let { data, branch } = await apiWithBranch(b => API_BASE + "/" + repo + "/contents/" + encPath + "?ref=" + encodeURIComponent(b), opts.branch);
      let arr = Array.isArray(data) ? data : [data];
      return { data: arr, json: arr, format, repo, path, branch, text: JSON.stringify(arr, null, 2) };
    }

    if(format === "tree") {
      let { data, branch } = await apiWithBranch(b => API_BASE + "/" + repo + "/git/trees/" + encodeURIComponent(b) + "?recursive=1", opts.branch);
      let paths = (data.tree || []).map(t => t.path);
      return { data: paths, json: paths, format, repo, path, branch, text: JSON.stringify(paths, null, 2) };
    }

    // text or json — fetch the raw file
    let { text, branch } = await rawFetch(repo, path, opts.branch);
    if(format === "json") {
      let parsed;
      try { parsed = JSON.parse(text); }
      catch(e) { throw new Error("File " + repo + "/" + path + " is not valid JSON"); }
      return { data: parsed, json: parsed, format, repo, path, branch, text: JSON.stringify(parsed, null, 2) };
    }
    return { data: text, json: null, format, repo, path, branch, text };
  }

  function makeResult(res) {
    let out = new String(res.text);
    out.text = res.text;
    out.data = res.data;
    out.json = res.json;
    out.format = res.format;
    out.repo = res.repo;
    out.path = res.path;
    out.branch = res.branch || null;
    out.error = null;
    out.ok = true;
    return out;
  }

  function makeError(err, opts) {
    let msg = (err && err.message) || String(err);
    let out = new String("(github-data-plugin error: " + msg + ")");
    out.text = String(out);
    out.data = null;
    out.json = null;
    out.error = msg;
    out.ok = false;
    out.format = opts ? opts.format : null;
    out.repo = opts ? opts.repo : null;
    out.path = opts ? opts.path : null;
    return out;
  }

  function main(repoOrOpts, pathOrOpts, maybeOpts) {
    let opts = {};
    if(repoOrOpts && typeof repoOrOpts === "object") {
      opts = Object.assign({}, repoOrOpts);
    } else {
      opts.repo = repoOrOpts;
      if(pathOrOpts && typeof pathOrOpts === "object") opts = Object.assign(opts, pathOrOpts);
      else opts.path = pathOrOpts;
      opts = Object.assign(opts, maybeOpts || {});
    }
    return fetchData(opts).then(makeResult).catch(err => makeError(err, opts));
  }

  // ---------- read helpers (throw on failure) ----------
  main.raw = async (repo, path, opts) => {
    let o = Object.assign({}, opts || {}, { repo, path, format: "text" });
    return (await fetchData(o)).text;
  };
  main.json = async (repo, path, opts) => {
    let o = Object.assign({}, opts || {}, { repo, path, format: "json" });
    return (await fetchData(o)).data;
  };
  main.info = async (repo, opts) => {
    let o = Object.assign({}, opts || {}, { repo, format: "info" });
    return (await fetchData(o)).data;
  };
  main.list = async (repo, path, opts) => {
    let o = Object.assign({}, opts || {}, { repo, path, format: "list" });
    return (await fetchData(o)).data;
  };
  main.tree = async (repo, branch, opts) => {
    let o = Object.assign({}, opts || {}, { repo, branch, format: "tree" });
    return (await fetchData(o)).data;
  };
  main.fetch = (url, init) => ghFetch(url, init);

  // render the result into a DOM element (by id or element) — no await needed
  main.into = (elOrId, repoOrOpts, pathOrOpts, maybeOpts) => {
    let el = typeof elOrId === "string" ? document.getElementById(elOrId) : elOrId;
    if(!el) return Promise.reject(new Error("Element not found: " + elOrId));
    let o = {};
    if(repoOrOpts && typeof repoOrOpts === "object") o = Object.assign({}, repoOrOpts);
    else {
      o.repo = repoOrOpts;
      if(pathOrOpts && typeof pathOrOpts === "object") o = Object.assign(o, pathOrOpts);
      else o.path = pathOrOpts;
      o = Object.assign(o, maybeOpts || {});
    }
    el.textContent = "Loading GitHub data…";
    return main(o).then(r => {
      if(!r.ok) {
        el.textContent = r.toString();
        return r;
      }
      if(typeof r.data === "string") {
        el.textContent = r.data;
      } else {
        let pre = document.createElement("pre");
        pre.style.margin = "0";
        pre.style.overflow = "auto";
        pre.style.maxHeight = "400px";
        pre.textContent = JSON.stringify(r.data, null, 2);
        el.textContent = "";
        el.append(pre);
      }
      return r;
    });
  };

  main.setToken = function(t) {
    _token = (typeof t === "string" && t.trim()) ? t.trim() : null;
    try {
      if(_token) localStorage.setItem("github-data-plugin-token", _token);
      else localStorage.removeItem("github-data-plugin-token");
    } catch(e) {}
    return _token;

  main.clearToken = function() { return main.setToken(null); };

  // ---------- push (Git Data API — one commit per push) ----------
  async function getBranchHead(repo, branch) {
    // plain fetch (not ghFetch) so we can inspect a 404 as "branch doesn't exist"
    let res = await fetch(API_BASE + "/" + repo + "/git/ref/heads/" + encodeURIComponent(branch), { headers: authHeaders() });
    if(res.status === 404) return null;
    if(!res.ok) {
      let detail = "";
      try { let b = await res.clone().json(); if(b && b.message) detail = " — " + b.message; } catch(e) {}
      throw new Error("HTTP " + res.status + " " + res.statusText + detail);
    }
    let ref = await res.json();
    let commit = await apiJson(ref.object.url);
    return { headSha: ref.object.sha, treeSha: commit.tree.sha };
  }

  async function pushCore(repo, files, opts) {
    if(!_token) throw new Error("No GitHub token set. Use gh.setToken(token) or pass {token: '...'}.");

    let fileList = [];
    if(Array.isArray(files)) fileList = files;
    else if(files && typeof files === "object") fileList = Object.keys(files).map(p => ({ path: p, content: String(files[p]) }));
    if(!fileList.length) throw new Error("No files to push.");
    fileList = fileList.map(f => {
      let path = String(f.path || "").replace(/^\/+/, "");
      if(!path) throw new Error("File path cannot be empty.");
      return { path, content: String(f.content === undefined ? "" : f.content) };
    });

    let branches = [];
    let b0 = normBranch(opts.branch);
    if(b0) branches.push(b0);
    branches.push("main");
    if(b0 !== "master") branches.push("master");

    let head = null;
    let usedBranch = null;
    for(let b of [...new Set(branches)]) {
      head = await getBranchHead(repo, b);
      if(head) { usedBranch = b; break; }
    }
    let branch = head ? usedBranch : (normBranch(opts.branch) || "main");

    let treeEntries = [];
    for(let f of fileList) {
      let blob = await apiJson(API_BASE + "/" + repo + "/git/blobs", {
        method: "POST",
        body: JSON.stringify({ content: f.content, encoding: "utf-8" })
      });
      treeEntries.push({ path: f.path, mode: "100644", type: "blob", sha: blob.sha });
    }

    let treeBody = { tree: treeEntries };
    if(head) treeBody.base_tree = head.treeSha;
    let tree = await apiJson(API_BASE + "/" + repo + "/git/trees", {
      method: "POST",
      body: JSON.stringify(treeBody)
    });

    let message = opts.message || "Update files via github-data-plugin";
    let commitBody = { message, tree: tree.sha };
    if(head) commitBody.parents = [head.headSha];
    let commit = await apiJson(API_BASE + "/" + repo + "/git/commits", {
      method: "POST",
      body: JSON.stringify(commitBody)
    });

    if(head) {
      await apiJson(API_BASE + "/" + repo + "/git/refs/heads/" + encodeURIComponent(branch), {
        method: "PATCH",
        body: JSON.stringify({ sha: commit.sha, force: !!opts.force })
      });
    } else {
      await apiJson(API_BASE + "/" + repo + "/git/refs", {
        method: "POST",
        body: JSON.stringify({ ref: "refs/heads/" + branch, sha: commit.sha })
      });
    }

    return {
      ok: true,
      branch,
      commitSha: commit.sha,
      treeSha: tree.sha,
      files: fileList.map(f => f.path),
      message,
      url: "https://github.com/" + repo + "/commit/" + commit.sha
    };
  }

  function push(repo, files, opts) {
    repo = parseRepo(repo);
    opts = opts || {};
    if(opts.token) {
      let prev = _token;
      main.setToken(opts.token);
      return pushCore(repo, files, opts).finally(() => { main.setToken(prev); });
    }
    return pushCore(repo, files, opts);
  }

  main.push = push;
  main.pushFile = (repo, path, content, opts) => push(repo, [{ path, content }], opts);

  // ---------- backup this generator (or any named one) ----------
  main.backupGenerator = async (repo, opts) => {
    repo = parseRepo(repo);
    opts = opts || {};
    let gen = opts.generatorName || window.generatorName;
    if(!gen) throw new Error("Could not determine the generator name.");

    let htmlRes = await fetch("https://perchance.org/api/getGeneratorHtml?generatorName=" + encodeURIComponent(gen));
    if(!htmlRes.ok) throw new Error("Failed to fetch the HTML panel of '" + gen + "' (HTTP " + htmlRes.status + "). Is it saved and public?");
    let html = await htmlRes.text();

    let codeRes = await fetch("https://perchance.org/api/getGeneratorsAndDependencies?generatorNames=" + encodeURIComponent(gen));
    if(!codeRes.ok) throw new Error("Failed to fetch the lists of '" + gen + "' (HTTP " + codeRes.status + ").");
    let codeJson = await codeRes.json();
    let code = codeJson.generators && codeJson.generators[gen] && codeJson.generators[gen].code;
    if(typeof code !== "string") throw new Error("Could not read the main.pjs of '" + gen + "'.");

    let message = opts.message || ("Backup of " + gen + " (main.pjs + index.html) via github-data-plugin");
    return push(repo, { "main.pjs": code, "index.html": html }, Object.assign({}, opts, { message }));
  };
