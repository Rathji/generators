<script>
(function(){
  "use strict";

  /* ── VGN helper API (games can use this) ── */
  var VGN = {
    keymap:{ up:"ArrowUp", down:"ArrowDown", left:"ArrowLeft", right:"ArrowRight",
             a:"1", b:"2", x:"3", y:"4", start:"Enter", select:"Escape" },
    setKeymap:function(map){ for(var k in map) this.keymap[k]=map[k]; },
    press:function(action){ dispatchKey(this.keymap[action], true);  },
    release:function(action){ dispatchKey(this.keymap[action], false); },
    hints:function(text){ hintsEl.textContent = text || ""; }
  };
  window.VGN = VGN;

  function keyCode(k){
    if(k===" ") return "Space";
    if(k.length===1) return "Key"+k.toUpperCase();
    return k;
  }
  function dispatchKey(key, down){
    if(!key) return;
    document.dispatchEvent(new KeyboardEvent(down?"keydown":"keyup", {key:key, code:keyCode(key), bubbles:true}));
    reflectPad(key, down);
  }

  /* ── toggles ── */
  var hintsEl = document.getElementById("hintsEl");
  var gamepad = document.getElementById("gamepad");
  var controlsBtn = document.getElementById("controlsBtn");
  var bezelBtn = document.getElementById("bezelBtn");
  var fullBtn = document.getElementById("fullBtn");
  var controlsOn = false;

  controlsBtn.addEventListener("click", function(){
    controlsOn = !controlsOn;
    controlsBtn.classList.toggle("on", controlsOn);
    controlsBtn.setAttribute("aria-pressed", String(controlsOn));
    gamepad.classList.toggle("on", controlsOn);
    hintsEl.classList.toggle("on", controlsOn);
  });
  bezelBtn.addEventListener("click", function(){
    var bare = document.body.classList.toggle("bare");
    bezelBtn.classList.toggle("on", bare);
    bezelBtn.setAttribute("aria-pressed", String(bare));
    window.dispatchEvent(new Event("resize"));
  });

  /* ── native browser fullscreen ── */
  var prevBare = false;
  function isFS(){
    return !!(document.fullscreenElement || document.webkitFullscreenElement);
  }
  function syncFullBtn(){
    var fs = isFS();
    fullBtn.classList.toggle("on", fs);
    fullBtn.setAttribute("aria-pressed", String(fs));
    if(!fs) document.body.classList.toggle("bare", prevBare);
  }
  document.addEventListener("fullscreenchange", syncFullBtn);
  document.addEventListener("webkitfullscreenchange", syncFullBtn);
  fullBtn.addEventListener("click", function(){
    if(!isFS()){
      prevBare = document.body.classList.contains("bare");
      document.body.classList.add("bare");
      var el = document.documentElement;
      var p = null;
      if(el.requestFullscreen) p = el.requestFullscreen();
      else if(el.webkitRequestFullscreen) p = el.webkitRequestFullscreen();
      if(p && p.catch) p.catch(function(){
        document.body.classList.toggle("bare", prevBare);
        syncFullBtn();
      });
      window.dispatchEvent(new Event("resize"));
    } else {
      if(document.exitFullscreen) document.exitFullscreen();
      else if(document.webkitExitFullscreen) document.webkitExitFullscreen();
    }
  });

  /* ── gamepad input → real keyboard events ── */
  function wire(btn){
    var action = btn.getAttribute("data-key");
    btn.addEventListener("pointerdown", function(e){
      e.preventDefault(); e.stopPropagation();
      btn.classList.add("on");
      VGN.press(action);
      try{ btn.setPointerCapture(e.pointerId); }catch(_){}
    });
    function off(e){
      if(!btn.classList.contains("on")) return;
      btn.classList.remove("on");
      VGN.release(action);
    }
    btn.addEventListener("pointerup", off);
    btn.addEventListener("pointercancel", off);
    btn.addEventListener("pointerleave", off);
  }
  document.querySelectorAll(".d,.pbtn,.pmin").forEach(wire);

  /* ── real keyboard presses light up the gamepad ── */
  function reflectPad(key, down){
    var action = null;
    for(var k in VGN.keymap){ if(VGN.keymap[k]===key) action=k; }
    if(!action) return;
    var el = document.querySelector('[data-key="'+action+'"]');
    if(el) el.classList.toggle("on", down);
  }
  window.addEventListener("keyup", function(e){ reflectPad(e.key, false); });

  /* ── default hint text ── */

var sfx = {
  ctx:null,
  init:function(){
    if (!this.ctx){
      try { this.ctx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e){}
    }
    if (this.ctx && this.ctx.state === "suspended"){ this.ctx.resume(); }
  },
  on:function(){ return !muted; },
  tone:function(freq, dur, type, vol){
    if (!this.on()) return;
    this.init();
    if (!this.ctx) return;
    var o = this.ctx.createOscillator(), g = this.ctx.createGain();
    o.type = type || "square";
    o.frequency.value = freq;
    g.gain.value = vol || 0.07;
    o.connect(g); g.connect(this.ctx.destination);
    var t0 = this.ctx.currentTime;
    o.start(t0);
    g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
    o.stop(t0 + dur);
  },
  blip:function(){ this.tone(880, 0.07, "square", 0.05); },
  type:function(){ this.tone(1200, 0.03, "square", 0.03); },

}
function finishAsk(value){
  printLine("? " + term.buf, "good");
  promptLine.textContent = "";
  term.active = false;
  keypadEl.hidden = true;
  term.resolve(value);
  term.resolve = null;
}
function submitAsk(){
  var val = parseInt(term.buf, 10);
  if (isNaN(val)){
    invalidAsk();
    return;
  }
  var err = term.cfg.valid ? term.cfg.valid(val) : null;
  if (err){
    invalidAsk(err);
    return;
  }
  sfx.ding();
  finishAsk(val);
}
function invalidAsk(msg){
  sfx.bad();
  print("COME ON, LET'S BE REASONABLE NOW!!!");
  if (msg) print(msg);
  print("TRY AGAIN");
  term.buf = "";
  renderAsk();
}
function cancelAsk(){
  if (!term.active) return;
  sfx.bad();
  finishAsk(null);
}
function adjustAsk(delta){
  if (!term.active) return;
  if (!term.buf){ term.buf = String((term.cfg.def !== undefined ? term.cfg.def : 0)); }
  var v = parseInt(term.buf, 10);
  if (isNaN(v)) v = term.cfg.def || 0;
  var min = term.cfg.min !== undefined ? term.cfg.min : 0;
  var max = term.cfg.max !== undefined ? term.cfg.max : 100;
  v = Math.max(min, Math.min(max, v + delta));
  term.buf = String(v);
  sfx.blip();
  renderAsk();
}
function keyTap(d){ term.buf += d; sfx.type(); renderAsk(); }
function keyBack(){ term.buf = term.buf.slice(0,-1); sfx.type(); renderAsk(); }
function keyOk(){ submitAsk(); }

function handleTermKey(e){
  var k = e.key;
  if (/^[0-9]$/.test(k)){
    if (term.buf.length < 5){ term.buf += k; sfx.type(); renderAsk(); }
  } else if (k === "Backspace"){
    term.buf = term.buf.slice(0,-1); sfx.type(); renderAsk();
  } else if (k === "ArrowUp"){ adjustAsk(+1); }
  else if (k === "ArrowDown"){ adjustAsk(-1); }
  else if (k === "Enter"){ submitAsk(); }
  else if (k === "Escape"){ cancelAsk(); }
  e.preventDefault();
}

  }

function settleDay(p, w, glasses, signs, price){
  var P9 = C.fairPrice, S2 = w.s2;
  var R1 = 1 - w.j/100;
  var N1;
  if (price <= P9) N1 = (P9 - price)/P9 * 0.8 * S2 + S2;
  else N1 = (P9*P9*S2) / (price*price);
  var V = 1 - Math.exp(-signs * C.adK);   // C2 = 1 in the original formula
  var customers = R1 * (N1 + N1 * V) * (0.75 + Math.random()*0.5);
  var sold = Math.max(0, Math.round(customers));
  if (sold > glasses) sold = glasses;

  var soldOut = false, ruined = false;
  if (w.stormy && Math.random() < C.stormChance){
    sold = glasses;
    soldOut = true;
  }
  if (S.day > 2 && Math.random() < C.eventChance){
    if (soldOut){} else if (Math.random() < 0.5){
      sold = glasses;
      soldOut = true;
    } else {
      sold = 0;
      ruined = true;
    }
  }

  var revenue = sold * price;
  var expenses = glasses * C.glassCost + signs * C.signCost;
  var profit = revenue - expenses;
  p.cash += profit;
  return { sold:sold, revenue:revenue, expenses:expenses, profit:profit, soldOut:soldOut, ruined:ruined };
}

  if (k !== "Escape"){ backToTitle(); }

document.addEventListener("keydown", function(e){
  if (e.target && (e.target.tagName === "INPUT" || e.target.tagName === "TEXTAREA")) return;
  if (term.active){ handleTermKey(e); return; }
  if (waitingKey){
    if (waitingKey.indexOf(e.key) >= 0){
      e.preventDefault();
      var k = waitingKey, r = waitingRes;
      waitingKey = null; waitingRes = null;
      r(e.key);
    } else if (e.key === "Escape" && waitingKey.indexOf("Escape") < 0){
      e.preventDefault();
      var k2 = waitingKey, r2 = waitingRes;
      waitingKey = null; waitingRes = null;
      r2("Escape");
    }
    return;
  }
  if (!S){
    if (e.key === "Enter" || e.key === " "){
      e.preventDefault();
      if (!continueBtn.hidden) continueGame();
      else newGame();
    }
  } else if (S.phase === "end"){
    if (e.key === "Enter" || e.key === " "){
      e.preventDefault();
      playAgain();
    }
  }
});
