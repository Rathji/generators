  if(!cands.length)return;
  var pick;
  if(game.selected&&game.puzzle[game.selected.r][game.selected.c]===0&&game.user[game.selected.r][game.selected.c]===0){
    pick=[game.selected.r,game.selected.c];
  }else{
    pick=cands[Math.floor(Math.random()*cands.length)];
  }
  pushHist(pick[0],pick[1]);
  game.history[game.history.length-1].wasHint=true;
  game.user[pick[0]][pick[1]]=game.solution[pick[0]][pick[1]];
  game.notes[pick[0]][pick[1]].clear();
  game.hints++;
  game.hintPenaltyMs+=HINT_PENALTY;
  sfx("hint");
  saveGame(); renderBoard();
  setStatus("Hint revealed (+"+Math.round(HINT_PENALTY/1000)+"s).");
  if(isSolved())win();
}

function serializeGame(){
  return {
    p:game.puzzle,s:game.solution,u:game.user,d:game.diff,
    m:game.mistakes,h:game.hints,hp:game.hintPenaltyMs,
    e:game.status==="playing"?Math.floor((Date.now()-game.startTs)/1000):game.elapsed,
    n:game.notes.map(function(row){ return row.map(function(set){ return Array.from(set); }); })
  };
}
function deserializeGame(d){
  if(!d||!d.p||!d.s)return false;
  game.puzzle=d.p; game.solution=d.s;
  game.user=d.u||emptyGrid();
  game.notes=d.n?d.n.map(function(row){ return row.map(function(arr){ return new Set(arr||[]); }); }):emptyNotes();
  game.diff=DIFFS[d.d]?d.d:"easy";
  game.mistakes=d.m||0; game.hints=d.h||0; game.hintPenaltyMs=d.hp||0;
  game.elapsed=d.e||0; game.history=[];
  game.startTs=Date.now()-game.elapsed*1000;
  game.status="playing";
  game.noteMode=false;
  return true;
}
function saveGame(){
  try{ localStorage.setItem(SAVE_KEY,JSON.stringify(serializeGame())); }catch(e){}
}
function restoreGame(){
  try{
    var raw=localStorage.getItem(SAVE_KEY);
    if(!raw)return false;
    return deserializeGame(JSON.parse(raw));
  }catch(e){ return false; }
}

function importJSON(str){
  var d;
  try{ d=JSON.parse(str); }catch(e){ return "That doesn't look like valid JSON."; }
  if(!d||typeof d!=="object"||Array.isArray(d))return "That doesn't look like a game export.";

function cloneWithStyles(src){
  var clone=src.cloneNode(true);
  clone.setAttribute("xmlns","http"+"://www.w3.org/1999/xhtml");
  var PROPS=["display","position","top","left","width","height","min-width","min-height","aspect-ratio","box-sizing",
    "padding-top","padding-right","padding-bottom","padding-left","margin-top","margin-right","margin-bottom","margin-left",
    "grid-template-columns","grid-template-rows","gap","grid-gap","align-items","justify-content","justify-items","align-content","flex-direction","flex-wrap",
    "background-color","background-image","background-size","background-position","background-repeat",
    "border-radius",
    "border-top-width","border-top-style","border-top-color","border-right-width","border-right-style","border-right-color",
    "border-bottom-width","border-bottom-style","border-bottom-color","border-left-width","border-left-style","border-left-color",
    "color","font-family","font-size","font-weight","font-style","line-height","letter-spacing","text-transform","text-align",
    "box-shadow","opacity","overflow","cursor","user-select","text-shadow","outline"];
  function walk(el,c){
    if(!el||!c)return;
    var cs=window.getComputedStyle(el),style="";
    for(var i=0;i<PROPS.length;i++){
      var v=cs.getPropertyValue(PROPS[i]);
      if(v)style+=PROPS[i]+":"+v+";";
    }
    c.setAttribute("style",style);
    for(var k=0;k<el.children.length;k++)walk(el.children[k],c.children[k]);
  }
  walk(src,clone);
  return clone;
}
function renderBoardImage(){
  var el=boardEl;
  var w=el.offsetWidth,h=el.offsetHeight;
  if(!w||!h)return Promise.resolve(null);
  var scale=3;
  var svg='<svg xmlns="http'+'://www.w3.org/2000/svg" width="'+w+'" height="'+h+'">'+
    '<foreignObject width="100%" height="100%">'+new XMLSerializer().serializeToString(cloneWithStyles(el))+'</foreignObject></svg>';
  return new Promise(function(resolve){
    var img=new Image();
    img.onload=function(){
      var c=document.createElement("canvas");
      c.width=Math.round(w*scale); c.height=Math.round(h*scale);

var VERSION = 1;
var SLOT_BASE = 16;
var SLOT_SIZE = 512;
var SLOT_LIMIT = Math.floor((state.length - SLOT_BASE) / SLOT_SIZE);
if (SLOT_LIMIT > 200) SLOT_LIMIT = 200;
if (SLOT_LIMIT < 1) SLOT_LIMIT = 1;

// Slot layout (offsets within a 512-byte slot):
//   0     status u8 (0 free 1 open 2 playing 3 done)
//   1     turn u8 (0 host 1 guest)
//   2     strikes u8 (shared lives)
//   3     hints u8
//   4     penaltySec u8
//   5     winner u8 (0 none 1 host 2 guest 3 shared-loss)
//   6-9   startTs u32
//   10-13 lastAct u32
//   14-15 reserved
//   16-96    puzzle (81 bytes, 0 = empty)
//   97-177   solution (81 bytes)
//   178-258  user entries (81 bytes)
//   259-339  note bitmasks (81 bytes; bit k = digit k+1)
//   340-363  hostName (24)
//   364-387  guestName (24)
//   388-403  hostToken (16)
//   404-419  guestToken (16)
//   420-424  room code (5)
//   425-511  reserved

var CODE_CHARS = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
var rooms = {};
var connRoom = {};
var rateMap = {};

var BANNED = ["nigger","nigga","faggot","kys","kill yourself","retard","heil hitler","tranny"];

function nowSec(){ return Math.floor(Date.now()/1000); }
function writeU32(o,v){ state[o]=v&255; state[o+1]=(v>>>8)&255; state[o+2]=(v>>>16)&255; state[o+3]=(v>>>24)&255; }
function readU32(o){ return ((state[o]|state[o+1]<<8|state[o+2]<<16|state[o+3]<<24)>>>0); }
function u8(s){
  var out=[], i, c; s=String(s);
  for(i=0;i<s.length;i++){
    c=s.charCodeAt(i);
    if(c<128){ out.push(c); }
    else if(c<2048){ out.push(192|c>>6, 128|c&63); }
    else if(c<55296||c>57343){ out.push(224|c>>12, 128|c>>6&63, 128|c&63); }
    else{ i++;
      var h=(c&1023)<<10 | s.charCodeAt(i)&1023;
      out.push(240|h>>18, 128|h>>12&63, 128|h>>6&63, 128|h&63);
    }
  }
  return out;
}
function d8(a, o, l){
  var s="", i=o, e=o+l, c;
  while(i<e){
    c=a[i++];
    if(c<128){ s+=String.fromCharCode(c); }
    else if(c>191&&c<224){ s+=String.fromCharCode((c&31)<<6|a[i++]&63); }
    else if(c>239){ var u=(c&7)<<18|(a[i++]&63)<<12|(a[i++]&63)<<6|a[i++]&63; s+=String.fromCharCode(55296+(u>>10), 56320+(u&1023)); }
    else{ s+=String.fromCharCode((c&15)<<12|(a[i++]&63)<<6|a[i++]&63); }
  }
  return s;
}
function putStr(base,o,s,max){
  var b=u8(s), i;
  for(i=0;i<max;i++) state[base+o+i] = i<b.length ? b[i] : 0;
}
function getStr(base,o,max){
  var i=o, end=o+max;
  while(i<end && state[base+i]!==0) i++;
  return d8(state, base+o, i-o);
}
function clean(s,max){ return String(s||"").trim().replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").slice(0,max); }
function isBanned(t){ var s=String(t||"").toLowerCase(); for(var i=0;i<BANNED.length;i++){ if(s.indexOf(BANNED[i])!==-1) return true; } return false; }
function randCode(n){ var s="", i; for(i=0;i<n;i++) s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)]; return s; }
function sweepRate(){
  var keys=Object.keys(rateMap), now=Date.now(), i;
  for(i=0;i<keys.length;i++){ var e=rateMap[keys[i]]; if(now-e.t>90000) delete rateMap[keys[i]]; }
}
function rateOk(conn,key,max,win){
  var k1=key+"|"+conn.id, k2=key+"|c"+(conn.net[2]|0);
  var a=rateMap[k1]||{t:0,n:0}, b=rateMap[k2]||{t:0,n:0}, now=Date.now();
  if(now-a.t>win*1000){ a={t:now,n:0}; }
  if(now-b.t>win*1000){ b={t:now,n:0}; }
  a.n++; b.n++; rateMap[k1]=a; rateMap[k2]=b;
  return a.n<=max && b.n<=max;
}
function clearRate(conn){ var i; for(i=0;i<3;i++){ var k=["mv","hv","ch"][i]+"|"+conn.id; delete rateMap[k]; } }
function roleOf(conn,r){
  if(r.hostConn && r.hostConn.id===conn.id) return "host";
  if(r.guestConn && r.guestConn.id===conn.id) return "guest";
  return null;
}
function slotOf(code){ var r=rooms[code]; return r?SLOT_BASE+r.slot*SLOT_SIZE:-1; }
function freeSlot(slot){
  var base=SLOT_BASE+slot*SLOT_SIZE, i;
  for(i=0;i<SLOT_SIZE;i++) state[base+i]=0;
}
function findFreeSlot(){
  var i, st;
  for(i=0;i<SLOT_LIMIT;i++){ if(state[SLOT_BASE+i*SLOT_SIZE]===0) return i; }
  var best=-1, bestTs=Infinity;
  for(i=0;i<SLOT_LIMIT;i++){
    st=state[SLOT_BASE+i*SLOT_SIZE];
    if(st===1||st===3){
      var ts=readU32(SLOT_BASE+i*SLOT_SIZE+10);
      if(ts<bestTs){ bestTs=ts; best=i; }
    }
  }
  if(best>=0) freeSlot(best);
  return best;
}
function snapOf(code){
  var base=slotOf(code); if(base<0) return null;
  var snap={ code:code, status:state[base], turn:state[base+1], strikes:state[base+2], hints:state[base+3], penaltySec:state[base+4], winner:state[base+5], el:Math.max(0,nowSec()-readU32(base+6)), p:[], s:[], u:[], n:[], hostName:getStr(base,340,24), guestName:getStr(base,364,24) };
  var i;
  for(i=0;i<81;i++){ snap.p.push(state[base+16+i]); snap.s.push(state[base+97+i]); snap.u.push(state[base+178+i]); snap.n.push(state[base+259+i]); }
  return snap;
}
function publishSnap(code){ try{ pubsub.publish("room:"+code, JSON.stringify({t:"snap", d:snapOf(code)})); }catch(e){} }
function publishLog(code,msg){ try{ pubsub.publish("room:"+code, JSON.stringify({t:"log", m:msg})); }catch(e){} }
function claimWin(code, role){
  var base=slotOf(code); if(base<0) return false;
  var i;
  for(i=0;i<81;i++){ var cur=state[base+178+i]||state[base+16+i]; if(cur!==state[base+97+i]) return false; }
  state[base+5] = role==="host" ? 1 : 2;
  return true;
}
function guestLeft(code){
  var r=rooms[code]; if(!r) return;
  var base=slotOf(code);
  if(r.guestConn){ try{ r.guestConn.unsubscribe("room:"+code); }catch(e){} if(connRoom[r.guestConn.id]===code) delete connRoom[r.guestConn.id]; r.guestConn=null; }
  var i;
  for(i=364;i<420;i++) state[base+i]=0;
  state[base]=1; state[base+1]=0; state[base+5]=0;
  writeU32(base+10, nowSec());
  publishLog(code, "Guest left — waiting for a new opponent.");
  publishSnap(code);
}
function closeRoom(code, reason){
  var r=rooms[code]; if(!r) return;
  try{ pubsub.publish("room:"+code, JSON.stringify({t:"closed", r:reason})); }catch(e){}
  if(r.hostConn){ try{ r.hostConn.unsubscribe("room:"+code); }catch(e){} if(connRoom[r.hostConn.id]===code) delete connRoom[r.hostConn.id]; }
  if(r.guestConn){ try{ r.guestConn.unsubscribe("room:"+code); }catch(e){} if(connRoom[r.guestConn.id]===code) delete connRoom[r.guestConn.id]; }
  freeSlot(r.slot);
  delete rooms[code];
}

if (state[0] !== VERSION){ state[0] = VERSION; }
(function rebuild(){
  rooms = {};
  var i, base, code;
  for(i=0;i<SLOT_LIMIT;i++){
    base=SLOT_BASE+i*SLOT_SIZE;
    if(state[base]===0) continue;
    code=getStr(base,420,5);
    if(code) rooms[code]={slot:i, hostConn:null, guestConn:null};
  }
})();

self.rpc = {
  createRoom: function({conn}, data){
    sweepRate();
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var name=clean(d.name,16) || "Host";
    var code=randCode(5), tries=0;
    while(rooms[code] && tries++<20) code=randCode(5);
    if(rooms[code]) return '{"ok":0,"err":"busy"}';
    var slot=findFreeSlot();
    if(slot<0) return '{"ok":0,"err":"full"}';
    var base=SLOT_BASE+slot*SLOT_SIZE;
    var token=randCode(10);
    state[base]=1; state[base+1]=0; state[base+2]=0; state[base+3]=0; state[base+4]=0; state[base+5]=0;
    writeU32(base+6, nowSec()); writeU32(base+10, nowSec());
    putStr(base,340,name,24); putStr(base,388,token,16); putStr(base,420,code,5);
    rooms[code]={slot:slot, hostConn:conn, guestConn:null};
    connRoom[conn.id]=code;
    try{ conn.subscribe("room:"+code); }catch(e){}
    var snap=snapOf(code); snap.myRole="host"; snap.token=token;
    return JSON.stringify({ok:1, code:code, token:token, role:"host", snap:snap});
  },
  joinRoom: function({conn}, data){
    sweepRate();
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var name=clean(d.name,16) || "Guest";
    var r=rooms[code];
    if(!r) return '{"ok":0,"err":"not_found"}';
    var base=slotOf(code);
    if(state[base]!==1 || r.guestConn) return '{"ok":0,"err":"full"}';
    if(r.hostConn && r.hostConn.id===conn.id) return '{"ok":0,"err":"already_host"}';
    var token=randCode(10);
    state[base]=2; state[base+1]=0; state[base+5]=0;
    putStr(base,364,name,24); putStr(base,404,token,16);
    writeU32(base+10, nowSec());
    r.guestConn=conn; connRoom[conn.id]=code;
    try{ conn.subscribe("room:"+code); }catch(e){}
    publishLog(code, name+" joined the room.");
    publishSnap(code);
    var snap=snapOf(code); snap.myRole="guest"; snap.token=token;
    return JSON.stringify({ok:1, code:code, token:token, role:"guest", snap:snap});
  },
  rejoin: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var token=clean(d.token,16);
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';
    var base=slotOf(code), snap;
    if(getStr(base,388,16)===token){
      r.hostConn=conn; connRoom[conn.id]=code;
      try{ conn.subscribe("room:"+code); }catch(e){}
      snap=snapOf(code); snap.myRole="host"; snap.token=token;
      return JSON.stringify({ok:1, role:"host", snap:snap});
    }
    if(getStr(base,404,16)===token){
      r.guestConn=conn; connRoom[conn.id]=code;
      try{ conn.subscribe("room:"+code); }catch(e){}
      snap=snapOf(code); snap.myRole="guest"; snap.token=token;
      return JSON.stringify({ok:1, role:"guest", snap:snap});
    }
    return '{"ok":0,"err":"not_found"}';
  },
  getSnap: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    if(!rooms[code]) return '{"ok":0,"err":"not_found"}';
    return JSON.stringify({t:"snap", d:snapOf(code)});
  },
  move: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';
    var role=roleOf(conn,r); if(!role) return '{"ok":0,"err":"not_in_room"}';
    var base=slotOf(code);
    if(state[base]!==2) return '{"ok":0,"err":"not_playing"}';
    if(state[base+5]!==0) return '{"ok":0,"err":"game_over"}';
    var turn=state[base+1];
    if((role==="host" && turn!==0) || (role==="guest" && turn!==1)) return '{"ok":0,"err":"not_your_turn"}';
    if(!rateOk(conn,"mv",25,10)) return '{"ok":0,"err":"slow_down"}';
    var cell=d.cell|0, v=d.val|0;
    if(cell<0 || cell>80) return '{"ok":0,"err":"bad"}';
    if(state[base+16+cell]!==0) return '{"ok":0,"err":"given"}';
    if(d.note){
      if(v<1||v>9) return '{"ok":0,"err":"bad"}';
      if(state[base+178+cell]===0) state[base+259+cell] ^= (1<<(v-1));
    } else if(v===0){
      state[base+178+cell]=0;
    } else if(v<1||v>9){
      return '{"ok":0,"err":"bad"}';
    } else if(v===state[base+97+cell]){
      state[base+178+cell]=v; state[base+259+cell]=0;
    } else {
      state[base+178+cell]=v; state[base+259+cell]=0;
      state[base+2]=Math.min(3,state[base+2]+1);
      if(state[base+2]>=3){ state[base+5]=3; }
    }
    if(!d.note && state[base+5]===0){
      state[base+1]=1-turn;
      claimWin(code, role);
    }
    if(state[base+5]===3) publishLog(code, "Three strikes — the puzzle is lost.");
    else if(state[base+5]===1) publishLog(code, "Host completed the puzzle — Host wins!");
    else if(state[base+5]===2) publishLog(code, "Guest completed the puzzle — Guest wins!");
    writeU32(base+10, nowSec());
    publishSnap(code);
    return JSON.stringify({ok:1, snap:snapOf(code)});
  },
  hint: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';

    if(state[base+5]!==0) return '{"ok":0,"err":"game_over"}';
    var turn=state[base+1];
    if((role==="host" && turn!==0) || (role==="guest" && turn!==1)) return '{"ok":0,"err":"not_your_turn"}';
    if(!rateOk(conn,"hv",6,10)) return '{"ok":0,"err":"slow_down"}';
    var cands=[], i;
    for(i=0;i<81;i++){ if(state[base+16+i]===0 && state[base+178+i]===0) cands.push(i); }
    if(!cands.length) return '{"ok":0,"err":"none"}';
    var pick=cands[Math.floor(Math.random()*cands.length)];
    state[base+178+pick]=state[base+97+pick];
    state[base+259+pick]=0;
    state[base+3]=Math.min(255,state[base+3]+1);
    state[base+4]=Math.min(255,state[base+4]+30);
    state[base+1]=1-turn;
    claimWin(code, role);
    if(state[base+5]===1) publishLog(code, "Host completed the puzzle — Host wins!");
    else if(state[base+5]===2) publishLog(code, "Guest completed the puzzle — Guest wins!");
    writeU32(base+10, nowSec());
    publishSnap(code);
    return JSON.stringify({ok:1, snap:snapOf(code)});
  },
  setGame: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';
    if(!(r.hostConn && r.hostConn.id===conn.id)) return '{"ok":0,"err":"not_host"}';
    var base=slotOf(code);
    var p=d.p||[], s=d.s||[];
    if(!Array.isArray(p)||!Array.isArray(s)||p.length!==81||s.length!==81) return '{"ok":0,"err":"bad"}';
    var i, pv, sv;
    for(i=0;i<81;i++){
      pv=p[i]|0; sv=s[i]|0;
      if(pv<0||pv>9||sv<1||sv>9) return '{"ok":0,"err":"bad"}';
      state[base+16+i]=pv; state[base+97+i]=sv; state[base+178+i]=0; state[base+259+i]=0;
    }
    state[base]=r.guestConn?2:1;
    state[base+1]=0; state[base+2]=0; state[base+3]=0; state[base+4]=0; state[base+5]=0;
    writeU32(base+6, nowSec()); writeU32(base+10, nowSec());
    publishLog(code, "A new puzzle was dealt.");
    publishSnap(code);
    return '{"ok":1}';
  },
  chat: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';
    var role=roleOf(conn,r); if(!role) return '{"ok":0,"err":"not_in_room"}';
    if(!rateOk(conn,"ch",6,10)) return '{"ok":0,"err":"slow_down"}';
    var msg=clean(d.msg,140);
    if(!msg) return '{"ok":1}';
    if(isBanned(msg)){ try{ conn.close(4003,"banned word"); }catch(e){} return '{"ok":0,"err":"banned"}'; }
    var base=slotOf(code);
    var name=getStr(base, role==="host"?340:364, 24) || (role==="host"?"Host":"Guest");
    try{ pubsub.publish("room:"+code, JSON.stringify({t:"chat", n:name, m:msg})); }catch(e){}
    return '{"ok":1}';
  },
  leave: function({conn}, data){
    var d; try{ d=JSON.parse(data); }catch(e){ return '{"ok":0,"err":"bad"}'; }
    var code=clean(d.code,5).toUpperCase();
    var r=rooms[code]; if(!r) return '{"ok":0,"err":"not_found"}';
    var role=roleOf(conn,r); if(!role) return '{"ok":0,"err":"not_in_room"}';
    if(role==="host") closeRoom(code, "host_left");
    else guestLeft(code);
    return '{"ok":1}';
  }
};

self.onclose = function({conn}){
  var code=connRoom[conn.id];
  if(code) delete connRoom[conn.id];
  clearRate(conn);
  if(!code) return;
  var r=rooms[code]; if(!r) return;
  if(r.hostConn && r.hostConn.id===conn.id){ closeRoom(code, "host_left"); }
  else if(r.guestConn && r.guestConn.id===conn.id){ guestLeft(code); }
};
</script>

function onOpen(){
  retryDelay=1000;
  netStatus("connected");
  tryRejoin();
}
function onClose(ev){
  online.socket=null;
  if(ev && ev.code===4403){ netStatus("online is unavailable on embedded pages"); return; }
  if(online.active || mode==="online"){
    netStatus("reconnecting…");
    setTimeout(ensureSocket,retryDelay);
    retryDelay=Math.min(retryDelay*2,15000);
  }
}
function onMessage(data){
  if(typeof data!=="string") return;
  var m; try{ m=JSON.parse(data); }catch(e){ return; }
  if(!m || !m.t) return;
  if(m.t==="snap"){ if(online.active && m.d && m.d.code===online.code) applyServerSnap(m.d); }
  else if(m.t==="chat"){ addChat(m.n,m.m); }
  else if(m.t==="log"){ addLog(m.m); }
  else if(m.t==="closed"){
    if(!online.active) return;
    exitRoomLocal();
    netStatus(m.r==="host_left" ? "The host left — room closed." : "Room closed.");
  }
}
function tryRejoin(){
  if(!online.active) return;
  var raw=null; try{ raw=localStorage.getItem("bgnSudokuRoom"); }catch(e){}
  if(!raw) return;
  var saved=null; try{ saved=JSON.parse(raw); }catch(e){}
  if(!saved || !saved.code || !saved.token) return;
  if(!online.socket || online.socket.readyState!==1) return;
  online.socket.rpc.rejoin(JSON.stringify({code:saved.code, token:saved.token})).then(function(res){
    var r; try{ r=JSON.parse(res); }catch(e){ return; }
    if(!r || !r.ok){ online.active=false; return; }
    online.code=saved.code; online.token=saved.token;
    enterRoom(r.role, r.snap);
  }).catch(function(){});
}

  ensureSocket();
  if(!online.socket || online.socket.readyState!==1){ netStatus("connecting… try again in a second"); return; }
  online.socket.rpc.createRoom(JSON.stringify({name:online.name})).then(function(res){
    var r; try{ r=JSON.parse(res); }catch(e){ return; }
    if(!r || !r.ok){ netStatus(r && r.err==="full" ? "no room slots free right now" : "couldn't create a room"); return; }
    online.code=r.code; online.token=r.token;
    saveRoom();

function joinRoom(code){
  var c=cleanText(code||"").toUpperCase().slice(0,5);
  if(c.length<5){ netStatus("enter a 5-character room code"); return; }
  ensureSocket();
  if(!online.socket || online.socket.readyState!==1){ netStatus("connecting… try again in a second"); return; }
  online.socket.rpc.joinRoom(JSON.stringify({code:c, name:online.name})).then(function(res){
    var r; try{ r=JSON.parse(res); }catch(e){ return; }
    if(!r || !r.ok){
      netStatus({not_found:"room not found", full:"room is full or already started", already_host:"that's your own room", bad:"bad request"}[r.err] || "couldn't join");
      return;
    }
    online.code=r.code; online.token=r.token;
    saveRoom();
    enterRoom(r.role, r.snap);
  }).catch(function(){ netStatus("server error"); });
}
function sendSetGame(p,s){

    if(r && !r.ok) netStatus(r.err==="not_host" ? "only the host deals puzzles" : "couldn't deal a puzzle");
  }).catch(function(){});
}

function firstEmpty(){
  var g=__s.game; if(!g) return null;
  for(var r=0;r<9;r++)for(var c=0;c<9;c++) if(g.puzzle[r][c]===0 && g.user[r][c]===0) return {r:r,c:c};
  return null;
}
function onlineSubmitMove(val){
  if(!online.active) return;
  var g=__s.game; if(!g || g.status!=="playing") return;
  if(!g.selected){ var em=firstEmpty(); if(!em) return; if(__s.select) __s.select(em.r,em.c); }
  var cell=g.selected.r*9+g.selected.c;
  var payload={code:online.code, token:online.token, cell:cell, val:val};
  if(g.noteMode && val!==0) payload.note=true;
  if(!online.socket || online.socket.readyState!==1){ netStatus("reconnecting…"); return; }
  online.socket.rpc.move(JSON.stringify(payload)).then(function(res){
    var r; try{ r=JSON.parse(res); }catch(e){ return; }
    if(!r || !r.ok){
      if(r && r.err==="not_your_turn") netStatus("It's not your turn.");
      else if(r && r.err==="not_in_room"){ netStatus("rejoining…"); rejoinAndRetryMove(payload); }
      else if(r && r.err==="game_over") netStatus("This puzzle is already finished.");
      else if(r && r.err==="not_playing") netStatus("Waiting for an opponent to join…");
      else netStatus((r && r.err) || "move rejected");
      return;
    }
    if(r.snap) applyServerSnap(r.snap);
  }).catch(function(){});
}
function rejoinAndRetryMove(payload){
  var saved=null; try{ saved=JSON.parse(localStorage.getItem("bgnSudokuRoom")||"null"); }catch(e){}
  if(!saved || !saved.code || !saved.token || !online.socket) return;
  online.socket.rpc.rejoin(JSON.stringify({code:saved.code, token:saved.token})).then(function(res){
    var r; try{ r=JSON.parse(res); }catch(e){ return; }
    if(!r || !r.ok) return;
    online.code=saved.code; online.token=saved.token;
    enterRoom(r.role, r.snap);
    setTimeout(function(){ onlineSubmitMove(payload.val); }, 250);
  }).catch(function(){});
}

/* ── apply server snapshot ── */
function gridFrom81(arr){
  var out=[], r, c;
  for(r=0;r<9;r++){ out[r]=[]; for(c=0;c<9;c++) out[r][c]=(arr && arr[r*9+c])||0; }
  return out;
}
function applyServerSnap(snap){
  if(!snap || !snap.p || !snap.s || !__s.game) return;
  var g=__s.game;
  g.puzzle=gridFrom81(snap.p);
  g.solution=gridFrom81(snap.s);
  g.user=gridFrom81(snap.u);
  var notes=[], i;
  for(i=0;i<81;i++){
    var r=(i/9)|0, c=i%9, bits=snap.n[i]||0, set=new Set(), b;
    if(!notes[r]) notes[r]=[];
    for(b=0;b<9;b++) if(bits&(1<<b)) set.add(b+1);
    notes[r][c]=set;
  }
  g.notes=notes;
  g.mistakes=snap.strikes||0;
  g.hints=snap.hints||0;
  g.hintPenaltyMs=(snap.penaltySec||0)*1000;
  g.elapsed=snap.el||0;
  g.startTs=Date.now()-g.elapsed*1000;
  if(snap.winner===3) g.status="lost";
  else if(snap.winner) g.status="won";
  else g.status="playing";
  if(__s.renderAll) __s.renderAll();
  updateTurnUi(snap);
}
function updateTurnUi(snap){
  if(!online.active || !turnEl) return;
  var hostName=snap.hostName||"Host", guestName=snap.guestName||"Guest";
  if(snap.status===1){ turnEl.textContent="Waiting for an opponent…"; if(roomStatusEl) roomStatusEl.textContent="Open room — share code "+snap.code; lockBoard(true); return; }
  if(snap.winner){
    if(snap.winner===3) turnEl.textContent="💥 Three strikes — nobody wins";
    else { var w=snap.winner===1?hostName:guestName; turnEl.textContent="🏆 "+w+" wins!"; }
    if(roomStatusEl) roomStatusEl.textContent="";
    lockBoard(true);
    return;
  }
  var yourTurn=(online.role==="host"&&snap.turn===0)||(online.role==="guest"&&snap.turn===1);
  var oppName=snap.turn===0?hostName:guestName;
  turnEl.textContent=yourTurn?"Your turn":(oppName+"'s turn");
  if(roomStatusEl) roomStatusEl.textContent=hostName+" vs "+guestName;
  lockBoard(!yourTurn);
}
