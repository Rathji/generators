     block, so the value is a real array. The pack (deck grid,

<!-- ═══ DUEL SERVER (server-plugin) ═══
     Requires `createServerSocket = {import:server-plugin}` in main.pjs.
     Public source — no secrets here. Rooms are in-memory only. -->
<script type="text/x-server-plugin">
/* ── BGN Beat-the-Card duel server ──────────────────────────────
   2-player rooms. The host shuffles its configured pack in the
   browser and uploads both private stacks + a rank table; the
   server is authoritative for the stacks and turn order. Host
   strikes first each round, then guest; higher rank wins. First
   to `rounds` (from the host, default 5) wins. Rooms live in
   memory only.                                                    */
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const MAX_ROOMS=400, NAME_MAX=16, WIN_ROUNDS=5, DECK_MAX=160;
let rooms=new Map();
const byConn=new Map();
const createTimes=new Map();
const connRate=new Map();
function nowSec(){ return Math.floor(Date.now()/1000); }
function clean(s,max){ return String(s||"").replace(/[\u0000-\u001f\u007f]/g,"").trim().slice(0,max); }
function makeCode(){ let c=""; for(let i=0;i<5;i++)c+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)]; return c; }
function rateOk(conn){ const n=nowSec(), last=connRate.get(conn.id); if(last===n)return false; connRate.set(conn.id,n); return true; }
function canCreate(conn){ const k=conn.net[2], n=nowSec(), e=createTimes.get(k); if(!e||e[1]<n-600){ createTimes.set(k,[1,n]); return true; } if(e[0]>=6)return false; e[0]++; return true; }
function idArr(a){ if(!Array.isArray(a))return []; const out=[]; for(let i=0;i<a.length&&i<DECK_MAX;i++){ const v=Number(a[i]); if(Number.isInteger(v)&&v>=0&&v<1000)out.push(v); } return out; }
function send(conn,obj){ try{ if(conn&&conn.readyState===1)conn.send(JSON.stringify(obj)); }catch(e){} }
self.rpc={
  create({conn},data){
    if(!rateOk(conn)||!canCreate(conn))return JSON.stringify({ok:false,err:"slow_down"});
    let d; try{ d=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad"}); }
    const name=clean(d.name,NAME_MAX)||"Player";
    const cur=byConn.get(conn.id);
    if(cur&&cur.role==="host"){ const r=rooms.get(cur.code); if(r){ r.hostConn=conn; r.hostName=name; return JSON.stringify({ok:true,code:r.code}); } }
    if(rooms.size>=MAX_ROOMS)return JSON.stringify({ok:false,err:"full"});
    let code=makeCode(), tries=0;
    while(rooms.has(code)&&tries++<25)code=makeCode();
    rooms.set(code,{code,hostConn:conn,hostName:name,guestConn:null,guestName:"",status:0,
      hostStack:[],guestStack:[],vals:{},hostWins:0,guestWins:0,turn:"host",round:0,rounds:WIN_ROUNDS,pending:null,winner:null});
    byConn.set(conn.id,{code,role:"host"});
    return JSON.stringify({ok:true,code});
  },
  join({conn},data){
    if(!rateOk(conn))return JSON.stringify({ok:false,err:"slow_down"});
    let d; try{ d=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad"}); }
    const code=String(d.code||"").toUpperCase().replace(/[^A-Z2-9]/g,"").slice(0,5);
    const name=clean(d.name,NAME_MAX)||"Player";
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_room"});
    if(room.status===1)return JSON.stringify({ok:false,err:"started"});
    if(room.guestConn)return JSON.stringify({ok:false,err:"full"});
    room.guestConn=conn; room.guestName=name;
    byConn.set(conn.id,{code,role:"guest"});
    send(room.hostConn,{t:"peer",name});
    return JSON.stringify({ok:true,code});
  },
  start({conn},data){
    if(!rateOk(conn))return JSON.stringify({ok:false,err:"slow_down"});
    let d; try{ d=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad"}); }
    const loc=byConn.get(conn.id);
    if(!loc||loc.role!=="host")return JSON.stringify({ok:false,err:"not_host"});
    const room=rooms.get(loc.code);
    if(!room)return JSON.stringify({ok:false,err:"no_room"});
    if(!room.guestConn)return JSON.stringify({ok:false,err:"no_opp"});
    const hostStack=idArr(d.hostStack), guestStack=idArr(d.guestStack);
    if(!hostStack.length||!guestStack.length)return JSON.stringify({ok:false,err:"bad_deck"});
    const vals={};
    if(d.vals&&typeof d.vals==="object"){ for(const k in d.vals){ const v=Number(d.vals[k]); if(Number.isFinite(v))vals[k]=v; } }
    room.hostStack=hostStack; room.guestStack=guestStack; room.vals=vals;
    room.hostWins=0; room.guestWins=0; room.turn="host"; room.round=0; room.pending=null; room.winner=null;
    room.rounds=Math.max(1,Math.min(20,Number(d.rounds)||WIN_ROUNDS));
    room.status=1;
    send(room.hostConn,{t:"start",you:"host",count:room.hostStack.length,oppName:room.guestName,rounds:room.rounds});
    send(room.guestConn,{t:"start",you:"guest",count:room.guestStack.length,oppName:room.hostName,rounds:room.rounds});
    return JSON.stringify({ok:true});
  },
  strike({conn},data){
    if(!rateOk(conn))return JSON.stringify({ok:false,err:"slow_down"});
    const loc=byConn.get(conn.id);
    if(!loc)return JSON.stringify({ok:false,err:"no_room"});
    const room=rooms.get(loc.code);
    if(!room||room.status!==1)return JSON.stringify({ok:false,err:"no_room"});
    const role=loc.role;
    if(room.turn!==role)return JSON.stringify({ok:false,err:"not_your_turn"});
    const stack=role==="host"?room.hostStack:room.guestStack;
    if(!stack.length)return JSON.stringify({ok:false,err:"no_cards"});
    const id=stack.shift();
    const rank=Number(room.vals[id]||0)||0;
    room.pending=room.pending||{};
    room.pending[role]=id; room.pending[role+"Rank"]=rank;
    send(room.hostConn,{t:"struck",by:role,id,rank});
    send(room.guestConn,{t:"struck",by:role,id,rank});
    if(room.pending.host!==undefined&&room.pending.guest!==undefined){
      const hr=room.pending.hostRank, gr=room.pending.guestRank;
      const tie=hr===gr, winner=tie?null:(hr>gr?"host":"guest");
      if(winner)room[winner==="host"?"hostWins":"guestWins"]++;
      room.round++;
      const done=room.hostWins>=room.rounds||room.guestWins>=room.rounds||!room.hostStack.length||!room.guestStack.length;
      const result={t:"round",aId:room.pending.host,bId:room.pending.guest,tie,winner,hostWins:room.hostWins,guestWins:room.guestWins,round:room.round,done};
      send(room.hostConn,result); send(room.guestConn,result);
      room.pending=null;
      if(done){
        room.winner=room.hostWins>room.guestWins?"host":(room.hostWins<room.guestWins?"guest":"draw");
        room.status=2;
        const msg={t:"done",winner:room.winner,hostWins:room.hostWins,guestWins:room.guestWins};
        send(room.hostConn,msg); send(room.guestConn,msg);
      }else{
        room.turn="host";
      }
    }else{
      room.turn=role==="host"?"guest":"host";
    }
    return JSON.stringify({ok:true});
  },
  leave({conn}){
    const loc=byConn.get(conn.id);
    if(!loc)return JSON.stringify({ok:true});
    byConn.delete(conn.id);
    const room=rooms.get(loc.code);
    if(!room)return JSON.stringify({ok:true});
    if(loc.role==="host"){ rooms.delete(loc.code); send(room.guestConn,{t:"peerleft"}); }
    else{ room.guestConn=null; room.guestName=""; send(room.hostConn,{t:"peerleft"}); }
    return JSON.stringify({ok:true});
  }
};
self.onclose=function({conn}){
  connRate.delete(conn.id);
  const loc=byConn.get(conn.id);
  if(!loc)return;
  const room=rooms.get(loc.code);
  byConn.delete(conn.id);
  if(!room)return;
  if(loc.role==="host"&&room.hostConn===conn){ rooms.delete(loc.code); send(room.guestConn,{t:"peerleft"}); }
  else if(loc.role==="guest"&&room.guestConn===conn){ room.guestConn=null; room.guestName=""; send(room.hostConn,{t:"peerleft"}); }
};
