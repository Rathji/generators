function writeU32(o, v){ state[o]=v&255; state[o+1]=(v>>>8)&255; state[o+2]=(v>>>16)&255; state[o+3]=(v>>>24)&255; }
function readU32(o){ return ((state[o]|state[o+1]<<8|state[o+2]<<16|state[o+3]<<24)>>>0); }

if (state[0] === 0){ state[0] = VERSION; writeU32(1, 0); }
else if (state[0] < VERSION){
  if (state[0] === 2) migrateSubs(); // v2→v3: add beta byte (iterate slots in decreasing order — new records are 1 byte wider and would clobber the next old slot)
  state[0] = VERSION;
}

function u8(s){
  var out=[], i, c; s=String(s);
  for(i=0;i<s.length;i++){
    c=s.charCodeAt(i);
    if(c<128){ out.push(c); }
    else if(c<2048){ out.push(192|c>>6, 128|c&63); }
    else if(c<55296||c>57343){ out.push(224|c>>12, 128|c>>6&63, 128|c&63); }
    else{ i++;
      var h=(c&1023)<<10 | s.charCodeAt(i)&1023;
      out.push(240|h>>18, 128|h>>12&63, 128|h>>6&63, 128|h&63);
    }
  }
  return out;
}
function d8(a, o, l){
  var s="", i=o, e=o+l, c;
  while(i<e){
    c=a[i++];
    if(c<128){ s+=String.fromCharCode(c); }
    else if(c>191&&c<224){ s+=String.fromCharCode((c&31)<<6|a[i++]&63); }
    else if(c>239){ var u=(c&7)<<18|(a[i++]&63)<<12|(a[i++]&63)<<6|a[i++]&63; s+=String.fromCharCode(55296+(u>>10), 56320+(u&1023)); }
    else{ s+=String.fromCharCode((c&15)<<12|(a[i++]&63)<<6|a[i++]&63); }
  }
  return s;
}

function rotr(x,n){ return (x>>>n)|(x<<(32-n)); }
function sha256hex(msg){
  var K=[0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2];
  var H=[0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19];
  var l=msg.length, n=(((l+8)>>6)+1)<<6, m=new Uint8Array(n), i;
  for(i=0;i<l;i++) m[i]=msg.charCodeAt(i)&255;
  m[l]=0x80;
  var bl=l*8;
  m[n-4]=(bl>>>24)&255; m[n-3]=(bl>>>16)&255; m[n-2]=(bl>>>8)&255; m[n-1]=bl&255;
  for(var blk=0;blk<n;blk+=64){
    var w=new Uint32Array(64);
    for(var t=0;t<16;t++) w[t]=(m[blk+t*4]<<24)|(m[blk+t*4+1]<<16)|(m[blk+t*4+2]<<8)|m[blk+t*4+3];
    for(t=16;t<64;t++){
      var s0=rotr(w[t-15],7)^rotr(w[t-15],18)^(w[t-15]>>>3);
      var s1=rotr(w[t-2],17)^rotr(w[t-2],19)^(w[t-2]>>>10);
      w[t]=(w[t-16]+s0+w[t-7]+s1)|0;
    }
    var a=H[0],b=H[1],c=H[2],d=H[3],e=H[4],f=H[5],g=H[6],h=H[7];
    for(t=0;t<64;t++){
      var S1=rotr(e,6)^rotr(e,11)^rotr(e,25);
      var ch=(e&f)^(~e&g);
      var t1=(h+S1+ch+K[t]+w[t])|0;
      var S0=rotr(a,2)^rotr(a,13)^rotr(a,22);
      var maj=(a&b)^(a&c)^(b&c);
      var t2=(S0+maj)|0;
      h=g;g=f;f=e;e=(d+t1)|0;d=c;c=b;b=a;a=(t1+t2)|0;
    }
    H[0]=(H[0]+a)|0;H[1]=(H[1]+b)|0;H[2]=(H[2]+c)|0;H[3]=(H[3]+d)|0;H[4]=(H[4]+e)|0;H[5]=(H[5]+f)|0;H[6]=(H[6]+g)|0;H[7]=(H[7]+h)|0;
  }
  var out="";
  for(i=0;i<8;i++){ var hv=H[i]>>>0; out+=("00000000"+hv.toString(16)).slice(-8); }
  return out;
}

function subOff(slot){ return SUB_BASE + slot*SUB_REC; }
function putStr(o, s, max){
  var b=u8(s); if(b.length>max) b=b.slice(0,max);
  state.set(b, o);
}
function subWrite(slot, ts, slug, title, genre, players, blurb, submitter, status, beta){
  var o=subOff(slot);
  state.fill(0, o, o+SUB_REC);
  state[o]=status; state[o+1]=beta?1:0; writeU32(o+2, ts);
  putStr(o+6, slug, 48); putStr(o+54, title, 48); putStr(o+102, genre, 24);
  putStr(o+126, players, 12); putStr(o+138, blurb, 160); putStr(o+298, submitter, 24);
}
function subFree(slot){ state.fill(0, subOff(slot), subOff(slot)+SUB_REC); }
function subRead(slot){
  var o=subOff(slot), status=state[o];
  if(!status) return null;
  function f(off,len){ return d8(state, o+off, len).replace(/\u0000+$/,"").trim(); }
  return {
    id:slot, status:status, beta:state[o+1]?1:0, ts:readU32(o+2),
    slug:f(6,48), title:f(54,48), genre:f(102,24),
    players:f(126,12), blurb:f(138,160), submitter:f(298,24)
  };
}
function migrateSubs(){
  var oldRec=321, cap=Math.min(1000, Math.floor((state.length-SUB_BASE)/oldRec));
  for(var i=cap-1;i>=0;i--){
    var o=SUB_BASE+i*oldRec, st=state[o];
    if(!st) continue;
    var slug=d8(state,o+5,48).replace(/\u0000+$/,"").trim();
    var title=d8(state,o+53,48).replace(/\u0000+$/,"").trim();
    var genre=d8(state,o+101,24).replace(/\u0000+$/,"").trim();
    var players=d8(state,o+125,12).replace(/\u0000+$/,"").trim();
    var blurb=d8(state,o+137,160).replace(/\u0000+$/,"").trim();
    var submitter=d8(state,o+297,24).replace(/\u0000+$/,"").trim();
    subWrite(i, readU32(o+1), slug, title, genre, players, blurb, submitter, st, 0);
  }
}

function subRateAllow(conn){
  var now=Date.now(), c=subRateConn[conn.id];
  if(!c || now-c[1]>600000){ subRateConn[conn.id]=[1,now]; }
  else{ if(c[0]>=2) return false; c[0]++; c[1]=now; }
  var key="s"+conn.net[2], n=subRateNet[key];
  if(!n || now-n[1]>3600000){ subRateNet[key]=[1,now]; }
  else{ if(n[0]>=5) return false; n[0]++; n[1]=now; }
  return true;
}

function appendMsg(name, text){
  var wc=readU32(1), rec=5+(wc%MAX_REC)*REC;
  writeU32(rec, nowSec());
  var nb=u8(name); if(nb.length>64) nb=nb.slice(0,64); state[rec+4]=nb.length; state.set(nb, rec+5);
  var tb=u8(text); if(tb.length>512) tb=tb.slice(0,512); state[rec+5+64]=tb.length; state.set(tb, rec+5+64+1);
  writeU32(1, wc+1);
  return wc;
}
function readHistory(maxN){
  var wc=readU32(1), total=wc<MAX_REC?wc:MAX_REC, start=wc-total, out=[];
  var from = total-maxN>0 ? total-maxN : 0;
  for(var k=from;k<total;k++){
    var rec=5+((start+k)%MAX_REC)*REC;
    out.push({i:start+k, ts:readU32(rec), name:d8(state,rec+5,state[rec+4]), text:d8(state,rec+5+64+1,state[rec+5+64])});
  }
  return out;
}

function targetOf(g){ return GAME_MIN[g] || 2; }
function broadcastQueue(game){
  var q=queues[game]||[], names=[];
  for(var i=0;i<q.length;i++) names.push(q[i].name);

function tryMatch(game){
  var target=targetOf(game), q=queues[game];
  if(!q || q.length<target) return;
  var batch=q.splice(0,target), names=[];
  for(var i=0;i<batch.length;i++) names.push(batch[i].name);
  var payload=JSON.stringify({t:"match", game:game, matchId:Date.now().toString(36)+"-"+Math.floor(Math.random()*0xFFFFFF).toString(36), players:names});
  for(var j=0;j<batch.length;j++){ try{ batch[j].conn.send(payload); }catch(e){} }
  broadcastQueue(game);
  tryMatch(game);
}
function rateAllow(conn, kind){
  var now=Date.now();
  if(kind==="net"){
    var key="n"+conn.net[2], b=rateNet[key];
    if(!b||now-b[1]>60000){ rateNet[key]=[1,now]; return true; }
    if(b[0]<30){ b[0]++; return true; } return false;
  }
  var b=rateConn[conn.id];
  if(!b||now-b[1]>15000){ rateConn[conn.id]=[1,now]; return true; }
  if(b[0]<6){ b[0]++; return true; } return false;
}
function sweep(){
  var now=Date.now(); if(now-lastSweep<60000) return; lastSweep=now;
  for(var k in rateConn){ if(now-rateConn[k][1]>30000) delete rateConn[k]; }
  for(var k2 in rateNet){ if(now-rateNet[k2][1]>120000) delete rateNet[k2]; }
  for(var k3 in subRateConn){ if(now-subRateConn[k3][1]>1800000) delete subRateConn[k3]; }
  for(var k4 in subRateNet){ if(now-subRateNet[k4][1]>7200000) delete subRateNet[k4]; }
  for(var k5 in authRate){ if(now-authRate[k5][1]>120000) delete authRate[k5]; }
  for(var k6 in bugRateConn){ if(now-bugRateConn[k6][1]>1800000) delete bugRateConn[k6]; }
  for(var k7 in bugRateNet){ if(now-bugRateNet[k7][1]>7200000) delete bugRateNet[k7]; }
}

self.onmessage = function({conn, data}){
  if(typeof data!=="string") return;
  var m; try{ m=JSON.parse(data); }catch(e){ return; }
  if(!m||!m.t) return;
  sweep();
  if(m.t==="hello"){
    var name=clean(m.name,16)||"Player";
    var had=conns.has(conn.id);
    conns.set(conn.id, {conn:conn, name:name});
    if(!had){ pubsub.publish("lobby", JSON.stringify({t:"sys", text:name+" joined the lobby"})); }
    broadcastPresence();
    return;
  }
  if(m.t==="msg"){
    var info=conns.get(conn.id);
    if(!info) return;
    if(!rateAllow(conn)) return;
    var text=clean(m.text,128);
    if(!text || isBanned(text)) return;
    var wc=appendMsg(info.name, text);
    pubsub.publish("lobby", JSON.stringify({t:"msg", i:wc, ts:nowSec(), name:info.name, text:text}));
    return;
  }
};

self.onclose = function({conn}){
  var info=conns.get(conn.id);
  var name=info?info.name:null;
  conns.delete(conn.id);
  delete rateConn[conn.id];
  for(var g in queues){
    var q=queues[g];
    for(var i=q.length-1;i>=0;i--){ if(q[i].conn.id===conn.id){ q.splice(i,1); } }
    broadcastQueue(g);
  }
  if(name){ pubsub.publish("lobby", JSON.stringify({t:"sys", text:name+" left the lobby"})); }
  broadcastPresence();
};

    var r=subRead(id);
    if(!r || r.status!==SUB_APPROVED) return JSON.stringify({ok:false, err:"not approved"});
    subFree(id);
    publishSubs();
    return JSON.stringify({ok:true});
  },
  reportBug: function({conn}, data){
    if(!rateAllow(conn)) return JSON.stringify({ok:false, err:"You're sending too fast — slow down a moment."});
    if(!bugRateAllow(conn)) return JSON.stringify({ok:false, err:"You've sent a few reports recently — give us a moment to catch up."});
    var d; try{ d=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false, err:"bad request"}); }
    var slug=clean(d.game,48).toLowerCase().replace(/[^a-z0-9-]/g,"");
    if(!/^[a-z0-9][a-z0-9-]{1,47}$/.test(slug)) return JSON.stringify({ok:false, err:"That game name doesn't look right."});
    var desc=clean(d.desc,300);
    if(desc.length<10) return JSON.stringify({ok:false, err:"Tell us a little more — at least a sentence about what happened."});
    var title=clean(d.title,48)||slug, reporter=clean(d.name,24)||"Anonymous";
    if(isBanned(desc+" "+reporter)) return JSON.stringify({ok:false, err:"That text isn't allowed here."});
    var slot=findFreeBug();
    if(slot<0) return JSON.stringify({ok:false, err:"The bug inbox is full right now — try again later."});
    bugWrite(slot, nowSec(), slug, title, desc, reporter, BUG_OPEN);
    publishSubs();
    return JSON.stringify({ok:true, id:slot});
  },
  bugs: function({conn}, data){
    if(!adminIds.has(conn.id)) throw new Error("unauthorized");
    return JSON.stringify({t:"bugs", list:listBugs()});
  },
  resolveBug: function({conn}, data){
    if(!adminIds.has(conn.id)) throw new Error("unauthorized");
    var id=parseInt(String(data),10);
    if(isNaN(id) || id<0 || id>=BUG_LIMIT) return JSON.stringify({ok:false, err:"bad id"});
    var b=bugRead(id);
    if(!b || b.status!==BUG_OPEN) return JSON.stringify({ok:false, err:"not open"});
    bugWrite(id, b.ts, b.slug, b.title, b.desc, b.reporter, BUG_DONE);
    publishSubs();
    return JSON.stringify({ok:true});
  }
};
</script>

    return "#"+(r<16?"0":"")+r.toString(16)+(g<16?"0":"")+g.toString(16)+(b<16?"0":"")+b.toString(16);
  }

  }

      '<div class="admin-item-blurb">'+esc(b.desc)+'</div>'+
      '<div class="admin-item-actions"><button class="admin-act approve" data-act="resolve" data-id="'+b.id+'">Resolve ✓</button></div>'+
    '</div>';
  }
  function refreshPending(){
    if(!socket || socket.readyState!==1 || !adminPw) return;
    socket.rpc.pendingSubs("").then(function(r){
      var m; try{ m=JSON.parse(r); }catch(e){ return; }
      pending=(m&&m.list)||[];
      renderPending();
    }).catch(function(){});
  }
  function renderPending(){
    setAdminCounts();
    if(adminTab!=="pending") return;
    if(!pending.length){ adminList.innerHTML='<div class="admin-empty">Nothing waiting for review. 🎉</div>'; return; }
    adminList.innerHTML=pending.map(function(g){ return adminItemHtml(g,true); }).join("");
  }
  function renderApproved(){
    setAdminCounts();
    if(adminTab!=="approved") return;
    if(!approved.length){ adminList.innerHTML='<div class="admin-empty">No approved community games yet.</div>'; return; }
    adminList.innerHTML=approved.map(function(g){ return adminItemHtml(g,false); }).join("");
  }
  function pjsBlock(g){
    var lines=["  game","    slug = "+g.slug,"    title = "+g.title,"    genre = "+g.genre];
    if(g.players) lines.push("    players = "+g.players);
    lines.push("    blurb = "+g.blurb);
    lines.push("    color = "+autoColor(g.slug));
    lines.push("    icon = "+autoGlyph(g.slug));
    if(g.beta) lines.push("    beta = true");
    return lines.join("\n");
  }
  function adminItemHtml(g, isPending){
    return '<div class="admin-item">'+
      '<div class="admin-item-head">'+
        '<span class="admin-item-title">'+esc(g.title)+'</span>'+
        (g.genre?'<span class="admin-item-genre">'+esc(g.genre)+'</span>':"")+
        (g.beta?'<span class="admin-item-genre beta-chip">Beta</span>':"")+
        '<span class="admin-item-by">by '+esc(g.submitter||"Anonymous")+' · '+fmtDate(g.ts)+'</span>'+
      '</div>'+
      '<div class="admin-item-slug">perchance.org/'+esc(g.slug)+'</div>'+
      '<div class="admin-item-blurb">'+esc(g.blurb)+(g.players?" · "+esc(g.players):"")+'</div>'+
      '<div class="admin-item-actions">'+
        (isPending
          ? '<button class="admin-act approve" data-act="approve" data-id="'+g.id+'">Approve ✓</button>'+
            '<button class="admin-act reject" data-act="reject" data-id="'+g.id+'">Reject ✕</button>'
          : '<button class="admin-act copy" data-act="copy" data-id="'+g.id+'">Copy pjs block</button>'+
            '<button class="admin-act reject" data-act="remove" data-id="'+g.id+'">Remove</button>')+
      '</div>'+
      (isPending?"":'<textarea class="admin-pjs" readonly onclick="this.select()">'+esc(pjsBlock(g))+'</textarea>')+
    '</div>';
  }
  adminList.addEventListener("click",function(e){
    var b=e.target.closest && e.target.closest(".admin-act");
    if(!b) return;
    var id=parseInt(b.dataset.id,10), act=b.dataset.act;
    if(act==="approve"){ socket.rpc.approveSub(String(id)).then(function(){ refreshAdmin(); }).catch(function(){}); }
    else if(act==="reject"){ socket.rpc.rejectSub(String(id)).then(function(){ refreshAdmin(); }).catch(function(){}); }
    else if(act==="remove"){ socket.rpc.removeSub(String(id)).then(function(){ refreshAdmin(); }).catch(function(){}); }
    else if(act==="resolve"){ socket.rpc.resolveBug(String(id)).then(function(){ refreshAdmin(); }).catch(function(){}); }
    else if(act==="copy"){
      var rec=approvedById[id];
      if(rec && navigator.clipboard){ navigator.clipboard.writeText(pjsBlock(rec)).catch(function(){}); }
    }
  });

  connect();
})();

  function makeCardHtml(g, fork, idx, label){

  function render(){
    var rows = GAMES.filter(function(g){
      var okG = state.genre==="All"||g.genre===state.genre;
      var hay = (g.title+" "+g.genre+" "+g.blurb).toLowerCase();
      var okQ = !state.q||hay.indexOf(state.q)!==-1;
      return okG&&okQ;
    });
    var live = rows.filter(function(g){ return !g.comingSoon && !g.beta; });
    var beta = rows.filter(function(g){ return g.beta && !g.comingSoon; });
    var soon = rows.filter(function(g){ return g.comingSoon; });
    var html = "";
    if(state.q){
      html += '<div class="search-note">Search results — <b>'+live.length+'</b> on the shelf · <b>'+beta.length+'</b> in beta'+(soon.length?' · <b>'+soon.length+'</b> coming soon':'')+'</div>';

      var genreOrder = [];
      var genreCount = {};
      live.forEach(function(g){ var gn=g.genre||"Games"; if(genreOrder.indexOf(gn)===-1){ genreOrder.push(gn); genreCount[gn]=0; } genreCount[gn]++; });
      if(state.q) beta.forEach(function(g){ var gn=g.genre||"Games"; if(genreOrder.indexOf(gn)===-1){ genreOrder.push(gn); genreCount[gn]=0; } genreCount[gn]++; });
      genreOrder = genreOrder.map(function(gn,i){ return {gn:gn,i:i}; })
        .sort(function(a,b){ return (genreCount[b.gn]-genreCount[a.gn]) || (a.i-b.i); })
        .map(function(x){ return x.gn; });
      genreOrder.forEach(function(gn){
        var gs = live.filter(function(g){ return (g.genre||"Games")===gn; });
        var bs = state.q ? beta.filter(function(g){ return (g.genre||"Games")===gn; }) : [];
        if(!gs.length && !bs.length) return;
        html += '<div class="shelf"><h3 class="shelf-head">'+esc(gn)+'<span class="count">'+(gs.length+bs.length)+'</span></h3><div class="grid">';

<!-- ═══════════ BGN LOBBY CLIENT — chat + matchmaking UI ═══════════ -->
<script>
(function(){
  "use strict";
  var GAMES = (window.BGN_GAMES || []).filter(function(g){ return !g.comingSoon; });
  function $(id){ return document.getElementById(id); }
  function esc(s){ return String(s).replace(/[&<>"']/g,function(c){return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c];}); }
  function sfx(n){ if(window.BGN&&BGN.sfx)BGN.sfx.play(n); }
  function fmtTime(ts){ var d=new Date(ts*1000); return String(d.getHours()).padStart(2,"0")+":"+String(d.getMinutes()).padStart(2,"0"); }

  var socket=null, queued=null, retryDelay=1000, closedByUs=false, lastSeenIdx=-1, myName="";
  try{ myName=localStorage.getItem("bgn_name")||""; }catch(e){}

  var ui = {
    btn:$("lobbyBtn"), badge:$("onlineBadge"), backdrop:$("lobbyBackdrop"), panel:$("lobbyPanel"),
    closeBtn:$("lobbyCloseBtn"), onlineEl:$("onlineEl"), namesEl:$("onlineNamesEl"),
    msgs:$("msgsEl"), input:$("msgInput"), sendBtn:$("sendBtn"), nameInput:$("nameInput"),
    matchSel:$("matchSel"), findBtn:$("findBtn"), queueEl:$("queueEl"),
    queueLine:$("queueLine"), queueNames:$("queueNames"), cancelBtn:$("cancelBtn"),
    matchBg:$("matchBg"), matchSub:$("matchSub"), matchPlayers:$("matchPlayers"),
    matchOpenBtn:$("matchOpenBtn"), matchLaterBtn:$("matchLaterBtn")
  };

  function status(txt, live){
    ui.onlineEl.innerHTML = '<span class="lobby-dot'+(live?" on":"")+'"></span> '+esc(txt);
  }

  /* ── build game dropdown ── */
  GAMES.forEach(function(g){
    if(!g.matchmaking && /^1\b/.test(g.players||"")) return; // solo games skip matchmaking unless opted in
    var o=document.createElement("option");
    o.value=g.slug; o.textContent=g.title; ui.matchSel.appendChild(o);
  });
  if(!GAMES.length){ ui.findBtn.disabled=true; ui.findBtn.textContent="No live games yet"; }

  /* ── socket lifecycle ── */
  function connect(){
    if(!window.root || !root.createServerSocket){ status("lobby offline"); return; }
    status("connecting…");
    var s=root.createServerSocket();
    s.binaryType="arraybuffer";
    s.addEventListener("open",function(){ onOpen(s); });
    s.addEventListener("message",function(ev){ onMessage(ev.data); });
    s.addEventListener("close",function(ev){ onClose(ev); });
    socket=s;
  }
  function onOpen(s){
    retryDelay=1000;
    s.send(JSON.stringify({t:"hello",name:currentName()}));
    if(queued){ try{ s.rpc.joinQueue(JSON.stringify({game:queued.game,min:2})); }catch(e){} }
    pullHistory();
  }
  function onMessage(data){
    if(typeof data!=="string") return;
    var m; try{ m=JSON.parse(data); }catch(e){ return; }
    if(!m||!m.t) return;
    if(m.t==="msg"){
      if(m.i!==undefined){ if(m.i<=lastSeenIdx) return; lastSeenIdx=m.i; }
      addMsg(m);
      sfx("draw");
    } else if(m.t==="sys"){ addSys(m.text); }
    else if(m.t==="hist"){
      var max=-1;
      (m.msgs||[]).forEach(function(x){ if(x.i>lastSeenIdx){ addMsg(x); if(x.i>max)max=x.i; } });
      if(max>lastSeenIdx) lastSeenIdx=max;
    } else if(m.t==="presence"){ setPresence(m); }
    else if(m.t==="queue"){ updateQueue(m); }
    else if(m.t==="match"){ showMatch(m); }
  }
  function onClose(ev){
    socket=null; status("reconnecting…");
    if(closedByUs || (ev&&ev.code===4403)){ status("lobby offline"); return; }
    setTimeout(connect,retryDelay);
    retryDelay=Math.min(retryDelay*2,15000);
  }
  async function pullHistory(){
    try{ await socket.opened; var r=await socket.rpc.history(""); onMessage(r); }catch(e){}
  }

  function currentName(){
    var v=ui.nameInput.value.trim().slice(0,16);
    if(v&&v!==myName){ myName=v; try{ localStorage.setItem("bgn_name",myName); }catch(e){} }
    return myName||"Player";
  }

  /* ── chat ── */
  function addMsg(m){
    var row=document.createElement("div"); row.className="lobby-msg";
    row.innerHTML='<span class="lobby-msg-name">'+esc(m.name)+'</span><span class="lobby-msg-time">'+fmtTime(m.ts||0)+'</span><div class="lobby-msg-text">'+esc(m.text)+'</div>';
    ui.msgs.appendChild(row);
    while(ui.msgs.children.length>250) ui.msgs.removeChild(ui.msgs.firstChild);
    ui.msgs.scrollTop=ui.msgs.scrollHeight;
  }
  function addSys(text){
    var row=document.createElement("div"); row.className="lobby-sys"; row.textContent=text;
    ui.msgs.appendChild(row);
    while(ui.msgs.children.length>250) ui.msgs.removeChild(ui.msgs.firstChild);
    ui.msgs.scrollTop=ui.msgs.scrollHeight;
  }
  function sendChat(){
    var text=ui.input.value.trim().slice(0,128);
    if(!text) return;
    if(!socket||socket.readyState!==1){ status("reconnecting…"); return; }
    socket.send(JSON.stringify({t:"msg",text:text}));
    ui.input.value=""; ui.input.focus();
  }

  /* ── presence ── */
  function setPresence(m){
    ui.badge.textContent=m.n||0;
    ui.badge.classList.toggle("on",!!m.n);
    status(m.n?m.n+" online":"nobody online",true);
    var names=(m.names||[]).slice(0,9);
    ui.namesEl.textContent=names.length?("Now here: "+names.join(", ")):"";
  }

  /* ── matchmaking ── */
  async function findMatch(){
    if(queued){ leaveQueue(); return; }
    var slug=ui.matchSel.value, title=ui.matchSel.selectedOptions[0].textContent;
    if(!socket||socket.readyState!==1){ status("reconnecting…"); return; }
    try{
      var r=await socket.rpc.joinQueue(JSON.stringify({game:slug,min:2}));
      var m=JSON.parse(r);
      if(m.t==="queued"){ queued={game:slug,title:title}; ui.findBtn.textContent="Waiting…"; ui.queueEl.hidden=false; updateQueue(m); }
      else ui.findBtn.textContent="Find a Match";
    }catch(e){}
  }
  function leaveQueue(){
    if(!queued) return;
    var g=queued.game; queued=null;
    ui.findBtn.textContent="Find a Match"; ui.queueEl.hidden=true;
    if(socket&&socket.readyState===1){ try{ socket.rpc.leaveQueue(JSON.stringify({game:g})); }catch(e){} }
  }
  function updateQueue(m){
    if(!queued||m.game!==queued.game) return;
    var need=Math.max(0,m.target-m.count);
    ui.queueLine.textContent = need===0 ? "Party's full — opening now…" : (need===1 ? "1 more player needed" : need+" more players needed");
    ui.queueNames.textContent = m.names.length?("In queue: "+m.names.join(", ")):"";
  }

  /* ── match modal ── */
  function showMatch(m){
    var g=GAMES.filter(function(x){return x.slug===m.game;})[0];
    var title=g?g.title:m.game;
    var others=(m.players||[]).filter(function(p){return p!==currentName();});
    ui.matchSub.textContent="A table opened up for "+title+".";
    ui.matchPlayers.textContent=others.length?("Sitting across: "+others.join(", ")):"";
    sfx("tada");
    ui.matchOpenBtn.href="https://perchance.org/"+encodeURIComponent(m.game);
    ui.matchBg.hidden=false;
    leaveQueue();
  }

  /* ── wiring ── */
  ui.btn.addEventListener("click",function(){
    var open=ui.panel.hidden; ui.panel.hidden=!open; ui.backdrop.hidden=!open;
    if(open){ ui.input.focus(); if(ui.nameInput.value==="") ui.nameInput.focus(); }
  });
  ui.backdrop.addEventListener("click",function(){ ui.panel.hidden=true; ui.backdrop.hidden=true; });
  ui.closeBtn.addEventListener("click",function(){ ui.panel.hidden=true; ui.backdrop.hidden=true; });
  ui.sendBtn.addEventListener("click",sendChat);
  ui.input.addEventListener("keydown",function(e){ if(e.key==="Enter") sendChat(); });
  ui.nameInput.addEventListener("change",function(){
    currentName();
    if(socket&&socket.readyState===1) socket.send(JSON.stringify({t:"hello",name:currentName()}));
  });
  ui.nameInput.value=myName;
  ui.findBtn.addEventListener("click",findMatch);
  ui.cancelBtn.addEventListener("click",leaveQueue);
  ui.matchOpenBtn.addEventListener("click",function(){ ui.matchBg.hidden=true; ui.panel.hidden=true; ui.backdrop.hidden=true; });
  ui.matchLaterBtn.addEventListener("click",function(){ ui.matchBg.hidden=true; });
  ui.matchBg.addEventListener("click",function(e){ if(e.target===ui.matchBg) ui.matchBg.hidden=true; });

  var tabs=document.querySelectorAll(".lobby-tab");
  Array.prototype.forEach.call(tabs,function(t){
    t.addEventListener("click",function(){
      Array.prototype.forEach.call(tabs,function(x){ x.classList.toggle("on",x===t); });
      Array.prototype.forEach.call(document.querySelectorAll(".lobby-view"),function(v){ v.hidden=v.dataset.view!==t.dataset.tab; });
    });
  });

  connect();
})();
</script>
