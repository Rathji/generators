var KINGDOM_COUNT=10, HAND_SIZE=5, START_COPPERS=7, START_ESTATES=3;
var POOL, POOL_BY_NAME;

function setPool(arr){
  POOL=[]; POOL_BY_NAME={};
  for(var i=0;i<arr.length;i++){
    var t=arr[i], o={};
    o.name=String(t.name||"");
    o.cost=Math.max(0,Math.floor(t.cost)||0);
    o.emoji=String(t.emoji||"");
    o.coin=Math.floor(t.coin)||0;
    o.vp=Math.floor(t.vp)||0;
    var ks=(typeof t.kinds==="string")?t.kinds.split(","):(Array.isArray(t.kinds)?t.kinds:[]);
    o.kinds=[];
    for(var k=0;k<ks.length;k++){ var s=String(ks[k]).trim(); if(s)o.kinds.push(s); }
    POOL.push(o); POOL_BY_NAME[o.name]=o;
  }
}
function cardInfo(n){ return BASE_CARDS[n]||POOL_BY_NAME[n]||null; }
function isAction(n){ var c=cardInfo(n); return !!(c&&c.kinds&&c.kinds.indexOf("action")>=0); }
function isTreasure(n){ var c=cardInfo(n); return !!(c&&c.kinds&&c.kinds.indexOf("treasure")>=0); }
function isVictory(n){ var c=cardInfo(n); return !!(c&&c.kinds&&c.kinds.indexOf("victory")>=0); }
function isCurse(n){ return n==="Curse"; }
function keepVal(c){ var ci=cardInfo(c); if(!ci)return 0; return (ci.vp||0)+(isTreasure(c)?(ci.coin||0)*0.5:0); }
function shuffle(a){ for(var i=a.length-1;i>0;i--){ var j=Math.floor(Math.random()*(i+1)); var t=a[i];a[i]=a[j];a[j]=t; } return a; }
function pickKingdom(rng){
  var names=[]; for(var i=0;i<POOL.length;i++)names.push(POOL[i].name);
  var r=rng||Math.random;
  for(var j=names.length-1;j>0;j--){ var k=Math.floor(r()*(j+1)); var t=names[j];names[j]=names[k];names[k]=t; }
  return names.slice(0,KINGDOM_COUNT);
}
function basePileCount(n){
  if(n==="Copper")return 46; if(n==="Silver")return 40; if(n==="Gold")return 30;

function makeSupply(kingdom){
  var s={};
  var bases=["Copper","Silver","Gold","Estate","Duchy","Province","Curse"];
  for(var i=0;i<bases.length;i++)s[bases[i]]=basePileCount(bases[i]);

}
function pileOrder(){ return ["Copper","Silver","Gold","Estate","Duchy","Province","Curse"].concat(POOL.map(function(c){return c.name;})); }
function current(g){ return g.turnIndex%2; }
function draw(g,p,n){
  var got=0;
  while(n>0){
    var pl=g.players[p];
    if(!pl.deck.length){
      if(!pl.discard.length)break;
      shuffle(pl.discard);
      pl.deck=pl.discard; pl.discard=[];
    }
    pl.hand.push(pl.deck.pop());
    n--; got++;
  }
  return got;
}
function gain(g,p,name){
  if(g.supply[name]>0){ g.supply[name]--; g.players[p].discard.push(name); return true; }
  return false;
}
function gainable(g,name,maxCost){
  if(g.supply[name]==null||g.supply[name]<=0)return false;
  var c=cardInfo(name); return !!(c&&c.cost<=maxCost);
}
function idxList(cards,len){
  if(!cards||!cards.length)return [];
  if(!Array.isArray(cards))return null;
  var seen={},out=[];
  for(var i=0;i<cards.length;i++){
    var v=cards[i];
    if(typeof v!=="number"||v<0||v>=len||seen[v])return null;
    seen[v]=1; out.push(v);
  }
  return out.sort(function(a,b){return a-b;});
}
function removeIdx(arr,idxs){
  var out=arr.slice();
  for(var i=idxs.length-1;i>=0;i--)out.splice(idxs[i],1);
  return out;
}
function actionIndices(g,p){ var pl=g.players[p],out=[]; for(var i=0;i<pl.hand.length;i++)if(isAction(pl.hand[i]))out.push(i); return out; }
function treasureIndices(g,p){ var pl=g.players[p],out=[]; for(var i=0;i<pl.hand.length;i++)if(isTreasure(pl.hand[i]))out.push(i); return out; }
function victoryIndices(g,p){ var pl=g.players[p],out=[]; for(var i=0;i<pl.hand.length;i++)if(isVictory(pl.hand[i]))out.push(i); return out; }
function score(g,p){ var pl=g.players[p],v=0; var zones=[pl.hand,pl.deck,pl.discard,pl.inPlay]; for(var z=0;z<zones.length;z++){ for(var i=0;i<zones[z].length;i++){ var c=cardInfo(zones[z][i]); if(c)v+=(c.vp||0); } } return v; }
function emptyPiles(g){ var c=0; for(var k in g.supply){ if(g.supply[k]<=0)c++; } return c; }
function ev(g,p,txt){ return {p:p,txt:txt}; }
function checkOver(g){
  if(g.over)return true;
  if(g.supply["Province"]<=0||emptyPiles(g)>=3){
    g.over=true; g.reason="supply";
    var s0=score(g,0),s1=score(g,1);
    g.scores=[s0,s1];
    g.winner=s0>s1?0:(s1>s0?1:"draw");
    g.events.push(ev(g,g.winner==="draw"?-1:g.winner,"Game over — "+(g.winner==="draw"?("tie "+s0+"–"+s1):(s0+"–"+s1))+"."));
    return true;
  }
  return false;
}
function pendingFor(g,p){ for(var i=0;i<g.pending.length;i++){ if(g.pending[i].player===p)return g.pending[i]; } return null; }
function pushPrompt(g,p,pd){
  g.pid=(g.pid||0)+1;
  var pr={player:p,type:pd.type,data:pd.data||{},id:g.pid};
  g.pending.push(pr);
  return pr;
}
function resolveAutoPrompts(g,p){
  if(!AUTO_RESOLVE)return;
  for(;;){
    var pr=pendingFor(g,p);
    if(!pr)break;
    if(pr.type==="militia"){
      var pl=g.players[p], h=pl.hand;
      if(h.length>3){
        var idxs=h.map(function(c,i){return i;}).sort(function(a,b){return keepVal(h[b])-keepVal(h[a]);});
        var cards=idxs.slice(3);
        for(var i=0;i<cards.length;i++)pl.discard.push(h[cards[i]]);
        pl.hand=removeIdx(pl.hand,cards);
        g.events.push(ev(g,p,"Discarded down to 3 (auto)."));
      }
    } else if(pr.type==="bureaucrat"){
      var vi=victoryIndices(g,p);
      if(vi.length){
        g.players[p].deck.push(g.players[p].hand[vi[0]]);
        g.players[p].hand=removeIdx(g.players[p].hand,[vi[0]]);
        g.events.push(ev(g,p,"Revealed a Victory card (auto)."));
      }
    }
    g.pending=g.pending.filter(function(x){return x.id!==pr.id;});
  }
}
function cleanupPlayer(g,p){
  var pl=g.players[p];
  for(var i=0;i<pl.hand.length;i++)pl.discard.push(pl.hand[i]);
  pl.hand=[];
  for(var j=0;j<pl.inPlay.length;j++)pl.discard.push(pl.inPlay[j]);
  pl.inPlay=[];
  draw(g,p,HAND_SIZE);
}
function startTurn(g,p){
  g.phase="action";
  g.t={actions:1,buys:1,coins:0};
  resolveAutoPrompts(g,p);
  g.events.push(ev(g,p,"Turn "+(Math.floor(g.turnIndex/2)+1)+" — drew 5."));
}
function newGame(kingdom, firstMover){
  var g={
    kingdom:kingdom.slice(),
    supply:makeSupply(kingdom),
    players:[{hand:[],deck:[],discard:[],inPlay:[]},{hand:[],deck:[],discard:[],inPlay:[]}],
    trash:[], turnIndex:0, phase:"action",
    t:{actions:1,buys:1,coins:0},
    pending:[], pid:0, over:false, winner:null, reason:"", scores:null,
    seq:0, events:[], lastAct:0
  };
  for(var p=0;p<2;p++){
    var d=[];
    for(var i=0;i<START_COPPERS;i++)d.push("Copper");
    for(var j=0;j<START_ESTATES;j++)d.push("Estate");
    shuffle(d);
    g.players[p].deck=d;
  }
  draw(g,0,HAND_SIZE);
  draw(g,1,HAND_SIZE);
  g.turnIndex=firstMover||0;
  startTurn(g,current(g));
  return g;
}
function resolveAction(g,p,name){
  var t=g.t;
  if(name==="Smithy"){ var got=draw(g,p,3); g.events.push(ev(g,p,"Smithy: drew "+got+".")); }
  else if(name==="Village"){ draw(g,p,1); t.actions+=2; g.events.push(ev(g,p,"Village: +1 card, +2 Actions.")); }
  else if(name==="Market"){ draw(g,p,1); t.actions+=1; t.buys+=1; t.coins+=1; g.events.push(ev(g,p,"Market: +1 card, +1 Action, +1 Buy, +$1.")); }
  else if(name==="Moat"){ var g2=draw(g,p,2); g.events.push(ev(g,p,"Moat: drew "+g2+".")); }
  else if(name==="Council Room"){ draw(g,p,4); t.buys+=1; var og=draw(g,1-p,1); g.events.push(ev(g,p,"Council Room: +4 cards, +1 Buy; opponent drew "+og+".")); }
  else if(name==="Militia"){ t.coins+=2; g.events.push(ev(g,p,"Militia: +$2; each other player discards to 3.")); if(g.players[1-p].hand.length>3)return pushPrompt(g,1-p,{type:"militia"}); }
  else if(name==="Witch"){ draw(g,p,2); var gd=gain(g,1-p,"Curse"); g.events.push(ev(g,p,"Witch: drew 2; opponent gained a Curse"+(gd?"":" (pile empty)")+".")); }
  else if(name==="Bureaucrat"){ var gs=gain(g,p,"Silver"); if(gs)g.players[p].deck.push("Silver"); g.events.push(ev(g,p,"Bureaucrat: Silver on top of your deck; opponent reveals a Victory card.")); return pushPrompt(g,1-p,{type:"bureaucrat"}); }
  else if(name==="Cellar"){ t.actions+=1; return pushPrompt(g,p,{type:"cellar"}); }
  else if(name==="Chapel"){ return pushPrompt(g,p,{type:"trashN",data:{max:4}}); }
  else if(name==="Moneylender"){ return pushPrompt(g,p,{type:"moneylender"}); }
  else if(name==="Workshop"){ return pushPrompt(g,p,{type:"gain",data:{maxCost:5}}); }
  else if(name==="Feast"){ var ip=g.players[p].inPlay; var fi=ip.lastIndexOf("Feast"); if(fi>=0)ip.splice(fi,1); g.trash.push("Feast"); return pushPrompt(g,p,{type:"gain",data:{maxCost:5}}); }
  else if(name==="Mine"){ if(!treasureIndices(g,p).length){ g.events.push(ev(g,p,"Mine: no Treasure in hand.")); } else return pushPrompt(g,p,{type:"mineStep1"}); }
  else if(name==="Remodel"){ if(!g.players[p].hand.length){ g.events.push(ev(g,p,"Remodel: no cards in hand.")); } else return pushPrompt(g,p,{type:"remodelStep1"}); }
  else if(name==="Throne Room"){ if(!actionIndices(g,p).length){ g.events.push(ev(g,p,"Throne Room: no Action in hand to double.")); } else return pushPrompt(g,p,{type:"throne"}); }
  return null;
}
function playCard(g,p,idx){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="action")return {ok:false,err:"not_action_phase"};
  if(g.t.actions<=0)return {ok:false,err:"no_actions"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  var pl=g.players[p];
  if(idx==null||idx<0||idx>=pl.hand.length)return {ok:false,err:"bad_card"};
  var name=pl.hand[idx];
  if(!isAction(name))return {ok:false,err:"not_action"};
  pl.hand.splice(idx,1);
  pl.inPlay.push(name);
  g.t.actions--;
  g.seq++;
  g.events.push(ev(g,p,"Played "+name+"."));
  var pr=resolveAction(g,p,name);
  checkOver(g);
  return {ok:true,prompt:pr,over:g.over};
}
function respondPrompt(g,p,req){
  if(g.over)return {ok:false,err:"over"};
  var pr=pendingFor(g,p);
  if(!pr)return {ok:false,err:"no_prompt"};
  if(req.id!==pr.id)return {ok:false,err:"bad_prompt"};
  var resp=req.resp||{};
  var pl=g.players[p];
  var newPrompt=null;
  if(pr.type==="cellar"){
    var cd=idxList(resp.cards,pl.hand.length);
    if(cd==null)return {ok:false,err:"bad_resp"};
    for(var a=0;a<cd.length;a++)pl.discard.push(pl.hand[cd[a]]);
    pl.hand=removeIdx(pl.hand,cd);
    var cg=draw(g,p,cd.length);
    g.events.push(ev(g,p,"Cellar: discarded "+cd.length+", drew "+cg+"."));
  } else if(pr.type==="trashN"){
    var tr=idxList(resp.cards,pl.hand.length);
    if(tr==null||tr.length>(pr.data.max||4))return {ok:false,err:"bad_resp"};
    for(var b=0;b<tr.length;b++)g.trash.push(pl.hand[tr[b]]);
    pl.hand=removeIdx(pl.hand,tr);
    g.events.push(ev(g,p,"Chapel: trashed "+(tr.length?tr.length:"no")+" card"+(tr.length===1?"":"s")+"."));
  } else if(pr.type==="moneylender"){
    if(resp.skip){ g.events.push(ev(g,p,"Moneylender: kept your Coppers.")); }
    else{
      var ml=idxList(resp.cards,pl.hand.length);
      if(ml==null||ml.length!==1||pl.hand[ml[0]]!=="Copper")return {ok:false,err:"bad_resp"};
      g.trash.push("Copper"); pl.hand=removeIdx(pl.hand,ml); g.t.coins+=3;
      g.events.push(ev(g,p,"Moneylender: trashed a Copper, +$3."));
    }
  } else if(pr.type==="gain"){
    var gp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!gainable(g,gp,pr.data.maxCost||5))return {ok:false,err:"bad_pile"};
      gain(g,p,gp);
      g.events.push(ev(g,p,"Gained "+gp+"."));
    }
  } else if(pr.type==="remodelStep1"){
    var rs=idxList(resp.cards,pl.hand.length);
    if(rs==null||rs.length!==1)return {ok:false,err:"bad_resp"};
    var rm=pl.hand[rs[0]];
    g.trash.push(rm); pl.hand=removeIdx(pl.hand,rs);
    g.events.push(ev(g,p,"Remodel: trashed "+rm+"."));
    newPrompt=pushPrompt(g,p,{type:"remodelStep2",data:{maxCost:(cardInfo(rm).cost||0)+2}});
  } else if(pr.type==="remodelStep2"){
    var rp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!gainable(g,rp,pr.data.maxCost||2))return {ok:false,err:"bad_pile"};
      gain(g,p,rp);
      g.events.push(ev(g,p,"Gained "+rp+"."));
    }
  } else if(pr.type==="mineStep1"){
    var ms=idxList(resp.cards,pl.hand.length);
    if(ms==null||ms.length!==1||!isTreasure(pl.hand[ms[0]]))return {ok:false,err:"bad_resp"};
    var mt=pl.hand[ms[0]];
    g.trash.push(mt); pl.hand=removeIdx(pl.hand,ms);
    g.events.push(ev(g,p,"Mine: trashed "+mt+"."));
    newPrompt=pushPrompt(g,p,{type:"mineStep2",data:{maxCost:(cardInfo(mt).cost||0)+3}});
  } else if(pr.type==="mineStep2"){
    var mp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!isTreasure(mp)||!gainable(g,mp,pr.data.maxCost||3))return {ok:false,err:"bad_pile"};
      gain(g,p,mp);
      g.events.push(ev(g,p,"Gained "+mp+"."));
    }
  } else if(pr.type==="throne"){
    var th=idxList(resp.cards,pl.hand.length);
    if(th==null||th.length!==1||!isAction(pl.hand[th[0]]))return {ok:false,err:"bad_resp"};
    var tn=pl.hand[th[0]];
    pl.hand=removeIdx(pl.hand,th);
    pl.inPlay.push(tn);
    var r1=resolveAction(g,p,tn);
    if(r1){ newPrompt=r1; newPrompt.data.doubleNext=tn; }
    else{
      var r2=resolveAction(g,p,tn);
      if(r2){ newPrompt=r2; newPrompt.data.doubleNext=tn; }
    }
    g.events.push(ev(g,p,"Throne Room: played "+tn+" twice."));
  } else if(pr.type==="militia"){
    var h=pl.hand;
    if(resp.moat){
      if(h.indexOf("Moat")<0)return {ok:false,err:"bad_resp"};
      g.events.push(ev(g,p,"Revealed Moat — immune to Militia."));
    }else{
      var md=idxList(resp.cards,h.length);
      if(md==null||h.length-md.length>3)return {ok:false,err:"bad_resp"};
      for(var e=0;e<md.length;e++)pl.discard.push(h[md[e]]);
      pl.hand=removeIdx(pl.hand,md);
      g.events.push(ev(g,p,"Militia: discarded to "+pl.hand.length+" cards."));
    }
  } else if(pr.type==="bureaucrat"){
    var hh=pl.hand;
    if(resp.none){
      if(victoryIndices(g,p).length)return {ok:false,err:"bad_resp"};
      g.events.push(ev(g,p,"No Victory card to reveal."));
    }else{
      var br=idxList(resp.cards,hh.length);
      if(br==null||br.length!==1||!isVictory(hh[br[0]]))return {ok:false,err:"bad_resp"};
      var bv=hh[br[0]];
      pl.hand=removeIdx(pl.hand,br);
      pl.deck.push(bv);
      g.events.push(ev(g,p,"Bureaucrat: revealed "+bv+", put on top of deck."));
    }
  } else return {ok:false,err:"bad_prompt"};
  g.pending=g.pending.filter(function(x){return x.id!==pr.id;});
  g.seq++;
  var dn=pr.data&&pr.data.doubleNext;
  if(dn&&!newPrompt){
    var r3=resolveAction(g,p,dn);
    if(r3){ newPrompt=r3; newPrompt.data.doubleNext=dn; }
  }
  checkOver(g);
  return {ok:true,prompt:newPrompt,over:g.over};
}
function endActions(g,p){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="action")return {ok:false,err:"bad_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  var pl=g.players[p], treas=[], keep=[];
  for(var i=0;i<pl.hand.length;i++){ if(isTreasure(pl.hand[i])){ treas.push(pl.hand[i]); pl.inPlay.push(pl.hand[i]); } else keep.push(pl.hand[i]); }
  pl.hand=keep;
  for(var j=0;j<treas.length;j++)g.t.coins+=cardInfo(treas[j]).coin;
  g.phase="buy";
  g.seq++;
  g.events.push(ev(g,p,"Buy phase — auto-played "+treas.length+" Treasure"+(treas.length===1?"":"s")+"."));
  return {ok:true,over:g.over};
}
function buyCard(g,p,pile){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="buy")return {ok:false,err:"not_buy_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  if(g.t.buys<=0)return {ok:false,err:"no_buys"};
  if(g.supply[pile]==null||g.supply[pile]<=0)return {ok:false,err:"pile_empty"};
  var c=cardInfo(pile); if(!c)return {ok:false,err:"bad_pile"};
  if(g.t.coins<c.cost)return {ok:false,err:"too_expensive"};
  g.t.buys--; g.t.coins-=c.cost;
  gain(g,p,pile);
  g.seq++;
  g.events.push(ev(g,p,"Bought "+pile+"."));
  checkOver(g);
  return {ok:true,over:g.over};
}
function endTurn(g,p){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="buy")return {ok:false,err:"not_buy_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  cleanupPlayer(g,p);
  g.events.push(ev(g,p,"Turn ended."));
  g.turnIndex++;
  g.seq++;
  startTurn(g,current(g));
  checkOver(g);
  return {ok:true,over:g.over};
}
function canPlay(g,p,idx){
  return !g.over&&current(g)===p&&g.phase==="action"&&g.t.actions>0&&!pendingFor(g,p)&&idx!=null&&idx>=0&&idx<g.players[p].hand.length&&isAction(g.players[p].hand[idx]);
}
function canBuyCard(g,p,pile){
  if(g.over||current(g)!==p||g.phase!=="buy"||pendingFor(g,p)||g.t.buys<=0)return false;
  if(g.supply[pile]==null||g.supply[pile]<=0)return false;
  var c=cardInfo(pile); return !!(c&&g.t.coins>=c.cost);
}
function drainEvents(g){
  var out=g.events.slice();
  g.events.length=0;
  return out;
}

function loadRooms(){

    const r=loadRoom(i);

  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}

    if(now-(lbNets.get(conn.net[3])||0)<6000)return JSON.stringify({ok:false,err:"rate"});
    lbNets.set(conn.net[3],now);
    const name=cleanField(p.name,24)||"Player";
    const score=Math.max(0,Math.min(0xffffffff,Math.floor(Number(p.score)||0)));
    const extra=cleanField(p.extra,11);
    lbSubmit(name,score,Math.floor(now/1000),extra,false);
    return JSON.stringify({ok:true});
  },
  getScores({conn},data){
    lbInit();
    const a=lbRead().sort(function(x,y){ return lbBetter(false,x,y)?-1:1; }).slice(0,10);
    return JSON.stringify({scores:a});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  chatTimes.delete(conn.id);
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

function resolveAutoPrompts(g,p){

function resolveAction(g,p,name){
  var t=g.t;
  if(name==="Smithy"){ var got=draw(g,p,3); g.events.push(ev(g,p,"Smithy: drew "+got+".")); }
  else if(name==="Village"){ draw(g,p,1); t.actions+=2; g.events.push(ev(g,p,"Village: +1 card, +2 Actions.")); }
  else if(name==="Market"){ draw(g,p,1); t.actions+=1; t.buys+=1; t.coins+=1; g.events.push(ev(g,p,"Market: +1 card, +1 Action, +1 Buy, +$1.")); }
  else if(name==="Moat"){ var g2=draw(g,p,2); g.events.push(ev(g,p,"Moat: drew "+g2+".")); }
  else if(name==="Council Room"){ draw(g,p,4); t.buys+=1; var og=draw(g,1-p,1); g.events.push(ev(g,p,"Council Room: +4 cards, +1 Buy; opponent drew "+og+".")); }
  else if(name==="Militia"){ t.coins+=2; g.events.push(ev(g,p,"Militia: +$2; each other player discards to 3.")); if(g.players[1-p].hand.length>3)return pushPrompt(g,1-p,{type:"militia"}); }
  else if(name==="Witch"){ draw(g,p,2); var gd=gain(g,1-p,"Curse"); g.events.push(ev(g,p,"Witch: drew 2; opponent gained a Curse"+(gd?"":" (pile empty)")+".")); }
  else if(name==="Bureaucrat"){ var gs=gain(g,p,"Silver"); if(gs)g.players[p].deck.push("Silver"); g.events.push(ev(g,p,"Bureaucrat: Silver on top of your deck; opponent reveals a Victory card.")); return pushPrompt(g,1-p,{type:"bureaucrat"}); }
  else if(name==="Cellar"){ t.actions+=1; return pushPrompt(g,p,{type:"cellar"}); }
  else if(name==="Chapel"){ return pushPrompt(g,p,{type:"trashN",data:{max:4}}); }
  else if(name==="Moneylender"){ return pushPrompt(g,p,{type:"moneylender"}); }
  else if(name==="Workshop"){ return pushPrompt(g,p,{type:"gain",data:{maxCost:5}}); }
  else if(name==="Feast"){ var ip=g.players[p].inPlay; var fi=ip.lastIndexOf("Feast"); if(fi>=0)ip.splice(fi,1); g.trash.push("Feast"); return pushPrompt(g,p,{type:"gain",data:{maxCost:5}}); }
  else if(name==="Mine"){ if(!treasureIndices(g,p).length){ g.events.push(ev(g,p,"Mine: no Treasure in hand.")); } else return pushPrompt(g,p,{type:"mineStep1"}); }
  else if(name==="Remodel"){ if(!g.players[p].hand.length){ g.events.push(ev(g,p,"Remodel: no cards in hand.")); } else return pushPrompt(g,p,{type:"remodelStep1"}); }
  else if(name==="Throne Room"){ if(!actionIndices(g,p).length){ g.events.push(ev(g,p,"Throne Room: no Action in hand to double.")); } else return pushPrompt(g,p,{type:"throne"}); }
  return null;
}
function playCard(g,p,idx){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="action")return {ok:false,err:"not_action_phase"};
  if(g.t.actions<=0)return {ok:false,err:"no_actions"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  var pl=g.players[p];
  if(idx==null||idx<0||idx>=pl.hand.length)return {ok:false,err:"bad_card"};
  var name=pl.hand[idx];
  if(!isAction(name))return {ok:false,err:"not_action"};
  pl.hand.splice(idx,1);
  pl.inPlay.push(name);
  g.t.actions--;
  g.seq++;
  g.events.push(ev(g,p,"Played "+name+"."));
  var pr=resolveAction(g,p,name);
  checkOver(g);
  return {ok:true,prompt:pr,over:g.over};
}
function respondPrompt(g,p,req){
  if(g.over)return {ok:false,err:"over"};
  var pr=pendingFor(g,p);
  if(!pr)return {ok:false,err:"no_prompt"};
  if(req.id!==pr.id)return {ok:false,err:"bad_prompt"};
  var resp=req.resp||{};
  var pl=g.players[p];
  var newPrompt=null;
  if(pr.type==="cellar"){
    var cd=idxList(resp.cards,pl.hand.length);
    if(cd==null)return {ok:false,err:"bad_resp"};
    for(var a=0;a<cd.length;a++)pl.discard.push(pl.hand[cd[a]]);
    pl.hand=removeIdx(pl.hand,cd);
    var cg=draw(g,p,cd.length);
    g.events.push(ev(g,p,"Cellar: discarded "+cd.length+", drew "+cg+"."));
  } else if(pr.type==="trashN"){
    var tr=idxList(resp.cards,pl.hand.length);
    if(tr==null||tr.length>(pr.data.max||4))return {ok:false,err:"bad_resp"};
    for(var b=0;b<tr.length;b++)g.trash.push(pl.hand[tr[b]]);
    pl.hand=removeIdx(pl.hand,tr);
    g.events.push(ev(g,p,"Chapel: trashed "+(tr.length?tr.length:"no")+" card"+(tr.length===1?"":"s")+"."));
  } else if(pr.type==="moneylender"){
    if(resp.skip){ g.events.push(ev(g,p,"Moneylender: kept your Coppers.")); }
    else{
      var ml=idxList(resp.cards,pl.hand.length);
      if(ml==null||ml.length!==1||pl.hand[ml[0]]!=="Copper")return {ok:false,err:"bad_resp"};
      g.trash.push("Copper"); pl.hand=removeIdx(pl.hand,ml); g.t.coins+=3;
      g.events.push(ev(g,p,"Moneylender: trashed a Copper, +$3."));
    }
  } else if(pr.type==="gain"){
    var gp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!gainable(g,gp,pr.data.maxCost||5))return {ok:false,err:"bad_pile"};
      gain(g,p,gp);
      g.events.push(ev(g,p,"Gained "+gp+"."));
    }
  } else if(pr.type==="remodelStep1"){
    var rs=idxList(resp.cards,pl.hand.length);
    if(rs==null||rs.length!==1)return {ok:false,err:"bad_resp"};
    var rm=pl.hand[rs[0]];
    g.trash.push(rm); pl.hand=removeIdx(pl.hand,rs);
    g.events.push(ev(g,p,"Remodel: trashed "+rm+"."));
    newPrompt=pushPrompt(g,p,{type:"remodelStep2",data:{maxCost:(cardInfo(rm).cost||0)+2}});
  } else if(pr.type==="remodelStep2"){
    var rp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!gainable(g,rp,pr.data.maxCost||2))return {ok:false,err:"bad_pile"};
      gain(g,p,rp);
      g.events.push(ev(g,p,"Gained "+rp+"."));
    }
  } else if(pr.type==="mineStep1"){
    var ms=idxList(resp.cards,pl.hand.length);
    if(ms==null||ms.length!==1||!isTreasure(pl.hand[ms[0]]))return {ok:false,err:"bad_resp"};
    var mt=pl.hand[ms[0]];
    g.trash.push(mt); pl.hand=removeIdx(pl.hand,ms);
    g.events.push(ev(g,p,"Mine: trashed "+mt+"."));
    newPrompt=pushPrompt(g,p,{type:"mineStep2",data:{maxCost:(cardInfo(mt).cost||0)+3}});
  } else if(pr.type==="mineStep2"){
    var mp=String(resp.pile||"");
    if(resp.skip){ g.events.push(ev(g,p,"Gained nothing.")); }
    else{
      if(!isTreasure(mp)||!gainable(g,mp,pr.data.maxCost||3))return {ok:false,err:"bad_pile"};
      gain(g,p,mp);
      g.events.push(ev(g,p,"Gained "+mp+"."));
    }
  } else if(pr.type==="throne"){
    var th=idxList(resp.cards,pl.hand.length);
    if(th==null||th.length!==1||!isAction(pl.hand[th[0]]))return {ok:false,err:"bad_resp"};
    var tn=pl.hand[th[0]];
    pl.hand=removeIdx(pl.hand,th);
    pl.inPlay.push(tn);
    var r1=resolveAction(g,p,tn);
    if(r1){ newPrompt=r1; newPrompt.data.doubleNext=tn; }
    else{
      var r2=resolveAction(g,p,tn);
      if(r2){ newPrompt=r2; newPrompt.data.doubleNext=tn; }
    }
    g.events.push(ev(g,p,"Throne Room: played "+tn+" twice."));
  } else if(pr.type==="militia"){
    var h=pl.hand;
    if(resp.moat){
      if(h.indexOf("Moat")<0)return {ok:false,err:"bad_resp"};
      g.events.push(ev(g,p,"Revealed Moat — immune to Militia."));
    }else{
      var md=idxList(resp.cards,h.length);
      if(md==null||h.length-md.length>3)return {ok:false,err:"bad_resp"};
      for(var e=0;e<md.length;e++)pl.discard.push(h[md[e]]);
      pl.hand=removeIdx(pl.hand,md);
      g.events.push(ev(g,p,"Militia: discarded to "+pl.hand.length+" cards."));
    }
  } else if(pr.type==="bureaucrat"){
    var hh=pl.hand;
    if(resp.none){
      if(victoryIndices(g,p).length)return {ok:false,err:"bad_resp"};
      g.events.push(ev(g,p,"No Victory card to reveal."));
    }else{
      var br=idxList(resp.cards,hh.length);
      if(br==null||br.length!==1||!isVictory(hh[br[0]]))return {ok:false,err:"bad_resp"};
      var bv=hh[br[0]];
      pl.hand=removeIdx(pl.hand,br);
      pl.deck.push(bv);
      g.events.push(ev(g,p,"Bureaucrat: revealed "+bv+", put on top of deck."));
    }
  } else return {ok:false,err:"bad_prompt"};
  g.pending=g.pending.filter(function(x){return x.id!==pr.id;});
  g.seq++;
  var dn=pr.data&&pr.data.doubleNext;
  if(dn&&!newPrompt){
    var r3=resolveAction(g,p,dn);
    if(r3){ newPrompt=r3; newPrompt.data.doubleNext=dn; }
  }
  checkOver(g);
  return {ok:true,prompt:newPrompt,over:g.over};
}
function endActions(g,p){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="action")return {ok:false,err:"bad_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  var pl=g.players[p], treas=[], keep=[];
  for(var i=0;i<pl.hand.length;i++){ if(isTreasure(pl.hand[i])){ treas.push(pl.hand[i]); pl.inPlay.push(pl.hand[i]); } else keep.push(pl.hand[i]); }
  pl.hand=keep;
  for(var j=0;j<treas.length;j++)g.t.coins+=cardInfo(treas[j]).coin;
  g.phase="buy";
  g.seq++;
  g.events.push(ev(g,p,"Buy phase — auto-played "+treas.length+" Treasure"+(treas.length===1?"":"s")+"."));
  return {ok:true,over:g.over};
}
function buyCard(g,p,pile){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="buy")return {ok:false,err:"not_buy_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  if(g.t.buys<=0)return {ok:false,err:"no_buys"};
  if(g.supply[pile]==null||g.supply[pile]<=0)return {ok:false,err:"pile_empty"};
  var c=cardInfo(pile); if(!c)return {ok:false,err:"bad_pile"};
  if(g.t.coins<c.cost)return {ok:false,err:"too_expensive"};
  g.t.buys--; g.t.coins-=c.cost;
  gain(g,p,pile);
  g.seq++;
  g.events.push(ev(g,p,"Bought "+pile+"."));
  checkOver(g);
  return {ok:true,over:g.over};
}
function endTurn(g,p){
  if(g.over)return {ok:false,err:"over"};
  if(current(g)!==p)return {ok:false,err:"not_your_turn"};
  if(g.phase!=="buy")return {ok:false,err:"not_buy_phase"};
  if(pendingFor(g,p))return {ok:false,err:"resolve_prompt"};
  cleanupPlayer(g,p);
  g.events.push(ev(g,p,"Turn ended."));
  g.turnIndex++;
  g.seq++;
  startTurn(g,current(g));
  checkOver(g);
  return {ok:true,over:g.over};
}
function canPlay(g,p,idx){
  return !g.over&&current(g)===p&&g.phase==="action"&&g.t.actions>0&&!pendingFor(g,p)&&idx!=null&&idx>=0&&idx<g.players[p].hand.length&&isAction(g.players[p].hand[idx]);
}
function canBuyCard(g,p,pile){
  if(g.over||current(g)!==p||g.phase!=="buy"||pendingFor(g,p)||g.t.buys<=0)return false;
  if(g.supply[pile]==null||g.supply[pile]<=0)return false;
  var c=cardInfo(pile); return !!(c&&g.t.coins>=c.cost);
}
function drainEvents(g){
  var out=g.events.slice();
  g.events.length=0;
  return out;
}

}

function promptTitle(pr){

  let who="";

  else who="You — ";
  if(pr.type==="cellar")return who+"Cellar: discard any number, then draw that many.";
  if(pr.type==="trashN")return who+"Chapel: trash up to "+pr.data.max+" cards.";
  if(pr.type==="moneylender")return who+"Moneylender: trash a Copper for +$3?";
  if(pr.type==="gain")return who+"Gain a card costing up to $"+pr.data.maxCost+".";
  if(pr.type==="remodelStep1")return who+"Remodel: choose a card to trash.";
  if(pr.type==="remodelStep2")return who+"Gain a card costing up to $"+pr.data.maxCost+".";

/* ══════════ AI ══════════ */
function aiRespond(g,p,pr){
  const pl=g.players[p];
  const resp={};
  if(pr.type==="cellar"){
    const drop=[];
    const coppers=pl.hand.filter(function(x){return x==="Copper";}).length;
    for(let i=0;i<pl.hand.length;i++){
      const c=pl.hand[i];
      if(c==="Curse"||c==="Estate")drop.push(i);
      else if(c==="Copper"&&coppers>1)drop.push(i);
    }
    resp.cards=drop;
  }
  else if(pr.type==="trashN"){
    const drop=[];
    for(let i=0;i<pl.hand.length;i++){
      const c=pl.hand[i];
      if(c==="Curse")drop.push(i);
      else if(c==="Estate"&&g.turnIndex<8)drop.push(i);
    }
    resp.cards=drop.slice(0,pr.data.max);
  }
  else if(pr.type==="moneylender"){
    const i=pl.hand.indexOf("Copper");
    if(i<0){ resp.skip=true; } else { resp.cards=[i]; }
  }
  else if(pr.type==="gain"||pr.type==="remodelStep2"||pr.type==="mineStep2"){
    const pile=aiPickGain(g,p,pr.data.maxCost,pr.type==="mineStep2");
    if(pile)resp.pile=pile; else resp.skip=true;
  }
  else if(pr.type==="remodelStep1"){
    let pick=0;
    const iC=pl.hand.indexOf("Copper");
    if(pl.hand.indexOf("Curse")>=0)pick=pl.hand.indexOf("Curse");
    else if(iC>=0&&pl.hand.filter(function(x){return x==="Copper";}).length>=2)pick=iC;
    else if(pl.hand.indexOf("Estate")>=0)pick=pl.hand.indexOf("Estate");
    resp.cards=[pick];
  }
  else if(pr.type==="mineStep1"){
    const iC=pl.hand.indexOf("Copper");
    resp.cards=[iC>=0?iC:0];
  }
  else if(pr.type==="throne"){
    const rank={"Smithy":5,"Market":5,"Witch":6,"Village":4,"Council Room":4,"Moneylender":4,"Moat":3,"Militia":3,"Mine":2,"Remodel":2,"Bureaucrat":2,"Workshop":1,"Cellar":1,"Feast":1,"Chapel":0,"Throne Room":0};
    let best=-1,bR=-1;
    for(let i=0;i<pl.hand.length;i++){
      const c=pl.hand[i];
      if(!isAction(c))continue;
      const r=(rank[c]||0);
      if(r>bR){ bR=r; best=i; }
    }
    resp.cards=[best];
  }
  else if(pr.type==="militia"){
    if(pl.hand.indexOf("Moat")>=0&&Math.random()<0.65){ resp.moat=true; }
    else{
      const idxs=pl.hand.map(function(c,i){return i;}).sort(function(a,b){return keepVal(pl.hand[b])-keepVal(pl.hand[a]);});
      resp.moat=false; resp.cards=idxs.slice(3);
    }
  }
  else if(pr.type==="bureaucrat"){
    const vi=[];
    for(let i=0;i<pl.hand.length;i++)if(isVictory(pl.hand[i]))vi.push(i);
    if(vi.length){
      vi.sort(function(a,b){return cardInfo(pl.hand[b]).vp-cardInfo(pl.hand[a]).vp;});
      resp.cards=[vi[0]];
    } else resp.none=true;
  }
  return resp;
}
function aiPickGain(g,p,maxCost,treasureOnly){
  const names=pileOrder();
  const late=g.supply["Province"]<=3||emptyPiles(g)>=2;
  let best="",bestV=-1e9;
  for(let i=0;i<names.length;i++){
    const n=names[i];
    if(g.supply[n]==null||g.supply[n]<=0)continue;
    const c=cardInfo(n);
    if(c.cost>maxCost)continue;
    if(treasureOnly&&!isTreasure(n))continue;
    let v=-5;
    if(isTreasure(n)){ v=(c.coin-1)*3+c.cost; }
    else if(isVictory(n)){ if(late&&(n==="Duchy"||n==="Estate"))v=2; else v=-3; }
    else if(isCurse(n)){ v=-99; }
    else{
      const a={"Smithy":4,"Market":5,"Village":3,"Witch":7,"Militia":4,"Moneylender":3,"Mine":3,"Remodel":2,"Workshop":3,"Feast":3,"Cellar":1,"Chapel":1,"Moat":2,"Council Room":4,"Throne Room":2,"Bureaucrat":1}[n]||0;
      v=late?a*0.3:a;
    }
    if(v>bestV){ bestV=v; best=n; }
  }
  return best||null;
}
function aiPickBuy(g,p){
  const coins=g.t.coins;
  const late=g.supply["Province"]<=4||emptyPiles(g)>=2;
  const early=!late&&g.turnIndex<12;
  const names=pileOrder();
  let best="",bs=-1e9;
  for(let i=0;i<names.length;i++){
    const n=names[i];
    if(g.supply[n]==null||g.supply[n]<=0)continue;
    const c=cardInfo(n);
    if(c.cost>coins)continue;
    let v=-10;
    if(isTreasure(n)){ v = n==="Copper"?-5:(n==="Silver"?2.2:(n==="Gold"?3.4:0)); }
    else if(isVictory(n)){
      if(n==="Province")v=coins>=8?8:(coins>=6?5:1);
      else if(n==="Duchy")v=late?2.5:-1;
      else v=(late&&g.supply["Province"]<=3)?0.6:-3;
    }
    else if(isCurse(n)){ v=-99; }
    else{
      const a={"Witch":6.5,"Militia":4,"Smithy":3,"Market":5.5,"Mine":3,"Remodel":2,"Village":3,"Moneylender":3,"Workshop":2.5,"Feast":2.5,"Throne Room":2.5,"Bureaucrat":1,"Cellar":1,"Chapel":1,"Moat":1,"Council Room":3}[n]||0;
      v=early?a:a*0.35;
    }
    if(v>bs){ bs=v; best=n; }
  }
  return best||null;
}
function aiBestAction(g,p){
  if(g.phase!=="action"||g.t.actions<=0||pendingFor(g,p))return null;
  const pl=g.players[p];
  const late=g.supply["Province"]<=4||emptyPiles(g)>=2;
  const SCORE={"Smithy":6,"Market":7,"Witch":8,"Village":6,"Council Room":6,"Militia":5,"Moat":4,"Moneylender":5,"Mine":4,"Bureaucrat":2,"Remodel":3,"Workshop":3,"Feast":3,"Cellar":3,"Chapel":3,"Throne Room":5};
  let best=-1,bestS=-1e9;
  for(let i=0;i<pl.hand.length;i++){
    const n=pl.hand[i];
    if(!isAction(n))continue;
    if(late&&(SCORE[n]||0)<5)continue;
    let s=SCORE[n]||0;
    if(n==="Cellar"){ const junk=pl.hand.filter(function(x){return x==="Curse"||x==="Estate";}).length; s=junk?junk+2:0; }
    if(n==="Chapel"){ const junk=pl.hand.filter(function(x){return x==="Curse"||x==="Estate";}).length; s=junk?junk*2:0; }
    if(n==="Moneylender"){ if(pl.hand.indexOf("Copper")<0)s=0; }
    if(n==="Mine"){ s=treasureIndices(g,p).length?s+1:0; }
    if(n==="Militia"){ if(g.players[1-p].hand.length<=3)s-=2; }
    if(n==="Bureaucrat"){ if(g.supply["Silver"]<=0)s=0; }
    if(n==="Throne Room"){ const good=["Smithy","Market","Witch","Village","Council Room","Moneylender","Militia","Moat"].filter(function(x){return pl.hand.indexOf(x)>=0;}); s=good.length?s+3:0; }
    if(n==="Remodel"){ if(pl.hand.length<2)s=0; }
    if(n==="Witch"){ if(g.supply["Curse"]<=0)s=1; }
    s+=Math.random()*(AI_DIFF==="easy"?6:(AI_DIFF==="hard"?0.4:2));
    if(s>bestS){ bestS=s; best=i; }
  }
  return best===-1?null:best;
}
function aiActions(g,p){
  for(let guard=0;guard<40;guard++){
    if(pendingFor(g,p)){
      const pr=pendingFor(g,p);
      const r=respondPrompt(g,p,{id:pr.id,resp:aiRespond(g,p,pr)});
      if(!r.ok)break;
      continue;
    }
    const i=aiBestAction(g,p);
    if(i==null)break;
    const r=playCard(g,p,i);
    if(!r.ok)break;
  }
}
function aiBuy(g,p){
  for(let guard=0;guard<20;guard++){
    if(g.phase!=="buy"||g.t.buys<=0||pendingFor(g,p))break;
    const pile=aiPickBuy(g,p);
    if(!pile)break;
    const r=buyCard(g,p,pile);
    if(!r.ok)break;
  }
}
function resolveAiPrompts(){
  if(game.mode!=="ai"||!game.g)return;
  for(let guard=0;guard<20;guard++){
    const pr=pendingFor(game.g,1);
    if(!pr)break;
    const r=respondPrompt(game.g,1,{id:pr.id,resp:aiRespond(game.g,1,pr)});
    if(!r.ok)break;
  }
}
function aiDoTurn(){
  game.aiThinking=true;
  updateBanner();
  setTimeout(function(){
    if(game.mode!=="ai"||!game.g||game.g.over)return;
    const g=game.g;
    aiActions(g,1);
    resolveAiPrompts();
    endActions(g,1);
    aiBuy(g,1);
    endTurn(g,1);
    game.aiThinking=false;
    flushLog();
    if(g.over){ showGameOver(); return; }
    renderAll();
    saveGame();
  },420);
}

/* ══════════ save / continue ══════════ */
const SAVE_KEY="bgn_save_dom";
function saveGame(){
  if(!game.g||game.mode==="online")return;
  if(game.g.over){ deleteSave(); return; }
  try{ localStorage.setItem(SAVE_KEY, JSON.stringify({v:1,mode:game.mode,g:game.g,logLines:game.logLines,savedAt:Date.now()})); }catch(e){}
}
function hasSave(){ try{ return !!localStorage.getItem(SAVE_KEY); }catch(e){ return false; } }
function deleteSave(){ try{ localStorage.removeItem(SAVE_KEY); }catch(e){} }
function refreshContinuePanel(){
  if(!continuePanel)return;
  const s=hasSave();
  continuePanel.hidden=!s;
  if(!s)return;
  let info="A saved game was found.";
  try{ const o=JSON.parse(localStorage.getItem(SAVE_KEY)); if(o&&o.savedAt)info="Saved "+new Date(o.savedAt).toLocaleTimeString()+" — "+(o.mode==="ai"?"vs the Machine":"hotseat")+"."; }catch(e){}
  continueInfoEl.textContent=info;
}
function continueGame(){
  let o; try{ o=JSON.parse(localStorage.getItem(SAVE_KEY)); }catch(e){ o=null; }
  if(!o||!o.g){ flashBanner("No saved game found."); return; }
  const mode=(o.mode==="ai")?"ai":"local";
  game.mode=mode; game.g=o.g;
  game.view=0; game.hand=[]; game.prompts=[];
  game.myPlayer=0; game.myRole=null;
  game.over=!!o.g.over; game.passShow=false; game.aiThinking=false; game.locked=false; game.selected=null;
  game.logLines=Array.isArray(o.logLines)?o.logLines.slice():[];
  myNameEl.textContent=mode==="ai"?"You":"Player 1";
  oppNameEl.textContent=mode==="ai"?"The Machine":"Player 2";
  roomBox.hidden=true; claimBtn.hidden=true; resignBtn.hidden=true; chatBox.hidden=true;
  lobbyCtn.hidden=true; gameCtn.hidden=false; passScreen.hidden=true; overModal.hidden=true; promptModal.hidden=true;
  sfx("deal");
  renderAll();
  saveGame();
  if(game.mode==="ai"&&activeP()===1&&!game.g.over)aiDoTurn();
}

/* ══════════ game over ══════════ */
function showGameOver(){
  const g=game.g; if(!g)return;
  const youWon=game.mode==="online"?g.winner===game.myPlayer:(game.mode==="ai"?g.winner===0:(g.winner===game.view));
  if(g.winner==="draw")sfx("draw2");
  else if(youWon)sfx("win");
  else sfx("lose");
  overTitleEl.textContent="Game over";
  let banner="";
  if(g.winner==="draw")banner="It's a tie!";
  else banner=youWon?"🏆 You win!":"🏆 "+npName(g.winner)+" wins!";
  overBannerEl.textContent=banner;
  const s0=g.scores?g.scores[0]:score(g,0);
  const s1=g.scores?g.scores[1]:score(g,1);
  const p0=game.mode==="online"?(game.myPlayer===0?"You":"Opponent"):npName(0);
  const p1=game.mode==="online"?(game.myPlayer===1?"You":"Opponent"):npName(1);
  overScoresEl.innerHTML='<div class="'+(g.winner===0?"win":"")+'"><span class="muted small">'+esc(p0)+'</span><b>'+s0+'</b><span class="pts-tag">VP</span>'+(g.winner===0?'<span class="win-tag">★ winner</span>':"")+'</div>'+
                          '<div class="'+(g.winner===1?"win":"")+'"><span class="muted small">'+esc(p1)+'</span><b>'+s1+'</b><span class="pts-tag">VP</span>'+(g.winner===1?'<span class="win-tag">★ winner</span>':"")+'</div>';
  overReasonEl.textContent=reasonText();
  playAgainBtn.textContent=game.mode==="online"?"Rematch":"Play again";
  overModal.hidden=false;
  promptModal.hidden=true; passScreen.hidden=true;
  renderAll();
  deleteSave();
  const w=g.winner;
  if(w!==null&&w!==undefined&&w!=="draw"){
    const sc=score(g,w);
    if(game.mode==="online"){ if(game.myRole==="host")lbSubmitScore(lbName(),sc,""); }
    else lbSubmitScore(lbName(),sc,"");
  }
}
function reasonText(){
  const g=game.g;
  const w=g.winner;
  const youWon=game.mode==="online"?w===game.myPlayer:(game.mode==="ai"?w===0:(w===game.view));
  if(w==="draw")return "It's a tie — most VP would have won.";
  if(g.reason==="resign")return npName(w)+" wins — "+(game.mode==="online"?(w===game.myPlayer?"your opponent resigned":"you resigned"):npName(1-w)+" resigned")+".";
  if(g.reason==="forfeit")return game.mode==="online"?(youWon?"You win — your opponent left!":"You lose — your opponent claimed the win."):npName(w)+" wins.";
  return "The supply ran out — "+npName(w)+" scored the most Victory Points.";
}

const TURN_TIMER_SECS=60;
let tmrRemain=0, tmrInt=null, tmrKey="", hintAt=-1;
function tmrFmt(s){ const m=Math.floor(s/60), r=s%60; return m+":"+String(r).padStart(2,"0"); }
function tmrStop(){ if(tmrInt){ clearInterval(tmrInt); tmrInt=null; } }
function tmrStart(){ if(TURN_TIMER_SECS<=0)return; tmrStop(); tmrRemain=TURN_TIMER_SECS; tmrInt=setInterval(tmrTick,1000); tmrTick(); }
function tmrTick(){
  tmrRemain--;
  if(tmrEl){ tmrEl.textContent="⏱ "+tmrFmt(Math.max(0,tmrRemain)); tmrEl.classList.toggle("low",tmrRemain<=10); }
  if(tmrRemain<=0){
    tmrStop();
    flashBanner("Time's up — turn auto-ended.");
    if(canActUI()){ if(game.g.phase==="action")onClickEndPhase(); else onClickEndTurn(); }
  }
}
function syncTurnTimer(){
  const g=game.g;
  const should = TURN_TIMER_SECS>0 && game.mode && g && !g.over && !game.locked && !game.aiThinking && !game.passShow && canActUI();
  if(!should){ tmrStop(); if(tmrEl)tmrEl.hidden=true; return; }
  if(tmrEl)tmrEl.hidden=false;
  const key = game.mode+":"+g.turnIndex+":"+activeP();
  if(key!==tmrKey){ tmrKey=key; tmrStart(); }
  else if(!tmrInt)tmrStart();
}

function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){ connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — share the code to start the match":"Connecting…"); }
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="hand"){ game.hand=msg.hand.slice(); if(game.g)game.g.players[msg.p].hand=game.hand; renderAll(); }
  else if(msg.t==="promptlist"){ game.prompts=Array.isArray(msg.prompts)?msg.prompts.slice():[]; renderAll(); }
  else if(msg.t==="chat")addChat(msg.role,msg.text,msg.ts);
}
function applyServerSnap(s){
  if(s.deckSig&&s.deckSig!==CFG.sig){
    if(Array.isArray(s.deck)&&s.deck.length){
      CFG.pool=s.deck.map(function(d){ return {name:String(d.name||""),cost:Math.floor(d.cost)||0,emoji:String(d.emoji||""),kinds:String(d.kinds||""),coin:0,vp:0}; });
      setPool(CFG.pool);
      CFG.sig=kingdomSig();
      if(!game.deckWarned){
        game.deckWarned=true;
        log("Note: the online server's kingdom pool differs from your main.pjs settings — playing with the room's pool. Update the KINGDOM_CFG mirror in index.html to match your main.pjs config.",true);
      }
    }
  }
  if(!game.g)game.g=newRoom(s.hostFirst);
  const g=game.g;
  g.turnIndex=s.turnIndex; g.phase=s.phase;
  g.t={actions:s.t.actions,buys:s.t.buys,coins:s.t.coins};
  g.supply=s.supply; g.kingdom=s.kingdom;
  for(let i=0;i<2;i++){
    g.players[i].deckLen=s.players[i].deckLen;
    g.players[i].discardLen=s.players[i].discardLen;
    g.players[i].discardTop=s.players[i].discardTop;
    g.players[i].inPlay=s.players[i].inPlay.slice();
  }
  g.trashLen=s.trashLen; g.trashTop=s.trashTop;
  g.scores=s.scores;
  g.over=s.status===3; g.winner=s.winner; g.reason=s.reason;
  g.pending=s.pending.map(function(pr){ return {player:pr.player,type:pr.type,id:pr.id}; });
  game.status=s.status;
  game.hostIn=s.hostIn; game.guestIn=s.guestIn;
  game.oppPresent=(game.myPlayer===0)?s.guestIn:s.hostIn;
  game.locked=false;
  if(s.seq!==game.lastSeq){
    if(Array.isArray(s.events)){
      for(let i=0;i<s.events.length;i++){
        const e=s.events[i];
        log((e.p===-1?"⚑ Table: ":npName(e.p)+": ")+e.txt,/wins|stole|Forfeit/.test(e.txt));
      }
    }
    game.lastSeq=s.seq;
  }
  if(g.over&&game.mode==="online")showGameOver();
  const ap=activeP();
  if(!g.over&&game.status===2&&ap===game.myPlayer&&game._lastAp!==ap)sfx("turn");
  game._lastAp=ap;
  renderAll();
}

async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  game.mode="online";
  game.myPlayer=0;
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{

    game.myRole=r.role;
    game.roomCode=r.code;
    myNameEl.textContent="You";
    oppNameEl.textContent="Opponent";
    game.logLines=[];

    enterOnlineGame();
