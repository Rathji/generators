/* ── Skull King server ────────────────────────────────
   Rooms: 5-letter codes. Host creates; a guest joining with the
   code starts round 1. The server keeps the deck, BOTH hands and
   sealed bids authoritatively (private hand pushed per-connection,
   never broadcast) and re-runs the rules engine to validate every
   bid and play. Public info — revealed bids, tricks in play, taken
   tricks, scores, totals — is broadcast via pubsub. Durable room
   state lives in the plugin `state` bytes (versioned layout).     */
const MAXR=10, NSUITS=5, NUMC=13, DECK_N=87;
const M0=65, P0=70, KING=72, TIG=73, TIDE0=74;
const SCHEMA=4;
const SLOT=512;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const O_STAT=0,O_CODE=1,O_SCHEMA=9,O_PHASE=10,O_ROUND=11,O_TURN=12,O_LEADER=13,O_FLAGS=14,O_WIN=15,O_REASON=16,O_LAST=17;
const O_TOTALS=21,O_RSCORE=25,O_TRICKS=29,O_BIDS=32,O_SPECIALS=34,O_TRICK=36,O_HANDS=45,O_DECK=67;

/* ── pure rules engine (mirrors client — keep in sync) ── */
function cardType(id){ if(id<M0)return {k:"color",suit:Math.floor(id/13),num:id%13+1}; if(id<P0)return {k:"mermaid"}; if(id<KING)return {k:"pirate"}; if(id===KING)return {k:"skullKing"}; if(id===TIG)return {k:"tigress"}; return {k:"tide"}; }
function isColor(id){ return id<M0; }
function suitOf(id){ return id<M0?Math.floor(id/13):-1; }
function numOf(id){ return id<M0?id%13+1:0; }
function pVal(k){ return k==="skullKing"?30:(k==="pirate"||k==="mermaid"||k==="tigress"?20:10); }
function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); const t=a[i]; a[i]=a[j]; a[j]=t; } return a; }
function buildDeck(){ const d=[]; for(let i=0;i<DECK_N;i++)d.push(i); return shuffle(d); }
function trickPower(card,role,ledSuit,hasKing){
  const t=cardType(card);
  if(t.k==="mermaid")return hasKing?6:0;
  if(t.k==="skullKing")return 5;
  if(t.k==="tigress")return role==="pirate"?4:0;
  if(t.k==="pirate")return 3;
  if(t.k==="color")return t.suit===ledSuit?t.num:0;
  return 0;
}
function resolveTrick(plays,ledSuit){
  const hasKing=plays.some(function(pl){return cardType(pl.card).k==="skullKing";});
  let wi=-1,wpow=-1;
  for(let i=0;i<plays.length;i++){
    const pow=trickPower(plays[i].card,plays[i].role,ledSuit,hasKing);
    if(pow>wpow){ wpow=pow; wi=plays[i].p; }
  }
  return {winner:wi,power:wpow};
}
function legalPlays(hand,ledSuit){
  if(ledSuit===-1){ const a=[]; for(let i=0;i<hand.length;i++)a.push(i); return a; }
  let hasSuit=false;
  for(let i=0;i<hand.length;i++){ if(isColor(hand[i])&&suitOf(hand[i])===ledSuit){ hasSuit=true; break; } }
  const out=[];
  for(let i=0;i<hand.length;i++){
    const id=hand[i];
    if(!isColor(id)){ out.push(i); continue; }
    if(suitOf(id)===ledSuit)out.push(i);
    else if(!hasSuit)out.push(i);
  }
  return out;
}
function specialPts(plays){ let s=0; for(const pl of plays){ const k=cardType(pl.card).k; if(k!=="color")s+=pVal(k); } return s; }
function roundScore(round,bid,tricks,spec){
  if(bid===0)return tricks===0?round*10:-round*10;
  if(tricks===bid)return 20*tricks+10+spec;
  return -10*Math.abs(tricks-bid);
}
function pName(p){ return p===0?"Host":"Guest"; }
function cardName(id,role){
  const t=cardType(id);
  if(t.k==="color")return {suit:["Red","Yellow","Green","Blue","Purple"][t.suit],num:t.num};
  const n={mermaid:"Mermaid",pirate:"Pirate",skullKing:"Skull King",tigress:"Tigress",tide:"Tide"}[t.k];
  return {suit:n,num:null};
}
function cardTxt(id,role){ const c=cardName(id,role); return c.num?c.suit+" "+c.num:c.suit+(tigRoleExtra(id,role)); }
function tigRoleExtra(id,role){ return cardType(id).k==="tigress"?(role==="pirate"?" (Pirate)":" (Escape)"):""; }

/* ── room helpers ── */
function newRoom(){ return {slot:-1,code:"",status:1,hostId:-1,guestId:-1,lastAct:0,
  round:1,phase:0,deck:[],hands:[[],[]],bids:[null,null],bidLocked:[false,false],
  turn:-1,leader:0,trick:{plays:[],ledSuit:-1,leadPlayer:-1},
  tricksTaken:[0,0],tricksPlayed:0,specials:[0,0],totals:[0,0],roundScores:null,
  over:false,winner:null,reason:"",seq:0,event:""}; }
function dealRound(g){ g.deck=buildDeck(); g.hands=[[],[]]; for(let p=0;p<2;p++){ for(let k=0;k<g.round;k++)g.hands[p].push(g.deck.pop()); } }
function startRound(g){
  g.phase=0; g.bids=[null,null]; g.bidLocked=[false,false]; g.turn=-1;
  g.leader=(g.round-1)%2; g.trick={plays:[],ledSuit:-1,leadPlayer:-1};
  g.tricksTaken=[0,0]; g.tricksPlayed=0; g.specials=[0,0]; g.roundScores=null;
  dealRound(g);
  g.event="Round "+g.round+": "+(g.round===1?"1 card":"cards")+" each — bid your tricks in secret!";
}
function finishRound(g){
  const rs=[0,0];
  for(let p=0;p<2;p++){
    const bid=g.bids[p]===null?0:g.bids[p];
    const tk=g.tricksTaken[p];
    const spec=(bid===tk&&bid>0)?g.specials[p]:0;
    rs[p]=roundScore(g.round,bid,tk,spec);
    g.totals[p]+=rs[p];
  }
  g.roundScores=rs; g.phase=2; g.turn=-1;
  g.event="Round "+g.round+" scored: Host "+(rs[0]>=0?"+":"")+rs[0]+", Guest "+(rs[1]>=0?"+":"")+rs[1]+". Totals: "+g.totals[0]+"–"+g.totals[1]+".";
  if(g.round>=MAXR){
    g.status=3; g.over=true;
    g.winner=g.totals[0]>g.totals[1]?0:g.totals[1]>g.totals[0]?1:"draw";
    g.reason="gameOver";
    g.event=(g.winner==="draw"?"It's a draw — "+g.totals[0]+" to "+g.totals[1]+"!":pName(g.winner)+" wins the sea — "+g.totals[0]+" to "+g.totals[1]+"!");
  }
}
function handlePlay(g,idx,ci,role){
  const id=g.hands[idx][ci];
  g.hands[idx].splice(ci,1);
  const t=cardType(id);
  g.trick.plays.push({p:idx,card:id,role:t.k==="tigress"?(role==="pirate"?"pirate":"escape"):null});
  if(g.trick.ledSuit===-1&&isColor(id))g.trick.ledSuit=suitOf(id);
  g.tricksPlayed++;
  const who=pName(idx);
  if(g.trick.plays.length<2){
    g.turn=1-idx;
    g.event=who+" plays "+cardTxt(id,role)+". Waiting for the next play…";
  }else{
    const r=resolveTrick(g.trick.plays,g.trick.ledSuit);
    g.tricksTaken[r.winner]++;
    g.specials[r.winner]+=specialPts(g.trick.plays);
    const winN=pName(r.winner);
    if(g.tricksPlayed>=g.round*2){ finishRound(g); }
    else{
      g.trick={plays:[],ledSuit:-1,leadPlayer:r.winner};
      g.leader=r.winner; g.turn=r.winner;
      g.event=winN+" takes the trick ("+g.tricksTaken[r.winner]+"/"+g.round+") — they lead next.";
    }
  }
}

/* ── durable state helpers ── */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+O_LAST]<<24)|(state[o+O_LAST+1]<<16)|(state[o+O_LAST+2]<<8)|state[o+O_LAST+3]; }
function slotStatus(i){ return state[off(i)+O_STAT]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function reasonCode(r){ return r==="gameOver"?1:r==="forfeit"?2:r==="resign"?3:0; }
function reasonName(c){ return c===1?"gameOver":c===2?"forfeit":c===3?"resign":""; }
function i16write(o,v){ const x=v|0; state[o]=(x>>8)&255; state[o+1]=x&255; }
function i16read(o){ const v=((state[o]<<8)|state[o+1]); return v>32767?v-65536:v; }
function saveRoom(g){
  const o=off(g.slot);
  state.fill(0,o,o+SLOT);
  state[o+O_STAT]=g.status;
  writeCode(o,g.code);
  state[o+O_SCHEMA]=SCHEMA;
  state[o+O_PHASE]=g.phase;
  state[o+O_ROUND]=g.round;
  state[o+O_TURN]=g.turn<0?255:g.turn;
  state[o+O_LEADER]=g.leader;
  state[o+O_FLAGS]=(g.bidLocked[0]?1:0)|(g.bidLocked[1]?2:0);
  state[o+O_WIN]=g.winner==="draw"?2:g.winner===1?1:g.winner===0?0:255;
  state[o+O_REASON]=reasonCode(g.reason);
  const la=Math.max(0,Math.min(0xFFFFFFFF,g.lastAct));
  state[o+O_LAST]=(la>>>24)&255; state[o+O_LAST+1]=(la>>>16)&255; state[o+O_LAST+2]=(la>>>8)&255; state[o+O_LAST+3]=la&255;
  i16write(o+O_TOTALS,g.totals[0]); i16write(o+O_TOTALS+2,g.totals[1]);
  i16write(o+O_RSCORE,g.roundScores?g.roundScores[0]:0); i16write(o+O_RSCORE+2,g.roundScores?g.roundScores[1]:0);
  state[o+O_TRICKS]=g.tricksTaken[0]; state[o+O_TRICKS+1]=g.tricksTaken[1]; state[o+O_TRICKS+2]=g.tricksPlayed;
  state[o+O_BIDS]=g.bids[0]===null?255:g.bids[0]; state[o+O_BIDS+1]=g.bids[1]===null?255:g.bids[1];
  state[o+O_SPECIALS]=g.specials[0]; state[o+O_SPECIALS+1]=g.specials[1];
  const tr=g.trick;
  state[o+O_TRICK]=tr.leadPlayer<0?255:tr.leadPlayer;
  state[o+O_TRICK+1]=tr.ledSuit;
  state[o+O_TRICK+2]=tr.plays.length;
  for(let i=0;i<tr.plays.length&&i<2;i++){ const q=o+O_TRICK+3+i*3; state[q]=tr.plays[i].p; state[q+1]=tr.plays[i].card; state[q+2]=tr.plays[i].role==="pirate"?1:(tr.plays[i].role==="escape"?2:0); }
  for(let pi=0;pi<2;pi++){
    const p=o+(pi===0?O_HANDS:O_HANDS+11); state[p]=g.hands[pi].length;
    for(let i=0;i<g.hands[pi].length&&i<11;i++)state[p+1+i]=g.hands[pi][i];
  }
  let p=o+O_DECK; state[p]=g.deck.length;
  for(let i=0;i<g.deck.length&&i<DECK_N;i++)state[p+1+i]=g.deck[i];
}
function loadRoom(i){
  const o=off(i), st=slotStatus(i);
  const readHand=function(bp){ const n=state[bp]; const a=[]; for(let k=0;k<n&&k<11;k++)a.push(state[bp+1+k]); return a; };
  const readRole=function(x){ return x===1?"pirate":x===2?"escape":null; };
  const g=newRoom();
  g.slot=i; g.code=codeAt(i); g.status=st; g.hostFirst=0;
  g.phase=state[o+O_PHASE]; g.round=state[o+O_ROUND];
  g.turn=state[o+O_TURN]===255?-1:state[o+O_TURN]; g.leader=state[o+O_LEADER];
  const fl=state[o+O_FLAGS]; g.bidLocked=[!!(fl&1),!!(fl&2)];
  const w=state[o+O_WIN]; g.winner=w===2?"draw":w===1?1:w===0?0:null;
  g.reason=reasonName(state[o+O_REASON]);
  g.lastAct=readLastAct(i);
  g.totals=[i16read(o+O_TOTALS),i16read(o+O_TOTALS+2)];
  const r0=i16read(o+O_RSCORE), r1=i16read(o+O_RSCORE+2);
  g.roundScores=(g.phase===2)?[r0,r1]:null;
  g.tricksTaken=[state[o+O_TRICKS],state[o+O_TRICKS+1]]; g.tricksPlayed=state[o+O_TRICKS+2];
  g.bids=[state[o+O_BIDS]===255?null:state[o+O_BIDS],state[o+O_BIDS+1]===255?null:state[o+O_BIDS+1]];
  g.trick.leadPlayer=state[o+O_TRICK]===255?-1:state[o+O_TRICK];
  g.trick.ledSuit=state[o+O_TRICK+1]===255?-1:state[o+O_TRICK+1];
  const npl=state[o+O_TRICK+2];
  for(let k=0;k<npl&&k<2;k++){ const q=o+O_TRICK+3+k*3; g.trick.plays.push({p:state[q],card:state[q+1],role:readRole(state[q+2])}); }
  g.hands=[readHand(o+O_HANDS),readHand(o+O_HANDS+11)];
  const dl=state[o+O_DECK], dp=o+O_DECK; for(let k=0;k<dl&&k<DECK_N;k++)g.deck.push(state[dp+1+k]);
  g.over=st===3; g.specials=[state[o+O_SPECIALS],state[o+O_SPECIALS+1]]; g.seq=0; g.event="";
  return g;
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(state[o+O_SCHEMA]!==SCHEMA){ state.fill(0,o,o+SLOT); continue; }
    if(st===1){ state.fill(0,o,o+SLOT); continue; }   // abandoned waiting rooms are reclaimed
    rooms.set(codeAt(i),loadRoom(i));
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── connection / broadcast ── */
let rooms=new Map();                 // code -> room
const conns=new Map();               // conn.id -> {code, role, conn}
function topic(code){ return "ch:"+code; }
function snap(g){
  const tr=g.trick;
  return {
    code:g.code, status:g.status, phase:g.phase, round:g.round,
    turn:g.turn, leader:g.leader,
    bidLocked:[g.bidLocked[0]?1:0,g.bidLocked[1]?1:0],
    bids:g.phase===0?null:[g.bids[0],g.bids[1]],
    totals:g.totals.slice(), roundScores:g.roundScores,
    tricksTaken:g.tricksTaken.slice(), tricksPlayed:g.tricksPlayed,
    trick:{plays:tr.plays.map(function(pl){return {p:pl.p,card:pl.card,role:pl.role};}), ledSuit:tr.ledSuit, leadPlayer:tr.leadPlayer},
    handsLen:[g.hands[0].length,g.hands[1].length],
    deckCount:g.deck.length,
    hostIn:g.hostId!==-1, guestIn:g.guestId!==-1,
    over:g.status===3, winner:g.winner, reason:g.reason,
    seq:g.seq, event:g.event||""
  };
}
function broadcast(g){ try{ pubsub.publish(topic(g.code), JSON.stringify({t:"snap",snap:snap(g)})); }catch(e){} }
function sendHands(g){
  const targets=[[g.hostId,0],[g.guestId,1]];
  for(const pair of targets){
    if(pair[0]===-1)continue;
    const info=conns.get(pair[0]);
    if(!info)continue;
    try{ info.conn.send(JSON.stringify({t:"hand",hand:g.hands[pair[1]].slice()})); }catch(e){}
  }
}
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }
function saveBroadcast(g){ saveRoom(g); broadcast(g); sendHands(g); }
function idxOfRole(role){ return role==="host"?0:1; }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const g=newRoom();
    g.slot=slot; g.code=code; g.status=1; g.hostId=conn.id; g.guestId=-1; g.lastAct=nowSec();
    rooms.set(code,g);
    saveRoom(g);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host",conn});
    return JSON.stringify({ok:true,code,role:"host",snap:snap(g)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const g=rooms.get(code);
    if(!g)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(g.hostId===-1)role="host";
    else if(g.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(g.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")g.hostId=conn.id; else g.guestId=conn.id;
    let started=false;
    if(g.status===1&&g.hostId!==-1&&g.guestId!==-1){
      g.status=2; startRound(g); g.seq++;
      started=true;
    }
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role,conn});
    g.lastAct=nowSec();
    saveBroadcast(g);
    return JSON.stringify({ok:true,role,snap:snap(g)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const g=rooms.get(code);
    if(!g)return JSON.stringify({ok:false,err:"no_such_room"});
    sendHands(g);
    return JSON.stringify({ok:true,snap:snap(g)});
  },
  setBid({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const g=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!g)return JSON.stringify({ok:false,err:"no_such_room"});
    if(g.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==g)return JSON.stringify({ok:false,err:"not_in_room"});
    if(g.phase!==0)return JSON.stringify({ok:false,err:"not_bidding"});
    if(obj.evt!==g.seq)return JSON.stringify({ok:false,err:"stale"});
    const idx=idxOfRole(role);
    if(g.bidLocked[idx])return JSON.stringify({ok:false,err:"already_locked"});
    const b=obj.bid;
    if(typeof b!=="number"||isNaN(b)||b<0||b>g.round)return JSON.stringify({ok:false,err:"bad_bid"});
    g.bids[idx]=b; g.bidLocked[idx]=true;
    g.seq++; g.lastAct=nowSec();
    if(g.bidLocked[0]&&g.bidLocked[1]){
      g.phase=1; g.turn=g.leader; g.trick={plays:[],ledSuit:-1,leadPlayer:g.leader};
      g.event="Bids revealed: Host "+g.bids[0]+" — Guest "+g.bids[1]+". "+pName(g.leader)+" leads.";
    }else{
      g.event=pName(idx)+" has locked their bid — waiting for the other.";
    }
    saveBroadcast(g);
    return JSON.stringify({ok:true});
  },
  play({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const g=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!g)return JSON.stringify({ok:false,err:"no_such_room"});
    if(g.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==g)return JSON.stringify({ok:false,err:"not_in_room"});
    const idx=idxOfRole(role);
    if(g.phase!==1)return JSON.stringify({ok:false,err:"not_playing"});
    if(g.turn!==idx)return JSON.stringify({ok:false,err:"not_your_turn"});
    if(obj.evt!==g.seq)return JSON.stringify({ok:false,err:"stale"});
    const ci=obj.card;
    if(typeof ci!=="number"||ci<0||ci>=g.hands[idx].length)return JSON.stringify({ok:false,err:"bad_card"});
    const id=g.hands[idx][ci];
    const t=cardType(id);
    if(t.k==="tigress"&&obj.role!=="pirate"&&obj.role!=="escape")return JSON.stringify({ok:false,err:"bad_role"});
    const legal=legalPlays(g.hands[idx],g.trick.ledSuit);
    let found=false;
    for(let k=0;k<legal.length;k++){ if(legal[k]===ci){ found=true; break; } }
    if(!found)return JSON.stringify({ok:false,err:"illegal"});
    handlePlay(g,idx,ci,obj.role);
    g.seq++; g.lastAct=nowSec();
    saveBroadcast(g);
    return JSON.stringify({ok:true});
  },
  nextRound({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const g=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!g)return JSON.stringify({ok:false,err:"no_such_room"});
    if(g.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    if(g.phase!==2)return JSON.stringify({ok:false,err:"not_scored"});
    if(obj.evt!==g.seq)return JSON.stringify({ok:false,err:"stale"});
    g.round++;
    startRound(g);
    g.seq++; g.lastAct=nowSec();
    saveBroadcast(g);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const g=roomOf(conn);
    if(!g)return JSON.stringify({ok:false,err:"not_in_room"});
    if(g.status===2){
      const idx=idxOfRole(roleOf(conn));
      g.status=3; g.over=true; g.winner=1-idx; g.reason="resign"; g.seq++;
      g.event=pName(1-idx)+" wins — "+pName(idx)+" resigned.";
      g.lastAct=nowSec(); saveBroadcast(g);
    }
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const g=roomOf(conn);
    if(!g)return JSON.stringify({ok:false,err:"not_in_room"});
    if(g.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const idx=idxOfRole(roleOf(conn));
    const oppPresent=idx===0?g.guestId!==-1:g.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    g.status=3; g.over=true; g.winner=idx; g.reason="forfeit"; g.seq++;
    g.event=pName(idx)+" wins — the opponent disconnected.";
    g.lastAct=nowSec(); saveBroadcast(g);
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const g=roomOf(conn);
    if(!g)return JSON.stringify({ok:false,err:"not_in_room"});
    if(g.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const idx=idxOfRole(roleOf(conn));
    const oppPresent=idx===0?g.guestId!==-1:g.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    g.status=2; g.over=false; g.winner=null; g.reason="";
    g.round=1; g.totals=[0,0];
    startRound(g);
    g.event="Rematch! Round 1 — bid in secret.";
    g.seq++; g.lastAct=nowSec();
    saveBroadcast(g);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const g=roomOf(conn);
    if(g&&g.status===2){
      const idx=idxOfRole(roleOf(conn));
      g.status=3; g.over=true; g.winner=1-idx; g.reason="forfeit"; g.seq++;
      g.event=pName(1-idx)+" wins — "+pName(idx)+" left.";
      g.lastAct=nowSec(); saveBroadcast(g);
    }
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  const g=rooms.get(info.code);
  if(!g)return;
  if(info.role==="host")g.hostId=-1; else g.guestId=-1;
  g.lastAct=nowSec();
  saveRoom(g);
  broadcast(g);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

function initTotals(g){ g.totals=[]; for(let p=0;p<g.n;p++)g.totals.push(0); }
function startLocalRound(g){
  g.deck=buildDeck();
  g.hands=[]; g.bids=[]; g.bidLocked=[];
  for(let p=0;p<g.n;p++){
    g.hands.push([]); g.bids.push(null); g.bidLocked.push(false);
    for(let k=0;k<g.round;k++)g.hands[p].push(g.deck.pop());
  }
  g.phase=0; g.turn=-1; g.leader=(g.round-1)%g.n;
  g.trick={plays:[],ledSuit:-1,leadPlayer:-1}; g.lastTrick=null;
  g.tricksTaken=[]; g.tricksPlayed=0; g.specials=[]; g.roundScores=null;
  for(let p=0;p<g.n;p++){ g.tricksTaken.push(0); g.specials.push(0); }
}
function applyLocalPlay(g,p,i,role){
  if(g.trick.plays.length===0)g.lastTrick=null;
  const id=g.hands[p][i];
  g.hands[p].splice(i,1);
  const t=cardType(id);
  g.trick.plays.push({p:p,card:id,role:t.k==="tigress"?(role==="pirate"?"pirate":"escape"):null});
  if(g.trick.ledSuit===-1&&isColor(id))g.trick.ledSuit=suitOf(id);
  g.tricksPlayed++;
  if(g.trick.plays.length<g.n){ g.turn=(p+1)%g.n; return; }
  const r=resolveTrick(g.trick.plays,g.trick.ledSuit);
  g.tricksTaken[r.winner]++;
  g.specials[r.winner]+=specialPts(g.trick.plays);
  g.lastTrick={plays:g.trick.plays.slice(),winner:r.winner};
  log(npName(r.winner)+" takes the trick ("+g.tricksTaken[r.winner]+"/"+g.round+")!",true);
  if(g.tricksPlayed>=g.round*g.n){ finishLocalRound(g); return; }
  g.trick={plays:[],ledSuit:-1,leadPlayer:r.winner};
  g.leader=r.winner; g.turn=r.winner;
}
function finishLocalRound(g){
  g.phase=2; g.turn=-1;
  const rs=[];
  for(let p=0;p<g.n;p++){
    const bid=g.bids[p]===null?0:g.bids[p];
    const tk=g.tricksTaken[p];
    const spec=(bid===tk&&bid>0)?g.specials[p]:0;
    const sc=roundScore(g.round,bid,tk,spec);
    rs.push(sc); g.totals[p]+=sc;
  }
  g.roundScores=rs;
  log("Round "+g.round+" scored — "+g.n+" players, totals updated.");
  if(g.round>=MAXR){
    g.over=true; g.reason="gameOver";
    const mx=Math.max.apply(null,g.totals);
    const winners=g.totals.map(function(t,p){return t===mx?p:-1;}).filter(function(p){return p!==-1;});
    g.winner=winners.length===1?winners[0]:"draw";
    showOverModal();
  }else{
    showRoundModal();
  }
}

/* ══════════ AI ══════════ */
function aiBid(g,p){
  const hand=g.hands[p], r=g.round;
  let strength=0;
  for(const c of hand){
    const t=cardType(c);
    if(t.k==="skullKing")strength+=2.1;
    else if(t.k==="pirate")strength+=1.5;
    else if(t.k==="tigress")strength+=1.0;
    else if(t.k==="mermaid")strength+=0.7;
    else if(t.k==="tide")strength+=0.05;
    else if(t.num>=12)strength+=1.15;
    else if(t.num>=10)strength+=0.85;
    else if(t.num>=8)strength+=0.55;
    else if(t.num>=6)strength+=0.3;
    else strength+=0.1;
  }
  const b=Math.floor(strength*0.7+Math.random()*1.3);
  return Math.max(0,Math.min(r,b));
}
function aiPlayLow(hand,legal){
  let best=null,bscore=1e9;
  for(const i of legal){
    const t=cardType(hand[i]);
    let s;
    if(t.k==="tide")s=0;
    else if(t.k==="mermaid")s=1;
    else if(t.k==="color")s=2+t.num/20;
    else if(t.k==="tigress")s=30;
    else if(t.k==="pirate")s=40;
    else s=50;
    if(s<bscore){ bscore=s; best=i; }
  }
  return best;
}
function aiLeadStrong(hand,legal){
  const colors=legal.filter(function(i){return isColor(hand[i]);}).map(function(i){return {i:i,n:numOf(hand[i])};});
  if(colors.length){
    colors.sort(function(a,b){return a.n-b.n;});
    const idx=Math.min(colors.length-1,Math.floor(colors.length*0.5+Math.random()*2));
    return colors[idx].i;
  }
  return legal[0];
}
function aiFindBeater(hand,legal,trick){
  const cur=resolveTrick(trick.plays,trick.ledSuit);
  const hasKing=trick.plays.some(function(pl){return cardType(pl.card).k==="skullKing";});
  let best=null,bestNum=1e9;
  for(const i of legal){
    const pow=trickPower(hand[i],null,trick.ledSuit,hasKing);
    if(pow>cur.power&&pow<bestNum){ bestNum=pow; best=i; }
  }
  return best;
}
function aiChooseCard(g,p){
  const hand=g.hands[p], trick=g.trick;
  const legal=legalPlays(hand,trick.ledSuit);
  if(!legal.length)return null;
  const targets=g.bids[p]-g.tricksTaken[p];
  let i;
  if(trick.plays.length===0){
    i=targets>0?aiLeadStrong(hand,legal):aiPlayLow(hand,legal);
  }else{
    if(targets>0){ const b=aiFindBeater(hand,legal,trick); i=b!==null?b:aiPlayLow(hand,legal); }
    else i=aiPlayLow(hand,legal);
  }
  if(i===null)i=legal[0];
  const role=cardType(hand[i]).k==="tigress"?"pirate":null;
  return {i:i,role:role};
}
function driveAI(){
  const g=game.g;
  if(game.mode!=="ai"||!g||g.over)return;
  if(g.phase===0){
    let pending=false;
    for(let p=1;p<g.n;p++){ if(!g.bidLocked[p]){ pending=true; break; } }
    if(!pending)return;
    game.aiThinking=true;
    setTimeout(function(){
      if(game.mode!=="ai"||!game.g)return;
      for(let p=1;p<g.n;p++){
        if(!g.bidLocked[p]){ g.bids[p]=aiBid(g,p); g.bidLocked[p]=true; log(npName(p)+" locked a bid ("+g.bids[p]+")."); }
      }
      game.aiThinking=false;
      if(g.bidLocked.every(function(b){return b;})){ revealBids(); }
    },650);
    return;
  }
  if(g.phase===1&&g.turn!==0){
    game.aiThinking=true;
    const p=g.turn;
    setTimeout(function(){
      if(game.mode!=="ai"||!game.g||game.g.over)return;
      const sel=aiChooseCard(game.g,p);
      if(sel!==null){
        const nm=cardName(game.g.hands[p][sel.i],sel.role);
        applyLocalPlay(game.g,p,sel.i,sel.role);
        log(npName(p)+" plays "+nm+".");
        game.aiThinking=false;
        renderAll();
        if(game.g.phase===2){ return; }
        driveAI();
      }else{
        game.aiThinking=false;
        driveAI();
      }
    },650+Math.random()*450);
  }
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    renderRoundBar();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){ connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — share the code to start the match":"Connecting…"); }
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="hand"){ game.hand=msg.hand.slice(); if(game.g&&game.mode==="online")game.g.hands[game.myPlayer]=game.hand; renderAll(); }
}
function applyServerSnap(s){
  if(!game.g){ game.g=newLocalGame(2); initTotals(game.g); }
  const g=game.g;
  g.phase=s.phase; g.round=s.round; g.turn=s.turn; g.leader=s.leader;
  g.bidLocked=[!!s.bidLocked[0],!!s.bidLocked[1]];
  g.bids=s.bids?s.bids.slice():[null,null];
  g.totals=s.totals.slice();
  g.roundScores=s.roundScores?s.roundScores.slice():null;
  g.tricksTaken=s.tricksTaken.slice(); g.tricksPlayed=s.tricksPlayed;
  g.trick={plays:s.trick.plays.map(function(pl){return {p:pl.p,card:pl.card,role:pl.role};}),ledSuit:s.trick.ledSuit,leadPlayer:s.trick.leadPlayer};
  g.handsLen=s.handsLen; g.deckCount=s.deckCount;
  g.over=s.over; g.winner=s.winner; g.reason=s.reason;
  game.status=s.status;
  game.hostIn=!!s.hostIn; game.guestIn=!!s.guestIn;
  game.oppPresent=(game.myPlayer===0)?game.guestIn:game.hostIn;
  game.locked=false;
  if(game.myPlayer>=0)g.hands[game.myPlayer]=game.hand;
  if(s.event&&s.seq!==game.lastSeq){ log(s.event.replace(/Host/g,npName(0)).replace(/Guest/g,npName(1)),/wins|takes the trick|Bids revealed/.test(s.event)); }
  game.lastSeq=s.seq;
  renderAll();
  if(g.over&&game.mode==="online"){ showOverModal(); }
  else if(g.phase===2&&g.roundScores&&game.mode==="online"){ showRoundModal(); }
  else{ roundModal.hidden=true; }
}

async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.myRole=r.role; game.myPlayer=0;
    game.roomCode=r.code;
    game.logLines=[];
    log("Room created — code "+r.code+". Share it with a friend to start.");
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    game.mode="online";
    game.myRole=r.role; game.myPlayer=r.role==="host"?0:1;
    game.roomCode=code;
    game.logLines=[];
    log("Joined room "+code+".");
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role; game.myPlayer=r.role==="host"?0:1;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
function sendSetBid(){

  updateButtons();

  rpc("setBid",JSON.stringify({code:game.roomCode,evt:evt,bid:game.bidVal})).then(function(r){

    }
  }).catch(function(){
    game.locked=false;
    flashBanner("Couldn't reach the server — reconnecting…");
  });
}

  });
}
