    setStatus('<span style="color:'+p.color+'">'+p.short+'</span> — Turn '+G.turn+' · <b>'+PHASE_LABEL[G.phase]+'</b>');
    toast(p.short+": "+PHASE_LABEL[G.phase]+" phase");
    render();
    if (G.mode==="online") sendState();
  } else if (ctrl==="remote"){
    setStatus('Waiting for <span style="color:'+p.color+'">'+p.short+'</span> to play '+PHASE_LABEL[G.phase]+'…');
    sendState();
  } else {
    setStatus(p.short+' (AI) — '+PHASE_LABEL[G.phase]+'…');
    runAI(p);
  }
}
function endTurn(){
  G.acted.add(G.phasePlayer);
  checkEliminations();
  if (G.over) return;
  const alive = aliveCount();
  if (G.acted.size >= alive){
    const pi = PHASES.indexOf(G.phase);
    if (pi < PHASES.length-1){ beginPhase(PHASES[pi+1]); return; }
    G.turn++;
    if (G.mode!=="online") autosave();
    toast("TURN "+G.turn);
    beginPhase("build");
    return;
  }
  let steps=0;
  do { G.phasePlayer=(G.phasePlayer+1)%6; steps++; } while(!G.powers[G.phasePlayer].alive && steps<6);
  startAction();
}
function endPhaseBtn(){

  if (G.mode==="online" && !G.net.isHost){

  } else {
    endTurn();
  }
}
function refreshMarket(){
  const m=G.market;
  for (const k of RES_KEYS) m[k] = Math.max(T.marketMin, Math.min(T.marketMax, Math.round(m[k]*(T.marketDrift+Math.random()*T.marketSpread))));
  log("World market prices fluctuate: "+RES_KEYS.map(k=>RES_META[k].name+" "+fmt(m[k])).join(", ")+".");
}

function capFor(p){ let cap=T.storageCap; for (const o of G.powers) if (o.alive && o.idx!==p.idx){ const s=G.zones.get(o.supply); if (s.owner===p.idx && !s.irr) cap+=T.supplyBonus; } return cap; }

function doBuildArmy(p, zoneId, sets){
  sfx("place");
  sets = Math.floor(sets);
  if (!p.alive) return "power eliminated";
  if (sets<1) return "bad amount";
  const z = G.zones.get(zoneId);
  if (!z || z.irr) return "zone unavailable";
  if (z.owner!==p.idx) return "you don't control this zone";
  const cost=T.buildCost*sets;
  if (p.money<cost) return "not enough money";
  if (RES_KEYS.some(k=>p.res[k]<sets*T.buildResourceCost)) return "need one of each resource per "+T.buildUnits+" units";
  p.money-=cost; RES_KEYS.forEach(k=> p.res[k]-=sets*T.buildResourceCost);
  z.armies += T.buildUnits*sets;
  log(p.short+" builds "+T.buildUnits*sets+" armies in "+zoneById[zoneId].name+".");
  return null;
}
function doBuildNavy(p, seaId, sets){
  sfx("place");
  sets=Math.floor(sets);
  if (!p.alive) return "power eliminated";
  if (sets<1) return "bad amount";
  const s = G.sea.get(seaId);
  if (!s) return "bad sea";
  if (!coastalSeaZones.has(seaId)) return "navies need a coastal sea zone";
  const access = s.owner===p.idx || [...(seaToLand.get(seaId)||[])].some(lz=>G.zones.get(lz).owner===p.idx);
  if (!access) return "no access to this sea";
  const cost=T.buildCost*sets;
  if (p.money<cost) return "not enough money";
  if (RES_KEYS.some(k=>p.res[k]<sets*T.buildResourceCost)) return "need one of each resource per "+T.buildUnits+" units";
  p.money-=cost; RES_KEYS.forEach(k=> p.res[k]-=sets*T.buildResourceCost);
  s.navies += T.buildUnits*sets;
  if (s.owner===-1) s.owner=p.idx;
  log(p.short+" builds "+T.buildUnits*sets+" navies in the "+zoneById[seaId].name+".");
  return null;
}
function doSell(p, r, qty){
  qty=Math.floor(qty);
  if (qty<1) return "bad amount";
  sfx("chip");
  if (p.res[r]<qty) return "not enough "+r;
  p.res[r]-=qty; p.money += qty*G.market[r];
  log(p.short+" sells "+qty+" "+r+" at "+fmt(G.market[r])+" each.");
  return null;
}
function doBuy(p, r, qty){
  qty=Math.floor(qty);
  if (qty<1) return "bad amount";
  sfx("chip");
  const cap=capFor(p);
  if (p.res[r]+qty>cap) return "storage full ("+cap+")";
  const cost=qty*G.market[r];
  if (p.money<cost) return "not enough money";
  p.res[r]+=qty; p.money-=cost;
  log(p.short+" buys "+qty+" "+r+" at "+fmt(G.market[r])+" each.");
  return null;
}
function doMoveArmy(p, from, to, count){
  sfx("step");
  count=Math.floor(count);
  if (!p.alive) return "power eliminated";
  const fz=G.zones.get(from), tz=G.zones.get(to);
  if (!fz||!tz) return "bad zone";
  if (fz.irr||tz.irr) return "irradiated zone";
  if (fz.owner!==p.idx) return "origin not yours";
  if (G.movedZones.has(from)) return "this zone has already moved";
  if (!(landAdj.get(from)||new Set()).has(to)) return "zones are not adjacent";
  if (count<1||count>fz.armies) return "bad count";
  if (tz.owner!==p.idx && count>fz.armies-1) return "must leave one army behind";
  fz.armies-=count; G.movedZones.add(from);
  if (tz.owner===p.idx){ tz.armies+=count; log(p.short+" reinforces "+zoneById[to].name+"."); return null; }
  if (tz.owner===-1||tz.armies===0){
    tz.owner=p.idx; tz.armies=count;
    log(p.short+" occupies "+zoneById[to].name+" with no resistance.");
    return null;
  }
  const defCount=tz.armies, defPower=POWERS[tz.owner].short;
  const r=resolveCombat(count,defCount);
  if (r.attackerWins){
    tz.owner=p.idx; tz.armies=r.attackers;
    log(p.short+" invades "+zoneById[to].name+" ("+count+" vs "+defCount+", "+r.rounds+" round"+(r.rounds===1?"":"s")+"). Victory! "+r.attackers+" armies advance.");
    checkEliminations();
  } else {
    tz.armies=r.defenders;
    log(p.short+"'s attack on "+zoneById[to].name+" fails ("+count+" vs "+defCount+"). "+defPower+" holds with "+r.defenders+".");
  }
  return null;
}
function doMoveNavy(p, from, to, count){
  sfx("step");
  count=Math.floor(count);
  if (!p.alive) return "power eliminated";
  const fs=G.sea.get(from), ts=G.sea.get(to);
  if (!fs||!ts) return "bad sea";
  if (fs.owner!==p.idx||fs.navies<count) return "origin not yours";
  if (G.movedSeas.has(from)) return "already moved this phase";
  if (count<1) return "bad count";
  const reach=new Set();
  let frontier=new Set([from]);
  for(let d=0;d<T.navyRange;d++){
    const next=new Set();
    for(const z of frontier) for(const n of (seaAdj.get(z)||new Set())) if(!reach.has(n) && n!==from) next.add(n);
    next.forEach(n=>reach.add(n));
    frontier=next;
  }
  if (!reach.has(to)) return "destination out of range (max "+T.navyRange+" sea zones)";
  fs.navies-=count; G.movedSeas.add(from);
  if (ts.owner===p.idx){ ts.navies+=count; log(p.short+" moves a fleet into the "+zoneById[to].name+"."); return null; }
  if (ts.owner===-1||ts.navies===0){ ts.owner=p.idx; ts.navies=count; log(p.short+" sails into the "+zoneById[to].name+"."); return null; }
  const defCount=ts.navies, defPower=POWERS[ts.owner].short;
  const r=resolveCombat(count,defCount);
  if (r.attackerWins){
    ts.owner=p.idx; ts.navies=r.attackers;
    log(p.short+" wins the naval battle in the "+zoneById[to].name+" ("+count+" vs "+defCount+").");
    checkEliminations();
  } else {
    ts.navies=r.defenders;
    log(p.short+" loses the naval battle in the "+zoneById[to].name+" ("+count+" vs "+defCount+"). "+defPower+" rules the waves.");
  }
  return null;
}
function doTransport(p, from, to, count){
  sfx("step");
  count=Math.floor(count);
  if (!p.alive) return "power eliminated";
  const fz=G.zones.get(from), tz=G.zones.get(to);
  if (!fz||!tz) return "bad zone";
  if (fz.irr||tz.irr) return "irradiated zone";
  if (fz.owner!==p.idx) return "origin not yours";
  if (G.movedZones.has(from)) return "already moved this phase";
  if (count<1||count>fz.armies) return "bad count";
  if (tz.owner!==p.idx && count>fz.armies-1) return "must leave one army behind";
  let seaId=null;
  for (const sId of (landToSea.get(from)||new Set())){
    if ((seaToLand.get(sId)||new Set()).has(to) && G.sea.get(sId).owner===p.idx && G.sea.get(sId).navies>0){ seaId=sId; break; }
  }
  if (seaId===null) return "no transport fleet in range";
  const cap = T.transportPerNavy*G.sea.get(seaId).navies - (G.usedCap.get(seaId)||0);
  if (cap<count) return "fleet capacity exceeded ("+T.transportPerNavy+" armies per navy)";
  G.usedCap.set(seaId,(G.usedCap.get(seaId)||0)+count);
  G.movedZones.add(from);
  fz.armies-=count;
  if (tz.owner===p.idx){ tz.armies+=count; log(p.short+" ferries "+count+" armies to "+zoneById[to].name+"."); return null; }
  if (tz.owner===-1||tz.armies===0){ tz.owner=p.idx; tz.armies=count; log(p.short+" lands in "+zoneById[to].name+" unopposed."); return null; }
  const defCount=tz.armies, defPower=POWERS[tz.owner].short;
  const r=resolveCombat(count,defCount);
  if (r.attackerWins){
    tz.owner=p.idx; tz.armies=r.attackers;
    log(p.short+" makes an amphibious assault on "+zoneById[to].name+" ("+count+" vs "+defCount+"). Victory!");
    checkEliminations();
  } else {
    tz.armies=r.defenders;
    log(p.short+"'s amphibious landing at "+zoneById[to].name+" fails. "+defPower+" holds.");
  }
  return null;
}

function doLaunch(p, targetId){
  if (G.clouds>=T.cloudsMax) return "the world is already doomed";
  const tz=G.zones.get(targetId);
  if (!tz) return "bad target";
  if (tz.irr) return "already destroyed";
  if (p.nukes<1) return "no nukes in arsenal";
  p.nukes--;
  for (const op of G.powers) if (op.alive && op.lstars>0){
    if (1+Math.floor(Math.random()*6)>=T.lstarIntercept){
      op.lstars--;
      log("🛰 "+op.short+"'s L-Star intercepts and destroys "+p.short+"'s missile!");
      sfx("warn");
      return null;
    }
  }
  tz.armies=0; tz.owner=-1; tz.irr=true; G.clouds++;
  log("💥 "+p.short+" NUKES "+zoneById[targetId].name+"! ("+G.clouds+" mushroom clouds on the board)");
  sfx("boom");
  checkEliminations();
  if (G.over) return null;
  if (G.clouds>=T.winterCheck){
    const need=(T.cloudsMax+1)-G.clouds;
    const roll=1+Math.floor(Math.random()*6);
    if (roll>=need){
      G.winter=true;
      log("The die shows "+roll+" (needed "+need+"). The sky darkens forever.");
      endGame(null,"NUCLEAR WINTER — THE EARTH IS DEAD. EVERYONE LOSES.");
      return null;
    }
    log("The die shows "+roll+" (needed "+need+"). The world shudders but survives… for now.");
  }
  return null;
}
function doAid(p, targetIdx, kind, qty){
  qty=Math.floor(qty);
  const t=G.powers[targetIdx];
  if (!t.alive) return "target eliminated";
  if (qty<1) return "bad amount";
  if (kind==="money"){

    log(p.short+" grants "+t.short+" "+fmt(qty)+" in foreign aid.");
  } else {
    if (p.res[kind]<qty) return "not enough "+kind;
    p.res[kind]-=qty; t.res[kind]+=qty;
    log(p.short+" ships "+qty+" "+kind+" to "+t.short+".");
  }

}
function doProspect(p){
  const deck=[];
  for (let id=1;id<=LZ.length;id++) for (const r of zoneById[id].res) deck.push([id,r]);
  shuffle(deck);
  const cap=capFor(p);
  const results=[];
  for (let i=0;i<T.prospectCards;i++){
    const [id,r]=deck[i];
    const z=G.zones.get(id);
    if (z.owner===p.idx && !z.irr && p.res[r]<cap){ p.res[r]++; results.push([id,r,true]); }
    else results.push([id,r,false]);
  }
  const got=results.filter(r=>r[2]).length;
  sfx("deal");
  log(p.short+" prospects "+got+"/"+T.prospectCards+" resource cards"+(got>0?" (+"+got+" resources)":"."));
  return results;
}
function checkEliminations(){
  if (G.over) return;
  for (const p of G.powers) if (p.alive){
    let zc=0, nc=0;
    for (const [,z] of G.zones) if (z.owner===p.idx && !z.irr) zc++;
    for (const [,s] of G.sea) if (s.owner===p.idx) nc+=s.navies;
    if (zc===0 && nc===0){
      p.alive=false;
      log("☠ "+p.name+" has been eliminated!");
    }
  }
  const alive=G.powers.filter(p=>p.alive);
  if (alive.length<=1){
    if (alive.length===1 && G.startingPowers>1) endGame(alive[0], alive[0].name+" ACHIEVES GLOBAL SUPREMACY!");
    else if (alive.length===0) endGame(null,"The last great power has fallen. The world descends into chaos…");
  }
}

function runAI(p){
  const delay = window.__fastAI ? (5+Math.random()*15) : (220+Math.random()*420);
  setTimeout(()=>{
    try {
      if (G.over || G.powers[G.phasePlayer].idx!==p.idx) return;
      aiPhase(p);
      if (G.mode==="online" && !G.net.isHost) return;
      if (!G.over) endTurn();
      else render();
    } catch(e){
      console.error("AI error:", e);
      try { if (!G.over) endTurn(); } catch(e2) { console.error("recovery failed", e2); }
    }
  }, delay);
}
function aiPhase(p){
  switch(G.phase){
    case "build": aiBuild(p); break;
    case "prospect": doProspect(p); break;
    case "economic": aiEconomic(p); break;
    case "political": break;
    case "military": if(AI_DIFF==="easy"&&Math.random()<0.25){break;} aiMilitary(p); break;
    case "rnd": aiRnd(p); break;
    case "nuclear": aiNuclear(p); break;
  }
  render();
}
function aiBuild(p){
  let sets = 0;
  if (p.money>=T.aiBuildTier3) sets=3; else if (p.money>=T.aiBuildTier2) sets=2; else if (p.money>=T.aiBuildTier1) sets=1;
  if (sets===0 || RES_KEYS.some(k=>p.res[k]<T.buildResourceCost)) return;
  let pool = [];
  for (const [id,z] of G.zones){
    if (z.owner===p.idx && !z.irr && [...(landAdj.get(id)||new Set())].some(t=>G.zones.get(t).owner!==p.idx)) pool.push(id);
  }
  const homePool = p.home.filter(id=>!G.zones.get(id).irr);
  if (pool.length && Math.random()<0.6) pool = pool;
  else pool = homePool;
  const coastalPool = pool.filter(coastalLand);
  const hasNavyNear = (coastalPool.length?coastalPool:pool).some(lzid=>[...(landToSea.get(lzid)||[])].some(s=>G.sea.get(s).owner===p.idx && G.sea.get(s).navies>0));
  if (!hasNavyNear && Math.random()<0.3){
    const s = p.navy;
    if (coastalSeaZones.has(s) && (G.sea.get(s).owner===p.idx||[...(seaToLand.get(s)||[])].some(lz=>G.zones.get(lz).owner===p.idx))){
      if (!doBuildNavy(p,s,Math.min(2,sets))) return;
    }
  }
  const zone = pool.slice().sort((a,b)=>G.zones.get(a).armies-G.zones.get(b).armies)[0];
  const err = doBuildArmy(p, zone, sets);
  if (err && sets>1){ doBuildArmy(p, zone, 1); }
}
function aiEconomic(p){
  for (const r of RES_KEYS){
    if (p.res[r]>=8 && G.market[r]>=28) doSell(p,r,p.res[r]-6);
  }
  if (p.money>1100){
    const need = p.nukeDev ? T.nukeResource : RES_KEYS[0];
    if (p.res[need] < 6) doBuy(p,need,Math.min(2,6-p.res[need]));
  }
}
function aiMilitary(p){
  let guard=0;
  while(guard++<50){
    const cands=[];
    for (const [id,z] of G.zones){
      if (z.owner===p.idx && z.armies>1 && !G.movedZones.has(id)){
        for (const t of (landAdj.get(id)||new Set())){
          const tz=G.zones.get(t);
          if (tz.owner!==p.idx && !tz.irr) cands.push({from:id,to:t,arm:z.armies});
        }
      }
    }
    if (!cands.length) break;
    for (const c of cands){
      const tz=G.zones.get(c.to);
      const op = tz.owner>=0 ? G.powers[tz.owner] : null;
      const isCap = op && (op.supply===c.to || op.home.includes(c.to));
      c.score = -100;
      const avail = c.arm-1;
      if (tz.owner===-1){ c.score = 100+Math.random()*20; }
      else if (tz.armies===0){ c.score = 60+Math.random()*15; }
      else if (avail >= tz.armies*1.3){ c.score = 30 + tz.armies*3 + (isCap?25:0) + Math.random()*10; }
      else if (avail >= tz.armies*0.9 && Math.random()<0.35){ c.score = 5 + tz.armies + (isCap?15:0) + Math.random()*5; }
    }
    cands.sort((a,b)=>b.score-a.score);
    const c=cands[0];
    if (c.score<0) break;
    const tz=G.zones.get(c.to);
    let send;
    if (tz.owner===-1||tz.armies===0) send=Math.max(1,c.arm-1);
    else send=Math.max(1,Math.min(c.arm-1,Math.ceil(tz.armies*1.4)));
    doMoveArmy(p,c.from,c.to,send);
  }
  for (const [id,z] of G.zones){
    if (z.owner===p.idx && z.armies>6 && !G.movedZones.has(id)){
      for (const n of (landAdj.get(id)||new Set())){
        const nz=G.zones.get(n);
        if (nz.owner===p.idx && [...(landAdj.get(n)||new Set())].some(t=>G.zones.get(t).owner!==p.idx)){
          doMoveArmy(p,id,n,Math.max(1,z.armies-5));
          break;
        }
      }
    }
  }
  for (const [sId,s] of G.sea){
    if (s.owner===p.idx && s.navies>0 && !G.movedSeas.has(sId)){
      for (const n of (seaAdj.get(sId)||new Set())){
        const ns=G.sea.get(n);
        if (ns.owner!==p.idx && ns.navies===0){ doMoveNavy(p,sId,n,s.navies); break; }
      }
    }
  }
}
function aiRnd(p){
  const anyNukes = G.powers.some(o=>o.alive && o.nukes>0 && o.idx!==p.idx);
  if (!p.nukeDev && p.money>=800 && (anyNukes || p.money>=1100)){
    if (!doResearch(p,"nuke")) { if (p.money>=650 && p.res[T.nukeResource]>=T.nukeResourceCost && p.nukes<2) doBuildNuke(p); }
  } else if (p.nukeDev && p.money>=650 && p.res[T.nukeResource]>=T.nukeResourceCost && p.nukes<2){
    doBuildNuke(p);
  }
  if (anyNukes && !p.lstarDev && p.money>=600){
    doResearch(p,"lstar");
  }
  if (p.lstarDev && p.lstars===0 && p.money>=350) doBuildLstar(p);
}
function aiNuclear(p){
  if (p.nukes<=0 || G.clouds>=11) return;
  let best=null;
  for (const [id,z] of G.zones){
    if (z.owner<0 || z.owner===p.idx || z.irr) continue;
    const op=G.powers[z.owner];
    const isCap = op.supply===id || op.home.includes(id);
    const score = (isCap?16:0)+z.armies;
    if (!best || score>best.score) best={id,score};
  }
  if (best && best.score>=15) doLaunch(p,best.id);
}

function rr(x,y,w,h,r){
  const ctx=boardCtx;
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
function hash2(x,y){ const n=(x*374761393+y*668265263); return ((n^(n>>13))*1274126177>>>0)/4294967296; }
function label(x,y,text,size,color,bold){
  const ctx=boardCtx;
  ctx.font=(bold===false?"":"bold ")+size+"px Georgia";
  ctx.textAlign="center"; ctx.textBaseline="middle";
  ctx.lineWidth=Math.max(3,size*0.5); ctx.lineJoin="round";
  ctx.strokeStyle="rgba(8,14,24,0.72)"; ctx.strokeText(text,x,y);
  ctx.fillStyle=color; ctx.fillText(text,x,y);
}
function drawStar(x,y,r,color){
  const ctx=boardCtx;
  ctx.beginPath();
  for(let i=0;i<10;i++){
    const a=-Math.PI/2+i*Math.PI/5;
    const rad=i%2?r*0.45:r;
    const px=x+Math.cos(a)*rad, py=y+Math.sin(a)*rad;
    i?ctx.lineTo(px,py):ctx.moveTo(px,py);
  }
  ctx.closePath();
  ctx.fillStyle=color; ctx.fill();
  ctx.strokeStyle="rgba(255,255,255,0.9)"; ctx.lineWidth=1; ctx.stroke();
}
function cubeToken(x,y,r,color,text){
  const ctx=boardCtx;
  ctx.fillStyle="rgba(0,0,0,0.35)";
  rr(x-r+2,y-r+2,r*2,r*2,r*0.4); ctx.fill();
  ctx.fillStyle=shade(color,-0.3);
  rr(x-r,y-r,r*2,r*2,r*0.4); ctx.fill();
  ctx.fillStyle=shade(color,0.28);
  rr(x-r*0.78,y-r*0.78,r*1.56,r*1.56,r*0.28); ctx.fill();
  ctx.strokeStyle="rgba(255,255,255,0.9)"; ctx.lineWidth=1;
  rr(x-r,y-r,r*2,r*2,r*0.4); ctx.stroke();
  ctx.fillStyle="#fff"; ctx.font="bold "+Math.round(r*1.2)+"px Arial";
  ctx.textAlign="center"; ctx.textBaseline="middle";
  ctx.fillText(text,x,y+1);
}
function shipToken(x,y,r,color,text){
  const ctx=boardCtx;
  const w=r*1.7, h=r*1.15;
  ctx.fillStyle="rgba(0,0,0,0.35)";
  rr(x-w/2+2,y-h/2+2,w,h,r*0.3); ctx.fill();
  ctx.fillStyle=shade(color,-0.14);
  rr(x-w/2,y-h/2,w,h,r*0.3); ctx.fill();
  ctx.fillStyle=shade(color,0.14);
  rr(x-w/2+2,y-h/2+1,w-4,h*0.42,r*0.16); ctx.fill();
  ctx.strokeStyle="rgba(255,255,255,0.9)"; ctx.lineWidth=1;
  rr(x-w/2,y-h/2,w,h,r*0.3); ctx.stroke();
  ctx.fillStyle="#fff"; ctx.font="bold "+Math.round(r*0.95)+"px Arial";
  ctx.textAlign="center"; ctx.textBaseline="middle";
  ctx.fillText(text,x,y+1);
}
function mushroom(x,y,s){
  const ctx=boardCtx;
  ctx.fillStyle="rgba(24,20,14,0.92)";
  rr(x-s*0.55,y-s*0.35,s*1.1,s*0.7,s*0.2); ctx.fill();
  ctx.beginPath();
  ctx.arc(x-s*0.8,y-s*0.5,s*0.5,0,7);
  ctx.arc(x,y-s*0.6,s*0.68,0,7);
  ctx.arc(x+s*0.8,y-s*0.5,s*0.5,0,7);
  ctx.fill();
}
function borderStyle(v,nv){
  const isLand=a=>a>0&&a<=LZ.length;
  if(v===250||nv===250) return {c:"rgba(245,241,225,0.3)",w:1};
  if(isLand(v)||isLand(nv)) return {c:"rgba(247,243,229,0.95)",w:2};
  return {c:"rgba(255,255,255,0.18)",w:1};
}
function drawMap(){
  const ctx=boardCtx;
  const OCEAN_DEEP="#1a5ba0", OCEAN="#2566b1", OCEAN_COAST="#4d93d2", OCEAN_LAND="#39493c", LAND_NEUTRAL="#7b8651", LAND_IRR="#5a5234";
  ctx.fillStyle=OCEAN_DEEP; ctx.fillRect(0,0,CW,CH);
  const isOcean=v=>v===0||v>=101;
  const OY=CROP_TOP*CELL, yBot=(GH-CROP_TOP-CROP_BOT)*CELL;
  ctx.strokeStyle="rgba(255,255,255,0.09)"; ctx.lineWidth=1;
  for(let gx=30;gx<GW;gx+=30){
    const x=gx*CELL; let y0=null;
    for(let y=CROP_TOP;y<GH-CROP_BOT;y++){
      if(isOcean(grid[y*GW+gx])){ if(y0===null)y0=y; }
      else if(y0!==null){ ctx.beginPath(); ctx.moveTo(x,(y0-CROP_TOP)*CELL); ctx.lineTo(x,(y-CROP_TOP)*CELL); ctx.stroke(); y0=null; }
    }
    if(y0!==null){ ctx.beginPath(); ctx.moveTo(x,(y0-CROP_TOP)*CELL); ctx.lineTo(x,yBot); ctx.stroke(); }
  }
  for(let gy=CROP_TOP;gy<GH-CROP_BOT;gy++){
    if(gy%30!==0) continue;
    const y=(gy-CROP_TOP)*CELL; let x0=null;
    for(let x=0;x<GW;x++){
      if(isOcean(grid[gy*GW+x])){ if(x0===null)x0=x; }
      else if(x0!==null){ ctx.beginPath(); ctx.moveTo(x0*CELL,y); ctx.lineTo(x*CELL,y); ctx.stroke(); x0=null; }
    }
    if(x0!==null){ ctx.beginPath(); ctx.moveTo(x0*CELL,y); ctx.lineTo(GW*CELL,y); ctx.stroke(); }
  }
  for (let y=CROP_TOP;y<GH-CROP_BOT;y++) for (let x=0;x<GW;x++){
    const v=grid[y*GW+x];
    const h=hash2(x,y);
    let col;
    if (v===0) col=shade(OCEAN,(h-0.5)*0.06);
    else if (v===250) col=shade(OCEAN_LAND,(h-0.5)*0.05);
    else if (v>=101) col = coastalSeaZones.has(v) ? shade(OCEAN_COAST,(h-0.5)*0.05) : shade(OCEAN,(h-0.5)*0.05);
    else {
      const z=G.zones.get(v);
      if (z.irr) col=shade(LAND_IRR,(h-0.5)*0.06);
      else if (z.owner>=0) col=shade(POWERS[z.owner].color,(h-0.5)*0.03);
      else col=shade(LAND_NEUTRAL,(h-0.5)*0.09);
    }
    ctx.fillStyle=col;
    ctx.fillRect(x*CELL,(y-CROP_TOP)*CELL,CELL+0.5,CELL+0.5);
  }
  for (let y=CROP_TOP;y<GH-CROP_BOT;y++) for (let x=0;x<GW;x++){
    const v=grid[y*GW+x];
    if (x<GW-1 && grid[y*GW+x+1]!==v){
      const st=borderStyle(v,grid[y*GW+x+1]); ctx.strokeStyle=st.c; ctx.lineWidth=st.w;
      ctx.beginPath(); ctx.moveTo((x+1)*CELL,(y-CROP_TOP)*CELL); ctx.lineTo((x+1)*CELL,(y-CROP_TOP+1)*CELL); ctx.stroke();
    }
    if (y<GH-CROP_BOT-1 && grid[(y+1)*GW+x]!==v){
      const st=borderStyle(v,grid[(y+1)*GW+x]); ctx.strokeStyle=st.c; ctx.lineWidth=st.w;
      ctx.beginPath(); ctx.moveTo(x*CELL,(y-CROP_TOP+1)*CELL); ctx.lineTo((x+1)*CELL,(y-CROP_TOP+1)*CELL); ctx.stroke();
    }
  }
  if (ui.mode!=="idle" && ui.origin!=null){
    for (const t of validTargets(ui.mode,ui.origin)) drawZoneOutline(t,"#7cff7c");
    drawZoneOutline(ui.origin,"#ffffff");
  }
  if (ui.selected!=null && zoneById[ui.selected]) drawZoneOutline(ui.selected,"#ffe066");
  for (let id=1;id<=LZ.length;id++){
    const c=MAP.centroids[id]; if (!c) continue;
    const z=G.zones.get(id);
    const cx=(c.x+0.5)*CELL, cy=(c.y+0.5)*CELL-OY;
    const isSupply=POWERS.some(p=>p.supply===id);
    const tokenR = z.armies>0 ? Math.min(26,Math.max(13,9+Math.sqrt(z.armies)*2.3)) : 0;
    if (isSupply){
      const sy = cy - (tokenR>0 ? tokenR+9 : 11);
      ctx.beginPath(); ctx.arc(cx,sy,8,0,7);
      ctx.fillStyle="rgba(8,14,24,0.45)"; ctx.fill();
      ctx.beginPath(); ctx.arc(cx,sy,6.5,0,7);
      ctx.strokeStyle="rgba(255,224,102,0.95)"; ctx.lineWidth=1.5; ctx.stroke();
      drawStar(cx,sy,5,"#ffe066");
    }
    if (z.irr){
      mushroom(cx,cy,7);
      continue;
    }
    if (tokenR>0){
      cubeToken(cx,cy,tokenR,POWERS[z.owner].color,String(z.armies));
    } else if (z.owner>=0){
      ctx.beginPath(); ctx.arc(cx,cy,2.6,0,7);
      ctx.fillStyle="#fff"; ctx.fill();
      ctx.strokeStyle="rgba(0,0,0,0.5)"; ctx.lineWidth=0.8; ctx.stroke();
    }
    if (c.n>=42){
      const fs=Math.max(7,Math.min(12,0.95*Math.sqrt(c.n)));
      label(cx,cy+(tokenR>0?tokenR+9:11),zoneById[id].name,fs,"#fff",false);
    }
  }
  for (let id=SEA_OFF+1;id<=SEA_OFF+SZ.length;id++){
    const s=G.sea.get(id);
    const c=seaCentroid[id];
    if (c && s.navies>0){
      const r=Math.min(20,Math.max(11,7+Math.sqrt(s.navies)*2.2));
      shipToken(c.x,c.y-OY,r,POWERS[s.owner].color,String(s.navies));
    }
    if (c && c.n>=1000){
      ctx.font="italic 10.5px Georgia";
      ctx.textAlign="center"; ctx.textBaseline="middle";
      ctx.lineWidth=3; ctx.lineJoin="round";
      ctx.strokeStyle="rgba(8,14,24,0.7)"; ctx.strokeText(zoneById[id].name,c.x,c.y-OY-16);
      ctx.fillStyle="rgba(255,255,255,0.78)"; ctx.fillText(zoneById[id].name,c.x,c.y-OY-16);
    }
  }
}
function drawZoneOutline(id,color){
  const ctx=boardCtx;
  const OY=CROP_TOP*CELL;
  ctx.strokeStyle=color; ctx.lineWidth=2;
  for (let y=CROP_TOP;y<GH-CROP_BOT;y++) for (let x=0;x<GW;x++){
    if (grid[y*GW+x]!==id) continue;
    const py=(y-CROP_TOP)*CELL;
    if (x===0||grid[y*GW+x-1]!==id){ ctx.beginPath(); ctx.moveTo(x*CELL,py); ctx.lineTo(x*CELL,py+CELL); ctx.stroke(); }
    if (x===GW-1||grid[y*GW+x+1]!==id){ ctx.beginPath(); ctx.moveTo((x+1)*CELL,py); ctx.lineTo((x+1)*CELL,py+CELL); ctx.stroke(); }
    if (y===CROP_TOP||grid[(y-1)*GW+x]!==id){ ctx.beginPath(); ctx.moveTo(x*CELL,py); ctx.lineTo((x+1)*CELL,py); ctx.stroke(); }
    if (y===GH-CROP_BOT-1||grid[(y+1)*GW+x]!==id){ ctx.beginPath(); ctx.moveTo(x*CELL,py+CELL); ctx.lineTo((x+1)*CELL,py+CELL); ctx.stroke(); }
  }
}

function validTargets(mode, origin){
  const p=G.powers[G.phasePlayer];
  if (mode==="army") return [...(landAdj.get(origin)||new Set())].filter(id=>!G.zones.get(id).irr);
  if (mode==="navy"){
    const reach=new Set();
    let frontier=new Set([origin]);
    for(let d=0;d<T.navyRange;d++){
      const next=new Set();
      for(const z of frontier) for(const n of (seaAdj.get(z)||new Set())) if(!reach.has(n) && n!==origin) next.add(n);
      next.forEach(n=>reach.add(n));
      frontier=next;
    }
    return [...reach];
  }
  if (mode==="transport"){
    const out=new Set();
    for (const sId of (landToSea.get(origin)||new Set())){
      const ss=G.sea.get(sId);
      if (ss.owner===p.idx && ss.navies>0) for (const lz of (seaToLand.get(sId)||new Set())) if (lz!==origin) out.add(lz);
    }
    return [...out].filter(id=>!G.zones.get(id).irr);
  }
  if (mode==="nuke") return [...G.zones.keys()].filter(id=>!G.zones.get(id).irr);
  return [];
}

function onZoneClick(id){
  if (!G||G.over) return;
  ui.selected=id;
  const p=G.powers[G.phasePlayer];
  const ctrl=turnController(G.phasePlayer);
  const isLand=id<=LZ.length;
  if (G.phase!=="military" || ctrl!=="local"){ render(); return; }
  const z=isLand?G.zones.get(id):null;
  const s=isLand?null:G.sea.get(id);
  if (ui.mode==="army"){
    if (id===ui.origin){ setMode("idle"); render(); return; }
    const from=ui.origin, fz=G.zones.get(from);
    const max=(G.zones.get(id).owner===p.idx)?fz.armies:Math.max(0,fz.armies-1);
    qtyDialog("Move "+zoneById[id].name, max, n=>{
      act({t:"moveArmy",from,to:id,count:n},()=>{
        const e=doMoveArmy(p,from,id,n);
        if(e)toast(e); ui.mode="idle"; ui.origin=null;
        render();
      });
    });
    return;
  }
  if (ui.mode==="navy"){
    if (id===ui.origin){ setMode("idle"); render(); return; }
    const from=ui.origin, fs=G.sea.get(from);
    qtyDialog("Move fleet to "+zoneById[id].name, fs.navies, n=>{
      act({t:"moveNavy",from,to:id,count:n},()=>{
        const e=doMoveNavy(p,from,id,n);
        if(e)toast(e); ui.mode="idle"; ui.origin=null;
        render();
      });
    });
    return;
  }
  if (ui.mode==="transport"){
    if (id===ui.origin){ setMode("idle"); render(); return; }
    const from=ui.origin, fz=G.zones.get(from);
    let cap=0;
    for (const sId of (landToSea.get(from)||new Set())){
      const ss=G.sea.get(sId);
      if (ss.owner===p.idx && ss.navies>0 && (seaToLand.get(sId)||new Set()).has(id)) cap+=T.transportPerNavy*ss.navies-(G.usedCap.get(sId)||0);
    }
    let max=Math.min(fz.armies,cap);
    if (G.zones.get(id).owner!==p.idx) max=Math.min(max,Math.max(0,fz.armies-1));
    qtyDialog("Transport to "+zoneById[id].name, max, n=>{
      act({t:"transport",from,to:id,count:n},()=>{
        const e=doTransport(p,from,id,n);
        if(e)toast(e); ui.mode="idle"; ui.origin=null;
        render();
      });
    });
    return;
  }
  if (ui.mode==="nuke"){
    if (isLand){
      act({t:"launch",target:id},()=>{
        const e=doLaunch(p,id);
        if(e)toast(e); ui.mode="idle"; ui.origin=null;
        render();
      });
    }
    return;
  }
  if (ui.mode==="idle"){
    if ((z&&z.owner===p.idx&&z.armies>0&&!z.irr) || (s&&s.owner===p.idx&&s.navies>0)){
      ui.origin=id;
    } else {
      ui.origin=null;
    }
    render();
  }
}

let netState={ myConnId:null, roomCode:null, isHost:false, myPowerIdx:-1, myName:"Player" };
const NET_SESSION_KEY="bgn_supremacy_net_session";
function saveNetSession(){
  if (!netState.roomCode) return;
  try{ localStorage.setItem(NET_SESSION_KEY, JSON.stringify({roomCode:netState.roomCode, name:netState.myName, isHost:netState.isHost, myPowerIdx:netState.myPowerIdx})); }catch(e){}
}
function clearNetSession(){ try{ localStorage.removeItem(NET_SESSION_KEY); }catch(e){} }

function netConnect(opts){
  closeModal();
  setStatus("Connecting…");
  const name = opts.name || "Player";
  try {
    socket = root.createServerSocket();
  } catch(e){ openModal('<h2>Network Error</h2><p>'+e+'</p><button onclick="openSetup()">Back</button>'); return; }
  socket.addEventListener("open", ()=>{
    const payload = JSON.stringify({name, max:CONFIG.maxPlayers});
    const p = opts.isHost ? socket.rpc.createRoom(payload) : socket.rpc.joinRoom(JSON.stringify({code:opts.code, name}));
    p.then(reply=>{
      let info; try{ info=JSON.parse(reply); }catch(e){ openModal('<h2>Error</h2><p>'+reply+'</p><button onclick="openSetup()">Back</button>'); return; }
      if (info.error){ openModal('<h2>Error</h2><p>'+info.error+'</p><button onclick="openSetup()">Back</button>'); return; }
      netState.myConnId=info.connId; netState.roomCode=info.code; netState.isHost=opts.isHost; netState.myName=name;
      saveNetSession();
      showLobby(info.players, info);
    }).catch(e=>{ openModal('<h2>Network Error</h2><p>'+(e&&e.message||e)+'</p><button onclick="openSetup()">Back</button>'); });
  });
  socket.addEventListener("message", ev=>{
    let msg; try{ msg=JSON.parse(ev.data); }catch(e){ return; }
    onNetMessage(msg);
  });
  socket.addEventListener("close", ev=>{
    toast("Connection closed"+(ev.code?(" ("+ev.code+")"):""));
  });
}
function rejoinNet(){
  let saved; try{ saved=JSON.parse(localStorage.getItem(NET_SESSION_KEY)||"null"); }catch(e){ saved=null; }
  if (!saved || !saved.roomCode) return false;
  closeModal();
  setStatus("Reconnecting to room "+saved.roomCode+"…");
  const name = String(saved.name||"Player").slice(0,20);
  const power = (typeof saved.myPowerIdx==="number") ? saved.myPowerIdx : -1;
  try { socket = root.createServerSocket(); }
  catch(e){ setStatus("Ready."); openSetup(); return true; }
  socket.addEventListener("open", ()=>{
    socket.rpc.rejoinRoom(JSON.stringify({code:saved.roomCode, name, power})).then(reply=>{
      let info; try{ info=JSON.parse(reply); }catch(e){ setStatus("Ready."); openSetup(); return; }
      if (info.error){ setStatus("Ready."); openSetup(); toast(info.error); return; }
      netState.myConnId=info.connId; netState.roomCode=info.code; netState.isHost=false; netState.myName=name; netState.myPowerIdx=power;
      saveNetSession();
      if (info.state){ closeModal(); applyNetState(info.state); }
      else showLobby(info.players, info);
    }).catch(()=>{ setStatus("Ready."); openSetup(); });
  });
  socket.addEventListener("message", ev=>{
    let msg; try{ msg=JSON.parse(ev.data); }catch(e){ return; }
    onNetMessage(msg);
  });
  socket.addEventListener("close", ev=>{
    toast("Connection closed"+(ev.code?(" ("+ev.code+")"):""));
  });
  return true;
}
function onNetMessage(msg){
  if (msg.f && msg.f===netState.myConnId) return;
  if (msg.t==="err"){ toast(typeof msg.d==="string"?msg.d:"Move rejected"); return; }
  if (msg.t==="lobby"){ showLobby(msg.players,{host:netState.isHost}); return; }
  if (msg.t==="start"){
    const me = msg.players.find(x=>x.connId===netState.myConnId);
    startOnlineGame(msg.players, me);
    return;
  }
  if (msg.t==="leave"){
    toast("A player left the room");
    if (G && G.net && G.net.isHost && msg.players){
      const rem = new Set(msg.players.filter(x=>x.connId!==netState.myConnId).map(x=>x.power));
      if (G.net) G.net.remote = rem;
      render();
    }
    if (msg.players) showLobby(msg.players,{host:netState.isHost});
    return;
  }
  if (msg.t==="hostchange"){
    toast("The host left the game");
    return;
  }
  if (msg.t==="chat"){ appendChat(msg.d); return; }
  if (msg.t==="rematch"){ doRematch(); return; }
  if (msg.t==="state"){
    applyNetState(msg.d);
    return;
  }
  if (msg.t==="result"){
    if (G && G.net && G.net.isHost) handleRemoteResult(msg.d);
    return;
  }
}

function copyRoomCode(){
  const code = netState.roomCode || "";
  if (navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(code).then(()=>toast("Code copied: "+code)).catch(()=>legacyCopy(code));
  } else legacyCopy(code);
  function legacyCopy(c){
    const t=document.createElement("textarea"); t.value=c; t.style.position="fixed"; t.style.opacity="0";
    document.body.appendChild(t); t.select();
    try{ document.execCommand("copy"); toast("Code copied: "+c); }catch(e){ toast("Copy failed — code: "+c); }
    document.body.removeChild(t);
  }
}

let lbSock=null, lbReady=false, lbBack=1000, lbQueue=null;
function lbConnect(){
  if(lbSock||!window.root||!window.root.createServerSocket)return;
  try{ lbSock=window.root.createServerSocket(); }catch(e){ return; }
  lbSock.addEventListener("open",function(){
    lbReady=true; lbBack=1000;
    if(lbQueue){ const q=lbQueue; lbQueue=null; lbSubmitNow(q.name,q.score,q.extra); }
  });
  lbSock.addEventListener("close",function(e){
    lbReady=false;
    if(e.code===4403)return;
    setTimeout(function(){ lbConnect(); }, Math.min(15000,lbBack)+Math.random()*500);
    lbBack=Math.min(15000,lbBack*2);
  });
}

function handleRemoteResult(d){
  if (G.over) return;
  if (d.player!==G.phasePlayer) return;
  if (turnController(G.phasePlayer)!=="remote") return;
  const p=G.powers[G.phasePlayer];
  for (const it of (d.intents||[])){
    const e=applyIntent(it);
    if (e) log("⚠ rejected "+it.t+" ("+e+")");
  }
  endTurn();
}
function applyIntent(it){
  const p=G.powers[G.phasePlayer];
  if (!p) return "no power";
  switch(it.t){
    case "buildArmy": return doBuildArmy(p,it.zone,it.sets);
    case "buildNavy": return doBuildNavy(p,it.sea,it.sets);
    case "sell": return doSell(p,it.r,it.qty);
    case "buy": return doBuy(p,it.r,it.qty);
    case "moveArmy": return doMoveArmy(p,it.from,it.to,it.count);
    case "moveNavy": return doMoveNavy(p,it.from,it.to,it.count);
    case "transport": return doTransport(p,it.from,it.to,it.count);
    case "research": return doResearch(p,it.kind);
    case "buildNuke": return doBuildNuke(p);
    case "buildLstar": return doBuildLstar(p);
    case "launch": return doLaunch(p,it.target);
    case "prospect": return doProspect(p);
    case "aid": return doAid(p,it.target,it.kind,it.qty);
  }
  return "unknown action";
}

function escapeHtml(s){
  return String(s).replace(/[&<>"']/g, function(c){
    return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c];
  });
}
function cleanText(s){
  return String(s).replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").trim().slice(0,200);
}

function zoneFromEvent(e){
  const rect=boardEl.getBoundingClientRect();
  const scaleX=CW/rect.width, scaleY=CH/rect.height;
  const x=(e.clientX-rect.left)*scaleX/CELL;
  const y=(e.clientY-rect.top)*scaleY/CELL;
  const gx=Math.floor(x), gy=Math.floor(y)+CROP_TOP;
  if (gx<0||gy<CROP_TOP||gx>=GW||gy>=GH-CROP_BOT) return null;
  const v=grid[gy*GW+gx];
  return (v===0||v===250)?null:v;
}
