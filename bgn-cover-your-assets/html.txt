/* ── Cover Your Assets server ────────────────────────────
   Rooms: 5-letter codes. Host creates and moves first; a guest
   joining with the code starts the match. The server keeps the
   deck, discard pile and BOTH hands authoritatively (private
   hands are pushed per-connection, never broadcast) and re-runs
   the full rules engine to validate every play. Public info —
   vaults, deck count, discard top, whose turn, scores — is
   broadcast via pubsub. Durable room state lives in the plugin
   `state` bytes (versioned binary layout).                       */
const CARD_N=44, NT=8, WILD_T=8;
const VAL=[5,10,15,20,25,30,40,50,15];
const NAMES=["Coins","Urn","Ring","Clock","Painting","Trophy","Diamond","Crown","Wild"];
const EMO=["🪙","⚱️","💍","🕰️","🖼️","🏆","💎","👑","⭐"];
const SCHEMA=3;
const SLOT=512;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const O_STAT=0,O_CODE=1,O_HF=9,O_SCHEMA=10,O_TURN=11,O_FLAGS=12,O_LAST=13,O_WIN=17,O_REASON=18,O_PLAYS=19,O_ENDTURN=20;
const O_DISCARD=24,O_DECK=69,O_HAND0=114,O_HAND1=159,O_VAULT0=204,O_VAULT1=300,O_SEQ=400;

/* ── pure rules engine (mirrors client — keep in sync) ── */
function cardType(id){ return id<40?Math.floor(id/5):8; }
function cardVal(id){ return VAL[cardType(id)]; }
function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); const t=a[i]; a[i]=a[j]; a[j]=t; } return a; }
function buildDeck(){ const d=[]; for(let i=0;i<CARD_N;i++)d.push(i); return shuffle(d); }
function newRoom(hostFirst){ return {turnIndex:-1,hostFirst:hostFirst||0,deck:buildDeck(),discard:[],hands:[[],[]],vaults:[[],[],[]],playsLeft:0,finalRound:false,endTurn:1e9,over:false,winner:null,reason:"",scores:null,seq:0,event:""}; }
function turnPlayer(g){ return (g.turnIndex+(g.hostFirst?1:0))%2; }
function vaultEmpty(g,p){ const v=g.vaults[p]; return !v[0]&&!v[1]&&!v[2]; }
function vaultVal(g,p){ let s=0; const v=g.vaults[p]; for(let i=0;i<3;i++){ const st=v[i]; if(st)for(let j=0;j<st.cards.length;j++)s+=cardVal(st.cards[j]); } return s; }
function stackVal(st){ let s=0; if(st)for(let j=0;j<st.cards.length;j++)s+=cardVal(st.cards[j]); return s; }
function draw(g,p,n){ let got=0; while(n>0&&g.deck.length){ g.hands[p].push(g.deck.pop()); n--; got++; } return got; }
function setFinalRound(g){ if(g.finalRound)return; g.finalRound=true; g.endTurn=g.turnIndex+(g.turnIndex%2===0?2:1); }
function legalKinds(g,p){
  const mine=g.vaults[p], opp=g.vaults[1-p], hand=g.hands[p], out=[];
  for(let i=0;i<hand.length;i++){
    const base=cardType(hand[i]);
    const types=base===8?[0,1,2,3,4,5,6,7]:[base];
    for(const ty of types){
      let emptySlot=-1, ownStack=-1, oppStack=-1;
      for(let s=0;s<3;s++){ const st=mine[s]; if(!st){ if(emptySlot===-1)emptySlot=s; } else if(st.type===ty) ownStack=s; }
      for(let s=0;s<3;s++){ const st=opp[s]; if(st&&st.type===ty){ oppStack=s; break; } }
      if(emptySlot!==-1&&ownStack===-1)out.push({i:i,kind:"start",type:ty,stack:emptySlot});
      if(ownStack!==-1)out.push({i:i,kind:"build",type:ty,stack:ownStack});
      if(oppStack!==-1&&(ownStack!==-1||emptySlot!==-1))out.push({i:i,kind:"cover",type:ty,stack:oppStack});
    }
  }
  return out;
}
function applyPlay(g,p,k){
  const hand=g.hands[p], id=hand[k.i];
  hand.splice(k.i,1);
  const mine=g.vaults[p], opp=g.vaults[1-p];
  if(k.kind==="start"){
    mine[k.stack]={type:k.type,cards:[id]};
  }else if(k.kind==="build"){
    mine[k.stack].cards.push(id);
  }else{
    const st=opp[k.stack];
    opp[k.stack]=null;
    let target=-1;
    for(let s=0;s<3;s++){ if(mine[s]&&mine[s].type===k.type){ target=s; break; } }
    if(target===-1){ for(let s=0;s<3;s++){ if(!mine[s]){ target=s; break; } } }
    if(target===-1)return false;
    if(mine[target]){ mine[target].cards.push.apply(mine[target].cards,st.cards); mine[target].cards.push(id); }
    else mine[target]={type:k.type,cards:st.cards.concat([id])};
  }
  return true;
}
function deal(g){ for(let k=0;k<5;k++){ g.hands[0].push(g.deck.pop()); g.hands[1].push(g.deck.pop()); } g.turnIndex=0; startTurn(g); }
function startTurn(g){ const got=draw(g,turnPlayer(g),2); if(got<2)setFinalRound(g); g.playsLeft=2; }
function finishTurn(g){
  const p=turnPlayer(g);
  if(vaultEmpty(g,p)){ const got=draw(g,p,5); if(got<5)setFinalRound(g); }
  g.turnIndex++;
  if(g.finalRound&&g.turnIndex>=g.endTurn){ endByScore(g); return; }
  startTurn(g);
}
function endByScore(g){
  g.over=true; g.reason="deckOut";
  const s0=vaultVal(g,0), s1=vaultVal(g,1);
  g.scores=[s0,s1];
  g.winner=s0>s1?0:s1>s0?1:"draw";
}
function pName(p){ return p===0?"Host":"Guest"; }
function cardTxt(ty){ return NAMES[ty]+" ("+EMO[ty]+")"; }

/* ── durable state helpers ── */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+O_LAST]<<24)|(state[o+O_LAST+1]<<16)|(state[o+O_LAST+2]<<8)|state[o+O_LAST+3]; }
function slotStatus(i){ return state[off(i)+O_STAT]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function reasonCode(r){ return r==="deckOut"?1:r==="forfeit"?2:r==="resign"?3:0; }
function reasonName(c){ return c===1?"deckOut":c===2?"forfeit":c===3?"resign":""; }
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o+O_STAT]=room.status;
  writeCode(o,room.code);
  state[o+O_HF]=room.hostFirst;
  state[o+O_SCHEMA]=SCHEMA;
  state[o+O_TURN]=room.turnIndex;
  state[o+O_FLAGS]=room.finalRound?1:0;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+O_LAST]=(la>>>24)&255; state[o+O_LAST+1]=(la>>>16)&255; state[o+O_LAST+2]=(la>>>8)&255; state[o+O_LAST+3]=la&255;
  state[o+O_WIN]=room.winner==="draw"?3:room.winner===1?2:room.winner===0?1:0;
  state[o+O_REASON]=reasonCode(room.reason);
  state[o+O_PLAYS]=room.playsLeft;
  state[o+O_ENDTURN]=Math.min(65535,room.endTurn)&255; state[o+O_ENDTURN+1]=(Math.min(65535,room.endTurn)>>8)&255;
  state[o+O_SEQ]=room.seq&255; state[o+O_SEQ+1]=(room.seq>>8)&255;
  let p=o+O_DISCARD; state[p]=room.discard.length; for(let i=0;i<room.discard.length&&i<44;i++)state[p+1+i]=room.discard[i];
  p=o+O_DECK; state[p]=room.deck.length; for(let i=0;i<room.deck.length&&i<44;i++)state[p+1+i]=room.deck[i];
  for(let pi=0;pi<2;pi++){
    p=o+(pi===0?O_HAND0:O_HAND1); state[p]=room.hands[pi].length;
    for(let i=0;i<room.hands[pi].length&&i<44;i++)state[p+1+i]=room.hands[pi][i];
    p=o+(pi===0?O_VAULT0:O_VAULT1);
    for(let s=0;s<3;s++){ const st=room.vaults[pi][s]; if(!st)continue; const q=p+s*32; state[q]=st.type; state[q+1]=st.cards.length; for(let j=0;j<st.cards.length&&j<30;j++)state[q+2+j]=st.cards[j]; }
  }
}
function loadRoom(i){
  const o=off(i), st=slotStatus(i);
  const readHand=function(bp){ const n=state[bp]; const a=[]; for(let k=0;k<n&&k<44;k++)a.push(state[bp+1+k]); return a; };
  const readVault=function(bp){ const v=[[],[],[]]; for(let s=0;s<3;s++){ const q=bp+s*32, ty=state[q], n=state[q+1]; if(!n)continue; const cards=[]; for(let j=0;j<n&&j<30;j++)cards.push(state[q+2+j]); v[s]={type:ty,cards:cards}; } return v; };
  const hands=[readHand(o+O_HAND0),readHand(o+O_HAND1)];
  const vaults=[readVault(o+O_VAULT0),readVault(o+O_VAULT1)];
  const endTurn=(state[o+O_ENDTURN+1]<<8)|state[o+O_ENDTURN];
  const seq=(state[o+O_SEQ+1]<<8)|state[o+O_SEQ];
  const win=state[o+O_WIN];
  const room={
    slot:i, code:codeAt(i), status:st, hostFirst:state[o+O_HF],
    turnIndex:state[o+O_TURN], finalRound:state[o+O_FLAGS]===1, playsLeft:state[o+O_PLAYS],
    endTurn:endTurn||1e9, seq, event:"",
    deck:[], discard:[], hands, vaults,
    over:st===3, winner:win===3?"draw":win===2?1:win===1?0:null, reason:reasonName(state[o+O_REASON]),
    hostId:-1, guestId:-1, lastAct:readLastAct(i)
  };
  const dl=state[o+O_DISCARD], dp=o+O_DISCARD; for(let k=0;k<dl&&k<44;k++)room.discard.push(state[dp+1+k]);
  const klen=state[o+O_DECK], kp=o+O_DECK; for(let k=0;k<klen&&k<44;k++)room.deck.push(state[kp+1+k]);
  room.scores=[vaultVal(room,0),vaultVal(room,1)];
  return room;
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(state[o+O_SCHEMA]!==SCHEMA){ state.fill(0,o,o+SLOT); continue; }
    if(st===1){ state.fill(0,o,o+SLOT); continue; }   // abandoned waiting rooms are reclaimed
    rooms.set(codeAt(i),loadRoom(i));
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── connection / broadcast ── */
let rooms=new Map();                 // code -> room
const conns=new Map();               // conn.id -> {code, role, conn}
function topic(code){ return "ch:"+code; }
function vaultsJ(vault){ const out=[]; for(let s=0;s<3;s++){ const st=vault[s]; out.push(st?{type:st.type,cards:st.cards.slice()}:null); } return out; }
function snap(room){
  const discTop=room.discard.length?room.discard[room.discard.length-1]:null;
  return {
    code:room.code, status:room.status, turnIndex:room.turnIndex,
    turn:room.over?-1:turnPlayer(room), playsLeft:room.playsLeft,
    deckCount:room.deck.length, discardTop:discTop, hostFirst:room.hostFirst,
    vaults:[vaultsJ(room.vaults[0]),vaultsJ(room.vaults[1])],
    handsLen:[room.hands[0].length,room.hands[1].length],
    finalRound:room.finalRound, winner:room.winner, reason:room.reason,
    scores:[vaultVal(room,0),vaultVal(room,1)],
    hostIn:room.hostId!==-1, guestIn:room.guestId!==-1, seq:room.seq, event:room.event||""
  };
}
function broadcast(room){ try{ pubsub.publish(topic(room.code), JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function sendHands(room){
  const targets=[[room.hostId,0],[room.guestId,1]];
  for(const pair of targets){
    if(pair[0]===-1)continue;
    const info=conns.get(pair[0]);
    if(!info)continue;
    try{ info.conn.send(JSON.stringify({t:"hand",hand:room.hands[pair[1]].slice()})); }catch(e){}
  }
}
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }
function saveBroadcast(room){ saveRoom(room); broadcast(room); sendHands(room); }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room=newRoom(0);
    room.slot=slot; room.code=code; room.status=1; room.hostId=conn.id; room.guestId=-1; room.lastAct=nowSec();
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host",conn});
    return JSON.stringify({ok:true,code,role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    let started=false;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1){
      room.status=2; room.turnIndex=-1; deal(room); room.seq++;
      room.event="Deal! 5 cards each — "+pName(turnPlayer(room))+" moves first and draws 2.";
      started=true;
    }
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role,conn});
    room.lastAct=nowSec();
    saveBroadcast(room);
    return JSON.stringify({ok:true,role,snap:snap(room)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    sendHands(room);
    return JSON.stringify({ok:true,snap:snap(room)});
  },
  play({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const idx=role==="host"?0:1;
    if(turnPlayer(room)!==idx)return JSON.stringify({ok:false,err:"not_your_turn"});
    if(obj.evt!==room.turnIndex)return JSON.stringify({ok:false,err:"stale"});
    if(!room.playsLeft)return JSON.stringify({ok:false,err:"no_plays_left"});
    const k=obj.kind, ty=obj.type, stk=obj.stack, ci=obj.card;
    if(k!=="start"&&k!=="build"&&k!=="cover")return JSON.stringify({ok:false,err:"bad_kind"});
    if(typeof ci!=="number"||ci<0||ci>=room.hands[idx].length)return JSON.stringify({ok:false,err:"bad_card"});
    if(typeof ty!=="number"||ty<0||ty>7)return JSON.stringify({ok:false,err:"bad_type"});
    if(typeof stk!=="number"||stk<0||stk>2)return JSON.stringify({ok:false,err:"bad_stack"});
    const id=room.hands[idx][ci];
    if(cardType(id)!==8&&cardType(id)!==ty)return JSON.stringify({ok:false,err:"illegal"});
    let found=false;
    const kinds=legalKinds(room,idx);
    for(const lk of kinds){ if(lk.i===ci&&lk.kind===k&&lk.type===ty&&lk.stack===stk){ found=true; break; } }
    if(!found)return JSON.stringify({ok:false,err:"illegal"});
    const stolen=k==="cover"?stackVal(room.vaults[1-idx][stk]):0;
    applyPlay(room,idx,{i:ci,kind:k,type:ty,stack:stk});
    room.playsLeft--;
    const by=pName(idx), at=pName(1-idx);
    if(k==="start")room.event=by+" placed "+cardTxt(ty)+" in a new vault stack.";
    else if(k==="build")room.event=by+" added "+cardTxt(ty)+" to a stack.";
    else room.event=by+" COVERED "+at+"'s "+NAMES[ty]+" stack — stole "+stolen+" pts!";
    room.seq++; room.lastAct=nowSec();
    if(room.playsLeft<=0){ finishTurn(room); }
    else if(!legalKinds(room,idx).length){ room.event+=" No plays left — turn ends."; finishTurn(room); }
    if(room.over)room.event="Game over — "+(room.winner==="draw"?"draw":pName(room.winner))+" wins "+room.scores[0]+"–"+room.scores[1]+"!";
    saveBroadcast(room);
    return JSON.stringify({ok:true});
  },
  discard({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const idx=role==="host"?0:1;
    if(turnPlayer(room)!==idx)return JSON.stringify({ok:false,err:"not_your_turn"});
    if(obj.evt!==room.turnIndex)return JSON.stringify({ok:false,err:"stale"});
    const ci=obj.card;
    if(typeof ci!=="number"||ci<0||ci>=room.hands[idx].length)return JSON.stringify({ok:false,err:"bad_card"});
    const id=room.hands[idx][ci];
    room.hands[idx].splice(ci,1);
    room.discard.push(id);
    room.event=pName(idx)+" discarded "+cardTxt(cardType(id))+".";
    room.seq++; room.lastAct=nowSec();
    saveBroadcast(room);
    return JSON.stringify({ok:true});
  },
  pass({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const idx=role==="host"?0:1;
    if(turnPlayer(room)!==idx)return JSON.stringify({ok:false,err:"not_your_turn"});
    if(obj.evt!==room.turnIndex)return JSON.stringify({ok:false,err:"stale"});
    room.event=pName(idx)+" ended their turn.";
    room.seq++; room.lastAct=nowSec();
    finishTurn(room);
    if(room.over)room.event="Game over — "+(room.winner==="draw"?"draw":pName(room.winner))+" wins "+room.scores[0]+"–"+room.scores[1]+"!";
    saveBroadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      const idx=roleOf(conn)==="host"?0:1;
      room.status=3; room.over=true; room.winner=1-idx; room.reason="resign"; room.seq++;
      room.event=pName(1-idx)+" wins — "+pName(idx)+" resigned.";
      room.lastAct=nowSec(); saveBroadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const idx=roleOf(conn)==="host"?0:1;
    const oppPresent=idx===0?room.guestId!==-1:room.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3; room.over=true; room.winner=idx; room.reason="forfeit"; room.seq++;
    room.event=pName(idx)+" wins — the opponent disconnected.";
    room.lastAct=nowSec(); saveBroadcast(room);
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const idx=roleOf(conn)==="host"?0:1;
    const oppPresent=idx===0?room.guestId!==-1:room.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    room.hostFirst=1-room.hostFirst;
    room.status=2; room.turnIndex=-1;
    room.deck=buildDeck(); room.discard=[]; room.hands=[[],[]]; room.vaults=[[],[],[]];
    room.playsLeft=0; room.finalRound=false; room.endTurn=1e9; room.over=false; room.winner=null; room.reason=""; room.scores=null;
    deal(room);
    room.event="Rematch! "+pName(turnPlayer(room))+" moves first.";
    room.seq++; room.lastAct=nowSec();
    saveBroadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      const idx=roleOf(conn)==="host"?0:1;
      room.status=3; room.over=true; room.winner=1-idx; room.reason="forfeit"; room.seq++;
      room.event=pName(1-idx)+" wins — "+pName(idx)+" left.";
      room.lastAct=nowSec(); saveBroadcast(room);
    }
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); const t=a[i]; a[i]=a[j]; a[j]=t; } return a; }

function legalKinds(g,p){
  const mine=g.vaults[p], opp=g.vaults[1-p], hand=g.hands[p], out=[];
  for(let i=0;i<hand.length;i++){
    const base=cardType(hand[i]);
    const types=base===8?[0,1,2,3,4,5,6,7]:[base];
    for(const ty of types){
      let emptySlot=-1, ownStack=-1, oppStack=-1;
      for(let s=0;s<3;s++){ const st=mine[s]; if(!st){ if(emptySlot===-1)emptySlot=s; } else if(st.type===ty) ownStack=s; }
      for(let s=0;s<3;s++){ const st=opp[s]; if(st&&st.type===ty){ oppStack=s; break; } }
      if(emptySlot!==-1&&ownStack===-1)out.push({i:i,kind:"start",type:ty,stack:emptySlot});
      if(ownStack!==-1)out.push({i:i,kind:"build",type:ty,stack:ownStack});
      if(oppStack!==-1&&(ownStack!==-1||emptySlot!==-1))out.push({i:i,kind:"cover",type:ty,stack:oppStack});
    }
  }

}

  }else if(k.kind==="build"){

    const st=opp[k.stack];

    else mine[target]={type:k.type,cards:st.cards.concat([id])};
  }
  return true;
}

}

}
function onDiscard(i){
  const me=game.mode==="online"?game.myPlayer:game.view;
  const hand=game.mode==="online"?game.hand:game.g.hands[me];
  if(i>=hand.length)return;
  const id=hand[i];
  if(game.mode==="online"){ sendDiscard(i); return; }
  game.g.hands[me].splice(i,1);
  game.g.discard.push(id);
  log(npName(me)+" discarded "+NAMES[cardType(id)]+" ("+EMO[cardType(id)]+").");
  if(game.selected&&game.selected.i===i)game.selected=null;
  else if(game.selected&&game.selected.i>i)game.selected.i--;
  renderAll();
}

function aiChoosePlay(g,p){
  const kinds=legalKinds(g,p);
  if(!kinds.length)return null;
  let best=null, bestS=-1e9;
  for(const k of kinds){
    const id=g.hands[p][k.i];
    let gain=cardVal(id);
    if(k.kind==="cover"){
      gain=stackVal(g.vaults[1-p][k.stack])+cardVal(id);
      gain*=1.15;
    }else if(k.kind==="build"){
      gain=cardVal(id)*1.2;
      let more=0;
      for(let j=0;j<g.hands[p].length;j++){ if(j!==k.i&&(cardType(g.hands[p][j])===k.type||cardType(g.hands[p][j])===8))more++; }
      gain+=more*cardVal(id)*0.35;
    }else{
      gain=cardVal(id)*0.85;
    }
    gain+=Math.random()*8;
    if(gain>bestS){ bestS=gain; best=k; }
  }
  return best;
}
function aiDoTurn(){
  game.aiThinking=true;
  updateBanner();
  setTimeout(function(){
    if(game.mode!=="ai"||game.over)return;
    const g=game.g;
    for(let step=0;step<2;step++){
      const k=aiChoosePlay(g,1);
      if(!k)break;
      if(k.kind==="cover")k.stolen=stackVal(g.vaults[0][k.stack]);
      applyPlay(g,1,k);
      g.playsLeft--;
      logPlay(1,k);
    }
    game.aiThinking=false;
    if(g.playsLeft>0&&!legalKinds(g,1).length)log("No legal plays left — turn ends.");
    endPlayerTurn();
  },480);
}

function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{ game.socket=rootRef.createServerSocket(); }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){ connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — share the code to start the match":"Connecting…"); }

function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="hand"){ game.hand=msg.hand.slice(); if(game.g&&game.mode==="online")game.g.hands[game.myPlayer]=game.hand; renderAll(); }
}
function applyServerSnap(s){
  if(!game.g)game.g=newRoom(s.hostFirst);
  const g=game.g;
  g.turnIndex=s.turnIndex;
  g.playsLeft=s.playsLeft;
  g.finalRound=s.finalRound;
  g.hostFirst=s.hostFirst;
  g.vaults=[s.vaults[0].map(function(v){return v?{type:v.type,cards:v.cards.slice()}:null;}),s.vaults[1].map(function(v){return v?{type:v.type,cards:v.cards.slice()}:null;})];
  g.handsLen=[s.handsLen[0],s.handsLen[1]];
  g.deckCount=s.deckCount; g.discardTop=s.discardTop;
  g.deck.length=0; g.discard.length=0;
  g.scores=s.scores;
  g.over=s.status===3;
  g.winner=s.winner; g.reason=s.reason;
  game.status=s.status;
  game.hostIn=s.hostIn; game.guestIn=s.guestIn;
  game.oppPresent=(game.myPlayer===0)?s.guestIn:s.hostIn;
  game.locked=false;
  if(game.myPlayer>=0)g.hands[game.myPlayer]=game.hand;
  if(s.event&&s.seq!==game.lastSeq){ log(s.event.replace(/Host/g,npName(0)).replace(/Guest/g,npName(1)),/COVERED|stole|wins/.test(s.event)); }
  game.lastSeq=s.seq;
  if(g.over&&game.mode==="online")showGameOver();
  renderAll();
}

async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.myRole=r.role; game.myPlayer=0;
    game.roomCode=r.code;
    myNameEl.textContent="You";
    oppNameEl.textContent="Opponent";
    game.logLines=[];
    log("Room created — code "+r.code+". Share it with a friend to start.");
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{

    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role; game.myPlayer=r.role==="host"?0:1;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
