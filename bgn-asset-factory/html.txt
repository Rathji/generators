function hexToRgb(hex){
  var m=String(hex).replace("#","").trim();
  if(m.length===3)m=m.split("").map(function(c){return c+c;}).join("");
  var n=parseInt(m,16); if(isNaN(n))n=0xd4af37;

function shade(hex,amt){
  var c=hexToRgb(hex),out=[],i;
  for(i=0;i<3;i++) out[i]=amt>=0?Math.round(c[i]+(255-c[i])*amt):Math.round(c[i]*(1+amt));
  return "rgb("+out[0]+","+out[1]+","+out[2]+")";
}
function rr(ctx,x,y,w,h,r){
  if(r>w/2)r=w/2; if(r>h/2)r=h/2;
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
function polyPath(ctx,cx,cy,r,sides,rot){
  ctx.beginPath();
  for(var i=0;i<=sides;i++){
    var a=(i/sides)*Math.PI*2+(rot===undefined?-Math.PI/2:rot);
    var x=cx+Math.cos(a)*r, y=cy+Math.sin(a)*r;
    if(i===0)ctx.moveTo(x,y); else ctx.lineTo(x,y);
  }
  ctx.closePath();
}
function starPath(ctx,cx,cy,outer,inner,points){
  ctx.beginPath();
  for(var i=0;i<points*2;i++){
    var r=i%2?inner:outer;
    var a=-Math.PI/2+(i/(points*2))*Math.PI*2;
    var x=cx+Math.cos(a)*r, y=cy+Math.sin(a)*r;
    if(i===0)ctx.moveTo(x,y); else ctx.lineTo(x,y);
  }
  ctx.closePath();
}

  ctx.beginPath();

  }
  ctx.closePath();
}
function fitText(ctx,text,x,y,maxW,font,fill){
  if(text===undefined||text===null||text==="")return;
  ctx.font=font; ctx.fillStyle=fill; ctx.textBaseline="middle"; ctx.textAlign="center";
  var t=String(text);
  if(ctx.measureText(t).width>maxW){
    while(t.length>1&&ctx.measureText(t+"…").width>maxW)t=t.slice(0,-1);
    t+="…";
  }
  ctx.fillText(t,x,y);
}
function wrapText(ctx,text,x,y,maxW,font,fill,lh,maxLines){
  if(!text)return;
  ctx.font=font; ctx.fillStyle=fill; ctx.textBaseline="top"; ctx.textAlign="center";
  var words=String(text).split(/\s+/),lines=[],cur="",i,t;
  for(i=0;i<words.length;i++){
    t=cur?cur+" "+words[i]:words[i];
    if(ctx.measureText(t).width>maxW&&cur){ lines.push(cur); cur=words[i]; }
    else cur=t;
  }
  if(cur)lines.push(cur);
  if(maxLines&&lines.length>maxLines){ lines=lines.slice(0,maxLines); lines[maxLines-1]+="…"; }
  for(i=0;i<lines.length;i++)ctx.fillText(lines[i],x,y+i*lh);
}

    ctx.beginPath();

    for(y=0;y<=h;y+=gs){ctx.moveTo(0,y);ctx.lineTo(w,y);}
    ctx.stroke();

    ctx.beginPath();
    for(x=0;x<=w;x+=gs*4){ctx.moveTo(x,0);ctx.lineTo(x,h);}
    for(y=0;y<=h;y+=gs*4){ctx.moveTo(0,y);ctx.lineTo(w,y);}

  }
}

function drawStamp(ctx,w,h,p,def,count,bg){

function sheetLayout(count,iw,ih,pad){
  var cols=Math.max(1,Math.ceil(Math.sqrt(count*(iw/ih))));
  if(cols>count)cols=count;
  var rows=Math.ceil(count/cols);
  return {cols:cols,rows:rows,cellW:iw+pad*2,cellH:ih+pad*2};
}
function sheetRender(def,p,scale,drawItem){
  var pad=def.pad,sz=def.itemSize(p),iw=sz[0],ih=sz[1];
  var count=clampNum(Math.round(p.count||1),1,def.maxCount||144);
  var L=sheetLayout(count,iw,ih,pad);
  var w=L.cols*L.cellW,h=L.rows*L.cellH;
  var maxDim=5000;
  if(w*scale>maxDim||h*scale>maxDim)scale=maxDim/Math.max(w,h);
  var canvas=document.createElement("canvas");
  canvas.width=Math.max(1,Math.round(w*scale)); canvas.height=Math.max(1,Math.round(h*scale));
  var ctx=canvas.getContext("2d");
  ctx.setTransform(scale,0,0,scale,0,0);
  var bg=p.bg||"blueprint";
  drawBg(ctx,w,h,bg);
  var guideCol=bg==="blueprint"?"rgba(255,255,255,.30)":"rgba(0,0,0,.28)";
  var i,col,row,cx,cy;
  for(i=0;i<count;i++){
    col=i%L.cols; row=(i/L.cols)|0;
    cx=col*L.cellW+L.cellW/2; cy=row*L.cellH+L.cellH/2;
    if(p.guides)drawGuides(ctx,cx,cy,iw,ih,guideCol,def.circle);
    drawItem(ctx,i,cx,cy,iw,ih,p);
  }
  if(p.stamp&&bg!=="transparent")drawStamp(ctx,w,h,p,def,count,bg);
  return {canvas:canvas,count:count,w:w,h:h,cols:L.cols,rows:L.rows};
}

function measureWrap(ctx,text,maxW){
  if(!text)return [];
  var words=String(text).split(/\s+/),lines=[],cur="",i,t;
  for(i=0;i<words.length;i++){
    t=cur?cur+" "+words[i]:words[i];
    if(ctx.measureText(t).width>maxW&&cur){ lines.push(cur); cur=words[i]; }
    else cur=t;
  }
  if(cur)lines.push(cur);
  return lines;
}

/* ══════════ 3D DICE — software raytracer (specular + refraction) ══════════ */
function norm3(x,y,z){ var l=Math.sqrt(x*x+y*y+z*z)||1; return [x/l,y/l,z/l]; }
function mixN(a,b,t){ return a+(b-a)*t; }
function rotMatrix(ax,ay){
  var cx=Math.cos(ax),sx=Math.sin(ax),cy=Math.cos(ay),sy=Math.sin(ay);
  var m=[cy,sy*sx,sy*cx, 0,cx,-sx, -sy,cy*sx,cy*cx];
  return {m:m,it:[m[0],m[3],m[6],m[1],m[4],m[7],m[2],m[5],m[8]]};
}
function rotPoint(p,m){ return [p[0]*m[0]+p[1]*m[3]+p[2]*m[6], p[0]*m[1]+p[1]*m[4]+p[2]*m[7], p[0]*m[2]+p[1]*m[5]+p[2]*m[8]]; }
function sdfBox(x,y,z,b,r){
  var qx=Math.abs(x)-b, qy=Math.abs(y)-b, qz=Math.abs(z)-b;
  var ox=qx>0?qx:0, oy=qy>0?qy:0, oz=qz>0?qz:0;
  var len=Math.sqrt(ox*ox+oy*oy+oz*oz);
  var inr=Math.min(Math.max(qx,Math.max(qy,qz)),0);
  return len+inr-r;
}
function gradN(x,y,z,b,r){
  var e=0.0012;
  var c=sdfBox(x,y,z,b,r);
  var nx=(c-sdfBox(x-e,y,z,b,r)), ny=(c-sdfBox(x,y-e,z,b,r)), nz=(c-sdfBox(x,y,z-e,b,r));
  var l=Math.sqrt(nx*nx+ny*ny+nz*nz);
  if(l<1e-9)return [0,1,0];
  return [nx/l,ny/l,nz/l];
}
function faceUV(x,y,z){
  var axx=Math.abs(x),ayy=Math.abs(y),azz=Math.abs(z),ax,sx;
  if(axx>=ayy&&axx>=azz)ax=0; else if(ayy>=azz)ax=1; else ax=2;
  sx=ax===0?x:(ax===1?y:z);
  var u,v;
  if(ax===0){u=z;v=-y;if(sx<0)u=-u;}
  else if(ax===1){u=x;v=z;if(sx<0)v=-v;}
  else {u=x;v=-y;if(sx<0)u=-u;}
  return {idx:ax*2+(sx<0?1:0),tu:Math.max(0,Math.min(1,u+0.5)),tv:Math.max(0,Math.min(1,0.5-v))};
}
function sampleTex(tex,u,v){
  var w=tex.w,h=tex.h,x=u*(w-1),y=v*(h-1);
  var x0=x|0,y0=y|0,x1=x0+1<w?x0+1:w-1,y1=y0+1<h?y0+1:h-1;
  var fx=x-x0,fy=y-y0,d=tex.data;
  var a00=d[(y0*w+x0)*4+3],a10=d[(y0*w+x1)*4+3],a01=d[(y1*w+x0)*4+3],a11=d[(y1*w+x1)*4+3];
  return ((a00*(1-fx)+a10*fx)*(1-fy)+(a01*(1-fx)+a11*fx)*fy)/255;
}
function refract3(I,N,eta){
  var cosi=clampNum(-(I[0]*N[0]+I[1]*N[1]+I[2]*N[2]),-1,1);
  var k=1-eta*eta*(1-cosi*cosi);
  if(k<0)return null;
  var kk=eta*cosi-Math.sqrt(k);
  return [eta*I[0]+kk*N[0],eta*I[1]+kk*N[1],eta*I[2]+kk*N[2]];
}
function reflect3(D,N){
  var d2=2*(D[0]*N[0]+D[1]*N[1]+D[2]*N[2]);
  return [d2*N[0]-D[0],d2*N[1]-D[1],d2*N[2]-D[2]];
}
function traceDie(ox,oy,oz,dx,dy,dz,it,b,r,tol){
  var lox=ox*it[0]+oy*it[3]+oz*it[6], loy=ox*it[1]+oy*it[4]+oz*it[7], loz=ox*it[2]+oy*it[5]+oz*it[8];
  var ldx=dx*it[0]+dy*it[3]+dz*it[6], ldy=dx*it[1]+dy*it[4]+dz*it[7], ldz=dx*it[2]+dy*it[5]+dz*it[8];
  var hb=b+r;
  var tmin=0.0005, tmax=1e9, t1,t2,tmp;
  if(ldx>-1e-9&&ldx<1e-9){ if(lox>hb||lox<-hb)return null; }
  else{
    t1=(-hb-lox)/ldx; t2=(hb-lox)/ldx;
    if(t1>t2){tmp=t1;t1=t2;t2=tmp;}
    if(t1>tmin)tmin=t1; if(t2<tmax)tmax=t2;
    if(tmin>tmax)return null;
  }
  if(ldy>-1e-9&&ldy<1e-9){ if(loy>hb||loy<-hb)return null; }
  else{
    t1=(-hb-loy)/ldy; t2=(hb-loy)/ldy;
    if(t1>t2){tmp=t1;t1=t2;t2=tmp;}
    if(t1>tmin)tmin=t1; if(t2<tmax)tmax=t2;
    if(tmin>tmax)return null;
  }
  if(ldz>-1e-9&&ldz<1e-9){ if(loz>hb||loz<-hb)return null; }
  else{
    t1=(-hb-loz)/ldz; t2=(hb-loz)/ldz;
    if(t1>t2){tmp=t1;t1=t2;t2=tmp;}
    if(t1>tmin)tmin=t1; if(t2<tmax)tmax=t2;
    if(tmin>tmax)return null;
  }
  if(tmax<=0)return null;
  var t=tmin, wx,wy,wz,lx,ly,lz,dd,k;
  for(k=0;k<9;k++){
    wx=ox+dx*t; wy=oy+dy*t; wz=oz+dz*t;
    lx=wx*it[0]+wy*it[3]+wz*it[6]; ly=wx*it[1]+wy*it[4]+wz*it[7]; lz=wx*it[2]+wy*it[5]+wz*it[8];
    dd=sdfBox(lx,ly,lz,b,r);
    if(dd<tol)return [wx,wy,wz,lx,ly,lz];
    t+=dd;
    if(t>tmax)return null;
  }
  return null;
}
function traceInside(x,y,z,dx,dy,dz,it,b,r){
  var t=0.002;
  for(var k=0;k<48;k++){
    var px=x+dx*t, py=y+dy*t, pz=z+dz*t;
    var lx=px*it[0]+py*it[3]+pz*it[6], ly=px*it[1]+py*it[4]+pz*it[7], lz=px*it[2]+py*it[5]+pz*it[8];
    var dd=sdfBox(lx,ly,lz,b,r);
    if(dd>0.0006)return {p:[px,py,pz],len:t};
    if(t>6)break;
    t+=Math.max(Math.abs(dd)*0.85,0.004);
  }
  return null;
}

function makeEnv(p){

    return {top:[250,249,244],hor:[228,224,213],floor:[212,206,193],key:[255,252,242],fill:[240,245,255],keyD:norm3(0.5,0.85,0.45),fillD:norm3(-0.55,0.35,-0.7)};

function envCol(E,dx,dy,dz){
  var hr=E.hor[0]/255,hg=E.hor[1]/255,hb=E.hor[2]/255;
  var cd=Math.max(0,-dz);
  if(dy<0){
    var tt=Math.min(1,Math.pow(-dy,0.7)),m=0.22+0.78*tt;
    var r=E.floor[0]/255*(1-m)+hr*m, g=E.floor[1]/255*(1-m)+hg*m, b=E.floor[2]/255*(1-m)+hb*m;
    var cw=Math.pow(cd,6);
    r+=E.key[0]/255*cw*0.32; g+=E.key[1]/255*cw*0.32; b+=E.key[2]/255*cw*0.32;
    return [r,g,b];
  }
  var s=Math.pow(Math.max(0,dy),0.55);
  var r=hr+(E.top[0]/255-hr)*s, g=hg+(E.top[1]/255-hg)*s, b=hb+(E.top[2]/255-hb)*s;
  var kd=Math.max(0,dx*E.keyD[0]+dy*E.keyD[1]+dz*E.keyD[2]);
  var kg=Math.pow(kd,18);
  r+=E.key[0]/255*kg*1.7; g+=E.key[1]/255*kg*1.7; b+=E.key[2]/255*kg*1.7;
  var fd=Math.max(0,dx*E.fillD[0]+dy*E.fillD[1]+dz*E.fillD[2]);
  var fg=Math.pow(fd,10);
  r+=E.fill[0]/255*fg*0.8; g+=E.fill[1]/255*fg*0.8; b+=E.fill[2]/255*fg*0.8;
  var cw2=Math.pow(cd,5);
  r+=E.key[0]/255*cw2*0.28; g+=E.key[1]/255*cw2*0.28; b+=E.key[2]/255*cw2*0.28;
  if(r>1)r=1; if(g>1)g=1; if(b>1)b=1;
  return [r,g,b];
}
function traceRefr(cc,wx,wy,wz, dx,dy,dz, nx,ny,nz, depth){
  var eta=1/cc.ior;
  var tdir=refract3([dx,dy,dz],[nx,ny,nz],eta);
  if(!tdir)return envCol(cc.E,dx,dy,dz);
  var res=traceInside(wx,wy,wz,tdir[0],tdir[1],tdir[2],cc.it,cc.b,cc.hr);
  if(!res)return envCol(cc.E,dx,dy,dz);
  var ex=res.p;
  var elx=ex[0]*cc.it[0]+ex[1]*cc.it[3]+ex[2]*cc.it[6];
  var ely=ex[0]*cc.it[1]+ex[1]*cc.it[4]+ex[2]*cc.it[7];
  var elz=ex[0]*cc.it[2]+ex[1]*cc.it[5]+ex[2]*cc.it[8];
  var en=gradN(elx,ely,elz,cc.b,cc.hr);
  var enx=en[0]*cc.m[0]+en[1]*cc.m[3]+en[2]*cc.m[6];
  var eny=en[0]*cc.m[1]+en[1]*cc.m[4]+en[2]*cc.m[7];
  var enz=en[0]*cc.m[2]+en[1]*cc.m[5]+en[2]*cc.m[8];
  var t2=refract3(tdir,[enx,eny,enz],cc.ior);
  var col;
  if(t2)col=envCol(cc.E,t2[0],t2[1],t2[2]);
  else{
    var rv=reflect3(tdir,[enx,eny,enz]);
    if(depth<2)col=traceRefr(cc,ex[0],ex[1],ex[2],rv[0],rv[1],rv[2],-enx,-eny,-enz,depth+1);
    else col=envCol(cc.E,rv[0],rv[1],rv[2]);
  }
  var a=Math.exp(-res.len*cc.absorb);
  var cr=col[0]*a*(mixN(cc.tr,1,cc.tintAmt));
  var cg=col[1]*a*(mixN(cc.tg,1,cc.tintAmt));
  var cb=col[2]*a*(mixN(cc.tb,1,cc.tintAmt));
  return [cr,cg,cb];
}
function renderOneDie(cell,idx,p,E,rayRes){

  var transp=mat==="glossy"?0:1;
  var ior=mat==="glass"?1.5:1.55;

  var py,px;
  for(py=0;py<S;py++){
    var ndcY=-((py+0.5)/S*2-1);

      if(!hit){ d[oi+3]=0; continue; }
      var wx=hit[0],wy=hit[1],wz=hit[2];
      var lx=hit[3],ly=hit[4],lz=hit[5];
      var nloc=gradN(lx,ly,lz,b,hr), nlx=nloc[0],nly=nloc[1],nlz=nloc[2];
      var fuv=faceUV(lx,ly,lz);
      var aa=sampleTex(texes[fuv.idx],fuv.tu,fuv.tv);
      var pipF=aa>0.02;
      if(pipF&&aa>0.06){
        var dep=aa*0.05;

        nlx=g2[0];nly=g2[1];nlz=g2[2];
      }
      var wnx=nlx*m[0]+nly*m[3]+nlz*m[6], wny=nlx*m[1]+nly*m[4]+nlz*m[7], wnz=nlx*m[2]+nly*m[5]+nlz*m[8];
      var wl=Math.sqrt(wnx*wnx+wny*wny+wnz*wnz)||1; wnx/=wl;wny/=wl;wnz/=wl;
      var ao=1;
      if(pipF){
        for(var kk=1;kk<=3;kk++){

          var sd=sdfBox(lx+nlx*ds,ly+nly*ds,lz+nlz*ds,b,hr);
          var occ=(ds-sd)/ds; if(occ<0)occ=0;

        }
      }
      var aRgb;

      else aRgb=[aR,aG,aB];
      var Vx=-rdx,Vy=-rdy,Vz=-rdz;

      var d1=Math.max(0,wnx*key[0]+wny*key[1]+wnz*key[2]);
      var d2=Math.max(0,wnx*fill[0]+wny*fill[1]+wnz*fill[2]);
      var amb=(pipF?0.11:0.035)+0.05*nv;
      var dr=aRgb[0]*(amb+d1*1.2+d2*0.5)*ao;
      var dg=aRgb[1]*(amb+d1*1.2+d2*0.5)*ao;
      var db2=aRgb[2]*(amb+d1*1.2+d2*0.5)*ao;
      var hx=key[0]+Vx, hy=key[1]+Vy, hz=key[2]+Vz;
      var hl=Math.sqrt(hx*hx+hy*hy+hz*hz)||1; hx/=hl;hy/=hl;hz/=hl;
      var sp=Math.pow(Math.max(0,wnx*hx+wny*hy+wnz*hz),shin)*1.35+Math.pow(Math.max(0,wnx*hx+wny*hy+wnz*hz),8)*0.30;
      var h2x=fill[0]+Vx,h2y=fill[1]+Vy,h2z=fill[2]+Vz;
      var h2l=Math.sqrt(h2x*h2x+h2y*h2y+h2z*h2z)||1; h2x/=h2l;h2y/=h2l;h2z/=h2l;
      var sp2=Math.pow(Math.max(0,wnx*h2x+wny*h2y+wnz*h2z),shin)*0.5+Math.pow(Math.max(0,wnx*h2x+wny*h2y+wnz*h2z),8)*0.15;
      var fs2=Math.pow(nv,7)*0.22;
      var rx=2*nv*wnx+rdx, ry=2*nv*wny+rdy, rz=2*nv*wnz+rdz;
      var rl2=Math.sqrt(rx*rx+ry*ry+rz*rz)||1; rx/=rl2;ry/=rl2;rz/=rl2;
      var e1=envCol(E,rx,ry,rz), e2=e1;
      if(blur>0.01){
        var bx=rx-(rx-wnx)*blur, by=ry-(ry-wny)*blur, bz=rz-(rz-wnz)*blur;
        var bl2=Math.sqrt(bx*bx+by*by+bz*bz)||1;
        e2=envCol(E,bx/bl2,by/bl2,bz/bl2);
      }
      var evr=e1[0]*(1-blur)+e2[0]*blur;
      var evg=e1[1]*(1-blur)+e2[1]*blur;
      var evb=e1[2]*(1-blur)+e2[2]*blur;
      var fres=f0+(1-f0)*Math.pow(1-nv,5);
      var diffWp=pipF?1:diffW;
      var specW=(0.4+0.6*gloss)*(pipF?1.7:1);
      var cr=dr*(1-fres)*diffWp+evr*fres*ao*0.6+(sp+sp2+fs2)*specW;
      var cg=dg*(1-fres)*diffWp+evg*fres*ao*0.6+(sp+sp2+fs2)*specW;
      var cb=db2*(1-fres)*diffWp+evb*fres*ao*0.6+(sp+sp2+fs2)*specW;
      if(pipF){ cr+=0.10; cg+=0.06; }
      var maxA=Math.abs(lx); var secA=Math.abs(ly);
      if(Math.abs(ly)>maxA){var sw2=maxA;maxA=Math.abs(ly);secA=sw2;}
      if(Math.abs(lz)>maxA){secA=maxA;maxA=Math.abs(lz);}
      var edgeL=(secA-b)/hr; if(edgeL<0)edgeL=0; if(edgeL>1)edgeL=1;
      var es=edgeL*edgeL*(0.35+0.45*gloss);
      cr+=es; cg+=es*0.92; cb+=es*0.72;
      if(transp>0&&!pipF){
        var refr=traceRefr(cc,wx,wy,wz,rdx,rdy,rdz,wnx,wny,wnz,0);
        cr+=refr[0]*(1-fres)*transp*0.6;
        cg+=refr[1]*(1-fres)*transp*0.6;
        cb+=refr[2]*(1-fres)*transp*0.6;
      }
      if(cr<0)cr=0; if(cr>1)cr=1;
      if(cg<0)cg=0; if(cg>1)cg=1;
      if(cb<0)cb=0; if(cb>1)cb=1;
      cr=Math.sqrt(cr); cg=Math.sqrt(cg); cb=Math.sqrt(cb);
      d[oi]=Math.round(cr*255); d[oi+1]=Math.round(cg*255); d[oi+2]=Math.round(cb*255); d[oi+3]=255;
    }
  }
  octx.putImageData(id,0,0);
  return off;
}
function renderDice3D(p,scale,rayRes){
  rayRes=rayRes||1;
  var count=Math.min(rayRes>1?4:6,clampNum(Math.round(p.count||1),1,12));
  var cols=count===1?1:Math.min(3,Math.ceil(Math.sqrt(count*0.7)));
  var rows=Math.ceil(count/cols);
  var budget=520000*rayRes*rayRes;
  var cap=(rayRes===1?190:260)*rayRes;
  var cell=Math.min(cap, Math.min(Math.sqrt(budget/count), Math.max(120*rayRes, Math.round((p.size||84)*3.2*rayRes))));
  var w=cols*cell, h=rows*cell;
  var canvas=document.createElement("canvas");
  canvas.width=Math.max(1,Math.round(w*scale)); canvas.height=Math.max(1,Math.round(h*scale));
  var ctx=canvas.getContext("2d");
  ctx.setTransform(scale,0,0,scale,0,0);
  var bg=p.bg||"blueprint";
  drawBg(ctx,w,h,bg);
  var E=makeEnv(p);
  var i;
  for(i=0;i<count;i++){
    var col=i%cols,row=(i/cols)|0;
    var cx=col*cell+cell/2, cy=row*cell+cell/2;
    var dieCanvas=renderOneDie(cell,i,p,E,rayRes);
    if(bg!=="transparent"){
      var sg=ctx.createRadialGradient(cx,cy+cell*0.24,2,cx,cy+cell*0.24,cell*0.32);
      sg.addColorStop(0,"rgba(0,0,0,.42)");
      sg.addColorStop(1,"rgba(0,0,0,0)");
      ctx.fillStyle=sg;
      ctx.beginPath(); ctx.ellipse(cx,cy+cell*0.24,cell*0.30,cell*0.15,0,0,Math.PI*2); ctx.fill();
    }
    ctx.drawImage(dieCanvas,cx-cell/2,cy-cell/2,cell,cell);
  }
  return {canvas:canvas,count:count,w:w,h:h,cols:cols,rows:rows};
}

    if(c.type==="color"){
      input='<input type="color" id="'+id+'" value="'+esc(val)+'" class="wb-color">'
        +'<input type="text" id="'+id+'_hex" value="'+esc(val)+'" class="wb-hex" maxlength="7" spellcheck="false">';
    }else if(c.type==="text"){
      input='<input type="text" id="'+id+'" class="field" value="'+esc(val)+'"'+(c.placeholder?' placeholder="'+esc(c.placeholder)+'"':'')+'>';
    }else if(c.type==="textarea"){
      input='<textarea id="'+id+'" class="field" rows="'+(c.rows||3)+'">'+esc(val)+'</textarea>';
    }else if(c.type==="number"){
      input='<input type="number" id="'+id+'" class="field" value="'+val+'" min="'+(c.min||0)+'" max="'+(c.max||999)+'">';
    }else if(c.type==="range"){
      input='<input type="range" id="'+id+'" value="'+val+'" min="'+c.min+'" max="'+c.max+'" step="'+(c.step||1)+'">'
        +'<span class="wb-range-val" id="'+id+'_v">'+val+'</span>';
    }else if(c.type==="select"){
      var opts="";
      c.options.forEach(function(o){ opts+='<option value="'+esc(o[0])+'"'+(String(val)===String(o[0])?' selected':'')+'>'+esc(o[1])+'</option>'; });
      input='<select id="'+id+'" class="field">'+opts+'</select>';
    }else if(c.type==="toggle"){
      input='<label class="wb-switch"><input type="checkbox" id="'+id+'"'+(val?' checked':'')+'><span></span></label>';
    }
    html+='<div class="wb-ctrl'+(c.type==="toggle"?" wb-ctrl-row":"")+'">'+label+input+'</div>';
  });
  controlsEl.innerHTML=html;
  var asset=cur;
  d.controls.forEach(function(c){
    var el=document.getElementById("c_"+c.k); if(!el)return;
    if(c.type==="color"){
      el.addEventListener("input",function(){
        asset.p[c.k]=el.value.toLowerCase();
        var h=document.getElementById("c_"+c.k+"_hex"); if(h)h.value=el.value.toLowerCase();
        afterChange(false);
      });
      var hx=document.getElementById("c_"+c.k+"_hex");
      if(hx)hx.addEventListener("input",function(){
        var v=hx.value.trim();
        if(!/^#([0-9a-f]{3}|[0-9a-f]{6})$/i.test(v))return;
        asset.p[c.k]=v.toLowerCase(); el.value=v.toLowerCase(); afterChange(false);
      });
    }else if(c.type==="range"){
      el.addEventListener("input",function(){
        asset.p[c.k]=parseInt(el.value,10)||0;
        var v=document.getElementById("c_"+c.k+"_v"); if(v)v.textContent=el.value;
        afterChange(false);
      });
    }else if(c.type==="number"){
      el.addEventListener("input",function(){
        var n=parseInt(el.value,10); if(isNaN(n))return;
        n=clampNum(n,c.min||1,c.max||999); el.value=n;
        asset.p[c.k]=n; afterChange(false);
      });
    }else if(c.type==="toggle"){
      el.addEventListener("change",function(){ asset.p[c.k]=el.checked; afterChange(false); });
    }else if(c.type==="select"){
      el.addEventListener("change",function(){ asset.p[c.k]=el.value; afterChange(true); });
    }else{
      el.addEventListener("input",function(){ asset.p[c.k]=el.value; afterChange(false,true); });
    }
  });
}

function itemCountLabel(){

}

}
