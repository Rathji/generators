function hexToRgb(hex){
  var m=String(hex).replace("#","").trim();
  if(m.length===3)m=m.split("").map(function(c){return c+c;}).join("");
  var n=parseInt(m,16); if(isNaN(n))n=0xd4af37;

function shade(hex,amt){
  var c=hexToRgb(hex),out=[],i;
  for(i=0;i<3;i++) out[i]=amt>=0?Math.round(c[i]+(255-c[i])*amt):Math.round(c[i]*(1+amt));
  return "rgb("+out[0]+","+out[1]+","+out[2]+")";
}
function rr(ctx,x,y,w,h,r){
  if(r>w/2)r=w/2; if(r>h/2)r=h/2;
  ctx.beginPath();
  ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);
  ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);
  ctx.arcTo(x,y,x+w,y,r);
  ctx.closePath();
}
function polyPath(ctx,cx,cy,r,sides,rot){
  ctx.beginPath();
  for(var i=0;i<=sides;i++){
    var a=(i/sides)*Math.PI*2+(rot===undefined?-Math.PI/2:rot);
    var x=cx+Math.cos(a)*r, y=cy+Math.sin(a)*r;
    if(i===0)ctx.moveTo(x,y); else ctx.lineTo(x,y);
  }
  ctx.closePath();
}
function starPath(ctx,cx,cy,outer,inner,points){
  ctx.beginPath();
  for(var i=0;i<points*2;i++){
    var r=i%2?inner:outer;
    var a=-Math.PI/2+(i/(points*2))*Math.PI*2;
    var x=cx+Math.cos(a)*r, y=cy+Math.sin(a)*r;
    if(i===0)ctx.moveTo(x,y); else ctx.lineTo(x,y);
  }
  ctx.closePath();
}

  ctx.beginPath();

  }
  ctx.closePath();
}
function fitText(ctx,text,x,y,maxW,font,fill){
  if(text===undefined||text===null||text==="")return;
  ctx.font=font; ctx.fillStyle=fill; ctx.textBaseline="middle"; ctx.textAlign="center";
  var t=String(text);
  if(ctx.measureText(t).width>maxW){
    while(t.length>1&&ctx.measureText(t+"…").width>maxW)t=t.slice(0,-1);
    t+="…";
  }
  ctx.fillText(t,x,y);
}
function wrapText(ctx,text,x,y,maxW,font,fill,lh,maxLines){
  if(!text)return;
  ctx.font=font; ctx.fillStyle=fill; ctx.textBaseline="top"; ctx.textAlign="center";
  var words=String(text).split(/\s+/),lines=[],cur="",i,t;
  for(i=0;i<words.length;i++){
    t=cur?cur+" "+words[i]:words[i];
    if(ctx.measureText(t).width>maxW&&cur){ lines.push(cur); cur=words[i]; }
    else cur=t;
  }
  if(cur)lines.push(cur);
  if(maxLines&&lines.length>maxLines){ lines=lines.slice(0,maxLines); lines[maxLines-1]+="…"; }
  for(i=0;i<lines.length;i++)ctx.fillText(lines[i],x,y+i*lh);
}

    ctx.beginPath();
    for(x=0;x<=w;x+=gs){ctx.moveTo(x,0);ctx.lineTo(x,h);}
    for(y=0;y<=h;y+=gs){ctx.moveTo(0,y);ctx.lineTo(w,y);}
    ctx.stroke();
    ctx.strokeStyle="rgba(212,175,55,.20)";
    ctx.beginPath();
    for(x=0;x<=w;x+=gs*4){ctx.moveTo(x,0);ctx.lineTo(x,h);}
    for(y=0;y<=h;y+=gs*4){ctx.moveTo(0,y);ctx.lineTo(w,y);}
    ctx.stroke();
  }
}
function drawGuides(ctx,px,py,iw,ih,color,circle){

function drawStamp(ctx,w,h,p,def,count,bg){

function sheetLayout(count,iw,ih,pad){
  var cols=Math.max(1,Math.ceil(Math.sqrt(count*(iw/ih))));
  if(cols>count)cols=count;
  var rows=Math.ceil(count/cols);
  return {cols:cols,rows:rows,cellW:iw+pad*2,cellH:ih+pad*2};
}
function sheetRender(def,p,scale,drawItem){
  var pad=def.pad,sz=def.itemSize(p),iw=sz[0],ih=sz[1];
  var count=clampNum(Math.round(p.count||1),1,def.maxCount||144);
  var L=sheetLayout(count,iw,ih,pad);
  var w=L.cols*L.cellW,h=L.rows*L.cellH;
  var maxDim=5000;
  if(w*scale>maxDim||h*scale>maxDim)scale=maxDim/Math.max(w,h);
  var canvas=document.createElement("canvas");
  canvas.width=Math.max(1,Math.round(w*scale)); canvas.height=Math.max(1,Math.round(h*scale));
  var ctx=canvas.getContext("2d");
  ctx.setTransform(scale,0,0,scale,0,0);
  var bg=p.bg||"blueprint";
  drawBg(ctx,w,h,bg);
  var guideCol=bg==="blueprint"?"rgba(255,255,255,.30)":"rgba(0,0,0,.28)";
  var i,col,row,cx,cy;
  for(i=0;i<count;i++){
    col=i%L.cols; row=(i/L.cols)|0;
    cx=col*L.cellW+L.cellW/2; cy=row*L.cellH+L.cellH/2;
    if(p.guides)drawGuides(ctx,cx,cy,iw,ih,guideCol,def.circle);
    drawItem(ctx,i,cx,cy,iw,ih,p);
  }
  if(p.stamp&&bg!=="transparent")drawStamp(ctx,w,h,p,def,count,bg);
  return {canvas:canvas,count:count,w:w,h:h,cols:L.cols,rows:L.rows};
}

function tokenBase(ctx,shape,s){
  if(shape==="square")rr(ctx,-s*0.46,-s*0.46,s*0.92,s*0.92,s*0.12);

function measureWrap(ctx,text,maxW){
  if(!text)return [];
  var words=String(text).split(/\s+/),lines=[],cur="",i,t;
  for(i=0;i<words.length;i++){
    t=cur?cur+" "+words[i]:words[i];
    if(ctx.measureText(t).width>maxW&&cur){ lines.push(cur); cur=words[i]; }
    else cur=t;
  }
  if(cur)lines.push(cur);
  return lines;
}

function renderBoard(p,scale){
  var rows=clampNum(Math.round(p.rows||8),2,26),cols=clampNum(Math.round(p.cols||8),2,26);
  var cell=clampNum(p.cell||52,28,76),margin=56;
  var w=cols*cell+margin*2,h=rows*cell+margin*2;
  var canvas=document.createElement("canvas");
  canvas.width=Math.round(w*scale); canvas.height=Math.round(h*scale);
  var ctx=canvas.getContext("2d");
  ctx.setTransform(scale,0,0,scale,0,0);
  ctx.fillStyle=p.frame||"#5b3a1e";
  rr(ctx,0,0,w,h,10); ctx.fill();
  ctx.fillStyle=shade(p.frame||"#5b3a1e",0.12);
  rr(ctx,margin*0.3,margin*0.3,w-margin*0.6,h-margin*0.6,8); ctx.fill();
  var plain=p.pattern==="plain",r,c,odd,col;
  for(r=0;r<rows;r++)for(c=0;c<cols;c++){
    odd=(r+c)%2===0;
    col=plain?p.colorA:(odd?p.colorA:p.colorB);
    ctx.fillStyle=col;
    ctx.fillRect(margin+c*cell,margin+r*cell,cell,cell);
    ctx.strokeStyle="rgba(0,0,0,.22)"; ctx.lineWidth=1;
    ctx.strokeRect(margin+c*cell+0.5,margin+r*cell+0.5,cell,cell);
  }
  if(p.coordinates){
    ctx.fillStyle="rgba(255,255,255,.85)"; ctx.font="700 15px Inter, system-ui, sans-serif";
    ctx.textAlign="center"; ctx.textBaseline="middle";
    for(c=0;c<cols;c++){ ctx.fillText(String.fromCharCode(65+c),margin+c*cell+cell/2,27); ctx.fillText(String.fromCharCode(65+c),margin+c*cell+cell/2,h-27); }
    for(r=0;r<rows;r++){ ctx.fillText(String(r+1),27,margin+r*cell+cell/2); ctx.fillText(String(r+1),w-27,margin+r*cell+cell/2); }
  }
  ctx.strokeStyle=shade(p.frame||"#5b3a1e",0.35); ctx.lineWidth=4;
  rr(ctx,2,2,w-4,h-4,8); ctx.stroke();
  return {canvas:canvas,count:rows*cols,w:w,h:h,cols:cols,rows:rows};
}

var DS=1024, DCX=512, DCY=512;
function diceRgbToHsl(r,g,b){
  r/=255; g/=255; b/=255;
  var max=Math.max(r,g,b),min=Math.min(r,g,b),h=0,s=0,l=(max+min)/2,d=max-min;
  if(d){
    s=l>0.5?d/(2-max-min):d/(max+min);
    if(max===r)h=(g-b)/d+(g<b?6:0); else if(max===g)h=(b-r)/d+2; else h=(r-g)/d+4;
    h*=60;
  }
  return [h,s*100,l*100];
}
function diceHslToHex(h,s,l){
  s/=100; l/=100;
  var k=function(n){ return (n+h/30)%12; };
  var a=s*Math.min(l,1-l);
  var f=function(n){ return l-a*Math.max(-1,Math.min(k(n)-3,Math.min(9-k(n),1))); };
  var to=function(x){ return Math.round(255*x).toString(16).padStart(2,"0"); };
  return "#"+to(f(0))+to(f(8))+to(f(4));
}
function rgba(hex,a){ var c=hexToRgb(hex); return "rgba("+c[0]+","+c[1]+","+c[2]+","+a+")"; }
function dicePalette(base){
  var c=hexToRgb(base),r=c[0],g=c[1],b=c[2];
  var hsl=diceRgbToHsl(r,g,b),h=hsl[0],s=hsl[1],l=hsl[2];
  var lum=(0.299*r+0.587*g+0.114*b)/255;
  return {
    mid:base,
    lit:diceHslToHex(h,clampNum(s-4,0,100),clampNum(l+24,0,96)),
    core:diceHslToHex(h,clampNum(s+4,0,100),clampNum(l-14,3,100)),
    deep:diceHslToHex(h,clampNum(s+8,0,100),clampNum(l-34,2,100)),
    mark:lum>0.62?"#1c1c1e":"#ffffff",
    markShade:lum>0.62?"#ffffff":"#000000",
    light:lum>0.62,
    sideDeep:diceHslToHex(h,clampNum(s+8,0,100),clampNum(l-38,2,100)),
    sideMid:diceHslToHex(h,clampNum(s+6,0,100),clampNum(l-26,3,100)),
    sideLit:diceHslToHex(h,clampNum(s+4,0,100),clampNum(l-18,3,100))
  };
}
var DICE_SHAPES={square:36,triangle:30,kite:26,pentagon:30,barrel:44,circle:0,d20view:30};
function diceAutoShape(n){
  if(n===2)return "circle";
  if(n===3||n===6)return "square";
  if(n===4||n===8)return "triangle";
  if(n===20)return "d20view";
  if(n===12)return "pentagon";
  if(n===10||n===14||n===16||n===18)return "kite";
  return "barrel";
}
var DICE_D20=(function(){
  var r3=Math.sqrt(3),phi=(1+Math.sqrt(5))/2,y0=(phi-1)/r3;
  var P={A:[1,1/r3-y0],B:[-1,1/r3-y0],C:[0,-2/r3-y0],X:[0,2*phi/r3-y0],D1:[phi,-phi/r3-y0],D2:[-phi,-phi/r3-y0]};
  P.topCentroid=[0,(P.A[1]+P.B[1]+P.C[1])/3];
  return P;
})();
var DICE_D20_SCALE=904/(DICE_D20.D1[0]-DICE_D20.D2[0]);
function dicePt(k){ return [DCX+DICE_D20[k][0]*DICE_D20_SCALE, DCY+DICE_D20[k][1]*DICE_D20_SCALE]; }
function dicePoly(count,R,rot){
  if(rot===undefined)rot=-90;
  var out=[],i;
  for(i=0;i<count;i++){
    var a=((rot+(360/count)*i)*Math.PI)/180;
    out.push([DCX+R*Math.cos(a),DCY+R*Math.sin(a)]);
  }
  return out;
}
function diceVertices(shape){
  switch(shape){
    case "square": return dicePoly(4,452,-45);
    case "triangle": return dicePoly(3,386);
    case "d20view":{
      var order=["C","D2","B","X","A","D1"],out=[],i;
      for(i=0;i<order.length;i++)out.push(dicePt(order[i]));
      return out;
    }
    case "pentagon": return dicePoly(5,356);
    case "kite": return [[DCX,176],[DCX+276,438],[DCX,848],[DCX-276,438]];
    case "barrel": return [[DCX-336,DCY-232],[DCX+336,DCY-232],[DCX+336,DCY+232],[DCX-336,DCY+232]];
    default: return null;
  }
}
var DICE_CIRCLE_R=336;
function diceFacePath(shape){
  if(shape==="circle")return "M "+(DCX-DICE_CIRCLE_R)+" "+DCY+" A "+DICE_CIRCLE_R+" "+DICE_CIRCLE_R+" 0 1 0 "+(DCX+DICE_CIRCLE_R)+" "+DCY+" A "+DICE_CIRCLE_R+" "+DICE_CIRCLE_R+" 0 1 0 "+(DCX-DICE_CIRCLE_R)+" "+DCY+" Z";
  var pts=diceVertices(shape),r=DICE_SHAPES[shape],n=pts.length,d="",i;
  for(i=0;i<n;i++){
    var prev=pts[(i-1+n)%n],cur=pts[i],next=pts[(i+1)%n];
    var v1=[prev[0]-cur[0],prev[1]-cur[1]],v2=[next[0]-cur[0],next[1]-cur[1]];
    var l1=Math.hypot(v1[0],v1[1])||1,l2=Math.hypot(v2[0],v2[1])||1;
    var rv=Math.min(r,l1/2.2,l2/2.2);
    var a=[cur[0]+v1[0]/l1*rv,cur[1]+v1[1]/l1*rv],b=[cur[0]+v2[0]/l2*rv,cur[1]+v2[1]/l2*rv];
    d+=i===0?"M "+a[0].toFixed(1)+" "+a[1].toFixed(1)+" ":"L "+a[0].toFixed(1)+" "+a[1].toFixed(1)+" ";
    d+="Q "+cur[0].toFixed(1)+" "+cur[1].toFixed(1)+" "+b[0].toFixed(1)+" "+b[1].toFixed(1)+" ";
  }
  return d+"Z";
}
function diceBounds(shape){
  if(shape==="circle")return {x0:DCX-DICE_CIRCLE_R,x1:DCX+DICE_CIRCLE_R,y0:DCY-DICE_CIRCLE_R,y1:DCY+DICE_CIRCLE_R};
  var pts=diceVertices(shape),xs=[],ys=[],i;
  for(i=0;i<pts.length;i++){xs.push(pts[i][0]);ys.push(pts[i][1]);}
  return {x0:Math.min.apply(Math,xs),x1:Math.max.apply(Math,xs),y0:Math.min.apply(Math,ys),y1:Math.max.apply(Math,ys)};
}
function diceTightView(shape){

  return {x0:Math.floor(bb.x0*10)/10,y0:Math.floor(bb.y0*10)/10,w:Math.ceil(bb.x1*10)/10-Math.floor(bb.x0*10)/10,h:Math.ceil(bb.y1*10)/10-Math.floor(bb.y0*10)/10};
}
function diceTightDim(shape,size){

  return {w:Math.max(1,Math.round(tv.w*s)),h:Math.max(1,Math.round(tv.h*s))};
}

function renderDiceSheet(p,scale){
  var sides=clampNum(Math.round(p.sides||6),2,20);
  var diceN=clampNum(Math.round(p.count||1),1,6);
  var total=sides*diceN;
  var cell=clampNum(p.size||200,120,340);
  var cols=Math.max(1,Math.ceil(Math.sqrt(total)));
  if(cols>total)cols=total;
  var rows=Math.ceil(total/cols);
  var pad=16;
  var w=cols*cell+pad*2,h=rows*cell+pad*2,maxDim=5000;
  if(w*scale>maxDim||h*scale>maxDim)scale=maxDim/Math.max(w,h);
  var canvas=document.createElement("canvas");
  canvas.width=Math.max(1,Math.round(w*scale)); canvas.height=Math.max(1,Math.round(h*scale));
  var ctx=canvas.getContext("2d");
  ctx.setTransform(scale,0,0,scale,0,0);
  var bg=p.bg||"blueprint";
  drawBg(ctx,w,h,bg);
  var shape=(p.shape&&p.shape!=="auto")?p.shape:diceAutoShape(sides);
  var pal=dicePalette(p.color||"#c40027");
  var marking=(p.marking==="pips"&&shape==="square"&&sides<=6)?"pips":"numerals";
  var guideCol=bg==="blueprint"?"rgba(255,255,255,.30)":"rgba(0,0,0,.28)";
  var k=(cell*0.86)/DS;
  var i,col,row,cx,cy;
  for(i=0;i<total;i++){
    col=i%cols; row=(i/cols)|0;
    cx=pad+col*cell+cell/2; cy=pad+row*cell+cell/2;
    ctx.save();
    ctx.translate(cx-512*k,cy-512*k);
    ctx.scale(k,k);
    diceStage(ctx,{value:(i%sides)+1,shape:shape,palette:pal,marking:marking,underline69:!!p.underline,clear:!!p.clear});
    ctx.restore();
    if(p.guides)drawGuides(ctx,cx,cy,cell*0.86,cell*0.86,guideCol,false);
  }
  if(p.stamp&&bg!=="transparent")drawStamp(ctx,w,h,p,TYPES.dice,sides*diceN,bg);
  return {canvas:canvas,count:sides*diceN,w:w,h:h,cols:cols,rows:rows};
}
function renderDiceFace(p,scale){
  var sides=clampNum(Math.round(p.sides||6),2,20);
  var shape=(p.shape&&p.shape!=="auto")?p.shape:diceAutoShape(sides);
  var face=clampNum(Math.round(p.face||1),1,sides);
  var pal=dicePalette(p.color||"#c40027");
  var marking=(p.marking==="pips"&&shape==="square"&&sides<=6)?"pips":"numerals";
  var tight=scale>=3;
  var size=tight?clampNum(parseInt(p.exportSize,10)||1024,128,4096):DS;
  var tv=diceTightView(shape);
  var dim=tight?diceTightDim(shape,size):{w:DS,h:DS};
  var mult=tight?1:scale;
  var canvas=document.createElement("canvas");
  canvas.width=Math.max(1,Math.round(dim.w*mult)); canvas.height=Math.max(1,Math.round(dim.h*mult));
  var ctx=canvas.getContext("2d");
  ctx.setTransform(mult,0,0,mult,0,0);
  if(tight){
    var s=size/Math.max(tv.w,tv.h);
    ctx.scale(s,s);
    ctx.translate(-tv.x0,-tv.y0);
  }
  diceStage(ctx,{value:face,shape:shape,palette:pal,marking:marking,underline69:!!p.underline,clear:!!p.clear,tight:tight});
  return {canvas:canvas,count:1,w:dim.w,h:dim.h,cols:1,rows:1};
}
function renderDice(p,scale){
  if((p.view||"sheet")==="face")return renderDiceFace(p,scale);
  return renderDiceSheet(p,scale);
}

  var i; for(i=0;i<project.assets.length;i++)if(project.assets[i].id===id)break;

}

var RAND_COLORS=["#c0272d","#d4af37","#2fae8c","#3a5cc9","#a04fc4","#e07b39","#5b8c2a","#2f6f8f","#7a5c2e","#b06a2a"];
function randomHex(){ return RAND_COLORS[Math.floor(Math.random()*RAND_COLORS.length)]; }
function randomizeAsset(){
  if(!cur)return;
  var d=TYPES[cur.type];
  var changed=false;
  d.controls.forEach(function(c){
    if(c.type==="color"&&c.k!=="ink"&&c.k!=="paper"&&c.k!=="glyphColor"&&c.k!=="labelColor"&&c.k!=="pip"&&c.k!=="hub"){
      cur.p[c.k]=randomHex(); changed=true;
    }
  });
  if(cur.type==="cards"){ cur.p.accent=randomHex(); }
  if(changed){ renderWorkbench(); sfx("dice"); saveSoon(); }
}
