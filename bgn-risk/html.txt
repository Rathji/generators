<script type="text/x-server-plugin">
// RISK online server - lobby, relay and state cache. All of this is PUBLIC source.
const rooms = new Map();      // code -> room
const connRooms = new Map();  // connId -> room code
const CODE_CHARS = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
function makeCode(){ let s=""; for(let i=0;i<4;i++) s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)]; return s; }
function roomOf(conn){ return rooms.get(connRooms.get(conn.id)); }
function broadcast(room, obj){ pubsub.publish("room:"+room.code, JSON.stringify(obj)); }
function playerList(room){ return room.players.map(x=>({connId:x.connId, name:x.name, power:x.power})); }

/* ── durable leaderboard (fastest conquests, lives at the end of `state`) ──
   Fewer turns = better (asc). The server otherwise keeps rooms in ephemeral
   Maps; only this region is persisted, so it survives restarts. */
const LB_VERSION=1, LB_CAP=50, LB_NAME_MAX=24, LB_EXTRA_MAX=11, LB_ENTRY=44, LB_HEAD=2;
const LB_BYTES=LB_HEAD+LB_CAP*LB_ENTRY;
const LB_BASE=Math.max(state.length-LB_BYTES, 0);
const lbTimes=new Map(), lbNets=new Map();
function cleanField(s,max){ return String(s||"").replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").trim().slice(0,max); }
function lbInit(){ if(state[LB_BASE]!==LB_VERSION){ state.fill(0,LB_BASE,state.length); state[LB_BASE]=LB_VERSION; } }
function lbRead(){
  const a=[];
  for(let i=0;i<LB_CAP;i++){
    const e=LB_BASE+LB_HEAD+i*LB_ENTRY;
    if(state[e]!==1)continue;
    let name=""; for(let k=1;k<=LB_NAME_MAX;k++){ const c=state[e+k]; if(!c)break; name+=String.fromCharCode(c); }
    let extra=""; for(let k=0;k<LB_EXTRA_MAX;k++){ const c=state[e+33+k]; if(!c)break; extra+=String.fromCharCode(c); }
    a.push({i,
      name,
      score:((state[e+25]<<24)|(state[e+26]<<16)|(state[e+27]<<8)|state[e+28])>>>0,
      ts:((state[e+29]<<24)|(state[e+30]<<16)|(state[e+31]<<8)|state[e+32])>>>0,
      extra});
  }
  return a;
}
function lbWriteSlot(i,name,score,ts,extra){
  const e=LB_BASE+LB_HEAD+i*LB_ENTRY;
  state.fill(0,e,e+LB_ENTRY);
  state[e]=1;
  const nb=[]; for(let k=0;k<name.length&&k<LB_NAME_MAX;k++)nb.push(name.charCodeAt(k));
  for(let k=0;k<nb.length;k++)state[e+1+k]=nb[k];
  const xb=[]; for(let k=0;k<extra.length&&k<LB_EXTRA_MAX;k++)xb.push(extra.charCodeAt(k));
  for(let k=0;k<xb.length;k++)state[e+33+k]=xb[k];
  state[e+25]=(score>>>24)&255; state[e+26]=(score>>>16)&255; state[e+27]=(score>>>8)&255; state[e+28]=score&255;
  state[e+29]=(ts>>>24)&255; state[e+30]=(ts>>>16)&255; state[e+31]=(ts>>>8)&255; state[e+32]=ts&255;
}
function lbBetter(asc,a,b){ if(asc)return a.score<b.score||(a.score===b.score&&a.ts<b.ts); return a.score>b.score||(a.score===b.score&&a.ts<b.ts); }
function lbSubmit(name,score,ts,extra,asc){
  lbInit();
  const a=lbRead();
  if(a.length<LB_CAP){
    for(let i=0;i<LB_CAP;i++){ if(state[LB_BASE+LB_HEAD+i*LB_ENTRY]!==1){ lbWriteSlot(i,name,score,ts,extra); return; } }
    return;
  }
  let wi=-1; const w={score:0,ts:0};
  for(const en of a){ if(wi===-1||lbBetter(asc,w,en)){ wi=en.i; w.score=en.score; w.ts=en.ts; } }
  if(lbBetter(asc,{score,ts},w))lbWriteSlot(wi,name,score,ts,extra);
}
self.rpc = {
  createRoom({conn}, data) {
    let info; try { info = JSON.parse(data); } catch(e){ info = {}; }
    let code; do { code = makeCode(); } while (rooms.has(code));
    const room = { code, hostId:conn.id, max: Math.max(2, Math.min(6, info.max||6)),
      players:[{connId:conn.id, name:String(info.name||"Host").slice(0,20), power:-1}],
      started:false, lastState:"" };
    if (rooms.size > 60) { const first = rooms.keys().next().value; rooms.delete(first); }
    rooms.set(code, room); connRooms.set(conn.id, code);
    conn.subscribe("room:"+code);
    return JSON.stringify({code, connId:conn.id, host:true, players:playerList(room)});
  },
  joinRoom({conn}, data) {
    let info; try { info = JSON.parse(data); } catch(e){ info = {}; }
    const code = String(info.code||"").trim().toUpperCase();
    const room = rooms.get(code);
    if (!room) return JSON.stringify({error:"Room not found"});
    if (room.started) return JSON.stringify({error:"This game has already started"});
    if (room.players.length >= room.max) return JSON.stringify({error:"Room is full"});
    if (room.players.some(p=>p.connId===conn.id)) return JSON.stringify({error:"Already in room"});
    room.players.push({connId:conn.id, name:String(info.name||"Player").slice(0,20), power:-1});
    connRooms.set(conn.id, code);
    conn.subscribe("room:"+code);
    broadcast(room, {t:"lobby", players:playerList(room)});
    return JSON.stringify({code, connId:conn.id, host:false, players:playerList(room)});
  },
  setPower({conn}, data) {
    const room = roomOf(conn); if (!room) return "{}";
    let info; try { info = JSON.parse(data); } catch(e){ return "{}"; }
    const pl = room.players.find(p=>p.connId===conn.id);
    if (pl) { pl.power = (typeof info.power === "number") ? info.power : -1; }
    broadcast(room, {t:"lobby", players:playerList(room)});
    return "{}";
  },
  setName({conn}, data) {
    const room = roomOf(conn); if (!room) return "{}";
    let info; try { info = JSON.parse(data); } catch(e){ return "{}"; }
    const pl = room.players.find(p=>p.connId===conn.id);
    if (pl) pl.name = String(info.name||pl.name).slice(0,20);
    broadcast(room, {t:"lobby", players:playerList(room)});
    return "{}";
  },
  startGame({conn}, data) {
    const room = roomOf(conn);
    if (!room || room.hostId !== conn.id) return JSON.stringify({error:"Only the host can start"});
    const used = new Set();
    for (const pl of room.players) {
      if (pl.power < 0 || pl.power > 5 || used.has(pl.power)) {
        let p = -1;
        for (let i=0;i<6;i++) if (!used.has(i)) { p = i; break; }
        if (p < 0) return JSON.stringify({error:"Not enough free colors"});
        pl.power = p;
      }
      used.add(pl.power);
    }
    room.started = true;
    broadcast(room, {t:"start", players:playerList(room)});
    return JSON.stringify({players:playerList(room)});
  },
  getState({conn}, data) {
    const room = roomOf(conn); if (!room) return "";
    return room.lastState;
  },
  submitScore({conn}, data) {
    let p; try { p = JSON.parse(data); } catch(e){ return '{"ok":false,"err":"bad"}'; }
    const now = Date.now();
    if (now-(lbTimes.get(conn.id)||0) < 1200) return '{"ok":false,"err":"rate"}';
    lbTimes.set(conn.id, now);
    if (now-(lbNets.get(conn.net[3])||0) < 6000) return '{"ok":false,"err":"rate"}';
    lbNets.set(conn.net[3], now);
    const name = cleanField(p.name,24) || "Player";
    const score = Math.max(0, Math.min(0xffffffff, Math.floor(Number(p.score)||0)));
    const extra = cleanField(p.extra,11);
    lbSubmit(name, score, Math.floor(now/1000), extra, true);
    return '{"ok":true}';
  },
  getScores({conn}, data) {
    lbInit();
    const a = lbRead().sort(function(x,y){ return lbBetter(true,x,y)?-1:1; }).slice(0,10);
    return JSON.stringify({scores:a});
  },
  rematch({conn}, data) {
    const room = roomOf(conn); if (!room) return '{"error":"Not in a room"}';
    if (!room.started) return '{"error":"The game has not started"}';
    if (!room.players.some(p=>p.connId!==conn.id)) return '{"error":"Your opponent left — start a new room"}';
    room.lastState = "";
    broadcast(room, {t:"rematch"});
    return '{"ok":true}';
  },
  leaveRoom({conn}, data) {
    const code = connRooms.get(conn.id); if (!code) return "{}";
    const room = rooms.get(code); if (!room) return "{}";
    room.players = room.players.filter(p=>p.connId!==conn.id);
    connRooms.delete(conn.id);
    if (room.players.length===0) { rooms.delete(code); return "{}"; }
    broadcast(room, {t:"leave", connId:conn.id, players:playerList(room)});
    return "{}";
  }
};
self.onmessage = ({conn, data}) => {
  if (typeof data !== "string") return;
  let msg; try { msg = JSON.parse(data); } catch(e){ return; }
  const room = roomOf(conn); if (!room) return;
  if (msg.t === "state") room.lastState = (typeof msg.d === "string") ? msg.d : "";
  broadcast(room, {f:conn.id, t:msg.t, d:msg.d});
};
self.onclose = ({conn}) => {
  const code = connRooms.get(conn.id); if (!code) return;
  const room = rooms.get(code); if (!room) { connRooms.delete(conn.id); return; }
  room.players = room.players.filter(p=>p.connId!==conn.id);
  connRooms.delete(conn.id);
  if (room.players.length===0) { rooms.delete(code); return; }
  broadcast(room, {t:"leave", connId:conn.id, players:playerList(room)});
  if (room.hostId === conn.id) {
    room.hostId = room.players[0].connId;
    room.started = false;
    broadcast(room, {t:"hostchange", connId:room.hostId});
  }
};

function el(id){ return document.getElementById(id); }
function shade(hex,amt){ const n=parseInt(hex.slice(1),16); const r=Math.max(0,Math.min(255,Math.round((n>>16)*(1+amt)))); const g=Math.max(0,Math.min(255,Math.round(((n>>8)&255)*(1+amt)))); const b=Math.max(0,Math.min(255,Math.round((n&255)*(1+amt)))); return "rgb("+r+","+g+","+b+")"; }
function shuffle(a){ for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; } return a; }
function rollDice(n){ const a=[]; for(let i=0;i<n;i++) a.push(1+Math.floor(Math.random()*6)); return a; }

}

let G = null;
let socket = null;
let ui = { mode:"idle", origin:null, selected:null };
let lastOrigin = null;

function newGame(cfg){
  const np = cfg.np;
  const players = [];
  for (let i=0;i<np;i++){
    const ci = cfg.colorIdx ? cfg.colorIdx[i] : i;
    players.push({ idx:i, name:cfg.names ? cfg.names[i] : ("Player "+(i+1)), colorIdx:ci, color:PLAYER_DEFS[ci].color, alive:true, human:cfg.humans.includes(i), cards:[] });
  }
  const zones = new Map();
  const order = shuffle([...Array(N)].map((_,i)=>i+1));
  for (let k=0;k<N;k++) zones.set(order[k],{owner:k%np, armies:0});
  const terrOf = players.map(p=>{ const l=[]; zones.forEach((z,id)=>{ if(z.owner===p.idx) l.push(id); }); return l; });
  const diffF = AI_DIFF==="easy" ? {human:1.15, ai:0.85} : AI_DIFF==="hard" ? {human:0.85, ai:1.2} : {human:1, ai:1};
  const base = initialArmiesFor(np);
  terrOf.forEach((list,i)=>{
    const pool = Math.max(list.length, Math.round(base*(players[i].human?diffF.human:diffF.ai)));
    for (const t of list) zones.get(t).armies = 1;
    let rem = pool - list.length, guard=0;
    while (rem>0 && guard++<100000){ zones.get(list[Math.floor(Math.random()*list.length)]).armies++; rem--; }
  });
  const deck = shuffle([...Array(N)].map((_,i)=>({type:CARD_TYPES[i%3], terr:i+1})));
  G = {
    cfg, np, players, zones, deck, discard:[],
    mode: cfg.mode||"solo",
    turn:1, phase:"place", phasePlayer:0, toPlace:0, tradeCount:0,
    captured:false, lastConqueror:null, fortified:false, lastBattle:null,
    winner:null, over:false, startingPlayers:np, log:[],
    net: cfg.mode==="online" ? {isHost:!!cfg.isHost, myPlayer:cfg.myPlayer, remote:new Set(cfg.remote||[]), myConnId:null, roomCode:null, myName:cfg.name||"Player"} : null,
  };
  ui = { mode:"idle", origin:null, selected:null };
  lastOrigin = null;
}

function countTerritories(idx){ let n=0; for (const [,z] of G.zones) if (z.owner===idx) n++; return n; }
function countArmies(idx){ let n=0; for (const [,z] of G.zones) if (z.owner===idx) n+=z.armies; return n; }
function reachableFriendly(from){
  const pIdx = G.zones.get(from).owner;
  const seen=new Set([from]), q=[from];
  while (q.length){
    const cur=q.shift();
    for (const n of adj.get(cur)){
      if (!seen.has(n) && G.zones.get(n).owner===pIdx){ seen.add(n); q.push(n); }
    }
  }
  return seen;
}
function tradeValue(){ return G.tradeCount<5 ? [4,6,8,10,12,15][G.tradeCount] : 15+(G.tradeCount-5)*5; }
function calcReinforcements(p){
  const tc=countTerritories(p.idx);
  let n=Math.max(3,Math.floor(tc/3));
  for (const c of CONT){
    if (c.zones.every(z=>G.zones.get(z).owner===p.idx)) n+=c.bonus;
  }
  return n;
}
function nextAlive(from){
  let n=(from+1)%G.np, guard=0;
  while (guard++<G.np){ if (G.players[n].alive) return n; n=(n+1)%G.np; }
  return from;
}
function findSet(cards){
  const has={infantry:-1,cavalry:-1,artillery:-1};
  cards.forEach((c,i)=>{ if (has[c.type]===-1) has[c.type]=i; });
  if (has.infantry>=0 && has.cavalry>=0 && has.artillery>=0) return [has.infantry,has.cavalry,has.artillery];
  for (const t of CARD_TYPES){
    const idxs=[];
    cards.forEach((c,i)=>{ if (c.type===t) idxs.push(i); });
    if (idxs.length>=3) return idxs.slice(0,3);
  }
  return null;
}
function turnController(pIdx){
  if (G.mode!=="online") return G.players[pIdx].human ? "local" : "ai";
  if (G.net.myPlayer===pIdx) return "local";
  if (!G.net.isHost) return "remote";
  if (G.net.remote.has(pIdx)) return "remote";
  return "ai";
}

// ============================== ACTIONS ==============================
function doPlace(p, zone, n){
  n=Math.floor(n);
  if (G.phase!=="place") return "Not the reinforce phase";
  if (n<1) return "Bad count";
  const z=G.zones.get(zone);
  if (!z) return "Bad territory";
  if (z.owner!==p.idx) return "You don't control this territory";
  if (n>G.toPlace) return "Only "+G.toPlace+" armies left to place";
  G.toPlace-=n; z.armies+=n;
  return null;
}
function doTrade(p, idxs){
  if (G.phase!=="place") return "Only during reinforce";
  if (idxs.length!==3) return "Select exactly 3 cards";
  const cards=idxs.map(i=>p.cards[i]);
  if (cards.some(c=>!c)) return "Bad selection";
  const types=cards.map(c=>c.type);
  if (!(types.every(t=>t===types[0]) || new Set(types).size===3)) return "Need 3 of a kind or one of each";
  idxs.sort((a,b)=>b-a).forEach(i=>{ G.discard.push(p.cards.splice(i,1)[0]); });
  const v=tradeValue(); G.tradeCount++;
  G.toPlace += v;
  log(p.name+" trades in "+types.map(t=>CARD_LABEL[t]).join(", ")+" for +"+v+" armies.");
  return null;
}
function doUseCard(p, ci){
  if (G.phase!=="place") return "Only during reinforce";
  const c=p.cards[ci]; if (!c) return "No such card";
  const z=G.zones.get(c.terr);
  if (z.owner!==p.idx) return "You don't control "+zoneById[c.terr].name+" (needed for this card's bonus)";
  G.discard.push(p.cards.splice(ci,1)[0]);
  z.armies+=2;
  log(p.name+" cashes a "+CARD_LABEL[c.type]+" card for +2 armies in "+zoneById[c.terr].name+".");
  return null;
}
function doAttack(p, from, to, dice){
  if (G.phase!=="attack") return "Not the attack phase";
  const fz=G.zones.get(from), tz=G.zones.get(to);
  if (!fz||!tz) return "Bad territory";
  if (fz.owner!==p.idx) return "Attack from your own territory";
  if (tz.owner===p.idx) return "Target is your own territory";
  if (!adj.get(from).has(to)) return "Not adjacent";
  if (fz.armies<2) return "Need at least 2 armies";
  const aDice=Math.max(1,Math.min(3,Math.floor(dice)||3, fz.armies-1));
  const dDice=Math.max(1,Math.min(2,tz.armies));
  const aRolls=rollDice(aDice).sort((a,b)=>b-a);
  const dRolls=rollDice(dDice).sort((a,b)=>b-a);
  const pairs=Math.min(aDice,dDice);
  let aLoss=0,dLoss=0;
  for (let i=0;i<pairs;i++){ if (aRolls[i]>dRolls[i]) dLoss++; else aLoss++; }
  fz.armies-=aLoss; tz.armies-=dLoss;
  G.lastBattle={from,to,aRolls,dRolls,aLoss,dLoss,conquered:false,attacker:p.name};
  let msg=p.name+" attacks "+zoneById[to].name+" from "+zoneById[from].name+": attack ["+aRolls.join(", ")+"] vs defend ["+dRolls.join(", ")+"].";
  if (tz.armies<=0){
    const move=Math.max(1,Math.min(fz.armies-1, aDice-aLoss));
    tz.owner=p.idx; tz.armies=move; fz.armies-=move;
    G.captured=true; G.lastConqueror=p.idx;
    G.lastBattle.conquered=true;
    msg+=" CONQUERED — "+p.name+" takes "+zoneById[to].name+" with "+move+" armies.";
  } else {
    msg+=" "+(aLoss?p.name+" loses "+aLoss+"":"no attacker losses")+" / "+(dLoss?zoneById[to].name+" loses "+dLoss+"":"no defender losses")+".";
  }
  log(msg);
  checkElims();
  checkWin();
  return null;
}
function doFortify(p, from, to, n){
  n=Math.floor(n);
  if (G.phase!=="fortify") return "Not the fortify phase";
  const fz=G.zones.get(from), tz=G.zones.get(to);
  if (!fz||!tz) return "Bad territory";
  if (fz.owner!==p.idx || tz.owner!==p.idx) return "Both territories must be yours";
  if (!reachableFriendly(from).has(to)) return "No friendly path between them";
  if (n<1 || n>fz.armies-1) return "Bad count (leave at least 1 behind)";
  fz.armies-=n; tz.armies+=n;
  G.fortified=true;
  log(p.name+" fortifies "+zoneById[to].name+" with "+n+" armies.");
  return null;
}

function earnCard(p){
  if (G.deck.length===0){
    if (G.discard.length===0) return;
    G.deck=shuffle(G.discard); G.discard=[];
  }
  const c=G.deck.pop();
  p.cards.push(c);
  log(p.name+" draws a Risk card: "+CARD_ICON[c.type]+" "+CARD_LABEL[c.type]+" — "+zoneById[c.terr].name+".");
}
function tryAutoTrade(p){
  const set=findSet(p.cards);
  if (!set) return false;
  const e=doTrade(p,set);
  return !e;
}
function checkElims(){
  if (G.over) return;
  for (const p of G.players){
    if (p.alive && countTerritories(p.idx)===0){
      p.alive=false;
      log("☠ "+p.name+" has been eliminated!");
      if (G.lastConqueror!=null && G.lastConqueror!==p.idx){
        const c=G.players[G.lastConqueror];
        const n=p.cards.length;
        if (n>0){ c.cards.push(...p.cards); p.cards=[]; log(c.name+" collects "+n+" Risk card"+(n===1?"":"s")+" from the fallen."); }
        while (c.cards.length>=5){ if(!tryAutoTrade(c)) break; }
      }
    }
  }
  const alive=G.players.filter(p=>p.alive);
  if (alive.length===1 && G.startingPlayers>1) endGame(alive[0], alive[0].name+" CONQUERS THE WORLD!");
}
function checkWin(){
  if (G.over) return;
  for (const p of G.players){
    if (p.alive && countTerritories(p.idx)===N) endGame(p, p.name+" CONQUERS THE WORLD!");
  }
}
function endGame(winner, msg){
  G.winner = winner?winner.idx:null;
  G.over = true;
  if (winner && (G.net ? winner.idx===G.net.myPlayer : winner.human)) sfx("win");
  else sfx("lose");
  log("🏁 "+msg);
  render();
  const isOnlineWin = G.mode==="online";
  openModal(
    '<h2>🏆 '+msg+'</h2>'+
    '<p>'+(winner?winner.name+' now rules every territory on the map.'+((isOnlineWin&&!G.net.isHost)?' The host will confirm the final state.':' You are the master of the world.'):'So passes the glory of the world.')+'</p>'+
    '<p class="muted">The world was conquered in '+G.turn+' turn'+(G.turn===1?"":"s")+'.</p>'+
    (isOnlineWin
      ? '<button class="primary" onclick="requestRematch()">⚔️ Rematch</button> <button onclick="closeModal()">View Board</button> <button onclick="showLb()">🏆 Leaderboard</button> <button onclick="openSetup()">Menu</button>'
      : '<button class="primary" onclick="openSetup()">New Game</button> <button onclick="showLb()">🏆 Leaderboard</button> <button onclick="closeModal()">View Board</button>')
  );
  if (winner){
    const extra=String(PLAYER_DEFS[winner.colorIdx].name).slice(0,11);
    const name=G.net?G.net.myName:lbName();
    if (G.mode==="online"){ if (G.net&&G.net.isHost) lbSubmitScore(name,G.turn,extra); }
    else lbSubmitScore(name,G.turn,extra);
  }
}

// ============================== TURN FLOW ==============================
function startGame(){
  G.turn=1; G.phasePlayer=0;
  log("RISK — the struggle for world domination begins.");
  toast("RISK — BEGIN!");
  beginTurn(0);
}
function beginTurn(pIdx){
  G.phasePlayer=pIdx;
  const p=G.players[pIdx];
  G.toPlace=calcReinforcements(p);
  G.captured=false; G.lastConqueror=null; G.fortified=false; G.lastBattle=null;
  G.phase="place";
  log("— Turn "+G.turn+": "+p.name+" —");
  log(p.name+" receives "+G.toPlace+" armies.");
  startPhase();
}
function startPhase(){
  const p=G.players[G.phasePlayer];
  const ctrl=turnController(G.phasePlayer);
  uiReset();
  if (G.over) return;
  if (ctrl==="local"){
    setStatus('<span style="color:'+p.color+'">'+p.name+'</span> — Turn '+G.turn+' · <b>'+PHASE_LABEL[G.phase]+'</b>');
    toast(p.name+": "+PHASE_LABEL[G.phase]);
    render();
  } else if (ctrl==="remote"){
    setStatus('Waiting for <span style="color:'+p.color+'">'+p.name+'</span> to '+PHASE_LABEL[G.phase]+'…');
    render();
  } else {
    setStatus('<span style="color:'+p.color+'">'+p.name+'</span> (AI) — '+PHASE_LABEL[G.phase]+'…');
    render();
    runAI(p);
  }
  if (G.mode==="online" && G.net.isHost) sendState();
}
function endPhase(){
  if (G.over) return;
  const p=G.players[G.phasePlayer];
  if (G.phase==="place"){
    if (p.cards.length>=5){ toast(p.name+" holds 5+ cards and must trade a set first"); return; }
    G.phase="attack";
  } else if (G.phase==="attack"){
    if (G.captured) earnCard(p);
    G.phase="fortify";
  } else {
    checkWin(); if (G.over) return;
    G.turn++;
    if (G.mode!=="online") autosave();
    beginTurn(nextAlive(G.phasePlayer));
    return;
  }
  log(p.name+" — "+PHASE_LABEL[G.phase]+".");
  startPhase();
}
function endPhaseBtn(){
  if (!G || G.over) return;
  const ctrl=turnController(G.phasePlayer);
  if (ctrl==="ai") return;
  if (G.mode==="online" && !G.net.isHost){
    if (ctrl!=="local") return;
    if (socket && socket.readyState===1){
      socket.send(JSON.stringify({t:"endPhase", d:{player:G.phasePlayer}}));
      setStatus("Submitted — waiting for the host…");
      toast("Submitted");
    }
    return;
  }
  if (ctrl!=="local") return;
  endPhase();
}

// ============================== ONLINE ==============================
let netState={ myConnId:null, roomCode:null, isHost:false, myPlayer:-1, myName:"Player" };

function netConnect(opts){
  closeModal();
  setStatus("Connecting…");
  const name = opts.name || "Player";
  try { socket = root.createServerSocket(); }
  catch(e){ openModal('<h2>Network Error</h2><p>'+e+'</p><button onclick="openSetup()">Back</button>'); return; }
  socket.addEventListener("open", ()=>{
    const p = opts.isHost ? socket.rpc.createRoom(JSON.stringify({name, max:6})) : socket.rpc.joinRoom(JSON.stringify({code:opts.code, name}));
    p.then(reply=>{
      let info; try{ info=JSON.parse(reply); }catch(e){ openModal('<h2>Error</h2><p>'+reply+'</p><button onclick="openSetup()">Back</button>'); return; }
      if (info.error){ openModal('<h2>Error</h2><p>'+info.error+'</p><button onclick="openSetup()">Back</button>'); return; }
      netState.myConnId=info.connId; netState.roomCode=info.code; netState.isHost=opts.isHost; netState.myName=name;
      showLobby(info.players, info);
    }).catch(e=>{ openModal('<h2>Network Error</h2><p>'+(e&&e.message||e)+'</p><button onclick="openSetup()">Back</button>'); });
  });
  socket.addEventListener("message", ev=>{
    let msg; try{ msg=JSON.parse(ev.data); }catch(e){ return; }
    onNetMessage(msg);
  });
  socket.addEventListener("close", ev=>{ toast("Connection closed"+(ev.code?(" ("+ev.code+")"):"")); });
}
function onNetMessage(msg){
  if (msg.f && msg.f===netState.myConnId) return;
  if (msg.t==="lobby"){ showLobby(msg.players,{host:netState.isHost}); return; }
  if (msg.t==="start"){
    const me = msg.players.find(x=>x.connId===netState.myConnId);
    startOnlineGame(msg.players, me);
    return;
  }
  if (msg.t==="leave"){
    toast("A player left the room");
    if (G && G.net && G.net.isHost && msg.players){
      const rem = new Set(msg.players.filter(x=>x.connId!==netState.myConnId).map(x=>x.power));
      if (G.net) G.net.remote = rem;
      render();
    }
    if (msg.players) showLobby(msg.players,{host:netState.isHost});
    return;
  }
  if (msg.t==="hostchange"){ toast("The host left the game"); return; }
  if (msg.t==="chat"){ appendChat(msg.d); return; }
  if (msg.t==="rematch"){ doRematch(); return; }
  if (msg.t==="state"){ applyNetState(msg.d); return; }
  if (msg.t==="act"){
    if (!G || G.over) return;
    if (G.mode!=="online" || !G.net.isHost) return;
    const d=msg.d||{};
    if (d.player!==G.phasePlayer) return;
    if (turnController(G.phasePlayer)!=="remote") return;
    const e=applyIntent(d.it);
    if (e) log("⚠ "+G.players[G.phasePlayer].name+": "+e);
    render(); sendState();
    return;
  }
  if (msg.t==="endPhase"){
    if (!G || G.over) return;
    if (G.mode!=="online" || !G.net.isHost) return;
    const d=msg.d||{};
    if (d.player!==G.phasePlayer) return;
    if (turnController(G.phasePlayer)!=="remote") return;
    endPhase();
    return;
  }
}
function showLobby(players, info){
  const self = players.find(x=>x.connId===netState.myConnId);
  const taken = new Set(players.map(x=>x.power));
  let rows = players.map(x=>{
    const isMe = x.connId===netState.myConnId;
    return '<div class="actRow" style="justify-content:space-between">'+
      '<span>'+(isMe?'<b>▶ ':'')+x.name+(isMe?' (you)</b>':'')+'</span>'+
      '<span>'+(x.power>=0?'<span style="color:'+PLAYER_DEFS[x.power].color+'">'+PLAYER_DEFS[x.power].name+'</span>':'<span class="muted">no color</span>')+'</span>'+
      '</div>';
  }).join("");
  let colorBtns = PLAYER_DEFS.map((pd,i)=>{
    const disabled = taken.has(i) && !(self && self.power===i);
    return '<button class="powerBtn" '+(disabled?"disabled":"")+' onclick="pickColor('+i+')">'+
      '<span class="swatch" style="background:'+pd.color+'"></span> '+pd.name+(self&&self.power===i?' ✔':'')+'</button>';
  }).join("");
  let hostRow = netState.isHost
    ? '<button class="primary" onclick="hostStart()" style="margin-top:8px">Start Game</button>'
    : '<p class="muted">Waiting for the host to start…</p>';
  openModal('<h2>War Room</h2>'+
    '<p>Room code:</p><div class="bigCode">'+netState.roomCode+'</div>'+
    '<h3>Generals ('+players.length+'/'+CONFIG.maxPlayers+')</h3><div id="lobbyPlayers">'+rows+'</div>'+
    '<h3>Choose your color</h3>'+colorBtns+
    '<div style="margin-top:10px">'+hostRow+'</div>'+
    '<button onclick="leaveRoom()" style="margin-top:8px">Leave Room</button>');
}
function pickColor(colorIdx){
  if (!netState.roomCode) return;
  socket.rpc.setPower(JSON.stringify({code:netState.roomCode, power:colorIdx})).then(()=>{});
}
function hostStart(){
  socket.rpc.startGame(JSON.stringify({code:netState.roomCode})).then(reply=>{
    let info; try{ info=JSON.parse(reply); }catch(e){ return; }
    if (info.error){ toast(info.error); return; }
  });
}
function leaveRoom(){
  if (socket) socket.rpc.leaveRoom(JSON.stringify({})).then(()=>{});
  try{ socket && socket.close(); }catch(e){}
  openSetup();
}
function startOnlineGame(players, me){
  if (!me){ toast("Could not find you in the game!"); return; }
  const myIdx = players.findIndex(x=>x.connId===netState.myConnId);
  if (myIdx<0){ toast("Could not find you in the game!"); return; }
  netState.myPlayer = myIdx;
  const n=players.length;
  const cfg={ mode:"online", np:n, humans:[...Array(n)].map((_,i)=>i),
    colorIdx:players.map((p,i)=> (p.power>=0?p.power:i)),
    names:players.map(p=>p.name),
    isHost:netState.isHost, myPlayer:myIdx, remote:players.map((p,i)=>i).filter(i=>i!==myIdx), name:netState.myName };
  closeModal();
  if (netState.isHost){
    newGame(cfg);
    G.net.myConnId=netState.myConnId; G.net.roomCode=netState.roomCode; G.net.myName=netState.myName;
    toast("Game started! Room "+netState.roomCode);
    startGame();
  } else {
    if (!G){
      newGame(cfg);
      G.net.myConnId=netState.myConnId; G.net.roomCode=netState.roomCode; G.net.myName=netState.myName;
    }
    setStatus("Waiting for the host…");
    render();
  }
}
function applyNetState(s){
  let o; try{ o=JSON.parse(s); }catch(e){ return; }
  if (!G || G.np!==o.players.length){
    const n=o.players.length;
    newGame({mode:"online", np:n, humans:[...Array(n)].map((_,i)=>i), colorIdx:o.players.map(p=>p.colorIdx), names:o.players.map(p=>p.name), isHost:netState.isHost, myPlayer:netState.myPlayer, remote:[], name:netState.myName});
  }
  deserializeInto(G,o);
  G.mode="online";
  if (!G.net) G.net = { isHost:netState.isHost, myPlayer:netState.myPlayer, remote:new Set(), myConnId:netState.myConnId, roomCode:netState.roomCode, myName:netState.myName };

function serialize(){
  return {
    v:1, turn:G.turn, phase:G.phase, phasePlayer:G.phasePlayer, tradeCount:G.tradeCount,
    toPlace:G.toPlace, captured:G.captured, lastConqueror:G.lastConqueror, lastBattle:G.lastBattle,
    log:G.log.slice(-150),
    players:G.players.map(p=>({idx:p.idx, name:p.name, colorIdx:p.colorIdx, alive:p.alive, human:p.human, cards:p.cards})),
    zones:[...G.zones].map(([id,z])=>[id,z.owner,z.armies]),
    deck:G.deck, discard:G.discard,
  };
}
function deserializeInto(G,o){
  G.turn=o.turn; G.phase=o.phase; G.phasePlayer=o.phasePlayer; G.tradeCount=o.tradeCount;
  G.toPlace=o.toPlace; G.captured=!!o.captured; G.lastConqueror=o.lastConqueror; G.lastBattle=o.lastBattle||null;
  G.log=o.log.slice(); G.deck=o.deck.slice(); G.discard=o.discard.slice();
  o.players.forEach(p=>{
    const q=G.players[p.idx];
    q.name=p.name; q.colorIdx=p.colorIdx; q.color=PLAYER_DEFS[p.colorIdx].color; q.alive=p.alive; q.human=p.human; q.cards=p.cards.slice();
  });
  for (const [id,owner,armies] of o.zones){
    const z=G.zones.get(id);
    if (z){ z.owner=owner; z.armies=armies; }
  }
}

// ============================== AI ==============================
function runAI(p){
  const delay = window.__fastAI ? (5+Math.random()*15) : (180+Math.random()*320);
  setTimeout(()=>{
    try {
      if (G.over || G.players[G.phasePlayer].idx!==p.idx) return;
      aiPhase(p);
      if (G.over){ render(); return; }
      endPhase();
    } catch(e){
      console.error("AI error:", e);
      try { if (!G.over) endPhase(); } catch(e2){ console.error("recovery failed", e2); }
    }
  }, delay);
}
function aiPhase(p){
  if (G.phase==="place") aiPlace(p);
  else if (G.phase==="attack") aiAttack(p);
  else if (G.phase==="fortify") aiFortify(p);
  render();
}
function aiPlace(p){
  while (p.cards.length>=3){ if (!tryAutoTrade(p)) break; }
  let guard=0;
  while (G.toPlace>0 && guard++<300){
    let best=null, bestScore=-1e9;
    for (const [id,z] of G.zones){
      if (z.owner!==p.idx) continue;
      let threat=0, enCount=0;
      for (const n of adj.get(id)){
        const nz=G.zones.get(n);
        if (nz.owner!==p.idx){ threat+=nz.armies; enCount++; }
      }
      if (enCount>0){
        const score = threat - z.armies;
        if (score>bestScore){ bestScore=score; best=id; }
      }
    }
    if (best==null){
      const own=[...G.zones.keys()].filter(id=>G.zones.get(id).owner===p.idx);
      if (!own.length) break;
      best=own[Math.floor(Math.random()*own.length)];
    }
    doPlace(p,best,1);
  }
}
function aiAttack(p){
  let guard=0;
  while (guard++<80){
    const cands=[];
    for (const [id,z] of G.zones){
      if (z.owner!==p.idx || z.armies<2) continue;
      for (const n of adj.get(id)){
        const nz=G.zones.get(n);
        if (nz.owner===p.idx) continue;
        cands.push({from:id,to:n, fz:z.armies, tz:nz.armies, adv:z.armies-nz.armies});
      }
    }
    if (!cands.length) break;
    cands.sort((a,b)=> (b.adv-a.adv) || (b.fz-a.fz));
    const c=cands[0];
    if (c.fz<2) break;
    if (c.adv<1 && Math.random()<0.75) break;
    const e=doAttack(p,c.from,c.to,3);
    if (e) break;
    if (G.over) break;
    const fz=G.zones.get(c.from);
    const tz=G.zones.get(c.to);
    if (tz.owner===p.idx){ if (Math.random()<0.5) continue; else break; }
    if (fz.armies<=1) continue;
    if (c.tz>=3 && fz.armies<=c.tz+1) break;
  }
}
function aiFortify(p){
  const frontier=new Set();
  for (const [id,z] of G.zones){
    if (z.owner===p.idx){
      for (const n of adj.get(id)) if (G.zones.get(n).owner!==p.idx){ frontier.add(id); break; }
    }
  }
  let best=null;
  for (const [id,z] of G.zones){
    if (z.owner!==p.idx || z.armies<=2 || frontier.has(id)) continue;
    const reach=reachableFriendly(id);
    let target=null;
    for (const f of frontier) if (reach.has(f)){ target=f; break; }
    if (target){
      const gain=z.armies-1;
      if (!best || gain>best.gain) best={from:id,to:target,gain};
    }
  }
  if (best) doFortify(p,best.from,best.to,best.gain);
}

function drawMap(){
  const ctx=boardCtx;
  ctx.fillStyle="#0d2438"; ctx.fillRect(0,0,CW,CH);
  for (let y=0;y<GH;y++) for (let x=0;x<GW;x++){
    const v=grid[y*GW+x];
    let col;
    if (v===0) col="#123052";
    else if (v===250) col="#2b3a44";
    else if (v>=101) col="#1f4a75";
    else {
      const z=G.zones.get(v);
      col = (z.owner>=0) ? G.players[z.owner].color : "#8f989f";
    }
    ctx.fillStyle=col;
    ctx.fillRect(x*CELL,y*CELL,CELL+0.4,CELL+0.4);
  }
  ctx.strokeStyle="rgba(5,12,22,0.4)"; ctx.lineWidth=1;
  for (let y=0;y<GH;y++) for (let x=0;x<GW;x++){
    const v=grid[y*GW+x];
    if (x<GW-1 && grid[y*GW+x+1]!==v){ ctx.beginPath(); ctx.moveTo((x+1)*CELL,y*CELL); ctx.lineTo((x+1)*CELL,(y+1)*CELL); ctx.stroke(); }
    if (y<GH-1 && grid[(y+1)*GW+x]!==v){ ctx.beginPath(); ctx.moveTo(x*CELL,(y+1)*CELL); ctx.lineTo((x+1)*CELL,(y+1)*CELL); ctx.stroke(); }

    drawZoneOutline(ui.origin,"#ffffff");

// ============================== UI ACTIONS ==============================
function act(it){
  if (!G || G.over) return;
  if (G.mode==="online"){
    if (G.net.isHost){
      const e=applyIntent(it);
      if (e) toast(e);
      render();
      sendState();
      return;
    }
    if (G.net.myPlayer===G.phasePlayer && socket && socket.readyState===1){
      socket.send(JSON.stringify({t:"act", d:{player:G.phasePlayer, it}}));
      toast("Sent…");
    }
    return;
  }
  const e=applyIntent(it);
  if (e) toast(e);
  render();
}
function applyIntent(it){
  const p=G.players[G.phasePlayer];
  if (!p) return "No player";
  switch(it.t){
    case "place": return doPlace(p,it.zone,it.n);
    case "trade": return doTrade(p,it.cards);
    case "useCard": return doUseCard(p,it.card);
    case "attack": return doAttack(p,it.from,it.to,it.dice);
    case "fortify": return doFortify(p,it.from,it.to,it.n);
  }
  return "Unknown action";
}

function autoAttack(from,to){
  let guard=0;
  while (guard++<60){
    if (G.over) return;
    const p=G.players[G.phasePlayer];
    const fz=G.zones.get(from), tz=G.zones.get(to);
    if (!fz||!tz||fz.owner!==p.idx||tz.owner===p.idx||fz.armies<2) return;
    act({t:"attack", from, to, dice:3});
    if (G.zones.get(to).owner===p.idx) return;
  }
}

function zoneFromEvent(e){
  const rect=boardEl.getBoundingClientRect();
  const scaleX=CW/rect.width, scaleY=CH/rect.height;
  const x=(e.clientX-rect.left)*scaleX/CELL;
  const y=(e.clientY-rect.top)*scaleY/CELL;
  const gx=Math.floor(x), gy=Math.floor(y);
  if (gx<0||gy<0||gx>=GW||gy>=GH) return null;
  const v=grid[gy*GW+gx];
  return (v===0||v===250||v>=101)?null:v;
}
boardEl.addEventListener("mousemove", e=>{
  const id=zoneFromEvent(e);
  if (id==null || !G){ el("tooltip").style.display="none"; return; }
  const z=G.zones.get(id);
  const o=z.owner>=0?G.players[z.owner]:null;
  let txt=zoneById[id].name+" — "+(o?o.name+" ("+z.armies+" armies)":"Unowned");
  const tip=el("tooltip"); tip.textContent=txt;
  tip.style.display="block";
  tip.style.left=Math.min(e.clientX+14, window.innerWidth-260)+"px";
  tip.style.top=Math.max(8, e.clientY+14)+"px";
});

});
function onZoneClick(id){
  if (!G || G.over) return;
  const p=G.players[G.phasePlayer];
  const ctrl=turnController(G.phasePlayer);
  const z=G.zones.get(id);
  if (ctrl!=="local"){ ui.selected=id; render(); return; }
  if (G.phase==="place"){
    if (z.owner===p.idx && G.toPlace>0) qtyDialog("Place armies in "+zoneById[id].name, G.toPlace, n=>act({t:"place", zone:id, n}));
    ui.selected=id; render(); return;
  }
  if (G.phase==="attack"){
    if (ui.mode==="attack" && ui.origin!=null && ui.origin!==id){
      if (adj.get(ui.origin).has(id) && z.owner!==p.idx){
        act({t:"attack", from:ui.origin, to:id, dice:3});
        return;
      }
    }
    if (z.owner===p.idx && z.armies>=2){
      ui.mode="attack"; ui.origin=id; lastOrigin=id;
    }
    ui.selected=id; render(); return;
  }
  if (G.phase==="fortify"){
    if (ui.mode==="fortify" && ui.origin!=null && ui.origin!==id){
      if (z.owner===p.idx && reachableFriendly(ui.origin).has(id)){
        const fz=G.zones.get(ui.origin);
        qtyDialog("Fortify "+zoneById[id].name, fz.armies-1, n=>act({t:"fortify", from:ui.origin, to:id, n}));
        return;
      }
    }
    if (z.owner===p.idx && z.armies>=2){
      ui.mode="fortify"; ui.origin=id;
    }
    ui.selected=id; render(); return;
  }
  ui.selected=id; render();
}

let lbSock=null, lbReady=false, lbBack=1000, lbQueue=null;
function lbConnect(){
  if(lbSock||!window.root||!window.root.createServerSocket)return;
  try{ lbSock=window.root.createServerSocket(); }catch(e){ return; }
  lbSock.addEventListener("open",function(){
    lbReady=true; lbBack=1000;
    if(lbQueue){ const q=lbQueue; lbQueue=null; lbSubmitNow(q.name,q.score,q.extra); }
  });
  lbSock.addEventListener("close",function(e){
    lbReady=false;
    if(e.code===4403)return;
    setTimeout(function(){ lbConnect(); }, Math.min(15000,lbBack)+Math.random()*500);
    lbBack=Math.min(15000,lbBack*2);
  });
}
function lbSubmitNow(name,score,extra){
  if(!lbSock||lbSock.readyState!==1)return;
  lbSock.rpc.submitScore(JSON.stringify({name:name,score:score,extra:extra})).catch(function(){});
}
function lbSubmitScore(name,score,extra){
  if(!lbSock)lbConnect();
  if(!lbReady){ lbQueue={name:name,score:score,extra:extra}; return; }
  lbSubmitNow(name,score,extra);
}
function lbGetScores(cb){
  if(!lbSock)lbConnect();

  lbSock.rpc.getScores("").then(function(r){
    let o; try{ o=JSON.parse(r); }catch(e){ cb([]); return; }
    cb(o.scores||[]);
  }).catch(function(){ cb([]); });
}

function requestRematch(){
  if (!socket || socket.readyState!==1){ toast("Not connected"); return; }
  socket.rpc.rematch(JSON.stringify({})).then(reply=>{

  closeModal();

    colorIdx:G.players.map(p=>p.colorIdx), names:G.players.map(p=>p.name),
    isHost:G.net.isHost, myPlayer:G.net.myPlayer, remote:[...G.net.remote], name:G.net.myName };

  } else {
    log("⚔️ Rematch accepted — waiting for the host…");
    setStatus("Rematch — waiting for the host…");

}
