/* ── BGN Checkers server ─────────────────────────────────────────
   Rooms: 5-letter codes. Host = RED and moves first; guest = BLACK.
   The server re-runs the full rules engine to validate every move,
   enforces turn order, detects game over / draws, and broadcasts a
   snapshot to both players on every change. Durable room state is
   kept in the plugin `state` bytes (versioned binary layout).     */
const RED=1, BLACK=2;
const EMPTY=0, M_RED=3, K_RED=4, M_BLK=5, K_BLK=6;
function hostColorToColor(h){ return h===0?RED:BLACK; }
const STATE_VERSION=1;
const SLOT=1024;
const MAX_ROOMS=Math.min(512, Math.max(16, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";
const DRAW_PLIES=60;

let rooms=new Map();            // code -> room object (live cache)
const conns=new Map();          // conn.id -> {code, role}
const moveTimes=new Map();      // conn.id -> last move ms (rate limit)
const chatTimes=new Map();      // conn.id -> last chat ms (rate limit)
const CHAT_BANNED=["nigger","nigga","faggot","kys","kill yourself","retard","heil hitler","tranny"];
function cleanText(s,max){ return String(s||"").trim().replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").slice(0,max); }
function isChatBanned(t){ const s=t.toLowerCase(); for(let i=0;i<CHAT_BANNED.length;i++){ if(s.indexOf(CHAT_BANNED[i])!==-1) return true; } return false; }

/* ── rules engine (mirrors client — keep in sync) ── */
function colorOf(p){ if(p===M_RED||p===K_RED)return RED; if(p===M_BLK||p===K_BLK)return BLACK; return EMPTY; }
function isKing(p){ return p===K_RED||p===K_BLK; }
function opp(p){ return colorOf(p)===RED?BLACK:RED; }
function kingRow(p){ return colorOf(p)===RED?7:0; }
function diag(sq,d){ const r=sq>>3,c=sq&7,tr=r+(d===7||d===9?1:-1),tc=c+((d===7||d===-9)?-1:1); if(tr<0||tr>7||tc<0||tc>7)return -1; return tr*8+tc; }
function dirsFor(p){ const col=colorOf(p),king=isKing(p),ds=[]; if(col===RED||king)ds.push(7,9); if(col===BLACK||king)ds.push(-9,-7); return ds; }
function initialBoard(){ const b=new Array(64).fill(0); for(let r=0;r<8;r++)for(let c=0;c<8;c++){ if(((r+c)&1)===0)continue; if(r<3)b[r*8+c]=M_RED; else if(r>4)b[r*8+c]=M_BLK; } return b; }
function captureChains(b,sq){
  const chains=[],pc=b[sq];
  (function rec(board,from,path){
    const piece=board[from]; let any=false;
    for(const d of dirsFor(piece)){
      const mid=diag(from,d); if(mid<0)continue;
      const land=diag(mid,d); if(land<0)continue;
      if(path.indexOf(land)!==-1)continue;
      if(colorOf(board[mid])!==opp(pc))continue;
      if(board[land]!==EMPTY)continue;
      any=true;
      const b2=board.slice();
      b2[from]=EMPTY; b2[mid]=EMPTY;
      let np=piece;
      const promo=(land>>3)===kingRow(piece)&&!isKing(piece);
      if(promo)np=colorOf(piece)===RED?K_RED:K_BLK;
      b2[land]=np;
      if(promo){ chains.push(path.concat([land])); }   // promotion ends the move (English draughts)
      else rec(b2,land,path.concat([land]));
    }
    if(!any&&path.length>=2)chains.push(path.slice());
  })(b,sq,[sq]);
  return chains;
}
function simpleMoves(b,sq){ const res=[],p=b[sq]; if(!p)return res; for(const d of dirsFor(p)){ const t=diag(sq,d); if(t>=0&&b[t]===EMPTY)res.push([sq,t]); } return res; }
function allMoves(b,col){ const caps=[],sims=[]; for(let i=0;i<64;i++)if(colorOf(b[i])===col){ const ch=captureChains(b,i); if(ch.length)for(const m of ch)caps.push(m); else for(const m of simpleMoves(b,i))sims.push(m); } return caps.length?caps:sims; }
function applyPath(b,path){ const b2=b.slice(); let from=path[0],p=b2[from]; for(let k=1;k<path.length;k++){ const to=path[k],fr=from>>3,fc=from&7,tr=to>>3,tc=to&7; if(Math.abs(tr-fr)===2){ const mr=(fr+tr)/2,mc=(fc+tc)/2; b2[mr*8+mc]=EMPTY; } b2[from]=EMPTY; if((to>>3)===kingRow(p)&&!isKing(p))p=colorOf(p)===RED?K_RED:K_BLK; b2[to]=p; from=to; } return b2; }
function boardFromMoveList(moves){ let b=initialBoard(); for(const m of moves)b=applyPath(b,decodePath(m)); return b; }
function decodePath(s){
  if(typeof s!=="string")return null;
  const parts=s.split(",");
  if(parts.length<2||parts.length>14)return null;
  const out=[];
  for(const x of parts){ if(!/^\d{1,2}$/.test(x))return null; const n=+x; if(n>63)return null; out.push(n); }
  for(let k=1;k<out.length;k++){ const a=out[k-1],b=out[k]; const ar=a>>3,ac=a&7,br=b>>3,bc=b&7; if(Math.abs(ar-br)!==Math.abs(ac-bc)||Math.abs(ar-br)===0||Math.abs(ar-br)>2)return null; }
  return out;
}
function isLegalMove(b,path){
  if(b[path[0]]===EMPTY)return false;
  const mv=allMoves(b,colorOf(b[path[0]]));
  for(const m of mv){ if(m.length!==path.length)continue; let eq=true; for(let i=0;i<path.length;i++)if(m[i]!==path[i]){eq=false;break;} if(eq)return true; }
  return false;
}
function isCapturePath(path){ const a=path[path.length-2],b=path[path.length-1]; return Math.abs((b>>3)-(a>>3))===2; }

/* ── durable state (slot layout) ──
   slot byte 0: status 0=free 1=waiting 2=playing 3=finished
   slot +1..+8: room code (8 ASCII, zero padded)
   slot +9: hostColor (0=RED)
   slot +10..+11: reserved
   slot +12..+15: uint32 lastActive (unix secs)
   slot +16..+17: uint16 moveCap (plies since last capture)
   slot +18..+19: uint16 moves bytes length
   slot +20: winner 0=none 1=red 2=black 3=draw
   slot +21..+23: reserved
   slot +24..: move bytes [len][sq...] per move                        */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readLastAct(i){ const o=off(i); return (state[o+12]<<24)|(state[o+13]<<16)|(state[o+14]<<8)|state[o+15]; }
function slotStatus(i){ return state[off(i)]; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(i){ const o=off(i); let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function encodeMoves(list){ const out=[]; for(const m of list){ const p=m.split(",").map(Number); if(p.length<2||p.length>14)continue; out.push(p.length); for(const x of p)out.push(x); } return out; }
function decodeMoves(bytes){ const out=[]; let i=0; while(i<bytes.length){ const len=bytes[i++]; if(!len)break; const s=[]; for(let k=0;k<len&&i<bytes.length;k++)s.push(bytes[i++]); out.push(s.join(",")); } return out; }
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+9]=room.hostColor;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  state[o+12]=(la>>>24)&255; state[o+13]=(la>>>16)&255; state[o+14]=(la>>>8)&255; state[o+15]=la&255;
  state[o+16]=(room.moveCap>>8)&255; state[o+17]=room.moveCap&255;
  const mb=encodeMoves(room.moves);
  const len=Math.min(SLOT-24,mb.length);
  state[o+18]=(len>>8)&255; state[o+19]=len&255;
  state[o+20]=room.winner==="red"?1:room.winner==="black"?2:room.winner==="draw"?3:0;
  if(len>0)state.set(mb.slice(0,len),o+24);
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)continue;
    const o=off(i);
    if(st===1){ state.fill(0,o,o+SLOT); continue; }   // abandoned waiting rooms are reclaimed
    const movesLen=(state[o+18]<<8)|state[o+19];
    const moves=decodeMoves(state.slice(o+24,o+24+movesLen));
    const win=state[o+20];
    rooms.set(codeAt(i),{
      slot:i, code:codeAt(i), status:st,
      hostColor:state[o+9], hostId:-1, guestId:-1,
      moves, winner:win===1?"red":win===2?"black":win===3?"draw":null, reason:null,
      moveCap:(state[o+16]<<8)|state[o+17],
      lastAct:readLastAct(i)
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const st=slotStatus(i);
    if(st===0)return i;
    const t=readLastAct(i);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── helpers ── */
function topic(code){ return "ck:"+code; }
function snap(room){ return {code:room.code,status:room.status,moves:room.moves,hostColor:room.hostColor,hostIn:room.hostId!==-1,guestIn:room.guestId!==-1,winner:room.winner,reason:room.reason}; }
function broadcast(room){ try{ pubsub.publish(topic(room.code), JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return JSON.stringify({ok:false,err:"already_in_room"});
    const slot=findSlot();
    if(slot<0)return JSON.stringify({ok:false,err:"server_full"});
    const code=genCode();
    const room={slot,code,status:1,hostColor:0,hostId:conn.id,guestId:-1,moves:[],winner:null,reason:null,moveCap:0,lastAct:nowSec()};
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host"});
    return JSON.stringify({ok:true,code,color:hostColorToColor(room.hostColor),role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return JSON.stringify({ok:false,err:"room_full"});
    if(room.status===3&&!existing)return JSON.stringify({ok:false,err:"game_over"});
    if(role==="host")room.hostId=conn.id; else room.guestId=conn.id;
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)room.status=2;
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role});
    room.lastAct=nowSec();
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true,color:hostColorToColor(role==="host"?room.hostColor:1-room.hostColor),role,snap:snap(room)});
  },
  getRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    return JSON.stringify({ok:true,snap:snap(room)});
  },
  move({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const expectedTurn=room.moves.length%2===0?"host":"guest";
    if(role!==expectedTurn)return JSON.stringify({ok:false,err:"not_your_turn"});
    const now=Date.now();
    if(now-(moveTimes.get(conn.id)||0)<300)return JSON.stringify({ok:false,err:"rate_limited"});
    moveTimes.set(conn.id,now);
    if(typeof obj.evt!=="number"||obj.evt!==room.moves.length)return JSON.stringify({ok:false,err:"stale"});
    if(typeof obj.move!=="string"||obj.move.length<1||obj.move.length>80)return JSON.stringify({ok:false,err:"bad_move"});
    const path=decodePath(obj.move);
    if(!path)return JSON.stringify({ok:false,err:"bad_move"});
    const board=boardFromMoveList(room.moves);
    const mover=colorOf(board[path[0]]);
    if(mover===EMPTY)return JSON.stringify({ok:false,err:"no_piece"});
    if(mover!==hostColorToColor(role==="host"?room.hostColor:1-room.hostColor))return JSON.stringify({ok:false,err:"not_your_piece"});
    if(!isLegalMove(board,path))return JSON.stringify({ok:false,err:"illegal"});
    room.moves.push(obj.move);
    room.moveCap=isCapturePath(path)?0:room.moveCap+1;
    room.lastAct=nowSec();
    const b2=applyPath(board,path);
    const next=mover===RED?BLACK:RED;
    if(allMoves(b2,next).length===0){
      room.status=3; room.winner=mover===RED?"red":"black"; room.reason="no_moves";
    } else if(room.moveCap>=DRAW_PLIES){
      room.status=3; room.winner="draw"; room.reason="draw";
    }
    saveRoom(room);
    broadcast(room);
    return JSON.stringify({ok:true});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"black":"red"; room.reason="resign";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  rematch({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==3)return JSON.stringify({ok:false,err:"not_finished"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(!oppPresent)return JSON.stringify({ok:false,err:"opp_gone"});
    room.status=2; room.moves=[]; room.moveCap=0; room.winner=null; room.reason=null;
    room.hostColor=1-room.hostColor;
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return JSON.stringify({ok:false,err:"not_in_room"});
    if(room.status!==2)return JSON.stringify({ok:false,err:"not_playing"});
    const role=roleOf(conn);
    const oppPresent=role==="host"?room.guestId!==-1:room.hostId!==-1;
    if(oppPresent)return JSON.stringify({ok:false,err:"opp_present"});
    room.status=3; room.winner=role==="host"?"red":"black"; room.reason="forfeit";
    room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    return JSON.stringify({ok:true});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2){
      room.status=3; room.winner=roleOf(conn)==="host"?"black":"red"; room.reason="forfeit";
      room.lastAct=nowSec(); saveRoom(room); broadcast(room);
    }
    return JSON.stringify({ok:true});
  },
  chat({conn},data){
    let obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    const room=rooms.get(String(obj.code||"").trim().toUpperCase());
    if(!room)return JSON.stringify({ok:false,err:"no_such_room"});
    const role=roleOf(conn);
    if(!role||roomOf(conn)!==room)return JSON.stringify({ok:false,err:"not_in_room"});
    const now=Date.now();
    if(now-(chatTimes.get(conn.id)||0)<700)return JSON.stringify({ok:false,err:"rate_limited"});
    chatTimes.set(conn.id,now);
    const text=cleanText(obj.text,160);
    if(!text)return JSON.stringify({ok:false,err:"empty"});
    if(isChatBanned(text))return JSON.stringify({ok:false,err:"blocked"});
    pubsub.publish(topic(room.code),JSON.stringify({t:"chat",role,text,ts:nowSec()}));
    return JSON.stringify({ok:true});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  moveTimes.delete(conn.id);
  chatTimes.delete(conn.id);
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host")room.hostId=-1; else room.guestId=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);   // opponent sees "disconnected" and may claim the win
};

loadRooms();
</script>

      if(promo)np=colorOf(piece)===RED?RED_K:BLACK_K;
      b2[land]=np;
      if(promo){ chains.push(path.concat([land])); }   // promotion ends the move (English draughts)
      else rec(b2,land,path.concat([land]));
    }
    if(!any&&path.length>=2)chains.push(path.slice());
  })(b,sq,[sq]);
  return chains;
}
function simpleMoves(b,sq){ const res=[],p=b[sq]; if(!p)return res; for(const d of dirsFor(p)){ const t=diag(sq,d); if(t>=0&&b[t]===EMPTY)res.push([sq,t]); } return res; }
function legalMoves(b,sq){ const ch=captureChains(b,sq); return ch.length?ch:simpleMoves(b,sq); }
function allMoves(b,col){ const caps=[],sims=[]; for(let i=0;i<64;i++)if(colorOf(b[i])===col){ const ch=captureChains(b,i); if(ch.length)for(const m of ch)caps.push(m); else for(const m of simpleMoves(b,i))sims.push(m); } return caps.length?caps:sims; }
function applyPath(b,path){ const b2=b.slice(); let from=path[0],p=b2[from]; for(let k=1;k<path.length;k++){ const to=path[k],fr=from>>3,fc=from&7,tr=to>>3,tc=to&7; if(Math.abs(tr-fr)===2){ const mr=(fr+tr)/2,mc=(fc+tc)/2; b2[mr*8+mc]=EMPTY; } b2[from]=EMPTY; if((to>>3)===kingRow(p)&&!isKing(p))p=colorOf(p)===RED?RED_K:BLACK_K; b2[to]=p; from=to; } return b2; }
function applyOneJump(b,from,to){
  const fr=from>>3,fc=from&7,tr=to>>3,tc=to&7;
  b[((fr+tr)/2)*8+((fc+tc)/2)]=EMPTY;
  const p=b[from];
  b[from]=EMPTY;
  let np=p, promoted=false;
  if((to>>3)===kingRow(p)&&!isKing(p)){ np=colorOf(p)===RED?RED_K:BLACK_K; promoted=true; }
  b[to]=np;
  return promoted;
}
function boardFromMoveList(moves){ let b=initialBoard(); for(const m of moves)b=applyPath(b,decodePath(m)); return b; }
function hasMoves(b,col){ return allMoves(b,col).length>0; }
function encodePath(path){ return path.join(","); }
function decodePath(s){ return s.split(",").map(Number); }
function isCapturePath(path){ const a=path[path.length-2],b=path[path.length-1]; return Math.abs((b>>3)-(a>>3))===2; }
function colorForHost(h){ return h===0?RED:BLACK; }
function evalBoard(b){
  let s=0;
  for(let i=0;i<64;i++){
    const p=b[i];
    if(p===RED)s+=100+(i>>3)*2;
    else if(p===RED_K)s+=175;
    else if(p===BLACK)s-=100+(7-(i>>3))*2;
    else if(p===BLACK_K)s-=175;
  }
  return s;
}

/* ── AI: alpha-beta search (plays BLACK) ── */
let aiBudget=0;
function search(b,depth,alpha,beta,col){
  aiBudget--;
  if(aiBudget<=0)return evalBoard(b);
  const moves=allMoves(b,col);
  if(!moves.length)return col===RED? -100000+depth : 100000-depth;
  if(depth===0)return evalBoard(b);
  moves.sort(function(a,x){return x.length-a.length;});
  if(col===RED){
    let best=-1e9;
    for(const m of moves){
      const s=search(applyPath(b,m),depth-1,alpha,beta,BLACK);
      if(s>best)best=s;
      if(s>alpha)alpha=s;
      if(beta<=alpha)break;
    }
    return best;
  }else{
    let best=1e9;
    for(const m of moves){
      const s=search(applyPath(b,m),depth-1,alpha,beta,RED);
      if(s<best)best=s;
      if(s<beta)beta=s;
      if(beta<=alpha)break;
    }
    return best;
  }
}
function aiChoose(b){
  const moves=allMoves(b,BLACK);
  if(!moves.length)return null;
  moves.sort(function(a,x){return x.length-a.length;});
  let depth=3;
  if(moves.length<=10)depth=4;
  if(moves.length<=5)depth=5;
  let best=null,bestScore=1e9;
  for(const m of moves){
    aiBudget=22000;
    const s=search(applyPath(b,m),depth-1,-1e9,1e9,RED);
    if(s<bestScore||(s===bestScore&&Math.random()<0.5)){ bestScore=s; best=m; }
  }
  return best;
}

/* ══════════ flow / turn logic ══════════ */
function isMyTurn(){
  if(game.over||game.locked)return false;
  if(game.mode==="local")return true;
  if(game.mode==="ai")return game.turn===RED&&!game.aiThinking;
  return game.sockOpen&&game.status===2&&game.turn===game.myColor;
}
function findMoveEndingAt(moves,i){
  for(const m of moves)if(m[m.length-1]===i)return m;
  return null;
}
function onCellClick(i){
  if(game.over||game.locked)return;
  if(!isMyTurn())return;
  if(game.pending){
    const last=game.pending[game.pending.length-1];
    const cont=captureChains(game.board,last);
    const landings={};
    for(const c of cont)landings[c[1]]=true;
    if(landings[i])continueCapture(i);
    return;
  }
  const b=game.board;
  if(colorOf(b[i])===game.turn){
    const mv=legalMoves(b,i);
    game.selected=-1; game.legal=[];
    if(mv.length){ game.selected=i; game.legal=mv; }
    renderBoard();
    return;
  }
  if(game.selected!==-1){
    if(game.legal.length&&game.legal[0].length>2){
      for(const m of game.legal){
        if(m[1]===i){
          game.pending=[m[0]];
          game.selected=-1; game.legal=[];
          continueCapture(m[1]);
          return;
        }
      }
    }else{
      const m=findMoveEndingAt(game.legal,i);
      if(m){ executeMove(m); return; }
    }
  }
  game.selected=-1; game.legal=[];
  renderBoard();
}
function executeMove(m){
  if(m.length===2){
    applyMoveCommit(m);
  }else{
    game.pending=[m[0]];
    game.selected=-1; game.legal=[];
    continueCapture(m[1]);
  }
}
function continueCapture(to){
  const from=game.pending[game.pending.length-1];
  const promoted=applyOneJump(game.board,from,to);
  game.pending.push(to);
  if(promoted||!captureChains(game.board,to).length){
    commitMove(game.pending);
  }else{
    renderBoard();
  }
}
function commitMove(path){
  const encoded=encodePath(path);
  game.pending=null; game.selected=-1; game.legal=[];
  if(game.mode==="online"){
    sendMove(encoded,path);
    return;
  }
  game.moves.push(encoded);
  game.lastPath=path;
  finishTurn();
}
function applyMoveCommit(m){
  if(game.mode==="online"){
    game.pending=null; game.selected=-1; game.legal=[];
    sendMove(encodePath(m),m);
    return;
  }
  game.board=applyPath(game.board,m);
  game.moves.push(encodePath(m));
  game.lastPath=m;
  game.pending=null; game.selected=-1; game.legal=[];
  finishTurn();
}
function lastCapturePos(){
  for(let i=game.moves.length-1;i>=0;i--)if(isCapturePath(decodePath(game.moves[i])))return i;
  return -1;
}
function drawTriggered(){
  const lp=lastCapturePos();
  return lp===-1?game.moves.length>=60:game.moves.length-1-lp>=60;
}
function finishTurn(){
  game.pending=null; game.selected=-1; game.legal=[];
  const mover=game.turn;
  game.turn=mover===RED?BLACK:RED;
  if(!hasMoves(game.board,game.turn)){ endGame(mover,"no_moves"); return; }
  if(drawTriggered()){ endGame("draw","draw"); return; }
  if(game.mode!=="online")sfx("turn");
  renderAll();
  if(game.mode!=="online")saveGame();
  if(game.mode==="ai"&&game.turn===BLACK&&!game.over)scheduleAI();
}
function scheduleAI(){
  game.aiThinking=true;
  updateBanner();
  syncTimer();
  setTimeout(function(){
    if(game.mode!=="ai"||game.turn!==BLACK||game.over)return;
    const m=aiChoose(game.board);
    game.aiThinking=false;
    if(game.over)return;
    if(!m){ endGame(RED,"no_moves"); return; }
    applyMoveCommit(m);
  },60);
}
function endGame(winner,reason){
  game.over=true; game.winner=winner; game.reason=reason;
  game.selected=-1; game.pending=null; game.legal=[];
  game.aiThinking=false;
  if(game.mode==="online"){
    if(winner==="draw")sfx("draw2");
    else if(winner===(game.myColor===RED?"red":"black"))sfx("win");
    else sfx("lose");
  }else{
    sfx(winner==="draw"?"draw2":"win");
  }
  if(game.mode!=="online")deleteSave();
  renderAll();
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{
    game.socket=rootRef.createServerSocket();
  }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    updateBanner();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — sharing code with a friend wins you the match":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applyServerSnap(msg.snap);
  else if(msg.t==="chat")addChat(msg.role,msg.text,msg.ts);
}
function applyServerSnap(s){
  const ms=s.moves.length;
  if(lastKnownMoveCount!==-1&&ms>lastKnownMoveCount)sfx("turn");
  lastKnownMoveCount=ms;
  game.hostColor=s.hostColor;
  game.myColor=colorForHost(game.myRole==="host"?s.hostColor:(1-s.hostColor));
  game.status=s.status;
  game.moves=s.moves.slice();
  game.board=boardFromMoveList(game.moves);
  game.turn=ms%2===0?RED:BLACK;
  game.locked=false;
  game.selected=-1; game.pending=null; game.legal=[];
  game.lastPath=ms?decodePath(s.moves[ms-1]):null;
  game.oppPresent=game.myColor===RED?s.guestIn:s.hostIn;
  updateNames();
  if(s.status===3){
    game.over=true; game.winner=s.winner; game.reason=s.reason||"";
    if(game.winner==="draw")sfx("draw2");
    else if(game.winner===(game.myColor===RED?"red":"black"))sfx("win");
    else sfx("lose");
  }else{
    game.over=false; game.winner=null; game.reason="";
  }
  renderAll();
}
function enterOnlineGame(){
  lobbyCtn.hidden=true;
  gameCtn.hidden=false;
  roomBox.hidden=false;
  chatBox.hidden=false;
  roomCodeEl.textContent=game.roomCode;
  resetMatch();
  updateNames();
  renderAll();
  updateButtons();
  sfx("deal");
}
async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=r.code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=code; game.myRole=r.role;
    setOnlineMsg("");
    if(r.snap)applyServerSnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role;
    if(r.snap)applyServerSnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
async function sendMove(encoded,path){
  game.locked=true;
  game.board=boardFromMoveList(game.moves);   // drop provisional multi-jump display
  game.selected=-1; game.pending=null; game.legal=[];
  renderAll();
  updateBanner();
  try{
    const r=await rpc("move",JSON.stringify({code:game.roomCode,evt:game.moves.length,move:encoded}));
    if(!r||!r.ok){
      game.locked=false;
      flashBanner("Move rejected ("+(r&&r.err?r.err:"error")+")");
      const s=await rpc("getRoom",game.roomCode);
      if(s&&s.snap)applyServerSnap(s.snap);
    }
  }catch(e){
    game.locked=false;
    flashBanner("Couldn't send your move — reconnecting…");
  }
}

    refreshContinuePanel();
  }
}

/* ══════════ fullscreen board (BGN house standard) ══════════
   window.bgnFullscreen — register(canvas/tile/meeple/resize) then open/close.
   The board element is MOVED (never cloned) into the overlay and restored
   to its original parent/position on close. */
let fsCanvas=null, fsTile=null, fsMeeple=null, fsResize=null;
const fsStash=[];
function isOpenFullscreen(){ return !fsOverlay.hidden; }
function fsRefit(){
  if(!isOpenFullscreen())return;
  const tb=document.querySelector(".fs-topbar");
  const h=window.innerHeight-(tb?tb.getBoundingClientRect().height:66)-24;
  document.documentElement.style.setProperty("--fs-avail-h",Math.max(80,h)+"px");
  if(fsResize)fsResize();
}
function fsRegister(o){
  if(!o||!o.canvas)return;
  fsCanvas=o.canvas;
  fsTile=o.tile||null;
  fsMeeple=o.meeple||null;
  fsResize=o.resize||null;
  fsOpenBtn.hidden=false;
}
function openFullscreen(){
  if(!fsCanvas||isOpenFullscreen())return;
  fsStash.length=0;
  const els=[fsCanvas];
  if(fsTile&&fsTile.parentNode)els.push(fsTile);
  if(fsMeeple&&fsMeeple.parentNode)els.push(fsMeeple);
  for(const el of els){
    el._fsOrig={parent:el.parentNode,next:el.nextSibling};
    fsStash.push(el);
  }
  fsOverlay.hidden=false;
  fsRefit();
  if(fsTile&&fsTileCtn)fsTileCtn.appendChild(fsTile);
  if(fsMeeple&&fsMeepleCtn)fsMeepleCtn.appendChild(fsMeeple);
  if(fsBoardWrap)fsBoardWrap.appendChild(fsCanvas);
  document.body.style.overflow="hidden";
  if(fsStatusEl)fsStatusEl.textContent=bannerEl.textContent;
  try{ if(fsOverlay.requestFullscreen)fsOverlay.requestFullscreen().catch(function(){}); }catch(e){}
}
function closeFullscreen(){
  if(!isOpenFullscreen())return;
  for(let i=fsStash.length-1;i>=0;i--){
    const el=fsStash[i], o=el._fsOrig;
    if(o&&o.parent)o.parent.insertBefore(el,o.next);
    delete el._fsOrig;
  }
  fsStash.length=0;
  fsOverlay.hidden=true;
  document.body.style.overflow="";
  if(fsResize)fsResize();
  try{ if(document.fullscreenElement)document.exitFullscreen().catch(function(){}); }catch(e){}
}
if(fsOpenBtn)fsOpenBtn.addEventListener("click",openFullscreen);
if(fsExitBtn)fsExitBtn.addEventListener("click",closeFullscreen);
window.addEventListener("resize",fsRefit);
window.bgnFullscreen={
  register: function(o){ fsRegister(o); },
  open: openFullscreen,
  close: closeFullscreen,
  isOpen: function(){ return isOpenFullscreen(); }
};
window.bgnFullscreen.register({ canvas:document.getElementById("board"), resize:renderAll });
})();
</script>
