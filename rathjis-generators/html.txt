    });
  });
  document.querySelectorAll(".nav-links a, a.btn-primary[href*='#']").forEach(a => {
    a.addEventListener("click", () => openSection(a.getAttribute("href").slice(1)));
  });

  // looping background music (autoplay needs a user gesture, so we retry on first interaction)
  const TRACKS = [
    "https://user.uploads.dev/file/91aaae36f23892054d14f8783c12a669.mp3",
    "https://user.uploads.dev/file/38486cbda72d2f2d9d2fb18453317286.mp3",
    "https://user.uploads.dev/file/c3cf1909da2ffee971758d1a89d95e0a.mp3",
    "https://user.uploads.dev/file/2a441c2c94e5decd48b2bab26cba9861.mp3",
  ];
  const music = new Audio(TRACKS[0]);
  music.loop = true;
  music.volume = 0.35;
  music.preload = "auto";
  let musicOn = false;
  const musicBtn = document.getElementById("musicBtn");
  const musicSelect = document.getElementById("musicSelect");
  function setMusicPlaying(on) {
    musicOn = on;
    musicBtn.classList.toggle("playing", on);
    musicBtn.setAttribute("aria-label", on ? "Pause music" : "Play music");
    musicBtn.title = on ? "Pause music" : "Play music";
  }
  musicBtn.addEventListener("click", () => {
    if (musicOn) { music.pause(); setMusicPlaying(false); }
    else { music.play().catch(() => {}); setMusicPlaying(true); }
  });
  musicSelect.addEventListener("change", () => {
    music.src = musicSelect.value;
    if (musicOn) music.play().catch(() => {});
  });
  document.getElementById("musicVolume").addEventListener("input", e => {
    music.volume = parseFloat(e.target.value);
  });
  function tryAutoplayMusic() {
    if (musicOn) return;
    music.play().then(() => setMusicPlaying(true)).catch(() => {});
  }
  tryAutoplayMusic();
  addEventListener("pointerdown", tryAutoplayMusic, { passive: true });
  addEventListener("keydown", tryAutoplayMusic, { passive: true });

  // reveal on scroll
  const io = new IntersectionObserver(entries => {
    for (const en of entries) if (en.isIntersecting) { en.target.classList.add("is-visible"); io.unobserve(en.target); }
  }, { threshold: 0.08 });
  document.querySelectorAll(".reveal").forEach(el => io.observe(el));

  // mouse-tracking card glow
  document.querySelectorAll(".grid").forEach(grid => {
    grid.addEventListener("mousemove", e => {
      const card = e.target.closest(".card");
      if (!card) return;
      const r = card.getBoundingClientRect();
      card.style.setProperty("--mx", (e.clientX - r.left) + "px");
    });
  });

  // scroll progress + active nav link
  const progressBar = document.getElementById("progressBar");
  const navLinks = [...document.querySelectorAll(".nav-links a")];
  const sections = [...document.querySelectorAll("section, #tutorials")];
  const secObserver = new IntersectionObserver(entries => {
    for (const en of entries) {
      if (!en.isIntersecting) continue;
      const id = en.target.id;
      navLinks.forEach(a => a.classList.toggle("active", a.getAttribute("href") === "#" + id));
    }
  }, { rootMargin: "-45% 0px -50% 0px" });
  sections.forEach(s => secObserver.observe(s));

  addEventListener("scroll", () => {
    const h = document.documentElement;
    progressBar.style.width = (h.scrollTop / (h.scrollHeight - h.clientHeight) * 100) + "%";
  }, { passive: true });
</script>
