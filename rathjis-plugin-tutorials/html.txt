  async function sendBuilderMessage(text) {
    var userText = ((text || "").trim() || chatInput.value.trim());
    if (!userText || building) return;
    chatInput.value = "";
    autosize();
    if (!root.generateText) {
      addBubble("builder", "⚠️ The AI plugin isn't loaded, so I can't answer. Check that `generateText = " + imp("ai-text-plugin") + "` is in main.pjs and reload.");
      return;
    }
    chatEmpty.style.display = "none";
    messages.push({ role: "user", text: userText });
    addBubble("user", userText);
    setBusy(true);
    showTyping();
    currentBubble = null;
    try {
      var reply = await root.generateText({
        instruction: buildPrompt("Write the next response as 'Builder'."),
        startWith: "Builder:",
        stopSequences: ["\nUser:"],
        onChunk: function(d) {
          if (!currentBubble) { hideTyping(); currentBubble = streamBubble(); }
          renderInto(currentBubble, d.fullTextSoFar.replace(/^Builder:\s*/, ""));
          chatMessages.scrollTop = chatMessages.scrollHeight;
        },
        onFinish: function(d) {
          if (!currentBubble) currentBubble = streamBubble();
          var finalText = (d.generatedText || d.text || "").replace(/^Builder:\s*/, "").trim();
          renderInto(currentBubble, finalText);
        }
      });
      hideTyping();
      var finalText = (reply.generatedText || "").trim();
      messages.push({ role: "assistant", text: finalText || currentBubble.textContent || "" });
      chatMessages.scrollTop = chatMessages.scrollHeight;
    } catch (err) {
      hideTyping();
      console.error(err);
      addBubble("builder", "⚠️ Something went wrong while generating (the AI service can occasionally fail — please try again).");
    } finally {
      setBusy(false);
      chatMessages.scrollTop = chatMessages.scrollHeight;
    }
    maybeCompact();
  }
