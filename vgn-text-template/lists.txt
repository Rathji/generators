// ════════════════════════════════════════════════════════════════
//  VGN TEXT TEMPLATE  (slug: vgn-text-template)
//  A text-based game template for the Video Game Network.
//  ────────────────────────────────────────────────────────────────
//  The ENGINE lives in index.html. This file is the GAME — edit it
//  to make your own adventure. The included sample ("THE LOST
//  CAVERN") is fully playable and demonstrates every feature.
//
//  QUICK START
//    1. Change `title`, `subtitle`, `byline` and the `titleArt` art.
//    2. Point `startScene` at the scene where your game begins.
//    3. Write your scenes under `scenes`. Each scene has:
//         title    — optional heading
//         lines    — the story text (one line per paragraph)
//         choices  — numbered options (see below)
//         to       — scenes WITHOUT choices auto-advance here
//         end      — "win" or "gameover" ends the game
//
//  A CHOICE looks like:
//      choiceName
//        text   = THE BUTTON LABEL
//        to     = nextScene          // where it leads
//        if     = has("lantern")     // optional: shown only when true
//        effect = v.gold += 5; add("key"); say("You found gold!")
//        once   = 1                  // optional: offered only once
//
//  STATE — available inside `if` / `effect` / `status` / `onEnter`:
//    v               your own variables, e.g. v.hp = 10    (persisted)
//    inv             inventory array, e.g. inv.push("sword")
//    has("x")  add("x")  remove("x")
//    say("text")      prints extra lines right after the choice
//    rand(2,12)       random integer between 2 and 12
//    win()  gameover()  end the game immediately
//
//  TIPS:
//    • `status`, `if`, `effect`, `onEnter` are JS code strings.
//      Keep them on ONE line, and don't use square brackets inside.
//    • Your story `lines` must not contain square brackets.
//    • The game auto-saves after every choice; CONTINUE resumes it.
//      ESC quits to title (saved).
// ════════════════════════════════════════════════════════════════

$meta
  title = VGN Text Template
  description = A text-based game template for the Video Game Network. Ships with a short sample adventure — edit the vgn list in main.pjs to write your own scenes, choices, inventory, variables and endings.
  image = https://user.uploads.dev/file/ebfe64214b8b6c0a69cb973ef0902cbc.jpg
  tags = text adventure, interactive fiction, template, vgn, text game, story, crt
  header
    mode = minimal

kv = {import:kv-plugin}

// ──────────────────────────────────────────────────────────────
//  THE GAME — edit everything below this line
// ──────────────────────────────────────────────────────────────

vgn
  title = THE LOST CAVERN
  subtitle = A VGN TEXT ADVENTURE
  byline = SAMPLE GAME · MADE WITH THE VGN TEXT TEMPLATE

  titleArt
    $output = [this.joinItems("\n")]
    ┌───────────────┐
    │               │
    │    ___        │
    │   /   \       │
    │  │  o  │      │
    │  │  o  │      │
    │   \___/       │
    │               │
    └───────────────┘

  startScene = start
  promptText = WHAT WILL YOU DO?

  // Runs once when a new game begins. Set your starting variables.
  init = v.hp = 5; v.gold = 3

  // Status bar shown above every scene (can be overridden per scene).
  status = "HP " + v.hp + " · GOLD " + v.gold + (inv.length ? " · ITEMS: " + inv.join(", ") : "")

  // Text shown by the HOW TO PLAY button on the title screen.
  help
    $output = [this.joinItems("\n")]
    The Lost Cavern is a short sample game built into the VGN
    Text Template. Each scene describes your surroundings, then
    lists your options with numbers.
    Press 1-9 to take an action directly, or use up/down to
    highlight a choice and press Enter / Space.
    On the gamepad: A/B/X/Y = choices 1-4, Start = Enter,
    Select = quit to title (the game autosaves).
    Your progress is saved after every choice. Use Continue on
    the title screen to pick up where you left off.
    To make this your own game, edit the vgn list in main.pjs.

  scenes
    start
      title = THE CAVE MOUTH
      lines
        $output = [this.joinItems("\n")]
        You wake to cold air and the smell of damp stone. Before you, the mouth of a great cavern gapes in the hillside.
        A winding trail leads down toward a distant village.
      choices
        enter
          text = ENTER THE CAVERN
          to = cavern
        checkPack
          text = CHECK YOUR PACK
          to = pack
          once = 1
        climb
          text = CLIMB THE RIDGE
          to = ridge

    pack
      title = YOUR PACK
      lines
        $output = [this.joinItems("\n")]
        You dig through your pack and find a battered oil lantern, and two gold coins rattling in a sock.
      choices
        take
          text = TAKE THE LANTERN AND COINS
          to = start
          effect = add("lantern"); v.gold += 2; say("You sling the lantern on your belt and pocket the coins.")
        leave
          text = LEAVE IT FOR NOW
          to = start

    ridge
      title = THE RIDGE
      lines
        $output = [this.joinItems("\n")]
        From the ridge you can see the whole valley. The village is a day's walk away, smoke curling from its chimneys. Directly below, the dark mouth of the cavern.
      choices
        back
          text = GO BACK DOWN
          to = start

    cavern
      title = THE CAVERN
      lines
        $output = [this.joinItems("\n")]
        The cavern is cold and damp. Beyond the entrance, the blackness swallows every scrap of daylight.
      choices
        light
          text = LIGHT THE LANTERN
          to = litCavern
          if = has("lantern")
        dark
          text = STUMBLE INTO THE DARK
          to = darkFall
          if = !has("lantern")
        out
          text = GO BACK OUTSIDE
          to = start

    darkFall
      title = THE DARK
      lines
        $output = [this.joinItems("\n")]
        You take three blind steps into the black, then trip over a root and tumble down a rocky slope before crawling back toward the dim daylight.
      onEnter = v.hp -= 1; say("You lose 1 HP.")
      to = start

    litCavern
      title = THE LANTERN CHAMBER
      lines
        $output = [this.joinItems("\n")]
        The lantern throws long shadows across a vast chamber. On the far wall stands an iron gate, locked fast. Hanging from a rusted hook beside it: a key.
      choices
        takeKey
          text = TAKE THE RUSTY KEY
          to = litCavern
          once = 1
          effect = add("key"); say("You pocket the key. It is cold and heavy.")
        tryGate
          text = TRY THE GATE
          to = gate
          if = has("key")
        examine
          text = EXAMINE THE WALLS
          effect = say("Nothing but damp rock — and one promising keyhole.")
        goOut
          text = GO BACK OUTSIDE
          to = start

    gate
      title = THE IRON GATE
      lines
        $output = [this.joinItems("\n")]
        The key turns with a groan. Beyond the gate, sprawled across a heap of gold, sleeps an enormous troll.
      choices
        creep
          text = CREEP PAST THE TROLL
          to = treasure
          once = 1
          effect = say("The troll grunts in its sleep but does not wake.")
        poke
          text = POKE THE TROLL
          to = troll
        back
          text = GO BACK
          to = litCavern

    troll
      title = THE TROLL
      lines
        $output = [this.joinItems("\n")]
        The troll wakes with a roar that shakes the walls. The last thing you see is a club the size of a tree trunk swinging toward your head.
      end = gameover

    treasure
      title = THE TREASURE
      lines
        $output = [this.joinItems("\n")]
        Gold. More gold than you ever imagined. You fill your pack until the leather strains, then back out the way you came.
      choices
        run
          text = RUN FOR THE VILLAGE
          to = village
          effect = v.gold += 50; say("Your pack clinks with fortune as you run for daylight.")

    village
      title = THE VILLAGE
      lines
        $output = [this.joinItems("\n")]
        The villagers gather as you stagger into the square, spilling gold across the cobbles. Cheers go up — no one has walked out of the Lost Cavern alive, let alone rich.
      end = win
