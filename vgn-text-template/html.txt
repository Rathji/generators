<script>
(function(){
  "use strict";

  /* ── VGN helper API (games can use this) ── */
  var VGN = {
    keymap:{ up:"ArrowUp", down:"ArrowDown", left:"ArrowLeft", right:"ArrowRight",
             a:"1", b:"2", x:"3", y:"4", start:"Enter", select:"Escape" },
    setKeymap:function(map){ for(var k in map) this.keymap[k]=map[k]; },
    press:function(action){ dispatchKey(this.keymap[action], true);  },
    release:function(action){ dispatchKey(this.keymap[action], false); },
    hints:function(text){ hintsEl.textContent = text || ""; }
  };
  window.VGN = VGN;

  function keyCode(k){
    if(k===" ") return "Space";
    if(k.length===1) return "Key"+k.toUpperCase();
    return k;
  }
  function dispatchKey(key, down){
    if(!key) return;
    document.dispatchEvent(new KeyboardEvent(down?"keydown":"keyup", {key:key, code:keyCode(key), bubbles:true}));
    reflectPad(key, down);
  }

  /* ── toggles ── */
  var hintsEl = document.getElementById("hintsEl");
  var gamepad = document.getElementById("gamepad");
  var controlsBtn = document.getElementById("controlsBtn");
  var bezelBtn = document.getElementById("bezelBtn");
  var fullBtn = document.getElementById("fullBtn");
  var controlsOn = false;

  controlsBtn.addEventListener("click", function(){
    controlsOn = !controlsOn;
    controlsBtn.classList.toggle("on", controlsOn);
    controlsBtn.setAttribute("aria-pressed", String(controlsOn));
    gamepad.classList.toggle("on", controlsOn);
    hintsEl.classList.toggle("on", controlsOn);
  });
  bezelBtn.addEventListener("click", function(){
    var bare = document.body.classList.toggle("bare");
    bezelBtn.classList.toggle("on", bare);
    bezelBtn.setAttribute("aria-pressed", String(bare));
    window.dispatchEvent(new Event("resize"));
  });

  /* ── native browser fullscreen ── */
  var prevBare = false;
  function isFS(){
    return !!(document.fullscreenElement || document.webkitFullscreenElement);
  }
  function syncFullBtn(){
    var fs = isFS();
    fullBtn.classList.toggle("on", fs);
    fullBtn.setAttribute("aria-pressed", String(fs));
    if(!fs) document.body.classList.toggle("bare", prevBare);
  }
  document.addEventListener("fullscreenchange", syncFullBtn);
  document.addEventListener("webkitfullscreenchange", syncFullBtn);
  fullBtn.addEventListener("click", function(){
    if(!isFS()){
      prevBare = document.body.classList.contains("bare");
      document.body.classList.add("bare");
      var el = document.documentElement;
      var p = null;
      if(el.requestFullscreen) p = el.requestFullscreen();
      else if(el.webkitRequestFullscreen) p = el.webkitRequestFullscreen();
      if(p && p.catch) p.catch(function(){
        document.body.classList.toggle("bare", prevBare);
        syncFullBtn();
      });
      window.dispatchEvent(new Event("resize"));
    } else {
      if(document.exitFullscreen) document.exitFullscreen();
      else if(document.webkitExitFullscreen) document.webkitExitFullscreen();
    }
  });

  /* ── gamepad input → real keyboard events ── */
  function wire(btn){
    var action = btn.getAttribute("data-key");
    btn.addEventListener("pointerdown", function(e){
      e.preventDefault(); e.stopPropagation();
      btn.classList.add("on");
      VGN.press(action);
      try{ btn.setPointerCapture(e.pointerId); }catch(_){}
    });
    function off(e){
      if(!btn.classList.contains("on")) return;
      btn.classList.remove("on");
      VGN.release(action);
    }
    btn.addEventListener("pointerup", off);
    btn.addEventListener("pointercancel", off);
    btn.addEventListener("pointerleave", off);
  }
  document.querySelectorAll(".d,.pbtn,.pmin").forEach(wire);

  /* ── real keyboard presses light up the gamepad ── */
  function reflectPad(key, down){
    var action = null;
    for(var k in VGN.keymap){ if(VGN.keymap[k]===key) action=k; }
    if(!action) return;
    var el = document.querySelector('[data-key="'+action+'"]');
    if(el) el.classList.toggle("on", down);
  }
  window.addEventListener("keyup", function(e){ reflectPad(e.key, false); });

  /* ── default hint text ── */
  VGN.hints("1-9 / A B X Y = CHOOSE · ↑↓ + ENTER = SELECT · ESC = TITLE · SPACE = CONTINUE");
})();
</script>

<script>
"use strict";

/* ════════════════════════════════════════════════════════════════
   VGN TEXT TEMPLATE ENGINE
   Reads the game from root.vgn (defined in main.pjs) and runs it:
   scenes → choices → state → save/continue → endings.

   Scene config (see main.pjs for the full authoring guide):
     title, lines, choices, to, end, onEnter, status (per scene)
   Choice config:
     text, to, if, effect, once
   Author JS (in `if`/`effect`/`status`/`onEnter`) runs with the
   player state in scope:  v, inv, has(), add(), remove(), say(),
   rand(), flag(), unflag(), win(), gameover().
   ════════════════════════════════════════════════════════════════ */

var C = null;            // loaded game definition
var S = null;            // live game state
var muted = false;
var choiceState = null;  // active choice input { list, sel, resolve }
var waitingKey = null, waitingRes = null;

var out, promptLine, keypadEl, muteBtn, continueBtn, titleCtn, titleCred, endOut, endBtns;

function $(id){ return document.getElementById(id); }
function print(text, cls){
  var d = document.createElement("div");
  d.className = "l" + (cls ? " " + cls : "");
  d.textContent = (text === undefined || text === null) ? "" : String(text);
  out.appendChild(d);
  scrollDown();
  return d;
}
function printLines(t){
  var arr = Array.isArray(t) ? t : String(t).split("\n");
  arr.forEach(function(l){ print(l); });
}
function scrollDown(){ if(out) out.scrollTop = out.scrollHeight; }
function err(e){
  console.error(e);
  print("⚠ SCRIPT ERROR: " + (e && e.message ? e.message : e), "bad");
}

/* ==================================================================

var sfx = {
  ctx:null,
  init:function(){
    if (!this.ctx){
      try { this.ctx = new (window.AudioContext || window.webkitAudioContext)(); } catch(e){}
    }
    if (this.ctx && this.ctx.state === "suspended"){ this.ctx.resume(); }
  },
  on:function(){ return !muted; },
  tone:function(freq, dur, type, vol){
    if (!this.on()) return;
    this.init();
    if (!this.ctx) return;
    var o = this.ctx.createOscillator(), g = this.ctx.createGain();
    o.type = type || "square";
    o.frequency.value = freq;
    g.gain.value = vol || 0.07;
    o.connect(g); g.connect(this.ctx.destination);
    var t0 = this.ctx.currentTime;
    o.start(t0);
    g.gain.exponentialRampToValueAtTime(0.001, t0 + dur);
    o.stop(t0 + dur);
  },
  blip:function(){ this.tone(880, 0.07, "square", 0.05); },
  type:function(){ this.tone(1200, 0.03, "square", 0.03); },
  bad:function(){ this.tone(140, 0.4, "sawtooth", 0.07); },
  ding:function(){ this.tone(1040, 0.14, "square", 0.06); },
  ching:function(){ this.tone(1568, 0.09, "square", 0.06); this.tone(1175, 0.12, "square", 0.06); this.tone(880, 0.18, "square", 0.05); },
  win:function(){ this.tone(523,0.15,"square",0.07); this.tone(659,0.15,"square",0.07); this.tone(784,0.15,"square",0.07); this.tone(1047,0.4,"square",0.08); }
};
function soundToggle(){
  muted = !muted;
  muteBtn.textContent = "SOUND: " + (muted ? "OFF" : "ON");
  if (!muted) sfx.ding();
}

/* ==================================================================
   2. PLAYER STATE & AUTHOR-JS EVALUATION
   ================================================================== */
function makeCtx(){
  return {
    v: S.v,
    inv: S.inv,
    add: function(item){ if(S.inv.indexOf(item) < 0) S.inv.push(item); },
    remove: function(item){ S.inv = S.inv.filter(function(x){ return x !== item; }); },
    has: function(item){ return S.inv.indexOf(item) >= 0; },
    say: function(text){ printLines(text); },
    rand: function(a, b){ return a + Math.floor(Math.random() * (b - a + 1)); },
    flag: function(n){ S.v[n] = true; },
    unflag: function(n){ delete S.v[n]; },
    win: function(){ finishGame("win"); },
    gameover: function(){ finishGame("gameover"); }
  };
}
function evalCode(code, ctx, isExpr){
  if(code === null || code === undefined) return undefined;
  code = String(code).trim();
  if(!code) return undefined;
  var fn;
  try {
    fn = new Function("s", isExpr ? "with(s){ return (" + code + "); }" : "with(s){ " + code + "; }");
  } catch(e){
    err(new Error("bad JS: " + code.slice(0, 60) + " — " + e.message));
    return undefined;
  }
  try {
    return fn(ctx || makeCtx());
  } catch(e){
    err(e);
    return undefined;
  }
}

/* ==================================================================
   3. LOAD THE GAME DEFINITION FROM main.pjs
   ================================================================== */
function val(node){
  if(node === null || node === undefined) return "";
  try {
    var v = node.evaluateItem;
    return (v === null || v === undefined) ? "" : v;
  } catch(e){ return ""; }
}
function loadGame(){
  var v = root.vgn;
  if(!v) return null;
  var game = {
    title: val(v.title),
    subtitle: val(v.subtitle),
    byline: val(v.byline),
    titleArt: val(v.titleArt),
    startScene: val(v.startScene) || "start",
    promptText: val(v.promptText) || "WHAT WILL YOU DO?",
    init: val(v.init),
    status: val(v.status),
    help: val(v.help),
    scenes: {}
  };
  if(v.scenes && v.scenes.selectAll){
    v.scenes.selectAll.forEach(function(sc){
      var name = sc.$text;
      var scene = {
        name: name,
        title: val(sc.title),
        lines: val(sc.lines),
        status: val(sc.status),
        onEnter: val(sc.onEnter),
        to: val(sc.to),
        end: val(sc.end),
        choices: []
      };
      if(sc.choices && sc.choices.selectAll){
        sc.choices.selectAll.forEach(function(ch){
          scene.choices.push({
            key: ch.$text,
            text: val(ch.text),
            to: val(ch.to),
            ifExpr: val(ch.if),
            effect: val(ch.effect),
            once: !!(parseFloat(val(ch.once)) || val(ch.once) === "true")
          });
        });
      }
      game.scenes[name] = scene;
    });
  }
  return game;
}
function statusText(sc){
  var expr = (sc && typeof sc.status === "string" && sc.status) ? sc.status : (C.status || "");
  if(!expr) return "";
  var t = evalCode(expr, makeCtx(), true);
  return (t === null || t === undefined) ? "" : String(t);
}
function onceKey(c){ return S.scene + "::" + (c.key || c.text); }
function visibleChoices(sc){
  if(!sc.choices || !sc.choices.length) return [];
  return sc.choices.filter(function(c){
    if(c.once && S.used[onceKey(c)]) return false;
    if(c.ifExpr){
      var ok = evalCode(c.ifExpr, makeCtx(), true);
      if(!ok) return false;
    }
    return true;
  });
}

/* ==================================================================
   4. SCREENS
   ================================================================== */
function showScreen(name){
  $("titleScreen").hidden = name !== "title";
  $("gameScreen").hidden = name !== "game";
  $("endScreen").hidden = name !== "end";
  if(name !== "game"){ promptLine.textContent = ""; keypadEl.hidden = true; }
}

/* ==================================================================
   5. THE GAME LOOP
   ================================================================== */
async function renderScene(name){
  S.scene = name;
  out.innerHTML = "";
  keypadEl.hidden = true;
  var sc = C.scenes[name];
  if(!sc){
    err(new Error("No scene named \"" + name + "\". Scenes: " + Object.keys(C.scenes).join(", ")));
    return;
  }
  var st = statusText(sc);
  if(st) print(st, "status dim");
  if(sc.title) print(sc.title, "hero good");
  if(sc.onEnter) evalCode(sc.onEnter, makeCtx(), false);
  if(sc.lines) printLines(sc.lines);
  print("");

  var choices = visibleChoices(sc);
  if(choices.length){
    print(C.promptText, "hero");
    choices.forEach(function(c, i){ print("  " + (i + 1) + "  " + c.text); });
    buildKeypad(choices.length);
    var picked = await pickChoice(choices);
    if(picked === "quit") return;
    await applyChoice(picked);
  } else if(sc.end){
    await finishGame(sc.end);
  } else {
    var k = await pause("PRESS SPACE TO CONTINUE");
    if(k === "Escape"){ quitToTitle(); return; }
    if(sc.to) renderScene(sc.to);
    else renderScene(name);
  }
}

async function applyChoice(c){
  print("> " + c.text, "good");
  S.moves = (S.moves || 0) + 1;
  if(c.once) S.used[onceKey(c)] = true;
  evalCode(c.effect, makeCtx(), false);
  saveGame();
  if(c.to) await renderScene(c.to);
  else await renderScene(S.scene);
}

async function newGame(){
  sfx.blip();
  S = {
    scene: C.startScene,
    v: {}, inv: [], used: {}, moves: 0, phase: "game"
  };
  evalCode(C.init, makeCtx(), false);
  clearSave();
  showScreen("game");
  out = $("out");
  await renderScene(S.scene);
}

function continueGame(){
  root.kv.vgnText.get("save").then(function(saved){
    if(!saved) return;
    var d;
    try { d = JSON.parse(saved); } catch(e){ return; }
    S = {
      scene: d.scene || C.startScene,
      v: d.v || {}, inv: d.inv || [], used: d.used || {}, moves: d.moves || 0, phase: "game"
    };
    showScreen("game");
    out = $("out");
    out.innerHTML = "";
    print("WELCOME BACK, ADVENTURER.", "hero good");
    print("YOUR ADVENTURE CONTINUES.", "dim");
    print("");
    renderScene(S.scene);
  });
}

async function finishGame(kind){
  S.phase = "end";
  clearSave();
  showScreen("end");
  out = endOut;
  out.innerHTML = "";
  var sc = C.scenes[S.scene] || {};
  if(sc.title) print(sc.title, "hero good");
  print("");
  print(kind === "win" ? "★★★★★  YOU WIN  ★★★★★" : "· · ·  GAME OVER  · · ·", kind === "win" ? "big hero good" : "big bad");
  print("");
  if(sc.lines) printLines(sc.lines);
  print("");
  print("MOVES TAKEN: " + (S.moves || 0));
  var st = statusText(sc);
  if(st) print(st, "status dim");
  print("");
  if(C.byline) print(C.byline, "dim");
  if(kind === "win") sfx.win(); else sfx.bad();
  endBtns.style.display = "";
  scrollDown();
}

function playAgain(){ newGame(); }
function backToTitle(){ quitToTitle(); }
function quitToTitle(){
  if(choiceState){
    var r = choiceState.resolve;
    choiceState = null;
    promptLine.textContent = "";
    keypadEl.hidden = true;
    if(r) r("quit");
  }
  if(waitingKey){
    var r2 = waitingRes;
    waitingKey = null; waitingRes = null;
    if(r2) r2("Escape");
  }
  if(S && S.phase !== "end") saveGame();
  S = null;
  showScreen("title");
  renderTitle();
  refreshContinueBtn();
}

/* ==================================================================
   6. CHOICE INPUT
   ================================================================== */
function pickChoice(choices){
  return new Promise(function(res){
    choiceState = { list: choices, sel: 1, resolve: res };
    renderChoicePrompt();
    sfx.blip();
  });
}
function renderChoicePrompt(){
  promptLine.textContent = "> " + choiceState.sel;
}
function confirmChoice(n){
  if(!choiceState) return;
  if(n === undefined) n = choiceState.sel;
  if(n < 1 || n > choiceState.list.length){ sfx.bad(); return; }
  var c = choiceState.list[n - 1];
  var r = choiceState.resolve;
  choiceState = null;
  promptLine.textContent = "";
  keypadEl.hidden = true;
  sfx.ding();
  r(c);
}
function handleChoiceKey(e){
  var k = e.key;
  if(/^[1-9]$/.test(k)){
    e.preventDefault();
    var n = parseInt(k, 10);
    if(n <= choiceState.list.length) confirmChoice(n);
    else sfx.bad();
  } else if(k === "ArrowUp" || k === "ArrowDown"){
    e.preventDefault();
    choiceState.sel = Math.max(1, Math.min(choiceState.list.length, choiceState.sel + (k === "ArrowUp" ? -1 : 1)));
    sfx.blip();
    renderChoicePrompt();
  } else if(k === "Enter" || k === " "){
    e.preventDefault();
    confirmChoice();
  } else if(k === "Escape"){
    e.preventDefault();
    quitToTitle();
  }
}

/* ==================================================================
   7. KEYPAD (touch) + GLOBAL KEYS
   ================================================================== */
function buildKeypad(count){
  keypadEl.innerHTML = "";
  for(var i = 1; i <= count; i++){
    (function(x){
      var b = document.createElement("button");
      b.textContent = String(x);
      b.onclick = function(){ keyTap(x); };
      keypadEl.appendChild(b);
    })(i);
  }
  var en = document.createElement("button");
  en.className = "kok primary";
  en.textContent = "ENTER";
  en.onclick = keyOk;
  keypadEl.appendChild(en);
  keypadEl.hidden = false;
}
function keyTap(n){ if(choiceState) confirmChoice(n); }
function keyOk(){ if(choiceState) confirmChoice(); }

function pause(text){
  print(text || "PRESS SPACE TO CONTINUE", "waitHint");
  return new Promise(function(res){
    waitingKey = [" ", "Enter", "Escape"];
    waitingRes = res;
  });
}
function waitForKey(keys){
  return new Promise(function(res){
    waitingKey = keys; waitingRes = res;
  });
}

document.addEventListener("keydown", function(e){
  if(e.target && (e.target.tagName === "INPUT" || e.target.tagName === "TEXTAREA")) return;
  if(choiceState){ handleChoiceKey(e); return; }
  if(waitingKey){
    if(waitingKey.indexOf(e.key) >= 0){
      e.preventDefault();
      var k = waitingKey, r = waitingRes;
      waitingKey = null; waitingRes = null;
      r(e.key);
    } else if(e.key === "Escape" && waitingKey.indexOf("Escape") < 0){
      e.preventDefault();
      var r2 = waitingRes;
      waitingKey = null; waitingRes = null;
      r2("Escape");
    }
    return;
  }
  if(!S){
    if(e.key === "Enter" || e.key === " "){
      e.preventDefault();
      if(!continueBtn.hidden) continueGame();
      else newGame();
    }
  } else if(S.phase === "end"){
    if(e.key === "Enter" || e.key === " "){
      e.preventDefault();
      playAgain();
    }
  }
});

/* ==================================================================
   8. SAVE / CONTINUE
   ================================================================== */
function saveGame(){
  if(!S || S.phase === "end") return;
  root.kv.vgnText.set("save", JSON.stringify({
    scene: S.scene, v: S.v, inv: S.inv, used: S.used, moves: S.moves
  })).catch(function(){});
}
function clearSave(){
  try { root.kv.vgnText.delete("save"); } catch(e){}
}
function refreshContinueBtn(){
  root.kv.vgnText.get("save").then(function(s){
    continueBtn.hidden = !s;
  }).catch(function(){ continueBtn.hidden = true; });
}
