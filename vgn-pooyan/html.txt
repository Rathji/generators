<!-- ═══════════ ONLINE LEADERBOARD (server-plugin) ═══════════
     PUBLIC SOURCE (authoritative, but anyone can read it — no
     secrets here). Durable `state` layout:
       0        version guard (42)
       1..      N=60 entries × ENTRY=32 bytes:
                  name   (24 utf8 bytes, zero-padded)
                  score  (u32 BE)
                  ts     (u32 LE, unix seconds)
     Sorted best-first; the top 10 are shown. Also tracks players
     online via the "lobby" topic (subscriber count).
     RPC:  get() → {ok, board(top-10), n(online)}
           submit({name, score}) → {ok, rank(1-based | -1)}
     Rate limits: 1 per 5s per connection, 5 per 10min per net[2].
     ═══════════════════════════════════════════════════════════ -->
<script type="text/x-server-plugin">
"use strict";
var N = 60, NAME = 24, ENTRY = 32, BASE = 1;
if (state[0] !== 42) { state.fill(0, 0, state.length); state[0] = 42; }

function r32(o){ return (state[o]|state[o+1]<<8|state[o+2]<<16|state[o+3]<<24)>>>0; }
function u32(o,v){ state[o]=v&255; state[o+1]=(v>>>8)&255; state[o+2]=(v>>>16)&255; state[o+3]=(v>>>24)&255; }
function putName(dst,s){
  s=String(s||"").replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").trim().slice(0,NAME);
  state.fill(0,dst,dst+NAME);
  for(var i=0;i<s.length;i++) state[dst+i]=s.charCodeAt(i);
}
function getName(o){ var s="",i=o,e=o+NAME,c; while(i<e){ c=state[i++]; if(!c)break; s+=String.fromCharCode(c); } return s; }
function getEntry(i){ var o=BASE+i*ENTRY; return { name:getName(o), score:r32(o+NAME), ts:r32(o+NAME+4) }; }
function saveEntry(i,name,score,ts){ var o=BASE+i*ENTRY; putName(o,name); u32(o+NAME,score); u32(o+NAME+4,ts); }
function top10(){ var a=[]; for(var i=0;i<N;i++){ var e=getEntry(i); if(e.score>0) a.push(e); if(a.length>=10) break; } return a; }
function nowSec(){ return Math.floor(Date.now()/1000); }

var submitTimes=new Map(), netLog=new Map();
function canSubmit(conn){
  var n=nowSec(), k="n"+conn.net[2], arr=netLog.get(k)||[];
  arr=arr.filter(function(t){ return t>n-600; });
  if(arr.length>=5){ netLog.set(k,arr); return false; }
  arr.push(n); netLog.set(k,arr);
  return true;
}
function onlineCount(){ try{ return pubsub.numSubscribers("lobby"); }catch(e){ return 0; } }
function publishCount(){ try{ pubsub.publish("lobby", "{\"t\":\"n\",\"n\":"+onlineCount()+"}"); }catch(e){} }

self.onopen = function({conn}){
  try{ conn.subscribe("lb:pub"); }catch(e){}
  try{ conn.subscribe("lobby"); }catch(e){}
  publishCount();
};
self.onclose = function(){ publishCount(); };

self.rpc = {
  get: function(){
    return JSON.stringify({ok:true, board:top10(), n:onlineCount()});
  },
  submit: function({conn}, data){
    var obj; try{ obj=JSON.parse(data); }catch(e){ return JSON.stringify({ok:false,err:"bad_request"}); }
    var name=String(obj.name||"").replace(/[\u0000-\u001f\u007f]/g,"").replace(/\s+/g," ").trim().slice(0,NAME);
    var score=Math.floor(Number(obj.score));
    if(!name) return JSON.stringify({ok:false,err:"name"});
    if(!isFinite(score)||score<=0||score>4294967295) return JSON.stringify({ok:false,err:"score"});
    var now=Date.now();
    if(now-(submitTimes.get(conn.id)||0)<5000) return JSON.stringify({ok:false,err:"rate_limited"});
    submitTimes.set(conn.id,now);
    if(!canSubmit(conn)) return JSON.stringify({ok:false,err:"rate_limited"});
    var idx=-1;
    for(var i=0;i<N;i++){ var e=getEntry(i); if(e.score===0){ idx=i; break; } if(score>e.score){ idx=i; break; } }
    if(idx===-1) return JSON.stringify({ok:true,rank:-1});
    for(var j=N-1;j>idx;j--){ var s=getEntry(j-1); saveEntry(j,s.name,s.score,s.ts); }
    saveEntry(idx,name,score,nowSec());
    try{ pubsub.publish("lb:pub", JSON.stringify({t:"lb",board:top10(),newEntry:{rank:idx+1,name:name,score:score}})); }catch(e){}
    return JSON.stringify({ok:true,rank:idx+1});
  }
};
</script>

    function tone(f0, f1, dur, type, vol, delay){
      if(!sfxOn) return; var a = ac(); if(!a) return;
      var t0 = a.currentTime + (delay||0);
      var o = a.createOscillator(), g = a.createGain();
      o.type = type || "square";
      o.frequency.setValueAtTime(f0, t0);
      if(f1 && f1 !== f0) o.frequency.exponentialRampToValueAtTime(f1, t0+dur);
      g.gain.setValueAtTime(vol||0.04, t0);
      g.gain.exponentialRampToValueAtTime(0.0001, t0+dur);
      o.connect(g); g.connect(a.destination);
      o.start(t0); o.stop(t0+dur+0.02);
    }
    function noise(dur, vol, delay){
      if(!sfxOn) return; var a = ac(); if(!a) return;
      var t0 = a.currentTime + (delay||0);
      var n = Math.max(1, Math.floor(a.sampleRate*dur));
      var buf = a.createBuffer(1, n, a.sampleRate);
      var d = buf.getChannelData(0);
      for(var i=0;i<n;i++) d[i] = (Math.random()*2-1)*(1-i/n);
      var src = a.createBufferSource(); src.buffer = buf;
      var f = a.createBiquadFilter(); f.type = "lowpass"; f.frequency.value = 900;
      var g = a.createGain(); g.gain.value = vol||0.06;
      src.connect(f); f.connect(g); g.connect(a.destination);
      src.start(t0);
    }

    /* ── update ── */
    function update(dt){
      time += dt;
      var i;

      if(paused) return;

      if(state === "intro"){
        introTimer -= dt;
        if(introTimer <= 0) state = "playing";
        updateFx(dt);
        return;
      }
      if(state === "bonusIntro"){
        bonusIntroTimer -= dt;
        if(bonusIntroTimer <= 0){ buildBonus(); state = "bonus"; bonusTimer = CONFIG.bonusTime; }
        updateFx(dt);
        return;
      }
      if(state === "roundclear"){
        clearTimer -= dt;
        if(clearTimer <= 0){ state = "bonusIntro"; bonusIntroTimer = 1.5; }
        updateFx(dt);
        return;
      }
      if(state === "roundfail"){
        clearTimer -= dt;
        if(clearTimer <= 0) loseLife();
        updateFx(dt);
        return;
      }
      if(state === "dying"){
        dieTimer -= dt;
        if(dieTimer <= 0) loseLife();
        updateFx(dt);
        return;
      }
      if(state !== "playing" && state !== "bonus") return;

      /* stars drift */
      for(i=0;i<stars.length;i++){
        var st = stars[i];
        st.y += (22 + st.s*14)*dt;
        if(st.y > H+4){ st.y = -4; st.x = Math.random()*W; }
      }

      /* elevator move */
      var mv = 0;
      if(keys["ArrowUp"]||keys["w"]||keys["W"]) mv -= 1;
      if(keys["ArrowDown"]||keys["s"]||keys["S"]) mv += 1;
      if(touch.active){
        var d = touch.y - pig.y;
        if(Math.abs(d) > 4) mv = d > 0 ? 1 : -1;
        if(Math.abs(d) < 6) mv = 0;
      }
      pig.y += mv * CONFIG.playerSpeed * dt;
      pig.y = Math.max(ELEV_TOP, Math.min(ELEV_BOT, pig.y));
      pig.invuln = Math.max(0, pig.invuln - dt);
      pig.fireL -= dt; pig.fireR -= dt;

      /* fire */
      var fireL = !!(keys[" "] || keys["a"] || keys["A"] || keys["z"] || keys["Z"] || keys["ArrowLeft"] || keys["c"] || keys["C"]);
      var fireR = !!(keys["x"] || keys["X"] || keys["d"] || keys["D"] || keys["ArrowRight"] || keys["y"] || keys["Y"]);
      if(touch.active){ if(touch.x < W/2) fireL = true; else fireR = true; }
      if(fireL && pig.fireL <= 0){ fireArrow(-1); pig.fireL = CONFIG.fireCooldown; }
      if(fireR && pig.fireR <= 0){ fireArrow(1); pig.fireR = CONFIG.fireCooldown; }

      /* arrows */
      for(i=arrows.length-1;i>=0;i--){
        var a = arrows[i];
        a.vy += CONFIG.arrowGravity*dt;
        a.x += a.vx*dt; a.y += a.vy*dt;
        if(a.x < -20 || a.x > W+20 || a.y < -30 || a.y > H+20){ arrows.splice(i,1); continue; }
        var hit = false;
        if(state === "playing"){
          for(var wi=0;wi<wolves.length;wi++){
            var w = wolves[wi];
            if(w.fallen) continue;
            if(w.climb){
              if(a.x > w.x-15 && a.x < w.x+15 && a.y > w.y-14 && a.y < w.y+14){ shootClimber(w); hit = true; break; }
              continue;
            }
            if(w.exit) continue;
            var dxx = a.x - w.x, dyy = a.y - (w.y-30);
            if(dxx*dxx + dyy*dyy < (CONFIG.balloonR+5)*(CONFIG.balloonR+5) ||
               (a.x > w.x-15 && a.x < w.x+15 && a.y > w.y-14 && a.y < w.y+14)){
              popWolf(w); hit = true; break;
            }
          }
        } else if(state === "bonus"){
          for(var bi=0;bi<bonusWolves.length;bi++){
            var bw = bonusWolves[bi];
            if(bw.fallen || bw.respawn > 0) continue;
            var dxx = a.x - bw.x, dyy = a.y - (bw.y-30);
            if(dxx*dxx + dyy*dyy < (CONFIG.balloonR+5)*(CONFIG.balloonR+5) ||
               (a.x > bw.x-15 && a.x < bw.x+15 && a.y > bw.y-14 && a.y < bw.y+14)){
              popBonusWolf(bw); hit = true; break;
            }
          }
        }
        if(hit) arrows.splice(i,1);
      }

      if(state === "playing"){
        updateWolves(dt);
        updateFalling(dt);
        updateRocks(dt);
        updateMeat(dt);
        if(piglets.length && resolvedCount() >= piglets.length) resolveRound();
      }
      if(state === "bonus") updateBonus(dt);

      updateFx(dt);
    }

    function updateWolves(dt){
      if(piglets.length && resolvedCount() < piglets.length){
        spawnTimer -= dt;
        if(spawnTimer <= 0){ spawnTimer = spawnInterval(); spawnWolf(); }
      }
      for(var i=wolves.length-1;i>=0;i--){
        var w = wolves[i];
        if(w.fallen){
          w.vy += 300*dt; w.y += w.vy*dt; w.x += w.vx*dt; w.rot += 3*dt;
          if(w.y > H+40){ wolves.splice(i,1); continue; }
          continue;
        }
        if(w.climb){
          if(w.pause > 0){ w.pause -= dt; continue; }
          w.y -= w.vy*dt;
          if(w.y <= ELEV_TOP - 6){ wolves.splice(i,1); continue; }
          if(pig.invuln <= 0 && Math.abs(w.y - pig.y) < 12){ killPlayer(); return; }
          continue;
        }
        if(w.exit) continue;
        w.x += Math.sin(time*w.freq + w.ph)*w.amp*dt;
        w.y += w.vy*dt;
        if(w.y >= BOTTOM_Y){
          if(w.carry){ w.carry.state = "lost"; w.carry = null; popup(CAVE_X, H-78, "-1 PIGLET", "#ff8a7a"); SFX.lost(); shake = Math.max(shake, 3); }
          w.climb = true; w.pause = 1.0;
          w.x = LADDER_X; w.vy = CONFIG.climbSpeed;
          continue;
        }
      }
    }

    function shootClimber(w){
      w.fallen = true; w.climb = false;
      w.vy = CONFIG.fallSpeed; w.vx = rnd(-30, 30); w.rot = rnd(2, 5);
      addScore(50);
      popup(w.x, w.y - 20, "+50", "#ffd319");
      explode(w.x, w.y - 10, WOLF_COL, 12);
      SFX.pop(); SFX.squeal();
    }

    function updateRocks(dt){
      var list = [];
      for(var i=0;i<wolves.length;i++){ var w=wolves[i]; if(!w.fallen && !w.exit && !w.climb) list.push(w); }
      if(list.length){
        rockTimer -= dt;
        if(rockTimer <= 0){
          rockTimer = Math.max(CONFIG.rockTimerMin, CONFIG.rockTimerBase - (round-1)*0.4) + Math.random()*2;
          var w = list[Math.floor(Math.random()*list.length)];
          var tx = TRACK_X, ty = pig.y - 10;
          var dx = tx - w.x, dy = ty - (w.y - 30);
          var T = Math.max(0.25, Math.sqrt(dx*dx + dy*dy) / CONFIG.rockSpeed);
          var g = CONFIG.rockGravity;
          var jx = (Math.random()*2-1)*40, jy = (Math.random()*2-1)*40;
          rocks.push({ x:w.x, y:w.y-30, vx:dx/T + jx, vy:dy/T - 0.5*g*T + jy, r:6 });
        }
      }
      for(i=rocks.length-1;i>=0;i--){
        var r = rocks[i];
        r.vy += CONFIG.rockGravity*dt;
        r.x += r.vx*dt; r.y += r.vy*dt;
        if(r.x < -20 || r.x > W+20 || r.y < -40 || r.y > H+20){ rocks.splice(i,1); continue; }
        if(pig.invuln <= 0 && Math.abs(r.x - TRACK_X) < 18 && r.y > pig.y - 16 && r.y < pig.y + 16){
          rocks.splice(i,1);
          killPlayer();
          return;
        }
      }
    }

    function updateMeat(dt){
      if(meat.timer > 0){
        meat.timer -= dt;
        if(meat.timer <= 0){ meat.active = true; meat.x = TRACK_X; meat.y = ELEV_TOP + 16; }
      }
      if(meat.active && Math.abs(pig.y - (meat.y - 8)) < 26){
        meat.active = false;
        hasMeat = true;
        meat.timer = CONFIG.meatEvery;
        popup(TRACK_X + 40, meat.y - 20, "MEAT!", "#ffd319");
        SFX.extra();
      }
      if(meatThrow){
        meatThrow.x += meatThrow.vx*dt;
        meatThrow.y += meatThrow.vy*dt;
        if(meatThrow.x < -20 || meatThrow.x > W+20 || meatThrow.y < -30 || meatThrow.y > H+20) meatThrow = null;
      }
    }

    function updateBonus(dt){
      bonusTimer -= dt;
      if(bonusTimer <= 0){
        popup(W/2, H/2 - 60, "BONUS COMPLETE", "#ffd319");
        round++;
        buildRound();
        state = "intro"; introTimer = 1.8;
        SFX.clear();
        return;
      }
      for(var i=bonusWolves.length-1;i>=0;i--){
        var bw = bonusWolves[i];
        if(bw.respawn > 0){ bw.respawn -= dt; continue; }
        if(bw.fallen){
          bw.vy += 300*dt; bw.y += bw.vy*dt; bw.x += bw.vx*dt; bw.rot += 3*dt;
          if(bw.y > H+30){ bw.fallen = false; bw.respawn = 0.7; bw.y = -24; bw.x = rnd(50, W-50); bw.vy = bonusWolfSpeed(); }
          continue;
        }
        bw.x += Math.sin(time*bw.freq + bw.ph)*bw.amp*dt;
        bw.y += bw.vy*dt;
        if(bw.y > H+8){ bw.y = -24; bw.x = rnd(50, W-50); bw.vy = bonusWolfSpeed(); }
      }
    }

    function updateFalling(dt){
      for(var i=falling.length-1;i>=0;i--){
        var p = falling[i];
        p.t += dt;
        if(p.state === "caught"){
          p.x = TRACK_X + 24;
          p.y = pig.y - 16 - Math.abs(Math.sin(p.t*9))*9;
          if(p.t > 1.0){ falling.splice(i,1); continue; }
          continue;
        }
        if(p.state === "ground"){
          p.y = GROUND_Y - 12 - Math.abs(Math.sin(p.t*9))*7;
          if(p.t > 0.7){ falling.splice(i,1); continue; }
          continue;
        }
        p.vy += 90*dt;
        p.y += p.vy*dt;
        p.x += p.vx*dt + Math.sin(time*2.5 + p.swayPh)*16*dt;
        if(Math.abs(p.x - TRACK_X) < 36 && p.y > pig.y - 20 && p.y < pig.y + 2 && pig.invuln <= 0){
          p.state = "caught"; p.t = 0;
          addScore(300 + Math.min(60*(round-1), 300));
          popup(TRACK_X + 30, pig.y - 40, "SAVED! +"+(300 + Math.min(60*(round-1), 300)), "#3ddc84");
          SFX.rescue();
        } else if(p.y >= GROUND_Y - 10){
          p.state = "ground"; p.t = 0;
          addScore(100);
          popup(p.x, GROUND_Y - 46, "SAFE +100", "#3ddc84");
          SFX.oink();
        }
      }
    }

      drawShake();

      else if(state === "over"){ drawOver(); }

        if(state === "roundfail") drawIntroOverlay("ROUND FAILED", "#ff5a3c", "SAVE MORE PIGLETS!");
        if(paused) drawPaused();
      }
      ctx.restore();
    }

    /* ── main loop (fixed timestep + watchdog) ── */
    var FIXED = 1000/60;
    var acc = 0, lastT = 0, __steps = 0, __ticks = 0;
    function tick(t){
      var now = (typeof t === "number") ? t : performance.now();
      if(lastT === 0) lastT = now;
      var elapsed = now - lastT;
      lastT = now;
      if(elapsed > 250) elapsed = 250;
      acc += elapsed;
      var steps = 0;
      while(acc >= FIXED && steps < 6){
        update(FIXED/1000);
        acc -= FIXED;
        steps++;
      }
      if(steps === 6) acc = 0;
      __steps += steps; __ticks++;
      draw(now);
      requestAnimationFrame(tick);
    }
    setInterval(function(){ tick(); }, 100);

    requestAnimationFrame(tick);

    /* public hooks for builders + testing */
    window.SHT = {
      info: function(){
        return { state:state, score:score, hiScore:hiScore, lives:lives, round:round, required:required,

                 wolves:wolves.length, arrows:arrows.length, falling:falling.length,
                 rocks:rocks.length, rockTimer:Math.round(rockTimer*10)/10,
                 hasMeat:hasMeat, meatActive:meat.active,
                 bonusWolves:bonusWolves.length, bonusTimer:Math.round(bonusTimer*10)/10,
                 pigY:Math.round(pig.y), spawnTimer:Math.round(spawnTimer*10)/10,
                 ticks:__ticks, steps:__steps, acc:Math.round(acc*10)/10,
                 net:{ connected:Net.connected, online:Net.online, ready:Net.ready,
                       top:Net.board ? Net.board.slice(0,3) : null } };
      },
      netRefresh: function(){ Net.refresh(); },
      netSubmit: function(name, score){ return Net.submit(name, score); },
      qualifies: function(s){ return Net.qualifies(s); },
      startRun: startRun,
      setTwoPlayer: function(b){ twoPlayer = !!b; },
      setPigY: function(y){ pig.y = Math.max(ELEV_TOP, Math.min(ELEV_BOT, y)); },
      holdKey: function(k, ms){
        keys[k] = true;
        setTimeout(function(){ keys[k] = false; }, ms||120);
      },
      spawnWolf: spawnWolf,
      spawnClimber: function(y){
        wolves.push({ x:LADDER_X, y:y, vy:CONFIG.climbSpeed, climb:true, pause:0, fallen:false, exit:false, carry:null, rot:0, amp:0, freq:0, ph:0 });
      },
      throwRock: function(){
        rocks.push({ x:400, y:300, vx:-160, vy:100, r:6 });
      },
      giveMeat: function(){ hasMeat = true; },
      spawnAt: function(x, y, vy, carry){
        var tree = piglets.filter(function(p){ return p.state === "tree"; });
        var p = (carry && tree.length) ? tree[Math.floor(Math.random()*tree.length)] : null;
        if(p){ p.state = "carried"; }
        var w = { x:x, y:y, vy:vy||60, amp:16, freq:1.5, ph:Math.random()*6.283, color:randBalloon(),
          pop:false, fallen:false, exit:false, carry:p, rot:0 };
        wolves.push(w);
        return p ? 1 : 0;
      },
      wolvesInfo: function(){
        return wolves.map(function(w){ return { x:Math.round(w.x), y:Math.round(w.y), carry:!!w.carry, fallen:w.fallen, exit:w.exit, pop:w.pop }; });
      },
      popAll: function(){
        for(var i=0;i<wolves.length;i++){ var w=wolves[i]; if(!w.fallen && !w.climb && !w.exit) popWolf(w); }
      },
      skipIntro: skipIntro,
      skipEntry: function(){ state = "over"; },
      forceGameOver: function(){
        if(state === "playing"){ score = 5000; lives = -1; resolveRound(); }
      }
    };
  })();

})();
