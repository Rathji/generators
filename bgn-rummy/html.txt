/* ── BGN RUMMY server ─────────────────────────────────────────
   Rooms: 5-letter codes. Host = player 0, guest = player 1.
   The server deals the 52-card deck itself (Math.random), keeps
   both hands private in its durable state, validates every draw /
   discard / knock, computes layoff + scoring, and broadcasts a
   public snapshot on every change. Hands travel to players only
   via directed messages (conn.send) or RPC replies.             */
const TARGET=100;      // match target
const R_KNOCK=10;      // knock with deadwood at or below this
const SLOT=256;
const MAX_ROOMS=Math.min(4096, Math.max(32, Math.floor((state.length-1)/SLOT)));
const CODE_CHARS="ABCDEFGHJKMNPQRSTUVWXYZ23456789";

let rooms=new Map();   // code -> room object (live cache)
const conns=new Map(); // conn.id -> {code, role}

/* ── card helpers ── */
function rankOf(id){ return id%13; }
function suitOf(id){ return Math.floor(id/13); }
function cardVal(id){ const r=rankOf(id); return r===0?1:(r>=10?10:r+1); }

/* ── meld engine (mirrors client — keep in sync) ── */
function listMelds(hand){
  const melds=[];
  const byRank=new Map();
  for(const id of hand){ const r=rankOf(id); if(!byRank.has(r))byRank.set(r,[]); byRank.get(r).push(id); }
  for(const g of byRank.values()){ if(g.length>=3)melds.push(g.slice()); }
  const bySuit=new Map();
  for(const id of hand){ const s=suitOf(id); if(!bySuit.has(s))bySuit.set(s,[]); bySuit.get(s).push(id); }
  for(const g of bySuit.values()){
    const ranks=g.map(rankOf).sort(function(a,b){return a-b;});
    let i=0;
    while(i<ranks.length){
      let j=i;
      while(j+1<ranks.length&&ranks[j+1]===ranks[j]+1)j++;
      if(j-i+1>=3){
        const block=new Set(ranks.slice(i,j+1));
        melds.push(g.filter(function(id){return block.has(rankOf(id));}));
      }
      i=j+1;
    }
  }
  return melds;
}
function bestPartition(hand){
  const melds=listMelds(hand);
  const used=new Array(melds.length).fill(false);
  let bestMelds=[],bestDead=1e9,bestCount=-1;
  function tryFrom(k){
    const usedCards=new Set();
    let dead=0,mcount=0;
    for(let m=0;m<melds.length;m++){ if(used[m]){ mcount++; for(const c of melds[m])usedCards.add(c); } }
    for(const id of hand)if(!usedCards.has(id))dead+=cardVal(id);
    if(dead<bestDead||(dead===bestDead&&mcount>bestCount)){
      bestDead=dead; bestCount=mcount; bestMelds=[];
      for(let m=0;m<melds.length;m++)if(used[m])bestMelds.push(melds[m].slice());
    }
    for(let m=k;m<melds.length;m++){
      if(used[m])continue;
      let ok=true;
      for(const c of melds[m])if(usedCards.has(c)){ ok=false; break; }
      if(ok){ used[m]=true; tryFrom(m+1); used[m]=false; }
    }
  }
  tryFrom(0);
  const inBest=new Set();
  for(const m of bestMelds)for(const c of m)inBest.add(c);
  return {melds:bestMelds, dead:hand.filter(function(id){return !inBest.has(id);}), deadVal:bestDead};
}
function fitsMeld(card,meld){
  if(!meld.length)return false;
  const r0=rankOf(meld[0]);
  let allSameRank=true, allSameSuit=true;
  for(const m of meld){ if(rankOf(m)!==r0)allSameRank=false; if(suitOf(m)!==suitOf(meld[0]))allSameSuit=false; }
  if(allSameRank)return rankOf(card)===r0;
  if(allSameSuit){
    if(suitOf(card)!==suitOf(meld[0]))return false;
    let mn=11,mx=0;
    for(const m of meld){ const r=rankOf(m); if(r<mn)mn=r; if(r>mx)mx=r; }
    const cr=rankOf(card);
    return cr===mn-1||cr===mx+1;
  }
  return false;
}
function applyLayoff(dead,melds){
  const out=dead.slice();
  let changed=true;
  while(changed){
    changed=false;
    for(let i=0;i<out.length;i++){
      let fits=false;
      for(const m of melds){ if(fitsMeld(out[i],m)){ fits=true; break; } }
      if(fits){ out.splice(i,1); changed=true; break; }
    }
  }
  return out;
}
function deadValOf(dead){ let v=0; for(const c of dead)v+=cardVal(c); return v; }
function scoreHand(kIdx,dIdx,hK,hD){
  const pk=bestPartition(hK);
  const isGin=pk.deadVal===0;
  const pd=bestPartition(hD);
  if(isGin){
    return {kind:"gin",winner:kIdx,points:pd.deadVal+25,kDead:0,dDead:pd.deadVal,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
  }
  const laid=applyLayoff(pd.dead,pk.melds);
  const dDead=deadValOf(laid);
  if(pk.deadVal<dDead){
    return {kind:"knock",winner:kIdx,points:dDead-pk.deadVal,kDead:pk.deadVal,dDead:dDead,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
  }
  return {kind:"undercut",winner:dIdx,points:(pk.deadVal-dDead)+25,kDead:pk.deadVal,dDead:dDead,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
}

/* ── durable state (slot layout) ──
   byte 0: status 0=free 1=waiting 2=playing 3=finished
   1-8: room code (8 ASCII, zero padded)
   9: reserved
   10: handNo | 11: dealer | 12: turn (255 none) | 13: phase
   14: lastDrawn (255 none) | 15: handState 0=none 1=playing 2=handover 3=matchover
   16-19: lastActive u32 | 20-21: score0 u16 | 22-23: score1 u16
   24: handWinner (255 none) | 25: matchWinner (255 none) | 26: kind
   27: reserved
   28: hand0Count | 29-39: hand0 (11 bytes) | 40: hand1Count | 41-51: hand1 (11 bytes)
   52: stockCount | 53-83: stock (31 bytes, [0]=top)
   84: discardCount | 85-116: discard (32 bytes, last=top)
   117-118: points0 u16 | 119-120: points1 u16 | 121: kDead | 122: dDead
   123-132: revealed hand0 | 133-142: revealed hand1 | 143: knocker (255 none)
   kind: 0 none 1 gin 2 knock 3 undercut 4 stockdraw                    */
function off(i){ return 1+i*SLOT; }
function nowSec(){ return Math.floor(Date.now()/1000); }
function readU32(o){ return (state[o]<<24)|(state[o+1]<<16)|(state[o+2]<<8)|state[o+3]; }
function writeU32(o,v){ state[o]=(v>>>24)&255; state[o+1]=(v>>>16)&255; state[o+2]=(v>>>8)&255; state[o+3]=v&255; }
function readU16(o){ return (state[o]<<8)|state[o+1]; }
function writeU16(o,v){ state[o]=(v>>8)&255; state[o+1]=v&255; }
function writeCode(o,code){ for(let k=0;k<8;k++)state[o+1+k]=k<code.length?code.charCodeAt(k):0; }
function codeAt(o){ let s=""; for(let k=0;k<8;k++){ const ch=state[o+1+k]; if(!ch)break; s+=String.fromCharCode(ch); } return s; }
function writeCards(o,cards){
  const n=Math.min(cards.length,11);
  state[o]=n;
  for(let k=0;k<11;k++)state[o+1+k]=k<n?cards[k]:255;
}
function readCards(o,limit){
  const n=Math.min(state[o]||0,limit||11);
  const out=[];
  for(let k=0;k<n;k++)out.push(state[o+1+k]);
  return out;
}
function saveRoom(room){
  const o=off(room.slot);
  state.fill(0,o,o+SLOT);
  state[o]=room.status;
  writeCode(o,room.code);
  state[o+10]=room.handNo;
  state[o+11]=room.dealer;
  state[o+12]=room.turn<0?255:room.turn;
  state[o+13]=room.phase;
  state[o+14]=room.lastDrawn<0?255:room.lastDrawn;
  state[o+15]=room.handState;
  const la=Math.max(0,Math.min(0xFFFFFFFF,room.lastAct));
  writeU32(o+16,la);
  writeU16(o+20,room.scores[0]);
  writeU16(o+22,room.scores[1]);
  state[o+24]=room.handWinner<0?255:room.handWinner;
  state[o+25]=room.matchWinner<0?255:room.matchWinner;
  state[o+26]=room.kind;
  writeCards(o+28,room.hands[0]);
  writeCards(o+40,room.hands[1]);
  const sc=Math.min(room.stock.length,31);
  state[o+52]=sc;
  for(let k=0;k<sc;k++)state[o+53+k]=room.stock[k];
  const dc=Math.min(room.discard.length,32);
  state[o+84]=dc;
  for(let k=0;k<dc;k++)state[o+85+k]=room.discard[k];
  writeU16(o+117,room.points[0]);
  writeU16(o+119,room.points[1]);
  state[o+121]=room.kDead<0?255:room.kDead;
  state[o+122]=room.dDead<0?255:room.dDead;
  state[o+143]=room.knocker<0?255:room.knocker;
  if(room.reveal){
    const a=room.reveal.hands[0],b=room.reveal.hands[1];
    for(let k=0;k<10;k++)state[o+123+k]=k<a.length?a[k]:255;
    for(let k=0;k<10;k++)state[o+133+k]=k<b.length?b[k]:255;
  }else{
    for(let k=0;k<10;k++){ state[o+123+k]=255; state[o+133+k]=255; }
  }
}
function loadRooms(){
  rooms=new Map();
  for(let i=0;i<MAX_ROOMS;i++){
    const o=off(i);
    if(state[o]===0)continue;
    if(state[o]===1){ state.fill(0,o,o+SLOT); continue; }
    const turn=state[o+12]===255?-1:state[o+12];
    const lastDrawn=state[o+14]===255?-1:state[o+14];
    const handWinner=state[o+24]===255?-1:state[o+24];
    const matchWinner=state[o+25]===255?-1:state[o+25];
    const sc=state[o+52]||0, dc=state[o+84]||0;
    const stock=[],discard=[];
    for(let k=0;k<sc;k++)stock.push(state[o+53+k]);
    for(let k=0;k<dc;k++)discard.push(state[o+85+k]);
    let reveal=null;
    const knocker=state[o+143]===255?-1:state[o+143];
    if(state[o+26]>=1&&state[o+26]<=3){
      const h0=[],h1=[];
      for(let k=0;k<10;k++){ if(state[o+123+k]!==255)h0.push(state[o+123+k]); }
      for(let k=0;k<10;k++){ if(state[o+133+k]!==255)h1.push(state[o+133+k]); }
      const kind=state[o+26]===1?"gin":state[o+26]===2?"knock":"undercut";
      const pK=knocker>=0?bestPartition(knocker===0?h0:h1):null;
      const pD=knocker>=0?bestPartition(knocker===0?h1:h0):null;
      reveal={hands:[h0,h1],kMelds:pK?pK.melds:[],dMelds:pD?pD.melds:[],
        kDead:state[o+121]===255?-1:state[o+121],
        dDead:state[o+122]===255?-1:state[o+122],
        kind:kind,winner:handWinner,
        points:(handWinner===0?readU16(o+117):readU16(o+119)),
        knocker:knocker};
    }
    rooms.set(codeAt(o),{
      slot:i, code:codeAt(o), status:state[o],
      hostId:-1, guestId:-1, hostConn:null, guestConn:null,
      scores:[readU16(o+20),readU16(o+22)],
      handNo:state[o+10], dealer:state[o+11], turn:turn, phase:state[o+13],
      lastDrawn:lastDrawn, handState:state[o+15],
      handWinner:handWinner, matchWinner:matchWinner, kind:state[o+26],
      points:[readU16(o+117),readU16(o+119)],
      kDead:state[o+121]===255?-1:state[o+121],
      dDead:state[o+122]===255?-1:state[o+122],
      knocker:knocker,
      hands:[readCards(o+28),readCards(o+40)],
      stock:stock, discard:discard, reveal:reveal,
      lastAct:readU32(o+16)
    });
  }
}
function findSlot(){
  let free=-1,bFin=-1,bWait=-1,bAny=-1,bFinT=0,bWaitT=0,bAnyT=0;
  for(let i=0;i<MAX_ROOMS;i++){
    const o=off(i),st=state[o];
    if(st===0)return i;
    const t=readU32(o+16);
    if(st===3){ if(bFin===-1||t<bFinT){bFin=i;bFinT=t;} }
    else if(st===1){ if(bWait===-1||t<bWaitT){bWait=i;bWaitT=t;} }
    else if(st===2){ if(bAny===-1||t<bAnyT){bAny=i;bAnyT=t;} }
  }
  if(bFin!==-1)return bFin;
  if(bWait!==-1)return bWait;
  return bAny;
}
function genCode(){
  for(let t=0;t<60;t++){
    let s="";
    for(let k=0;k<5;k++)s+=CODE_CHARS[Math.floor(Math.random()*CODE_CHARS.length)];
    if(!rooms.has(s))return s;
  }
  return "AAAAA";
}

/* ── helpers ── */
function topic(code){ return "rm:"+code; }
function roleOf(conn){ const i=conns.get(conn.id); return i?i.role:null; }
function roomOf(conn){ const i=conns.get(conn.id); return i?rooms.get(i.code):null; }
function idxOf(conn){ return roleOf(conn)==="host"?0:1; }
function oppPresent(room,idx){ return idx===0?room.guestId!==-1:room.hostId!==-1; }
function connOf(room,idx){ return idx===0?room.hostConn:room.guestConn; }
function snap(room){
  return {code:room.code,status:room.status,handNo:room.handNo,dealer:room.dealer,
    turn:room.turn,phase:room.phase,stockCount:room.stock.length,
    discard:room.discard.slice(),handCounts:[room.hands[0].length,room.hands[1].length],
    scores:[room.scores[0],room.scores[1]],handState:room.handState,
    handWinner:room.handWinner,matchWinner:room.matchWinner,kind:room.kind,
    points:[room.points[0],room.points[1]],
    kDead:room.kDead,dDead:room.dDead,
    hostIn:room.hostId!==-1,guestIn:room.guestId!==-1,
    reveal:room.reveal||null};
}
function broadcast(room){ try{ pubsub.publish(topic(room.code),JSON.stringify({t:"snap",snap:snap(room)})); }catch(e){} }
function sendHand(room,idx){
  const c=connOf(room,idx);
  if(c)try{ c.send(JSON.stringify({t:"hand",hand:room.hands[idx].slice(),handNo:room.handNo,turn:room.turn,phase:room.phase})); }catch(e){}
}
function shuffledDeck(){
  const a=[];
  for(let i=0;i<52;i++)a.push(i);
  for(let i=51;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); const t=a[i]; a[i]=a[j]; a[j]=t; }
  return a;
}
function startHand(room){
  const d=shuffledDeck();
  room.hands=[d.slice(0,10),d.slice(10,20)];
  room.stock=d.slice(20,51);
  room.discard=[d[51]];
  room.handNo++;
  room.turn=1-room.dealer;
  room.phase=0;
  room.lastDrawn=-1;
  room.handState=1;
  room.handWinner=-1;
  room.kind=0;
  room.points=[0,0];
  room.kDead=-1; room.dDead=-1; room.knocker=-1;
  room.reveal=null;
  room.lastAct=nowSec();
  saveRoom(room);
  sendHand(room,0); sendHand(room,1);
  broadcast(room);
}
function endHandDraw(room){
  room.handState=2;
  room.kind=4;
  room.handWinner=-1;
  room.points=[0,0];
  room.turn=-1;
  room.reveal=null;
  room.knocker=-1;
  room.kDead=-1; room.dDead=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);
}
function endMatch(room,winner){
  room.handState=3;
  room.matchWinner=winner;
  room.handWinner=winner;
  room.kind=0;
  room.turn=-1;
  room.knocker=-1;
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);
}
function E(err){ return JSON.stringify({ok:false,err:err}); }
function OK(obj){ return JSON.stringify(Object.assign({ok:true},obj||{})); }

/* ── rate limiting (light) ── */
const createTimes=new Map(); // net3 -> timestamp

self.rpc={
  createRoom({conn},data){
    if(conns.has(conn.id))return E("already_in_room");
    const n3=conn.net[3];
    const t=Date.now();
    if((createTimes.get(n3)||0)+2000>t)return E("slow_down");
    createTimes.set(n3,t);
    const slot=findSlot();
    if(slot<0)return E("server_full");
    const code=genCode();
    const room={slot,code,status:1,hostId:conn.id,guestId:-1,hostConn:conn,guestConn:null,
      scores:[0,0],handNo:0,dealer:0,turn:-1,phase:0,lastDrawn:-1,handState:0,
      handWinner:-1,matchWinner:-1,kind:0,points:[0,0],kDead:-1,dDead:-1,knocker:-1,
      hands:[[],[]],stock:[],discard:[],reveal:null,lastAct:nowSec()};
    rooms.set(code,room);
    saveRoom(room);
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role:"host"});
    return OK({code,role:"host",snap:snap(room)});
  },
  joinRoom({conn},data){
    const code=String(data||"").trim().toUpperCase();
    const room=rooms.get(code);
    if(!room)return E("no_such_room");
    if(room.status===3&&!conns.has(conn.id))return E("game_over");
    const existing=conns.get(conn.id);
    let role=null;
    if(existing&&existing.code===code)role=existing.role;
    else if(room.hostId===-1)role="host";
    else if(room.guestId===-1)role="guest";
    else return E("room_full");
    if(role==="host"){ room.hostId=conn.id; room.hostConn=conn; }
    else { room.guestId=conn.id; room.guestConn=conn; }
    conn.subscribe(topic(code));
    conns.set(conn.id,{code,role});
    room.lastAct=nowSec();
    if(room.status===1&&room.hostId!==-1&&room.guestId!==-1)room.status=2;
    saveRoom(room);
    if(room.status===2&&room.handState===0)startHand(room);
    else if(room.handState===1)sendHand(room,role==="host"?0:1);
    broadcast(room);
    return OK({role,snap:snap(room),hand:room.handState===1?room.hands[role==="host"?0:1].slice():null});
  },
  getRoom({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    return OK({snap:snap(room),hand:room.handState===1?room.hands[idxOf(conn)].slice():null});
  },
  draw({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status!==2)return E("not_playing");
    if(room.handState!==1)return E("no_hand");
    if(room.turn!==idxOf(conn))return E("not_your_turn");
    if(room.phase!==0)return E("bad_phase");
    let obj; try{ obj=JSON.parse(data); }catch(e){ return E("bad_request"); }
    const idx=idxOf(conn);
    if(obj.t==="discard"){
      if(!room.discard.length)return E("empty_pile");
      const c=room.discard.pop();
      room.hands[idx].push(c);
      room.phase=2; room.lastDrawn=c;
      room.lastAct=nowSec();
      saveRoom(room); broadcast(room);
      return OK({card:c,phase:2});
    }
    if(!room.stock.length)return E("empty_stock");
    const c=room.stock.shift();
    room.hands[idx].push(c);
    room.phase=1; room.lastDrawn=-1;
    room.lastAct=nowSec();
    saveRoom(room); broadcast(room);
    return OK({card:c,phase:1});
  },
  discard({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status!==2)return E("not_playing");
    if(room.handState!==1)return E("no_hand");
    if(room.turn!==idxOf(conn))return E("not_your_turn");
    if(room.phase!==1&&room.phase!==2)return E("bad_phase");
    let obj; try{ obj=JSON.parse(data); }catch(e){ return E("bad_request"); }
    const c=obj.card;
    if(typeof c!=="number"||c<0||c>51)return E("bad_card");
    if(room.phase===2&&c===room.lastDrawn)return E("same_card");
    const idx=idxOf(conn);
    const p=room.hands[idx].indexOf(c);
    if(p===-1)return E("not_in_hand");
    room.hands[idx].splice(p,1);
    room.discard.push(c);
    room.lastDrawn=-1;
    room.lastAct=nowSec();
    if(room.stock.length<=2){ saveRoom(room); endHandDraw(room); return OK({}); }
    room.turn=1-idx;
    room.phase=0;
    saveRoom(room); broadcast(room);
    return OK({});
  },
  knock({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status!==2)return E("not_playing");
    if(room.handState!==1)return E("no_hand");
    if(room.turn!==idxOf(conn))return E("not_your_turn");
    if(room.phase!==1&&room.phase!==2)return E("bad_phase");
    const idx=idxOf(conn);
    const pk=bestPartition(room.hands[idx]);
    if(pk.deadVal>R_KNOCK)return E("too_much_deadwood");
    const res=scoreHand(idx,1-idx,room.hands[idx],room.hands[1-idx]);
    room.handWinner=res.winner;
    room.kind=res.kind==="gin"?1:res.kind==="knock"?2:3;
    room.points=[res.winner===0?res.points:0,res.winner===1?res.points:0];
    room.scores[res.winner]+=res.points;
    room.kDead=res.kDead; room.dDead=res.dDead; room.knocker=idx;
    room.reveal={hands:[room.hands[0].slice(),room.hands[1].slice()],kMelds:res.kMelds,dMelds:res.dMelds,
      kDead:res.kDead,dDead:res.dDead,kind:res.kind,winner:res.winner,points:res.points,knocker:idx};
    room.turn=-1;
    if(room.scores[res.winner]>=TARGET){ room.handState=3; room.matchWinner=res.winner; }
    else room.handState=2;
    room.lastAct=nowSec();
    saveRoom(room); broadcast(room);
    return OK({result:{kind:res.kind,winner:res.winner,points:res.points,kDead:res.kDead,dDead:res.dDead}});
  },
  nextHand({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status!==2)return E("not_playing");
    if(room.handState!==2&&room.handState!==3)return E("no_result");
    if(!oppPresent(room,idxOf(conn)))return E("opp_gone");
    if(room.handState===3){
      room.scores=[0,0];
      room.handNo=0;
      room.dealer=1-room.dealer;
      room.matchWinner=-1;
      room.handWinner=-1;
    }else if(room.handWinner>=0){
      room.dealer=room.handWinner;
    }else{
      room.dealer=1-room.dealer;
    }
    startHand(room);
    return OK({});
  },
  claimWin({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status!==2)return E("not_playing");
    const idx=idxOf(conn);
    if(oppPresent(room,idx))return E("opp_present");
    if(room.handState===0)return E("no_hand");
    endMatch(room,idx);
    return OK({});
  },
  resign({conn},data){
    const room=roomOf(conn);
    if(!room)return E("not_in_room");
    if(room.status===2&&room.handState>=1){
      endMatch(room,1-idxOf(conn));
    }
    return OK({});
  },
  leave({conn},data){
    const room=roomOf(conn);
    if(room&&room.status===2&&room.handState>=1){
      endMatch(room,1-idxOf(conn));
    }
    return OK({});
  }
};

self.onopen=({conn})=>{};
self.onclose=({conn})=>{
  const info=conns.get(conn.id);
  if(!info)return;
  conns.delete(conn.id);
  const room=rooms.get(info.code);
  if(!room)return;
  if(info.role==="host"){ room.hostId=-1; room.hostConn=null; }
  else { room.guestId=-1; room.guestConn=null; }
  room.lastAct=nowSec();
  saveRoom(room);
  broadcast(room);
};

/* ══════════ gin rummy rules engine (mirrors the server) ══════════ */
function listMelds(hand){
  const melds=[];
  const byRank=new Map();
  for(const id of hand){ const r=rankOf(id); if(!byRank.has(r))byRank.set(r,[]); byRank.get(r).push(id); }
  for(const g of byRank.values()){ if(g.length>=3)melds.push(g.slice()); }
  const bySuit=new Map();
  for(const id of hand){ const s=suitOf(id); if(!bySuit.has(s))bySuit.set(s,[]); bySuit.get(s).push(id); }
  for(const g of bySuit.values()){
    const ranks=g.map(rankOf).sort(function(a,b){return a-b;});
    let i=0;
    while(i<ranks.length){
      let j=i;
      while(j+1<ranks.length&&ranks[j+1]===ranks[j]+1)j++;
      if(j-i+1>=3){
        const block=new Set(ranks.slice(i,j+1));
        melds.push(g.filter(function(id){return block.has(rankOf(id));}));
      }
      i=j+1;
    }
  }
  return melds;
}
function bestPartition(hand){
  const melds=listMelds(hand);
  const used=new Array(melds.length).fill(false);
  let bestMelds=[],bestDead=1e9,bestCount=-1;
  function tryFrom(k){
    const usedCards=new Set();
    let dead=0,mcount=0;
    for(let m=0;m<melds.length;m++){ if(used[m]){ mcount++; for(const c of melds[m])usedCards.add(c); } }
    for(const id of hand)if(!usedCards.has(id))dead+=cardValueId(id);
    if(dead<bestDead||(dead===bestDead&&mcount>bestCount)){
      bestDead=dead; bestCount=mcount; bestMelds=[];
      for(let m=0;m<melds.length;m++)if(used[m])bestMelds.push(melds[m].slice());
    }
    for(let m=k;m<melds.length;m++){
      if(used[m])continue;
      let ok=true;
      for(const c of melds[m])if(usedCards.has(c)){ ok=false; break; }
      if(ok){ used[m]=true; tryFrom(m+1); used[m]=false; }
    }
  }
  tryFrom(0);
  const inBest=new Set();
  for(const m of bestMelds)for(const c of m)inBest.add(c);
  return {melds:bestMelds, dead:hand.filter(function(id){return !inBest.has(id);}), deadVal:bestDead};
}
function fitsMeld(card,meld){
  if(!meld.length)return false;
  const r0=rankOf(meld[0]);
  let allSameRank=true, allSameSuit=true;
  for(const m of meld){ if(rankOf(m)!==r0)allSameRank=false; if(suitOf(m)!==suitOf(meld[0]))allSameSuit=false; }
  if(allSameRank)return rankOf(card)===r0;
  if(allSameSuit){
    if(suitOf(card)!==suitOf(meld[0]))return false;
    let mn=11,mx=0;
    for(const m of meld){ const r=rankOf(m); if(r<mn)mn=r; if(r>mx)mx=r; }
    const cr=rankOf(card);
    return cr===mn-1||cr===mx+1;
  }
  return false;
}
function applyLayoff(dead,melds){
  const out=dead.slice();
  let changed=true;
  while(changed){
    changed=false;
    for(let i=0;i<out.length;i++){
      let fits=false;
      for(const m of melds){ if(fitsMeld(out[i],m)){ fits=true; break; } }
      if(fits){ out.splice(i,1); changed=true; break; }
    }
  }
  return out;
}
function deadValOf(dead){ let v=0; for(const c of dead)v+=cardValueId(c); return v; }
function scoreHand(kIdx,dIdx,hK,hD){
  const pk=bestPartition(hK);
  const isGin=pk.deadVal===0;
  const pd=bestPartition(hD);
  if(isGin){
    return {kind:"gin",winner:kIdx,points:pd.deadVal+25,kDead:0,dDead:pd.deadVal,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
  }
  const laid=applyLayoff(pd.dead,pk.melds);
  const dDead=deadValOf(laid);
  if(pk.deadVal<dDead){
    return {kind:"knock",winner:kIdx,points:dDead-pk.deadVal,kDead:pk.deadVal,dDead:dDead,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
  }
  return {kind:"undercut",winner:dIdx,points:(pk.deadVal-dDead)+25,kDead:pk.deadVal,dDead:dDead,kMelds:pk.melds,dMelds:pd.melds,knocker:kIdx};
}

/* ══════════ turn/action gating ══════════ */
function isHumanTurn(){
  if(game.over||game.locked)return false;
  if(game.mode==="online")return game.sockOpen&&game.status===2&&game.handState===1&&game.turn===game.myIndex;
  if(game.mode==="ai")return game.turn===0;

}
function turnActive(){
  if(game.over||game.locked)return false;
  if(game.mode==="online")return game.sockOpen&&game.status===2&&game.handState===1&&game.turn===game.myIndex;
  return true;
}

  if(game.mode==="online")return game.myHand;
  return game.hands[game.turn];
}

  if(!turnActive()||(game.phase!==1&&game.phase!==2))return false;

}

function discardCard(id){
  if(!canDiscard())return;
  if(game.mode==="online"){ onlineDiscard(id); return; }
  const h=game.hands[game.turn];
  const i=h.indexOf(id);
  if(i===-1)return;
  if(game.phase===2&&id===game.lastDrawn)return;
  h.splice(i,1);
  game.discard.push(id);
  game.lastDrawn=-1;
  game.selected=null;
  logMsg(nameOf(game.turn)+" discard"+v3(nameOf(game.turn))+" "+cardShort(id)+".");
  if(game.stock.length<=2){ endHandDraw(); return; }
  game.turn=1-game.turn;
  game.phase=0;
  renderAll();
  maybeAi();
}
function knock(){
  if(!canKnock())return;
  if(game.mode==="online"){ onlineKnock(); return; }
  const t=game.turn;
  const res=scoreHand(t,1-t,game.hands[t],game.hands[1-t]);
  finishHand(t,res);
}
function finishHand(t,res){
  game.over=true;
  game.kind=res.kind; game.handWinner=res.winner;
  game.points=[res.winner===0?res.points:0,res.winner===1?res.points:0];
  game.scores[res.winner]+=res.points;
  game.reveal={hands:[game.hands[0].slice(),game.hands[1].slice()],kMelds:res.kMelds,dMelds:res.dMelds,
    kDead:res.kDead,dDead:res.dDead,kind:res.kind,winner:res.winner,points:res.points,knocker:res.knocker};
  game.kDead=res.kDead; game.dDead=res.dDead;
  if(game.scores[res.winner]>=TARGET){ game.matchOver=true; game.matchWinner=res.winner; }
  logMsg(resultText(res));
  renderAll();
}
function endHandDraw(){
  game.over=true; game.kind="draw"; game.handWinner=null; game.reveal=null;
  logMsg("The stock runs out — the hand is a draw (no score).");
  renderAll();
}
function nextDealerFor(){
  if(game.kind==="draw"||game.handWinner==null)return 1-game.dealer;
  return game.handWinner;
}
function nextHandLocal(){
  game.dealer=nextDealerFor();
  dealLocal();
  renderAll();
  maybeAi();
}
function newMatchLocal(){
  game.scores=[0,0]; game.handNo=0; game.dealer=1-game.dealer;
  game.matchOver=false; game.matchWinner=null;
  game.handWinner=null; game.kind=null;
  nextHandLocal();
}
function maybeAi(){
  if(game.mode==="ai"&&game.turn===1&&!game.over&&!game.locked&&!game.aiThinking)scheduleAI();
}

/* ══════════ AI ══════════ */
function nearMeldCount(hand){
  let near=0;
  for(const id of hand){
    const r=rankOf(id), s=suitOf(id);
    let meldable=false;
    let same=0;
    for(const o of hand)if(o!==id&&rankOf(o)===r)same++;
    if(same>=1)meldable=true;
    const have=function(rr){ for(const o of hand)if(o!==id&&rankOf(o)===rr&&suitOf(o)===s)return true; return false; };
    if(have(r-1)&&have(r+1))meldable=true;
    else if(have(r-1)&&have(r-2))meldable=true;
    else if(have(r+1)&&have(r+2))meldable=true;
    else if(have(r-1)||have(r+1))meldable=true;
    if(meldable)near++;
  }
  return near;
}
function pickDiscard(hand,exclude){
  const p=bestPartition(hand);
  let cands=p.dead;
  if(exclude)cands=cands.filter(function(c){return !exclude.has(c);});
  if(!cands.length)return null;
  const nearSet=new Set();
  for(const c of cands){
    for(const o of cands){ if(o!==c&&rankOf(o)===rankOf(c))nearSet.add(c); }
    for(const o of cands){ if(o!==c&&suitOf(o)===suitOf(c)&&Math.abs(rankOf(o)-rankOf(c))<=2)nearSet.add(c); }
  }
  let best=cands[0],bs=-1;
  for(const c of cands){
    let sc=cardValueId(c)+(nearSet.has(c)?-8:0);
    if(sc>bs){ bs=sc; best=c; }
  }
  return best;
}
function scheduleAI(){
  game.aiThinking=true;
  renderAll();
  setTimeout(function(){
    if(game.mode!=="ai"||game.turn!==1||game.over)return;
    if(game.phase===0){
      const hand=game.hands[1];
      const oldDead=bestPartition(hand).deadVal;
      const top=game.discard[game.discard.length-1];
      const h2=hand.concat([top]);
      const d2=pickDiscard(h2,null);
      if(d2!==null){
        const newDead=bestPartition(h2.filter(function(c){return c!==d2;})).deadVal;
        if(newDead<oldDead)takeDiscard();
        else drawStock();
      }else drawStock();
    }
    if(game.over){ game.aiThinking=false; return; }
    const hand=game.hands[1];
    const p=bestPartition(hand);
    const stuck=nearMeldCount(p.dead)===0;
    if(p.deadVal<=6||(p.deadVal<=10&&stuck)){ knock(); }
    else{
      const ex=game.phase===2?new Set([game.lastDrawn]):null;
      const c=pickDiscard(hand,ex);
      if(c==null)knock();
      else discardCard(c);
    }
    game.aiThinking=false;
    renderAll();
  },480);
}

/* ══════════ rendering ══════════ */
function resultText(res){
  const nm=nameOf(res.winner);
  if(res.kind==="gin")return nm+" go"+(nm==="You"?"":"es")+" GIN! +"+res.points;
  if(res.kind==="knock")return nm+" knock"+v3(nm)+" and win"+v3(nm)+" +"+res.points;
  return nm+" undercut"+v3(nm)+"! +"+res.points;
}
function logMsg(msg){
  const div=document.createElement("div");
  div.className="log-item";
  div.innerHTML="<b>"+(game.handNo>0?("H"+game.handNo):"·")+"</b> "+msg.replace(/&/g,"&amp;").replace(/</g,"&lt;");
  logEl.insertBefore(div,logEl.firstChild);
  while(logEl.children.length>60)logEl.removeChild(logEl.lastChild);
}

function renderHands(){
  oppHand.innerHTML="";
  meHand.innerHTML="";
  const b=bottomIdx(), t=topIdx();
  const tb=handOf(t), tcount=handCountOf(t);
  for(let i=0;i<tcount;i++){
    const el=tb?renderGameCard(tb[i],false):renderGameCard(-1,false);
    oppHand.appendChild(el);
  }
  const bh=handOf(b);
  if(bh){
    bh.slice().sort(function(a,c){ return rankOf(a)-rankOf(c)||suitOf(a)-suitOf(c); }).forEach(function(id,i){
      const el=renderGameCard(id,true);
      if(isHumanTurn()&&canDiscard()){
        el.addEventListener("click",function(){ onCardClick(id); });
      }
      if(game.selected===id)el.classList.add("sel");
      meHand.appendChild(el);
    });
  }else{
    const bcount=handCountOf(b);
    for(let i=0;i<bcount;i++){
      const el=renderGameCard(-1,true);
      meHand.appendChild(el);
    }
  }
}

  discardPile.innerHTML="";
  stockCountEl.textContent=game.stock.length+" in stock";
  discardCountEl.textContent=game.discard.length+" in pile";
  if(game.stock.length>0){
    const s=renderGameCard(-1,false);
    if(canDrawStock()&&isHumanTurn()){
      s.classList.add("pickable");
      s.addEventListener("click",function(){ drawStock(); });
    }
    stockPile.appendChild(s);
  }
  const d=game.discard;
  const show=d.slice(-3);
  show.forEach(function(id,i){
    const el=renderGameCard(id,true);
    const isTop=i===show.length-1;
    el.style.marginRight=isTop?"0":"-64px";
    el.style.zIndex=i+1;
    if(!isTop)el.style.opacity=".5";
    if(isTop&&canTakeDiscard()&&isHumanTurn()){
      el.classList.add("pickable");
      el.addEventListener("click",function(){ takeDiscard(); });
    }
    discardPile.appendChild(el);
  });
}

  const b=bottomIdx(), t=topIdx();
  if(game.over){

  }else if(game.mode==="online"){

    turnEl.textContent=game.aiThinking?"The Machine is thinking…":(game.turn===0?"Your move"+(game.phase===0?" — draw":"") :"The Machine's move");
  }else{
    turnEl.textContent=nameOf(game.turn)+("'s turn")+(game.phase===0?" — draw":"");
  }
  const h=actingHand();
  if(h&&!game.over&&(game.mode!=="ai"||game.turn===0)&&(game.mode!=="online"||game.turn===game.myIndex)){

  }
}

  discardBtn.hidden=!(canDiscard()&&isHumanTurn()&&game.selected!=null);
  knockBtn.hidden=!(canKnock()&&isHumanTurn());
  const hasHand=game.mode==="online"?game.handState>=1:!game.over||game.handNo>0;
  if(game.mode==="online"){
    nextHandBtn.hidden=!(game.status===2&&(game.handState===2||game.handState===3));
    nextHandBtn.textContent=game.handState===3?"Rematch":"Next hand";
    claimBtn.hidden=!(game.status===2&&!game.oppPresent&&game.handState>=1);
    menuBtn.textContent="Exit match";
  }else{
    if(game.over&&game.matchOver){ nextHandBtn.textContent="New match"; nextHandBtn.hidden=false; }
    else if(game.over){ nextHandBtn.textContent="Next hand"; nextHandBtn.hidden=false; }
    else nextHandBtn.hidden=true;
    claimBtn.hidden=true;
    menuBtn.textContent="Main menu";
  }
}
function renderBanner(){
  let msg="";
  if(game.mode==="online"){
    if(game.handState===3)msg="Match over — "+(game.matchWinner===game.myIndex?"you win!":"your opponent wins.");
    else if(game.handState===2&&game.reveal)msg="Hand over — "+(game.reveal.winner===game.myIndex?"you ":"opponent ")+"+"+game.reveal.points+".";
    else if(game.handState===2)msg="Stock ran out — hand is a draw.";
    else if(game.status===1)msg="Waiting for an opponent to join…";
    else if(!game.oppPresent)msg="Your opponent disconnected — they can rejoin with the code. Claim the win anytime.";
    else if(game.turn===game.myIndex)msg=game.phase===0?"Draw from the stock, or take the top discard.":"Discard a card — or knock if your deadwood is ≤ 10.";
    else msg="Waiting for your opponent…";
  }else if(game.over){
    msg=game.matchOver
      ?(nameOf(game.matchWinner)+" wins the match "+game.scores[0]+"–"+game.scores[1]+"!")
      :(game.kind==="draw"?"No score — the hand is a draw."
        :nameOf(game.handWinner)+" takes the hand! ("+resultText({kind:game.kind,winner:game.handWinner,points:game.points[game.handWinner]})+")");
  }else if(game.mode==="ai"){
    msg=game.aiThinking?"The Machine is studying the discard…":(game.turn===0?(game.phase===0?"Draw from the stock, or take the top discard.":"Discard a card — or knock if your deadwood is ≤ 10."):"");
  }else{
    msg=game.phase===0?(nameOf(game.turn)+" — draw from the stock, or take the top discard."):(nameOf(game.turn)+" — discard a card, or knock if deadwood is ≤ 10.");
  }
  bannerEl.textContent=msg;
}
function renderReveal(){
  const r=game.reveal;
  revealWrap.hidden=!r;
  if(!r)return;
  const kindName=r.kind==="gin"?"GIN":r.kind==="knock"?"Knock":"Undercut";
  const w=nameOf(r.winner);
  if(r.kind==="gin")revealTitleEl.textContent=w+" go"+(w==="You"?"":"es")+" GIN — +"+r.points+"!";
  else if(r.kind==="knock")revealTitleEl.textContent=w+" knocks ("+r.kDead+" deadwood) — +"+r.points+"!";
  else revealTitleEl.textContent=w+" undercuts! +"+r.points+" ("+r.kDead+" vs "+r.dDead+")";
  revealRows.innerHTML="";
  const order=r.winner===r.knocker?[r.knocker,1-r.knocker]:[r.winner,1-r.winner];
  for(const p of order){
    const row=document.createElement("div");
    row.className="bgn-revrow";
    const nm=document.createElement("div");
    nm.className="bgn-revname";
    nm.textContent=nameOf(p)+(p===r.winner?" ✓":"");
    row.appendChild(nm);
    const melds=p===r.knocker?r.kMelds:r.dMelds;
    const cards=r.hands[p];
    const meldSet=new Set();
    for(const m of melds)for(const c of m)meldSet.add(c);
    const dead=cards.filter(function(c){return !meldSet.has(c);});
    for(const m of melds){
      const grp=document.createElement("div");
      grp.className="bgn-revgroup";
      for(const c of m){ const el=renderGameCard(c,true); el.classList.add("meldcard"); grp.appendChild(el); }
      row.appendChild(grp);
    }
    if(dead.length){
      if(melds.length){ const plus=document.createElement("span"); plus.className="bgn-revplus"; plus.textContent="+"; row.appendChild(plus); }
      const grp=document.createElement("div");
      grp.className="bgn-revdead";
      for(const c of dead){ const el=renderGameCard(c,true); el.classList.add("deadcard"); grp.appendChild(el); }
      row.appendChild(grp);
    }
    const tot=document.createElement("span");
    tot.className="bgn-revplus";
    tot.textContent="= "+(p===r.knocker?r.kDead:r.dDead)+" deadwood";
    row.appendChild(tot);
    revealRows.appendChild(row);
  }
}

function onCardClick(id){
  if(game.over||game.locked)return;
  if(!isHumanTurn()||!canDiscard())return;
  if(id===game.lastDrawn&&game.phase===2)return;
  if(game.selected===id)game.selected=null;
  else game.selected=id;
  renderAll();
}

  if(game.selected==null)return;

  game.selected=null;
  renderAll();

function resetMatch(){
  game.scores=[0,0]; game.handNo=0; game.dealer=Math.random()<0.5?0:1;
  game.matchOver=false; game.matchWinner=null;
  game.over=false; game.handWinner=null; game.kind=null; game.points=[0,0];
  game.reveal=null; game.selected=null; game.aiThinking=false;
}
function startMode(mode){
  if(game.mode==="online"&&game.socket)exitOnline(false);
  resetMatch();
  game.mode=mode;
  roomBox.hidden=true;
  claimBtn.hidden=true;
  lobbyCtn.hidden=true;
  gameCtn.hidden=false;
  dealLocal();
  renderAll();
  maybeAi();
}
function exitOnline(showLobby){
  if(game.socket){ game.wantReconnect=false; try{ game.socket.close(1000,"bye"); }catch(e){} game.socket=null; }
  game.sockOpen=false;
  game.roomCode=null; game.myIndex=null; game.myRole=null;
  if(showLobby){ gameCtn.hidden=true; lobbyCtn.hidden=false; }
}

/* ══════════ online client ══════════ */
function openedP(){
  return new Promise(function(res){
    if(game.sockOpen)res(true);
    else game.openResolvers.push(res);
  });
}
function connectSocket(){
  if(game.socket)return;
  if(!rootRef.createServerSocket){ setConnLabel("Online unavailable."); return; }
  try{
    game.socket=rootRef.createServerSocket();
  }catch(e){ setConnLabel("Online unavailable."); return; }
  game.socket.addEventListener("open",function(){
    game.sockOpen=true;
    game.retryDelay=800;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(true); });
    setConnLabel();
    if(game.roomCode)rejoinRoom();
    renderAll();
  });
  game.socket.addEventListener("message",function(ev){
    if(game.mode==="online")onServerMsg(ev.data);
  });
  game.socket.addEventListener("close",function(ev){
    game.sockOpen=false;
    const rs=game.openResolvers; game.openResolvers=[];
    rs.forEach(function(r){ r(false); });
    setConnLabel("Disconnected");
    renderAll();
    if(!game.wantReconnect||game.mode!=="online")return;
    if(ev.code===4403){ game.wantReconnect=false; setConnLabel("Online is unavailable on this page."); return; }
    setTimeout(function(){
      if(game.wantReconnect&&game.mode==="online"){
        game.retryDelay=Math.min(game.retryDelay*1.7,8000);
        game.socket=null;
        connectSocket();
      }
    },game.retryDelay);
  });
}
function rpc(method,data){
  return new Promise(function(res,rej){
    if(!game.socket||!game.sockOpen)return rej(new Error("socket not open"));
    game.socket.rpc[method](data).then(function(r){
      try{ res(JSON.parse(r)); }catch(e){ res({ok:false,err:"parse"}); }
    }).catch(rej);
  });
}
function setConnLabel(txt){
  connEl.textContent=txt!==undefined?txt:(game.sockOpen?"Connected — share the code with a friend.":"Connecting…");
}
function setOnlineMsg(t){ onlineMsgEl.textContent=t; }
function onServerMsg(raw){
  let msg;
  try{ msg=JSON.parse(raw); }catch(e){ return; }
  if(msg.t==="snap")applySnap(msg.snap);
  else if(msg.t==="hand")applyHandMsg(msg);
}
function applyHandMsg(msg){
  game.myHand=msg.hand;
  game.handNo=msg.handNo;
  game.selected=null;
  renderAll();
}
function applySnap(s){
  game.status=s.status;
  game.handNo=s.handNo;
  game.dealer=s.dealer;
  game.turn=s.turn;
  game.phase=s.phase;
  game.handCounts=[s.handCounts[0],s.handCounts[1]];
  game.scores=[s.scores[0],s.scores[1]];
  game.handState=s.handState;
  game.handWinner=s.handWinner;
  game.matchWinner=s.matchWinner;
  game.kind=s.kind;
  game.points=[s.points[0],s.points[1]];
  game.stock=s.stockCount===0?[]:new Array(s.stockCount); // contents hidden
  for(let i=0;i<game.stock.length;i++)game.stock[i]=-1;
  game.discard=s.discard.slice();
  game.kDead=s.kDead; game.dDead=s.dDead;
  game.oppPresent=game.myIndex===0?s.guestIn:s.hostIn;
  game.over=(s.handState===2||s.handState===3);
  game.matchOver=(s.handState===3);
  game.matchWinner=s.matchWinner;
  game.reveal=s.reveal;
  if(game.myHand&&s.handCounts[game.myIndex]!==game.myHand.length){
    rpc("getRoom",game.roomCode).then(function(r){
      if(r&&r.ok&&r.hand){ game.myHand=r.hand; game.selected=null; renderAll(); }
    }).catch(function(){});
  }
  renderAll();
}
function enterOnlineGame(){
  lobbyCtn.hidden=true;
  gameCtn.hidden=false;
  roomBox.hidden=false;
  roomCodeEl.textContent=game.roomCode;
  resetMatch();
  renderAll();
}
async function doCreateRoom(){
  setOnlineMsg("Creating room…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("createRoom","");
    if(!r.ok){ setOnlineMsg("Couldn't create a room ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=r.code; game.myRole=r.role; game.myIndex=r.role==="host"?0:1;
    setOnlineMsg("");
    if(r.snap)applySnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
async function doJoinRoom(code){
  setOnlineMsg("Joining room "+code+"…");
  const ok=await openedP();
  if(!ok){ setOnlineMsg("Couldn't reach the server."); return; }
  try{
    const r=await rpc("joinRoom",code);
    if(!r.ok){ setOnlineMsg("Couldn't join ("+r.err+")."); return; }
    game.mode="online";
    game.roomCode=code; game.myRole=r.role; game.myIndex=r.role==="host"?0:1;
    setOnlineMsg("");
    if(r.hand)game.myHand=r.hand;
    if(r.snap)applySnap(r.snap);
    enterOnlineGame();
  }catch(e){ setOnlineMsg("Server error — please retry."); }
}
function rejoinRoom(){
  if(!game.roomCode)return;
  rpc("joinRoom",game.roomCode).then(function(r){
    if(!r.ok){
      setConnLabel("Rejoin failed ("+r.err+").");
      if(r.err==="room_full"&&game.mode==="online"&&game.sockOpen){
        setTimeout(function(){ if(game.mode==="online"&&game.sockOpen)rejoinRoom(); },1500);
      }
      return;
    }
    game.myRole=r.role; game.myIndex=r.role==="host"?0:1;
    if(r.hand)game.myHand=r.hand;
    if(r.snap)applySnap(r.snap);
    setConnLabel();
  }).catch(function(){ setConnLabel("Rejoin failed."); });
}
async function onlineDraw(which){
  game.locked=true;
  renderAll();
  try{
    const r=await rpc("draw",JSON.stringify({code:game.roomCode,t:which}));
    if(r&&r.ok){
      if(game.myHand)game.myHand.push(r.card);
      game.phase=r.phase;
      if(which==="discard")game.lastDrawn=r.card;
      game.selected=null;
      logMsg("You "+(which==="discard"?"take the discard ("+cardShort(r.card)+").":"draw from the stock."));
    }else{
      flashBanner("Move rejected ("+(r&&r.err?r.err:"error")+")");
    }
  }catch(e){ flashBanner("Couldn't send your move — reconnecting…"); }
  game.locked=false;
  renderAll();
}
async function onlineDiscard(id){
  game.locked=true;
  renderAll();
  try{
    const r=await rpc("discard",JSON.stringify({code:game.roomCode,card:id}));
    if(r&&r.ok){
      const i=game.myHand.indexOf(id);
      if(i!==-1)game.myHand.splice(i,1);
      game.selected=null;
      logMsg("You discard "+cardShort(id)+".");
    }else{
      flashBanner("Move rejected ("+(r&&r.err?r.err:"error")+")");
    }
  }catch(e){ flashBanner("Couldn't send your move — reconnecting…"); }
  game.locked=false;
  renderAll();
}
async function onlineKnock(){
  game.locked=true;
  renderAll();
  try{
    const r=await rpc("knock",JSON.stringify({code:game.roomCode}));
    if(!r||!r.ok){
      flashBanner("Knock rejected ("+(r&&r.err?r.err:"error")+")");
    }
  }catch(e){ flashBanner("Couldn't send your move — reconnecting…"); }
  game.locked=false;
  renderAll();
}
async function onlineNextHand(){
  game.locked=true;
  try{
    const r=await rpc("nextHand",JSON.stringify({code:game.roomCode}));
    if(!r.ok)flashBanner("Couldn't start ("+(r.err||"error")+").");
  }catch(e){ flashBanner("Couldn't start the next hand."); }
  game.locked=false;
}
function flashBanner(msg){
  bannerEl.textContent=msg;
  setTimeout(function(){ renderBanner(); },2600);
}
function exitToLobby(){
  if(game.mode==="online"){
    if(game.socket&&game.sockOpen){
      rpc("leave",game.roomCode).catch(function(){});
    }
    exitOnline(true);
  }else{
    game.mode=null;
    gameCtn.hidden=true;
    lobbyCtn.hidden=false;
  }
}
